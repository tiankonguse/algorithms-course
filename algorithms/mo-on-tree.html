树上莫队

一条路径也能变成一段区间。dfs 时每个点进出各记一次得到括号序，路径对应其中连续一段，段里出现两次的点不算数。

零、背景

树上两点之间的路径查询，能合并的信息一般用树链剖分配线段树。碰上「路径上有几种不同的颜色」这类不好合并的询问，那条路就走不通了。

这类查询往往是离线的，于是可以把莫队搬上来。要做的事只有一件：把路径变成数组上的一段区间。

一、问题

一共有 n 个节点，每个节点给定一个颜色值，给定 n - 1 条边，所有节点连成一棵树。一共 m 条查询 u v，打印点 u 到点 v 的简单路径上有几种不同的颜色。

1 <= n <= 4 * 10^4，1 <= m <= 10^5，1 <= 颜色值 <= 2 * 10^9。测试链接：SP10707。

二、朴素做法

每条查询从 u 走一遍到 v，路上用 set 收集颜色。一次最坏 O(n)，m 条就是 O(nm)。n = 4 * 10^4、m = 10^5 时是 4 * 10^9，超时。

树链剖分配数据结构这条路也堵着：颜色种类数没法由两段路径的种类数合并出来，除非给每种颜色开一个 bitset 做或运算，那是 O(n / 64) 一次合并，n 大时同样吃力。

三、括号序

从根开始 dfs，进入节点 x 时把 x 记一次，离开 x 时再记一次。记下来的序列叫括号序，记为 seg，长度是 2n。进入那次的位置叫 st[x]，离开那次的位置叫 ed[x]。

注意这不是欧拉序。欧拉序是每走一条边记一次，括号序是每个点记两次。

四、路径变成区间

任取两个点 u、v，不妨设 st[u] < st[v]，记 lca 是它们的最低公共祖先。分两种情况。

如果 u 或者 v 本身就是 lca，路径就对应 seg[st[u]..st[v]] 这一段。

如果都不是，路径对应 seg[ed[u]..st[v]] 这一段，另外单独把 lca 这个点加上。

段里的点可能出现两次，出现两次说明这个点的子树被完整地走完了，它不在路径上，应当忽略；出现一次才算生效。

第二条里的 lca 为什么要单独加，画一遍就清楚：lca 的进入发生在 ed[u] 之前，离开发生在 st[v] 之后，它在这一段里一次都没出现。

于是每条路径查询都变成括号序上的一段区间，剩下的事情普通莫队全包了：窗口两端滑动，进来一个节点就把它的状态取反，奇数次算生效、偶数次算无效。

六、题目1 的代码

颜色值到 2 * 10^9，先离散化。括号序用递归 dfs 生成，lca 用倍增表，之后就是普通莫队的 invert：状态取反，计数到 0 减一，从 0 变 1 加一。

#include <bits/stdc++.h>

    using namespace std;

    struct Query {
        int l, r, lca, id;
    };

    const int MAXN = 40001;
    const int MAXM = 100001;
    const int MAXP = 20;

    int n, m;
    int color[MAXN];
    int sorted[MAXN];
    int cntv;

    Query query[MAXM];

    int head[MAXN];
    int to[MAXN << 1];
    int nxt[MAXN << 1];
    int cntg;

    int dep[MAXN];
    int seg[MAXN << 1];
    int st[MAXN];
    int ed[MAXN];
    int stjump[MAXN][MAXP];
    int cntd;

    int bi[MAXN << 1];

    bool vis[MAXN];
    int cnt[MAXN];
    int kind;

    int ans[MAXM];

    void addEdge(int u, int v) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        head[u] = cntg;
    }

    int kth(int num) {
        int left = 1, right = cntv, mid, ret = 0;
        while (left <= right) {
            mid = (left + right) >> 1;
            if (sorted[mid] <= num) {
                ret = mid;
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        return ret;
    }

    void dfs(int u, int fa) {
        dep[u] = dep[fa] + 1;
        seg[++cntd] = u;
        st[u] = cntd;
        stjump[u][0] = fa;
        for (int p = 1; p < MAXP; p++) {
            stjump[u][p] = stjump[stjump[u][p - 1]][p - 1];
        }
        for (int e = head[u], v; e > 0; e = nxt[e]) {
            v = to[e];
            if (v != fa) {
                dfs(v, u);
            }
        }
        seg[++cntd] = u;
        ed[u] = cntd;
    }

    int lca(int a, int b) {
        if (dep[a] < dep[b]) {
            swap(a, b);
        }
        for (int p = MAXP - 1; p >= 0; p--) {
            if (dep[stjump[a][p]] >= dep[b]) {
                a = stjump[a][p];
            }
        }
        if (a == b) {
            return a;
        }
        for (int p = MAXP - 1; p >= 0; p--) {
            if (stjump[a][p] != stjump[b][p]) {
                a = stjump[a][p];
                b = stjump[b][p];
            }
        }
        return stjump[a][0];
    }

    bool QueryCmp(Query &a, Query &b) {
        if (bi[a.l] != bi[b.l]) {
            return bi[a.l] < bi[b.l];
        }
        return a.r < b.r;
    }

    void invert(int node) {
        int val = color[node];
        if (vis[node]) {
            if (--cnt[val] == 0) {
                kind--;
            }
        } else {
            if (++cnt[val] == 1) {
                kind++;
            }
        }
        vis[node] = !vis[node];
    }

    void compute() {
        int winl = 1, winr = 0;
        for (int i = 1; i <= m; i++) {
            int jobl = query[i].l;
            int jobr = query[i].r;
            int lca = query[i].lca;
            int id = query[i].id;
            while (winl > jobl) {
                invert(seg[--winl]);
            }
            while (winr < jobr) {
                invert(seg[++winr]);
            }
            while (winl < jobl) {
                invert(seg[winl++]);
            }
            while (winr > jobr) {
                invert(seg[winr--]);
            }
            if (lca > 0) {
                invert(lca);
            }
            ans[id] = kind;
            if (lca > 0) {
                invert(lca);
            }
        }
    }

    void prepare() {
        for (int i = 1; i <= n; i++) {
            sorted[i] = color[i];
        }
        sort(sorted + 1, sorted + n + 1);
        cntv = 1;
        for (int i = 2; i <= n; i++) {
            if (sorted[cntv] != sorted[i]) {
                sorted[++cntv] = sorted[i];
            }
        }
        for (int i = 1; i <= n; i++) {
            color[i] = kth(color[i]);
        }
        int blen = (int) sqrt(cntd);
        for (int i = 1; i <= cntd; i++) {
            bi[i] = (i - 1) / blen + 1;
        }
        sort(query + 1, query + m + 1, QueryCmp);
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            cin >> color[i];
        }
        for (int i = 1, u, v; i < n; i++) {
            cin >> u >> v;
            addEdge(u, v);
            addEdge(v, u);
        }
        dfs(1, 0);
        for (int i = 1, u, v, uvlca; i <= m; i++) {
            cin >> u >> v;
            if (st[v] < st[u]) {
                swap(u, v);
            }
            uvlca = lca(u, v);
            if (u == uvlca) {
                query[i].l = st[u];
                query[i].r = st[v];
                query[i].lca = 0;
            } else {
                query[i].l = ed[u];
                query[i].r = st[v];
                query[i].lca = uvlca;
            }
            query[i].id = i;
        }
        prepare();
        compute();
        for (int i = 1; i <= m; i++) {
            cout << ans[i] << '\n';
        }
        return 0;
    }

七、糖果公园：树上带修莫队

一共有 n 个公园，n - 1 条边连成一棵树，c[i] 是 i 号公园的糖果型号。一共有 m 种糖果，v[y] 是 y 号糖果的美味指数，w 是长度为 n 的数组。游客第 x 次遇到 y 号糖果时，愉悦值增加 v[y] * w[x]。接下来 q 条操作。

操作 0 x y：把 x 号公园的糖果型号改成 y。操作 1 x y：游客从 x 走到 y，打印最终的愉悦值。1 <= n、m、q <= 10^5，1 <= v[i]、w[i] <= 10^6。测试链接：P4074。

路径照样用括号序转成区间，多出来的是修改。带修莫队加一个时间指针 wint，表示窗口处在第几次修改之后；查询带上 t，排序时多一个关键字。

排序规则变成三级：先按 l 所在块，再按 r 所在块，最后按 t。块长也不再是 √n，而是 n^(2/3)。

时间指针的移动用「交换」来做：update[t] 记下位置和要改成的值，前进时把当前值换进去、旧值存回 update[t]，后退时再换一次就回到原样。如果修改的那个点此刻在窗口里，先把它从答案里摘掉，换完颜色再放回去。

lca 的处理和上一题一样，答完立刻取反还原。

这里有个坑：写 query[++cntq] = {st[x], st[y], cntu, 0, cntq} 时，C++17 起赋值号右边先算，cntq 还是旧值，id 会填成上一个下标。下面拆成两行写。

#include <bits/stdc++.h>

    using namespace std;

    struct Query {
        int l, r, t, lca, id;
    };

    struct Update {
        int pos, val;
    };

    const int MAXN = 100001;
    const int MAXP = 20;
    int n, m, q;
    int v[MAXN];
    int w[MAXN];
    int c[MAXN];

    int head[MAXN];
    int to[MAXN << 1];
    int nxt[MAXN << 1];
    int cntg;

    Query query[MAXN];
    Update update[MAXN];
    int cntq, cntu;

    int dep[MAXN];
    int seg[MAXN << 1];
    int st[MAXN];
    int ed[MAXN];
    int stjump[MAXN][MAXP];
    int cntd;

    int bi[MAXN << 1];
    bool vis[MAXN];
    int cnt[MAXN];
    long long happy;
    long long ans[MAXN];

    void addEdge(int u, int v) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        head[u] = cntg;
    }

    void dfs(int u, int fa) {
        dep[u] = dep[fa] + 1;
        seg[++cntd] = u;
        st[u] = cntd;
        stjump[u][0] = fa;
        for (int p = 1; p < MAXP; p++) {
            stjump[u][p] = stjump[stjump[u][p - 1]][p - 1];
        }
        for (int e = head[u], v; e > 0; e = nxt[e]) {
            v = to[e];
            if (v != fa) {
                dfs(v, u);
            }
        }
        seg[++cntd] = u;
        ed[u] = cntd;
    }

    int lca(int a, int b) {
        if (dep[a] < dep[b]) {
            swap(a, b);
        }
        for (int p = MAXP - 1; p >= 0; p--) {
            if (dep[stjump[a][p]] >= dep[b]) {
                a = stjump[a][p];
            }
        }
        if (a == b) {
            return a;
        }
        for (int p = MAXP - 1; p >= 0; p--) {
            if (stjump[a][p] != stjump[b][p]) {
                a = stjump[a][p];
                b = stjump[b][p];
            }
        }
        return stjump[a][0];
    }

    bool QueryCmp(Query &a, Query &b) {
        if (bi[a.l] != bi[b.l]) {
            return bi[a.l] < bi[b.l];
        }
        if (bi[a.r] != bi[b.r]) {
            return bi[a.r] < bi[b.r];
        }
        return a.t < b.t;
    }

    void invert(int node) {
        int candy = c[node];
        if (vis[node]) {
            happy -= 1LL * v[candy] * w[cnt[candy]--];
        } else {
            happy += 1LL * v[candy] * w[++cnt[candy]];
        }
        vis[node] = !vis[node];
    }

    void moveTime(int tim) {
        int pos = update[tim].pos;
        int oldVal = c[pos];
        int newVal = update[tim].val;
        if (vis[pos]) {
            invert(pos);
            c[pos] = newVal;
            update[tim].val = oldVal;
            invert(pos);
        } else {
            c[pos] = newVal;
            update[tim].val = oldVal;
        }
    }

    void compute() {
        int winl = 1, winr = 0, wint = 0;
        for (int i = 1; i <= cntq; i++) {
            int jobl = query[i].l;
            int jobr = query[i].r;
            int jobt = query[i].t;
            int lca = query[i].lca;
            int id = query[i].id;
            while (winl > jobl) {
                invert(seg[--winl]);
            }
            while (winr < jobr) {
                invert(seg[++winr]);
            }
            while (winl < jobl) {
                invert(seg[winl++]);
            }
            while (winr > jobr) {
                invert(seg[winr--]);
            }
            while (wint < jobt) {
                moveTime(++wint);
            }
            while (wint > jobt) {
                moveTime(wint--);
            }
            if (lca > 0) {
                invert(lca);
            }
            ans[id] = happy;
            if (lca > 0) {
                invert(lca);
            }
        }
    }

    void prapare() {
        int blen = max(1, (int)pow(cntd, 2.0 / 3.0));
        for (int i = 1; i <= cntd; i++) {
            bi[i] = (i - 1) / blen + 1;
        }
        sort(query + 1, query + cntq + 1, QueryCmp);
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m >> q;
        for (int i = 1; i <= m; i++) {
            cin >> v[i];
        }
        for (int i = 1; i <= n; i++) {
            cin >> w[i];
        }
        for (int i = 1, u, v; i < n; i++) {
            cin >> u >> v;
            addEdge(u, v);
            addEdge(v, u);
        }
        for (int i = 1; i <= n; i++) {
            cin >> c[i];
        }
        dfs(1, 0);
        for (int i = 1, op, x, y; i <= q; i++) {
            cin >> op >> x >> y;
            if (op == 0) {
                cntu++;
                update[cntu].pos = x;
                update[cntu].val = y;
            } else {
                if (st[x] > st[y]) {
                    swap(x, y);
                }
                int xylca = lca(x, y);
                if (x == xylca) {
                    query[++cntq] = {st[x], st[y], cntu, 0, 0};
                    query[cntq].id = cntq;
                } else {
                    query[++cntq] = {ed[x], st[y], cntu, xylca, 0};
                    query[cntq].id = cntq;
                }
            }
        }
        prapare();
        compute();
        for (int i = 1; i <= cntq; i++) {
            cout << ans[i] << '\n';
        }
        return 0;
    }

八、复杂度分析

环节复杂度说明
括号序与倍增表O(n log n)一次 dfs 生成 seg、st、ed，顺带填 stjump 表
每条查询转区间O(log n)一次 lca
树上莫队（无修改）O((n + m)√n)括号序长 2n，与普通莫队同量级
树上带修莫队O(n^(5/3))块长取 n^(2/3)，三级排序后指针总移动是这个量级
额外空间O(n + m)seg、颜色计数、查询与修改数组

括号序的长度是 2n，代公式时把 n 换成 2n 即可，量级不变。

九、最后

树上莫队能做的事取决于括号序这一段上能维护什么。不同颜色数、路径上的众数、路径上满足某个条件的点对数，都能套进来；带修改就再加一个时间维。

回滚莫队同样能搬到树上：括号序也是数组，add 写不出来的就走只增那套，del 写不出来的走只删那套。
