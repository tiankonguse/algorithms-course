卡特兰数：四个公式与进出栈模型

卡特兰数序列是 1, 1, 2, 5, 14, 42, 132, 429…，一列数装下进出栈顺序、圆上连线、红黑前缀、二叉树结构、凸多边形剖分这么多互不相干的计数问题。这一页先把序列和四个计算公式摆清楚，再重点图解进出栈模型怎么落到公式1。

零、卡特兰数与四个公式

下标从 0 开始：C0=1, C1=1, C2=2, C3=5, C4=14。增长很快，n=20 已经 6564120420，超出 int 范围，n 稍大就要考虑取模。

四个公式算出来的全是卡特兰数，区别在时间复杂度和用不用除法同余：

公式1、公式2、公式3 都是 O(n)，公式4 是 O(n²)。公式4 没有除法，所以不需要阶乘余数表、阶乘逆元表、连续数逆元表，代价是慢。怎么选，看数据规模和取模要求：p 是质数才能用费马小定理求逆元，p 是合数时公式1、2、3 里的除法全部失效，只能退回公式4，或者用下一页的因子计数法。

一、进出栈模型

进栈顺序规定为 1、2、3…n，问有多少种不同的出栈顺序。把 push 记成 +1、pop 记成 −1，一个合法操作序列就是 2n 个 ±1 且任意前缀和 ≥ 0 的数列，这样的数列恰好有 C(2n,n) − C(2n,n−1) 个，也就是卡特兰数公式1。

同一套双射还能数别的问题。卖票人没钱找零，n 个游客只带 5 元、n 个只带 10 元，顺利找零的排队方法数；n 个 1 和 n 个 −1 组成任意前缀和 ≥ 0 的数列数；圆上 2n 个点两两连成不相交线段的连接方法数。全部是卡特兰数，后面两页会逐个落地。

二、公式怎么选

做题时按数据状况挑公式。n ≤ 35 时答案不溢出 long，常用公式4 直接算真实值；n 是 10³ 规模且 p 为质数，公式1、2、3、4 都行；n 是 10³ 规模但 p 可能为合数，只能用公式4；n 是 10⁶ 规模且 p 为质数，用公式1、2、3；n 是 10⁶ 规模且 p 可能为合数，公式2 + 因子计数法，这就是本页后面的重点。n 到 10³ 还要打印完整真实答案的，见树屋阶梯那页的大数做法。

三、题目1：卡特兰数模版

洛谷 P1044。进栈顺序 1..n，求不同出栈顺序数，答案对 1e9+7 取模。下面是完整四公式实现，主函数走公式1。公式1 要两张表：阶乘余数表 fac 和阶乘逆元表 inv1；公式3 要连续数逆元表 inv2；公式4 什么都不用。

这里有个细节：inv1 的生成是从大到小倒推的，inv1[n] 先用快速幂求出，再递推 inv1[i] = (i+1)·inv1[i+1]，比逐个求逆元省一个 O(n log MOD)。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;

    // 卡特兰数模版：进栈顺序 1..n，返回不同出栈顺序数，答案对 1000000007 取模
    const int MOD = 1000000007;
    const int MAXN = 1000001;

    ll fac[MAXN];  // 阶乘余数表
    ll inv1[MAXN]; // 阶乘逆元表
    ll inv2[MAXN]; // 连续数逆元表

    ll power(ll x, ll p) {
    	ll ans = 1;
    	while (p > 0) {
    		if (p & 1) {
    			ans = ans * x % MOD;
    		}
    		x = x * x % MOD;
    		p >>= 1;
    	}
    	return ans;
    }

    // 生成阶乘余数表、阶乘逆元表
    void build1(int n) {
    	fac[0] = inv1[0] = 1;
    	fac[1] = 1;
    	for (int i = 2; i <= n; i++) {
    		fac[i] = (ll)i * fac[i - 1] % MOD;
    	}
    	inv1[n] = power(fac[n], MOD - 2);
    	for (int i = n - 1; i >= 1; i--) {
    		inv1[i] = (ll)(i + 1) * inv1[i + 1] % MOD;
    	}
    }

    // 生成连续数逆元表
    void build2(int n) {
    	inv2[1] = 1;
    	for (int i = 2; i <= n + 1; i++) {
    		inv2[i] = MOD - inv2[MOD % i] * (MOD / i) % MOD;
    	}
    }

    ll c(int n, int k) {
    	return fac[n] * inv1[k] % MOD * inv1[n - k] % MOD;
    }

    // 公式1
    ll compute1(int n) {
    	build1(2 * n);
    	return (c(2 * n, n) - c(2 * n, n - 1) + MOD) % MOD;
    }

    // 公式2
    ll compute2(int n) {
    	build1(2 * n);
    	return c(2 * n, n) * power(n + 1, MOD - 2) % MOD;
    }

    // 公式3
    ll compute3(int n) {
    	build2(n);
    	ll f[MAXN];
    	f[0] = f[1] = 1;
    	for (int i = 2; i <= n; i++) {
    		f[i] = f[i - 1] * (4 * i - 2) % MOD * inv2[i + 1] % MOD;
    	}
    	return f[n];
    }

    // 公式4
    ll compute4(int n) {
    	ll f[MAXN];
    	f[0] = f[1] = 1;
    	for (int i = 2; i <= n; i++) {
    		for (int l = 0, r = i - 1; l < i; l++, r--) {
    			f[i] = (f[i] + f[l] * f[r] % MOD) % MOD;
    		}
    	}
    	return f[n];
    }

    int main() {
    	int n;
    	cin >> n;
    	cout << compute1(n) << endl;
    	return 0;
    }

四、题目4：不同结构的二叉树数量

n 个无差别节点能形成多少种不同结构的二叉树。这属于划分左右相乘模型：根的左子树有 i 个节点、右子树有 n−1−i 个节点，左右方案数相乘再求和，正是公式4 的形状。n ≤ 19，int 直接装得下，不用取模。

#include <bits/stdc++.h>
    using namespace std;

    // 不同结构的二叉树数量：n 个无差别节点能形成多少种不同结构
    // 1 <= n <= 19，数据量小不用考虑溢出和取模，直接使用公式4
    int numTrees(int n) {
    	int f[20] = {};
    	f[0] = f[1] = 1;
    	for (int i = 2; i <= n; i++) {
    		for (int l = 0, r = i - 1; l < i; l++, r--) {
    			f[i] += f[l] * f[r];
    		}
    	}
    	return f[n];
    }

    int main() {
    	int n;
    	cin >> n;
    	cout << numTrees(n) << endl;
    	return 0;
    }

总结

卡特兰数的四个公式共享同一串数：公式1 是 C(2n,n) − C(2n,n−1)，公式2 是 C(2n,n)/(n+1)，公式3 是一阶递推 f(n)=f(n−1)(4n−2)/(n+1)，公式4 是左右相乘求和。进出栈、圆上连线、红黑前缀、二叉树、凸多边形剖分都是同一串数。选公式看两条：n 的规模定 O(n) 还是 O(n²)，p 是否质数定能不能用除法同余。下一页的路径计数模型把题 6、题 2、题 3 三个变体一起落地。
