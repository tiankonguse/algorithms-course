割边与边双连通分量

这一讲讲的是无向图上最基础的一件事：删掉哪条边会让图断开。这样的边叫割边，也常叫桥。把桥都找出来，剩下的极大块就是边双连通分量；把每个块缩成一个点，图就成了一棵树。整讲七道题，前四道只用到割边，后三道在缩点后的树上做。这一页先把两个模版（求割边、求边双并缩点）讲清楚，题目下一节集中看。

零、背景

无向图里，「删掉这条边，连通块变多」的边叫割边，也叫桥。等价的说法是：这条边不属于任何一个环。环上的边删掉一条，环上其余的点还能绕另一侧走通；连接两个环、或者一条挂在环外的尾巴，删掉就断了。

先把无向图建图的技巧摆出来，后面两页都靠它。边的编号从 2 开始，正反两条边编号相邻：如果一条边是 e，它的反向边就是 e ^ 1，这条无向边的编号是 e >> 1。

这样建图有两个好处。一是 tarjan 里判断「来时那条边」只要比较 (e ^ 1) == preEdge，不用记父点，重边和自环也不会误判；二是按 e >> 1 就能直接给无向边编号存标记。

一、Tarjan 求割边

求割边是 tarjan 求强连通分量的一处微调，还是那两个数组：dfn[u] 是 u 第一次被访问的时间，low[u] 是 u 及其子树能回到的最小 dfn。

dfs 时遇到一条边，分三种情况。到没访问过的点是树边，递归下去，回来用 low[v] 更新 low[u]；到访问过、且不是来时边的点是回边，用 dfn[v] 更新 low[u]；来时那条边直接跳过。

判定的位置在树边回溯处：如果 low[v] > dfn[u]，说明 v 的子树里没有任何一条回边能回到 u 或更早，删掉 u-v 就断了，它是割边。

这里有个细节：更新 low[u] 时必须传 preEdge 而不是父点。重边的情况下，指向父点的那条平行边是一条合法的回边，用父点判断会把它当成来时边跳过，从而漏掉该环、把边误判成割边。

模版题一的完整代码。多连通区、重边、自环都直接可过：自环的反向边就是自己，遍历时被 (e ^ 1) == preEdge 跳过，不会带来影响；重边靠边编号判断父边，不会被误跳。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 500001;
    const int MAXM = 2000001;
    int n, m;

    int head[MAXN];
    int nxt[MAXM << 1];
    int to[MAXM << 1];
    int cntg;

    int dfn[MAXN];
    int low[MAXN];
    int cntd;

    bool cutEdge[MAXM];

    void addEdge(int u, int v) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        head[u] = cntg;
    }

    void tarjan(int u, int preEdge) {
        dfn[u] = low[u] = ++cntd;
        for (int e = head[u]; e > 0; e = nxt[e]) {
            if ((e ^ 1) == preEdge) {
                continue;
            }
            int v = to[e];
            if (dfn[v] == 0) {
                tarjan(v, e);
                low[u] = min(low[u], low[v]);
                if (low[v] > dfn[u]) {
                    cutEdge[e >> 1] = true;
                }
            } else {
                low[u] = min(low[u], dfn[v]);
            }
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cntg = 1;
        cin >> n >> m;
        for (int i = 1, u, v; i <= m; i++) {
            cin >> u >> v;
            addEdge(u, v);
            addEdge(v, u);
        }
        for (int i = 1; i <= n; i++) {
            if (dfn[i] == 0) {
                tarjan(i, 0);
            }
        }
        int ansCnt = 0;
        for (int i = 1; i <= m; i++) {
            if (cutEdge[i]) {
                ansCnt++;
            }
        }
        cout << ansCnt << "\n";
        for (int i = 1; i <= m; i++) {
            if (cutEdge[i]) {
                cout << i << " ";
            }
        }
        cout << "\n";
        return 0;
    }

模版题二是同一个算法换成 LeetCode 的接口，输出每条割边的两个端点。核心只有 tarjan 一处判定，把 cutEdge[e >> 1] 记为真，最后扫一遍边表输出。

class Solution {
    public:
        static const int MAXN = 100001;
        static const int MAXM = 100001;
        int n, m;

        int a[MAXM];
        int b[MAXM];

        int head[MAXN];
        int nxt[MAXM << 1];
        int to[MAXM << 1];
        int cntg;

        int dfn[MAXN];
        int low[MAXN];
        int cntd;

        bool cutEdge[MAXM];

        void prepare() {
            cntg = 1;
            cntd = 0;
            for (int i = 1; i <= n; i++) {
                head[i] = dfn[i] = low[i] = 0;
            }
            for (int i = 1; i <= m; i++) {
                cutEdge[i] = false;
            }
        }

        void addEdge(int u, int v) {
            nxt[++cntg] = head[u];
            to[cntg] = v;
            head[u] = cntg;
        }

        void tarjan(int u, int preEdge) {
            dfn[u] = low[u] = ++cntd;
            for (int e = head[u]; e > 0; e = nxt[e]) {
                if ((e ^ 1) == preEdge) {
                    continue;
                }
                int v = to[e];
                if (dfn[v] == 0) {
                    tarjan(v, e);
                    low[u] = min(low[u], low[v]);
                    if (low[v] > dfn[u]) {
                        cutEdge[e >> 1] = true;
                    }
                } else {
                    low[u] = min(low[u], dfn[v]);
                }
            }
        }

        vector<vector<int>> criticalConnections(int nodeCnt, vector<vector<int>>& connections) {
            n = nodeCnt;
            m = (int)connections.size();
            prepare();
            for (int i = 1; i <= m; i++) {
                a[i] = connections[i - 1][0] + 1;
                b[i] = connections[i - 1][1] + 1;
                addEdge(a[i], b[i]);
                addEdge(b[i], a[i]);
            }
            tarjan(1, 0);
            vector<vector<int>> ans;
            for (int i = 1; i <= m; i++) {
                if (cutEdge[i]) {
                    ans.push_back({a[i] - 1, b[i] - 1});
                }
            }
            return ans;
        }
    };

二、边双连通分量与缩点

两个点 u、v 是边双连通的，指随便删掉图上一条边之后它们仍然连通。一个极大的边双连通点集就是一个边双连通分量，简称边双（E-BCC）。

换成割边的语言：两个点边双连通，当且仅当它们之间的所有路径都不经过桥。所以删掉全部桥之后，每个连通块正好是一个边双。桥的两侧必然落在不同边双里，边双内部一条桥都没有。

求法和强连通分量几乎一样，用一个栈。dfn 与 low 照旧，访问到 u 时把 u 压栈；当 dfn[u] == low[u] 时，u 就是它所在边双里 dfn 最小的点，从栈顶一直弹到 u，弹出的这些点同属一个边双。

把每个边双缩成一个点，桥就变成连接这些新点的边。桥不在任何环上，所以缩点后不会再有环；原图连通时，缩点结果正好是一棵树。后面所有题都在这棵树上做。

边双模版题的完整代码，弹栈时顺便记录每个边双的大小和成员。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 500001;
    const int MAXM = 2000001;
    int n, m;

    int head[MAXN];
    int nxt[MAXM << 1];
    int to[MAXM << 1];
    int cntg;

    int dfn[MAXN];
    int low[MAXN];
    int cntd;

    int sta[MAXN];
    int top;

    int ebccSiz[MAXN];
    int ebccArr[MAXN];
    int ebccl[MAXN];
    int ebccr[MAXN];
    int idx;
    int ebccCnt;

    void addEdge(int u, int v) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        head[u] = cntg;
    }

    void tarjan(int u, int preEdge) {
        dfn[u] = low[u] = ++cntd;
        sta[++top] = u;
        for (int e = head[u]; e > 0; e = nxt[e]) {
            if ((e ^ 1) == preEdge) {
                continue;
            }
            int v = to[e];
            if (dfn[v] == 0) {
                tarjan(v, e);
                low[u] = min(low[u], low[v]);
            } else {
                low[u] = min(low[u], dfn[v]);
            }
        }
        if (dfn[u] == low[u]) {
            ebccCnt++;
            ebccSiz[ebccCnt] = 0;
            ebccl[ebccCnt] = idx + 1;
            int pop;
            do {
                pop = sta[top--];
                ebccSiz[ebccCnt]++;
                ebccArr[++idx] = pop;
            } while (pop != u);
            ebccr[ebccCnt] = idx;
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cntg = 1;
        cin >> n >> m;
        for (int i = 1, u, v; i <= m; i++) {
            cin >> u >> v;
            addEdge(u, v);
            addEdge(v, u);
        }
        for (int i = 1; i <= n; i++) {
            if (dfn[i] == 0) {
                tarjan(i, 0);
            }
        }
        cout << ebccCnt << "\n";
        for (int i = 1; i <= ebccCnt; i++) {
            cout << ebccSiz[i] << "\n";
            for (int j = ebccl[i]; j <= ebccr[i]; j++) {
                cout << ebccArr[j] << " ";
            }
            cout << "\n";
        }
        return 0;
    }
Tarjan 一边求割边，一边把边双弹出来

用一个小图走一遍：两个三角形由一条边相连。点「下一步」看 dfn / low 怎么记、回边怎么把 low 压小、哪条边被判成桥，最后两个边双各自弹出并缩成树上的一个点。

三、复杂度分析

建图和 tarjan 都是每条边遍历常数次，缩点再扫一遍边表，整体一遍线性。

操作时间复杂度空间复杂度
链式前向星建图（n 点 m 边）O(n + m)O(n + m)（前向星 2m 条有向边 + dfn / low / 栈）
tarjan 求割边（每个连通区一次 dfs）O(n + m)
tarjan 求边双并弹栈O(n + m)
缩点重建树（扫边表、按 belong 连边）O(n + m)

求割边那份代码里，head / nxt / to 三个数组各开 2m + 1，dfn / low / 栈都只开 n 个。规模到 5×10⁵ 点、2×10⁶ 边时，这是这一类题里最省的一套写法。

四、最后

割边和边双都只是两个数组加一个栈，真正的分界只有两条：割边看树边回溯时 low[v] > dfn[u] 是否成立，边双看 dfn[u] == low[u] 时弹栈。其余全是同一段 tarjan。

模版讲完了，下一讲换这些模版在题目里的用法：一条边的两侧各有多少服务、无桥时怎么给边定向、加边后桥数怎么变、最少补几条边能让整张图没有桥。
