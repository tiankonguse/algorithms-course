可持久化左偏树：版本不变的堆与 Y 的积木

普通左偏树的 merge 会改写沿途节点的指针，可持久化版本在合并时沿途拷贝出全新节点：旧版本的树一根指针都不动。每个版本只要记住自己的头节点编号，就能随时查询、合并、弹出。合并一趟 O(log n) 时间、O(log n) 空间。

零、版本就是头节点

可持久化左偏树不再配合并查集。每个版本的左偏树用一个头节点编号 rt[i] 代表，操作全都显式传入头节点：查询、合并、弹出都返回新版本的头。生成一个新版本，消耗 O(log n) 的时间和空间。

为什么左偏树能持久化，普通二叉堆不能？因为左偏树的一切操作都发生在右链上，右链长度 O(log n)。合并时把右链上的节点各拷贝一份，就得到一棵新树，老树还完整保留在原地。二叉堆的合并要把两个数组的元素逐个搬，数组是共享的，没法留版本。

合并的递归逻辑与普通左偏树完全一致，只多了一步 clone：取 num 较小的根，克隆成新节点 h，递归 merge(h 的右儿子, 另一棵树)，回溯时按 dist 交换左右。克隆出的节点与老节点共享左右子树指针，但老节点本身不可变。

弹出堆顶也保持持久化：左右子树先按普通方式合并（沿途拷贝），头节点被丢弃。代码里有个小优化：左儿子或右儿子为空时，直接 clone 那个非空儿子返回，省掉一次递归 merge。

一、对数器验证（题3）

网上几乎没有可持久化左偏树的模板题，正确性怎么保证？课程直接写了一个对数器：随机生成 9999 个版本，每个版本对应一棵左偏树，同时用一个 priority_queue 快照保存同样的内容。每次生成新版本，就用同样的操作更新快照，最后把左偏树里的元素挨个弹出，与快照逐项比对。

三个版本操作对应三个函数：treeAdd（旧版本加一个新值）、treeMerge（两个旧版本合并）、treePop（旧版本弹出堆顶）。每个函数都返回新版本的头，旧版本的头继续保留，验证时可以随时再拿来用。

#include <iostream>
    #include <vector>
    #include <queue>
    #include <algorithm>
    #include <cstdlib>
    #include <ctime>

    using namespace std;

    const int MAXN = 10000;
    const int MAXV = 100000;
    const int MAXT = 2000001;

    int rt[MAXN];
    int num[MAXT];
    int ls[MAXT];
    int rs[MAXT];
    int dist[MAXT];
    int siz[MAXT];
    int cnt = 0;

    int init(int v) {
        num[++cnt] = v;
        ls[cnt] = rs[cnt] = dist[cnt] = 0;
        return cnt;
    }

    int clone(int i) {
        num[++cnt] = num[i];
        ls[cnt] = ls[i];
        rs[cnt] = rs[i];
        dist[cnt] = dist[i];
        return cnt;
    }

    int merge(int i, int j) {
        if (i == 0 || j == 0) {
            return i + j;
        }
        if (num[i] > num[j]) {
            swap(i, j);
        }
        int h = clone(i);
        rs[h] = merge(rs[h], j);
        if (dist[ls[h]] < dist[rs[h]]) {
            swap(ls[h], rs[h]);
        }
        dist[h] = dist[rs[h]] + 1;
        return h;
    }

    int pop(int i) {
        if (ls[i] == 0 && rs[i] == 0) {
            return 0;
        }
        if (ls[i] == 0 || rs[i] == 0) {
            return clone(ls[i] + rs[i]);
        }
        return merge(ls[i], rs[i]);
    }

    void treeAdd(int x, int y, int i) {
        rt[i] = merge(rt[x], init(y));
        siz[rt[i]] = siz[rt[x]] + 1;
    }

    void treeMerge(int x, int y, int i) {
        if (rt[x] == 0 && rt[y] == 0) {
            rt[i] = 0;
        } else if (rt[x] == 0 || rt[y] == 0) {
            rt[i] = clone(rt[x] + rt[y]);
        } else {
            rt[i] = merge(rt[x], rt[y]);
        }
        siz[rt[i]] = siz[rt[x]] + siz[rt[y]];
    }

    void treePop(int x, int i) {
        if (siz[rt[x]] == 0) {
            rt[i] = 0;
        } else {
            rt[i] = pop(rt[x]);
            siz[rt[i]] = siz[rt[x]] - 1;
        }
    }

    vector<priority_queue<int, vector<int>, greater<int>>> verify;

    void verifyAdd(int x, int y) {
        priority_queue<int, vector<int>, greater<int>> pre = verify[x];
        vector<int> tmp;
        while (!pre.empty()) {
            tmp.push_back(pre.top());
            pre.pop();
        }
        priority_queue<int, vector<int>, greater<int>> cur;
        for (int number : tmp) {
            cur.push(number);
        }
        cur.push(y);
        verify.push_back(cur);
    }

    void verifyMerge(int x, int y) {
        priority_queue<int, vector<int>, greater<int>> h1 = verify[x];
        priority_queue<int, vector<int>, greater<int>> h2 = verify[y];
        vector<int> tmp;
        priority_queue<int, vector<int>, greater<int>> cur;
        while (!h1.empty()) {
            int number = h1.top();
            h1.pop();
            tmp.push_back(number);
            cur.push(number);
        }
        for (int number : tmp) {
            h1.push(number);
        }
        tmp.clear();
        while (!h2.empty()) {
            int number = h2.top();
            h2.pop();
            tmp.push_back(number);
            cur.push(number);
        }
        for (int number : tmp) {
            h2.push(number);
        }
        verify.push_back(cur);
    }

    void verifyPop(int x) {
        priority_queue<int, vector<int>, greater<int>> pre = verify[x];
        priority_queue<int, vector<int>, greater<int>> cur;
        if (pre.empty()) {
            verify.push_back(cur);
        } else {
            int top = pre.top();
            pre.pop();
            vector<int> tmp;
            while (!pre.empty()) {
                tmp.push_back(pre.top());
                pre.pop();
            }
            for (int number : tmp) {
                pre.push(number);
                cur.push(number);
            }
            pre.push(top);
            verify.push_back(cur);
        }
    }

    bool check(int i) {
        int h1 = rt[i];
        priority_queue<int, vector<int>, greater<int>> h2 = verify[i];
        if (siz[h1] != h2.size()) {
            return false;
        }
        bool ans = true;
        vector<int> tmp;
        while (!h2.empty()) {
            int o1 = num[h1];
            h1 = pop(h1);
            int o2 = h2.top();
            h2.pop();
            tmp.push_back(o2);
            if (o1 != o2) {
                ans = false;
                break;
            }
        }
        for (int v : tmp) {
            h2.push(v);
        }
        return ans;
    }

    int main() {
        cout << "test begin" << endl;
        dist[0] = -1;
        rt[0] = siz[0] = 0;
        verify.emplace_back(priority_queue<int, vector<int>, greater<int>>());
        srand(time(nullptr));
        for (int i = 1, op, x, y; i < MAXN; i++) {
            op = i == 1 ? 1 : (rand() % 3 + 1);
            x = rand() % i;
            if (op == 1) {
                y = rand() % MAXV;
                treeAdd(x, y, i);
                verifyAdd(x, y);
            } else if (op == 2) {
                do {
                    y = rand() % i;
                } while (y == x);
                treeMerge(x, y, i);
                verifyMerge(x, y);
            } else {
                treePop(x, i);
                verifyPop(x);
            }
            if (!check(i)) {
                cout << "err!" << endl;
            }
        }
        for (int i = 1; i < MAXN; i++) {
            if (!check(i)) {
                cout << "err!" << endl;
            }
        }
        cout << "test finish" << endl;
        return 0;
    }

注意 check 函数：它把版本 i 的树头 rt[i] 逐次 pop，与快照 verify[i] 依次弹出比较。这恰好验证了「旧版本还能继续弹」：如果某个版本被后来的操作改坏了，这里立刻对不上。

二、Y 的积木（P2409）

n 个数组，每个数组必须选恰好一个数字，得到 n 个数字的累加和，这算一个方案。要求输出累加和最小的前 k 个方案各自的和。n、mi 不超过 100，k 不超过 104，数字不超过 100。朴素枚举所有方案是乘积级，k 又很大，分组背包能过但不是最优；课程用可持久化左偏树做到 O(k log k)。

思路是增量展开。先把每个数组排好序，首方案是每个数组取最小值，累加和记 first。从首方案出发，每次把一个「最后一个被选数字」向后推一位，就得到一个严格变大的新方案。所有可能的推进被组织成一棵候选树，用左偏树维护。

具体地，init(i, j) 表示「数组 i 当前选到了下标 j」，这个状态的代价增量是 arr[j+1] - arr[j]，即把下标推进一步要多花的钱。可持久化左偏树按代价增量排小根堆，手写的小根堆存「候选方案」。每弹出一个候选：

（1）它本身就是一个新方案，代价增量累加进答案，记为 ans。

（2）把增量堆的堆顶 pop 掉，得到 h1：代表数组 i 继续推进下一位的候选。

（3）如果数组 i 还有下一位，把 h1 与「数组 i 从 j+1 起步」的新增量节点 merge，得到以新基线推进的候选。

#include <iostream>
    #include <algorithm>

    using namespace std;

    const int MAXN = 101;
    const int MAXM = 10001;
    const int MAXK = 10001;
    const int MAXT = 1000001;
    const int INF = 10000001;

    int n, k;

    int arr[MAXM];
    int start[MAXN];
    int boundary[MAXN];

    int idx[MAXT];
    int jdx[MAXT];
    int cost[MAXT];
    int ls[MAXT];
    int rs[MAXT];
    int dist[MAXT];
    int pre[MAXT];
    int cnt = 0;

    int heap[MAXK];
    int heapSize = 0;

    int ans[MAXK];

    int init(int i, int j) {
        idx[++cnt] = i;
        jdx[cnt] = j;
        cost[cnt] = (j + 1 < boundary[i]) ? (arr[j + 1] - arr[j]) : INF;
        ls[cnt] = rs[cnt] = dist[cnt] = 0;
        return cnt;
    }

    int clone(int i) {
        idx[++cnt] = idx[i];
        jdx[cnt] = jdx[i];
        cost[cnt] = cost[i];
        ls[cnt] = ls[i];
        rs[cnt] = rs[i];
        dist[cnt] = dist[i];
        return cnt;
    }

    int merge(int i, int j) {
        if (i == 0 || j == 0) return i + j;
        if (cost[i] > cost[j]) {
            swap(i, j);
        }
        int h = clone(i);
        rs[h] = merge(rs[h], j);
        if (dist[ls[h]] < dist[rs[h]]) {
            swap(ls[h], rs[h]);
        }
        dist[h] = dist[rs[h]] + 1;
        return h;
    }

    int pop(int i) {
        if (ls[i] == 0 && rs[i] == 0) return 0;
        if (ls[i] == 0 || rs[i] == 0) return clone(ls[i] + rs[i]);
        return merge(ls[i], rs[i]);
    }

    bool compare(int i, int j) {
        return pre[i] + cost[i] < pre[j] + cost[j];
    }

    void heapAdd(int i) {
        heap[++heapSize] = i;
        int cur = heapSize, up = cur / 2;
        while (cur > 1 && compare(heap[cur], heap[up])) {
            swap(heap[cur], heap[up]);
            cur = up;
            up = cur / 2;
        }
    }

    int heapPop() {
        int top = heap[1];
        heap[1] = heap[heapSize--];
        int cur = 1, l = 2, r = 3, best;
        while (l <= heapSize) {
            best = (r <= heapSize && compare(heap[r], heap[l])) ? r : l;
            best = compare(heap[best], heap[cur]) ? best : cur;
            if (best == cur) break;
            swap(heap[cur], heap[best]);
            cur = best;
            l = cur * 2;
            r = l + 1;
        }
        return top;
    }

    void compute() {
        int first = 0;
        for (int i = 1; i <= n; i++) {
            sort(arr + start[i], arr + boundary[i]);
            first += arr[start[i]];
        }
        dist[0] = -1;
        int head = 0;
        for (int i = 1; i <= n; i++) {
            head = merge(head, init(i, start[i]));
        }
        pre[head] = first;
        ans[1] = first;
        heapAdd(head);
        for (int ansi = 2, h1, h2; ansi <= k; ++ansi) {
            head = heapPop();
            ans[ansi] = pre[head] + cost[head];
            h1 = pop(head);
            if (h1 != 0) {
                pre[h1] = pre[head];
                heapAdd(h1);
            }
            if (jdx[head] + 1 < boundary[idx[head]]) {
                h2 = merge(h1, init(idx[head], jdx[head] + 1));
                pre[h2] = ans[ansi];
                heapAdd(h2);
            }
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> k;
        int ai = 0;
        for (int i = 1; i <= n; i++) {
            int m;
            cin >> m;
            start[i] = ai + 1;
            for (int j = 1; j <= m; j++) {
                cin >> arr[++ai];
            }
            boundary[i] = start[i] + m;
        }
        compute();
        for (int i = 1; i <= k; i++) {
            cout << ans[i] << " ";
        }
        cout << endl;
        return 0;
    }

ans[1] 就是首方案的和 first。后续每弹一个候选，答案就是上一个答案加这个候选的增量 cost。手写小根堆比较的是 pre + cost，即前缀方案的和加上当前推进增量。

三、可持久化合并演示

merge 沿途拷贝，旧版本不动

版本 1 是树 5(3, 9)，版本 2 是树 8(2, 7)。点「下一步」看 merge 怎样产生版本 3。

蓝 = 拷贝出的新节点，灰 = 旧版本节点，橙 / 绿 = 旧节点的左右子树

合并版本 1 与版本 2：5 和 8 比较，5 小，克隆 5 成 5'，递归合并 5' 的右儿子 9 与 8；9 与 8 比较，9 小，克隆 9 成 9'，8 挂到 9' 的右儿子；回溯时 9' 右子树 dist 1 大于左子树 dist 0，交换左右，8 整棵树成为 9' 的左子树。最后版本 3 的头是 5'，版本 1、2 的节点全部保持原样。

总结

可持久化左偏树把版本记成头节点编号，merge、pop 都沿途 clone 右链上的节点，旧版本完整保留。合并与弹出各 O(log n) 时间、O(log n) 空间，这个特性正是 Y 的积木前 k 小方案的保障：候选树的分支共享前缀，展开一个分支不会破坏另一个。题3 用对数器证明了这套实现与标准堆行为一致。下节课把可持久化左偏树用在图论里，做出 k 短路的最优解。
