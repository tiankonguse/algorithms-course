整体二分：陨石雨

前两页问的都是「第 k 小」，答案范围是排序下标。这一页换个形状：问每个国家「第几场陨石雨之后能达标」，答案范围变成时间轴上的场次编号，树状数组也要反过来用，范围修改、单点查询。

零、背景

题目 3 是 POI 的陨石雨。它仍然满足整体二分的四条前提：每个国家的答案单调（雨只会越下越多），达标之后场次往后依然达标，不达标时把场次往前推也不可能达标，而且所有国家可以离线一起算。

一、题目3：陨石雨

一共有 n 个国家，给定 n 个数字，表示每个国家希望收集到的陨石数量。一共有 m 个区域，1 号区顺时针到 2 号区，一直顺时针回到 1 号区，即环形相连。每个区域只属于某一个国家，给定 m 个数字，表示每个区域归属给哪个国家。

接下来会依次发生 k 场陨石雨，格式 l r num：从 l 号区顺时针到 r 号区发生了陨石雨，每个区域都增加 num 个陨石。打印每个国家经历前几场陨石雨可以达到收集要求，如果无法满足，打印 NIE。

1 ≤ n、m、k ≤ 3 × 10⁵，1 ≤ 陨石数量 ≤ 10⁹。测试链接：洛谷 P3527。

二、判定用什么数据结构

一场雨是范围修改（一段区域都加上 num），问国家收了多少是单点查询（把这个国家拥有的区域各查一次）。所以树状数组要用「范围修改、单点查询」的形态：加雨时 add(l, num)、add(r + 1, -num)，查区域 i 时取前缀和。

国家拥有的区域要能快速枚举，建一个链表：addEdge(nation, i) 把区域 i 挂到国家 nation 的头上。一个国家的陨石数就是链上每个区域的 query(to[e]) + query(to[e] + m) 之和。

这里有个细节：一个国家的区域可能很多，累加到已经不少于 need 就可以停，不必把链表走完。这一句剪枝在实际数据上很值钱。

三、答案范围与递归

答案范围是场次编号 [1, k]，但还得表示「永远达不到」，所以范围取 [1, k + 1]，落到 k + 1 的国家打印 NIE。初始调用是 compute(1, n, 1, k + 1)，问题编号就是国家编号。

执行 [vl..mid] 上的操作，就是把第 vl 场到第 mid 场陨石雨全都下下来。判定时看每个国家是否已经收集够：够了说明达标场次不晚于 mid，去左组；不够去右组，同时把 need 减掉已经收集到的部分。

四、写法一：范围操作 + 撤销

    
#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 300001;
    int n, m, k;

    int qid[MAXN];
    int need[MAXN];

    int rainl[MAXN];
    int rainr[MAXN];
    int num[MAXN];

    int head[MAXN];
    int nxt[MAXN];
    int to[MAXN];
    int cnt = 0;

    long long tree[MAXN << 1];

    int lset[MAXN];
    int rset[MAXN];

    int ans[MAXN];

    void addEdge(int i, int v) {
        nxt[++cnt] = head[i];
        to[cnt] = v;
        head[i] = cnt;
    }

    int lowbit(int i) {
        return i & -i;
    }

    void add(int i, int v) {
        int siz = m * 2;
        while (i <= siz) {
            tree[i] += v;
            i += lowbit(i);
        }
    }

    void add(int l, int r, int v) {
        add(l, v);
        add(r + 1, -v);
    }

    long long query(int i) {
        long long ret = 0;
        while (i > 0) {
        	ret += tree[i];
            i -= lowbit(i);
        }
        return ret;
    }

    void compute(int ql, int qr, int vl, int vr) {
        if (ql > qr) {
            return;
        }
        if (vl == vr) {
            for (int i = ql; i <= qr; i++) {
                ans[qid[i]] = vl;
            }
        } else {
            int mid = (vl + vr) >> 1;
            for (int i = vl; i <= mid; i++) {
                add(rainl[i], rainr[i], num[i]);
            }
            int lsiz = 0, rsiz = 0;
            for (int i = ql; i <= qr; i++) {
                int id = qid[i];
                long long satisfy = 0;
                for (int e = head[id]; e > 0; e = nxt[e]) {
                    satisfy += query(to[e]) + query(to[e] + m);
                    if (satisfy >= need[id]) {
                        break;
                    }
                }
                if (satisfy >= need[id]) {
                    lset[++lsiz] = id;
                } else {
                    need[id] -= satisfy;
                    rset[++rsiz] = id;
                }
            }
            for (int i = 1; i <= lsiz; i++) {
                qid[ql + i - 1] = lset[i];
            }
            for (int i = 1; i <= rsiz; i++) {
                qid[ql + lsiz + i - 1] = rset[i];
            }
            for (int i = vl; i <= mid; i++) {
                add(rainl[i], rainr[i], -num[i]);
            }
            compute(ql, ql + lsiz - 1, vl, mid);
            compute(ql + lsiz, qr, mid + 1, vr);
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1, nation; i <= m; i++) {
            cin >> nation;
            addEdge(nation, i);
        }
        for (int i = 1; i <= n; i++) {
            qid[i] = i;
            cin >> need[i];
        }
        cin >> k;
        for (int i = 1; i <= k; i++) {
            cin >> rainl[i] >> rainr[i] >> num[i];
            if (rainr[i] < rainl[i]) {
                rainr[i] += m;
            }
        }
        compute(1, n, 1, k + 1);
        for (int i = 1; i <= n; i++) {
            if (ans[i] == k + 1) {
                cout << "NIE" << '\n';
            } else {
                cout << ans[i] << '\n';
            }
        }
        return 0;
    }

vl == vr 时答案就是 vl，等于 k + 1 的那些国家输出 NIE。撤销这一层用的还是「把做过的事反着做一遍」：add(rainl[i], rainr[i], -num[i])。

五、写法二：指针滚动

    
int used = 0;

    void compute(int ql, int qr, int vl, int vr) {
        if (ql > qr) {
            return;
        }
        if (vl == vr) {
            for (int i = ql; i <= qr; i++) {
                ans[qid[i]] = vl;
            }
        } else {
            int mid = (vl + vr) >> 1;
            int lsiz = 0, rsiz = 0;
            while (used < mid) {
                used++;
                add(rainl[used], rainr[used], num[used]);
            }
            while (used > mid) {
                add(rainl[used], rainr[used], -num[used]);
                used--;
            }
            for (int i = ql; i <= qr; i++) {
                int id = qid[i];
                long long satisfy = 0;
                for (int e = head[id]; e > 0; e = nxt[e]) {
                    satisfy += query(to[e]) + query(to[e] + m);
                    if (satisfy >= need[id]) {
                        break;
                    }
                }
                if (satisfy >= need[id]) {
                    lset[++lsiz] = id;
                } else {
                    rset[++rsiz] = id;
                }
            }
            for (int i = 1; i <= lsiz; i++) {
                qid[ql + i - 1] = lset[i];
            }
            for (int i = 1; i <= rsiz; i++) {
                qid[ql + lsiz + i - 1] = rset[i];
            }
            compute(ql, ql + lsiz - 1, vl, mid);
            compute(ql + lsiz, qr, mid + 1, vr);
        }
    }

和前两页一样，写法二的 need 全程不减，因为树状数组里始终是「前 used 场雨」的累积，计数单调。其余部分完全相同。

七、复杂度分析

记 n 为国家数、m 为区域数、k 为场次，树状数组开 2m，单次修改或查询 O(log m)。一个国家的判定要遍历它自己的区域链表，所有国家的区域总数恰好是 m，所以一层里全部国家的检查是 O(m log m)。

操作时间复杂度空间复杂度
建链表，把每个区域挂到所属国家头上O(m)O(n + m + k)：链表 O(m)、每场雨的参数 O(k)、左右组暂存 O(n)，树状数组 O(m)
一层的全部下雨操作（写法一加一遍撤一遍 / 写法二挪指针）O(k log m)
一层的全部国家检查（区域总数 m，够了就停）O((n + m) log m)
递归层数（答案范围 k + 1）O(log k)
题目3 总代价O((n + m + k) log k log m)
朴素做法：每个国家各自二分场次，每轮重放一遍雨O(n · k log k)O(n + m + k)

三个量都是 3 × 10⁵ 时，整体二分大约是 9 × 10⁵ × 19 × 19 ≈ 3 × 10⁸ 次基本操作，剪枝之后实际更少。

八、最后

三道题放在一起看，整体二分只规定了「答案范围怎么切、查询怎么分组」，判定那一步用的是哪种数据结构完全自由：一维树状数组、二维树状数组、范围修改单点查询的树状数组，各换各的。
