康托展开：排列到名次的树状数组

1 到 n 的全排列有 n! 个，按字典序排好，每个排列都有一个名次。n 到 10^6 时枚举 n! 个排列是做梦，康托展开用树状数组 O(n log n) 算出名次；反过来，名次加一个大数 m 后再还原成排列，需要把名次写进位制再拆回去，这就是逆康托展开。

零、康托展开公式

排列 S 的字典序名次（从 0 开始）是：

rank(S) = Σi=1..n rightSmall(S[i]) × (n−i)!

rightSmall(S[i]) 是「S[i] 右边比它小的数字个数」。算它不用真的去看右边：从左往右扫，维护「还没用掉的数字集合」，rightSmall(S[i]) 就是集合里比 S[i] 小的个数。初始集合是 {1,2,…,n}，处理完 S[i] 把它从集合删掉。用树状数组存集合，单点删 + 前缀和查，每一位 O(log n)。

名次从 0 开始是公式自带的：最小的排列 1 2 3 … n 每位 rightSmall 都是 0，算出来是 0。洛谷 P5367 要的是第几名的名次，从 1 数，所以最后加 1。

一、康托展开实现

fac 存 (n−i)! 的余数，树状数组初始每个数字一个 1。逐位取前缀和、删数字，贡献是 rightSmall × (n−i)!，全部模 998244353。n 有 10^6，阶乘余数表直接线性递推。

题目链接：P5367 【模板】康托展开，n ≤ 10^6，答案对 998244353 取模。

#include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 1000001;
    const int MOD = 998244353;

    int n;
    int arr[MAXN];
    int fac[MAXN];
    int tree[MAXN];

    int lowbit(int i) {
    	return i & -i;
    }

    int sum(int i) {
    	int ans = 0;
    	while (i > 0) {
    		ans = (ans + tree[i]) % MOD;
    		i -= lowbit(i);
    	}
    	return ans;
    }

    void add(int i, int v) {
    	while (i <= n) {
    		tree[i] += v;
    		i += lowbit(i);
    	}
    }

    int compute() {
    	fac[0] = 1;
    	for (int i = 1; i < n; i++) {
    		fac[i] = (long long)fac[i - 1] * i % MOD;
    	}
    	for (int i = 1; i <= n; i++) {
    		add(i, 1);
    	}
    	long long ans = 0;
    	for (int i = 1; i <= n; i++) {
    		ans = (ans + (long long)sum(arr[i] - 1) * fac[n - i] % MOD) % MOD;
    		add(arr[i], -1);
    	}
    	// 康托公式给的是 0 开始的名次，题目要 1 开始
    	ans = (ans + 1) % MOD;
    	return ans;
    }

    int main() {
    	ios::sync_with_stdio(false);
    	cin.tie(nullptr);
    	cin >> n;
    	for (int i = 1; i <= n; i++) {
    		cin >> arr[i];
    	}
    	cout << compute() << endl;
    	return 0;
    }

二、逆康托展开

给一个排列 s，问第 x+m 名的排列是什么。x 是 s 的名次，m 能到 10^15，名次总和可能远超 64 位还能不能表示？不能取余，因为名次本身要拿来定位。解法是把名次写成「阶乘进制」：

rank = a1·(n−1)! + a2·(n−2)! + … + an·0!

每一位 ai 恰好是 rightSmall(S[i])，范围 0..n−i。第 i 位的进制是 (n−i+1)（ai 最大 n−i，进一位就是 (n−i+1)·(n−i)! = (n−i+1)!）。所以「名次加 m」就是最低位加 m 再向上进位，跟十进制加法一模一样，只是每一位的进位规则不同。

还原排列用线段树二分：数字集合里找「第 ai+1 小的数字」就是第 i 位的值，找到后删掉。线段树维护可用数字个数，getAndDelete 递归：左子树个数够就往左，不够就去右边找剩余的第 k−left 个。这就是「线段树二分」，O(log n) 一次。

三、逆康托展开实现

题目链接：U72177 逆康托展开，n ≤ 10^5，m ≤ 10^15，题目保证 x+m 不超排列总数。

三段的顺序不能乱：先 build 出全 1 的线段树，把 s 每一位转成阶乘进制（边查边删）；然后最低位加 m 从下往上进位；最后再 build 一次，逐位 getAndDelete 还原新排列。

#include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 100001;

    int n;
    long long m;
    long long arr[MAXN];
    int seg[MAXN << 2];

    void build(int l, int r, int i) {
    	if (l == r) {
    		seg[i] = 1;
    	} else {
    		int mid = (l + r) >> 1;
    		build(l, mid, i << 1);
    		build(mid + 1, r, i << 1 | 1);
    		seg[i] = seg[i << 1] + seg[i << 1 | 1];
    	}
    }

    void add(int jobi, int jobv, int l, int r, int i) {
    	if (l == r) {
    		seg[i] += jobv;
    	} else {
    		int mid = (l + r) >> 1;
    		if (jobi <= mid) {
    			add(jobi, jobv, l, mid, i << 1);
    		} else {
    			add(jobi, jobv, mid + 1, r, i << 1 | 1);
    		}
    		seg[i] = seg[i << 1] + seg[i << 1 | 1];
    	}
    }

    int sum(int jobl, int jobr, int l, int r, int i) {
    	if (jobl <= l && r <= jobr) {
    		return seg[i];
    	}
    	int mid = (l + r) >> 1;
    	int ans = 0;
    	if (jobl <= mid) {
    		ans += sum(jobl, jobr, l, mid, i << 1);
    	}
    	if (jobr > mid) {
    		ans += sum(jobl, jobr, mid + 1, r, i << 1 | 1);
    	}
    	return ans;
    }

    int getAndDelete(int k, int l, int r, int i) {
    	int ans;
    	if (l == r) {
    		seg[i]--;
    		ans = l;
    	} else {
    		int mid = (l + r) >> 1;
    		if (seg[i << 1] >= k) {
    			ans = getAndDelete(k, l, mid, i << 1);
    		} else {
    			ans = getAndDelete(k - seg[i << 1], mid + 1, r, i << 1 | 1);
    		}
    		seg[i] = seg[i << 1] + seg[i << 1 | 1];
    	}
    	return ans;
    }

    void compute() {
    	// 当前排列转成阶乘进制：每一位是「右边比它小的个数」
    	build(1, n, 1);
    	for (int i = 1, x; i <= n; i++) {
    		x = (int)arr[i];
    		if (x == 1) {
    			arr[i] = 0;
    		} else {
    			arr[i] = sum(1, x - 1, 1, n, 1);
    		}
    		add(x, -1, 1, n, 1);
    	}
    	// 名次加 m：最低位开始进位
    	arr[n] += m;
    	for (int i = n; i >= 1; i--) {
    		arr[i - 1] += arr[i] / (n - i + 1);
    		arr[i] %= n - i + 1;
    	}
    	// 阶乘进制还原排列
    	build(1, n, 1);
    	for (int i = 1; i <= n; i++) {
    		arr[i] = getAndDelete((int)arr[i] + 1, 1, n, 1);
    	}
    }

    int main() {
    	ios::sync_with_stdio(false);
    	cin.tie(nullptr);
    	cin >> n >> m;
    	for (int i = 1; i <= n; i++) {
    		cin >> arr[i];
    	}
    	compute();
    	for (int i = 1; i <= n; i++) {
    		cout << arr[i] << " ";
    	}
    	cout << endl;
    	return 0;
    }

演示 · 康托展开逐位求和：rightSmall × (n−i)!

总结

康托展开把排列映射成名次，核心是每位 rightSmall(S[i]) × (n−i)!。rightSmall 用「还没用掉的数字集合」维护，树状数组负责查小于 S[i] 的剩余个数和删除，整体 O(n log n)。名次从 0 开始算，题目要 1 开始就加 1。逆康托展开面对 10^15 的 m：名次写阶乘进制，每一位的进制是 n−i+1，加 m 逐位进位，再用线段树二分按序取出数字还原排列。树状数组查前缀和快，但「查剩余数字里第 k 小」得二分套树状数组变 O(log²n)；线段树二分一次递归 O(log n) 直接定位，这就是逆康托展开不推荐树状数组的原因。
