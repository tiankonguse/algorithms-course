可持久化 FHQ Treap：版本树与强制在线

普通平衡树做完操作，旧版本就没了。可持久化要求每次操作之后旧版本原样保留，随时能查任意历史版本。暴力做法是每版整棵树拷贝一份，一次 O(n)；FHQ Treap 的办法是只复制操作路径上碰到的 O(log n) 个节点，没碰到的子树直接共享，新旧版本共用同一批节点。这套设计和可持久化线段树（主席树）几乎一模一样。

零、背景

可持久化的核心就一句话：旧版本不能被破坏。split 和 merge 递归时都会改写沿途节点的左右孩子和 siz，所以每一步动手前先 copy 一个一模一样的节点，在新节点上改。没走到的子树完全不动，多个版本共享它们。

每次操作复制 O(log n) 个节点，时间空间代价都是 O(log n)。每个版本的头节点记录在 head[v] 里，第 i 号版本就存 head[i]。0 版本约定永远是空树：split 过程走完，临时槽 left[0] = right[0] = 0 必须保证。

一、可持久化平衡树 P3835：版本树

P3835 的每个操作都基于某个 v 版本：增加、删除生成新版本；查询也生成一个「状况等于 v 版本」的新版本，所以查询操作一样要把 head[i] 记成 head[v]。删除不存在的前驱时返回 -2^31 + 1，后继不存在返回 +2^31 - 1，这两个值用 INT_MIN + 1 和 INT_MAX 表达。

实现上 split 和 merge 内部先 copy 再递归，copy 函数把 key、左右孩子、siz、priority 全部搬到新槽位。add 建新节点后 merge 两棵树，remove 两次 split 取出 m 后把 m 的左右孩子合并回去。查询用的 small、index、pre、post 只读不写，不需要 copy。

merge 也要加可持久化设计：merge 里同样先 copy 再改孩子，这样可以直接合并两个不同版本的树而不污染它们。

二、题5 完整实现（P3835 可持久化平衡树）

n 到 5e5，x 到 ±1e9。节点数组开 MAXN 的几十倍，因为每次操作都会新造节点。输入先给 v、op、x，增删查共用一套 split/merge。

#include <iostream>
    #include <cstdio>
    #include <cstdlib>
    #include <cstring>
    #include <algorithm>
    #include <climits>

    using namespace std;

    const int MAXN = 500001;
    const int MAXM = MAXN * 50;

    int cnt = 0;
    int head[MAXN];
    int key[MAXM];
    int ls[MAXM];
    int rs[MAXM];
    int siz[MAXM];
    double priority[MAXM];

    int copy(int i) {
        ++cnt;
        key[cnt] = key[i];
        ls[cnt] = ls[i];
        rs[cnt] = rs[i];
        siz[cnt] = siz[i];
        priority[cnt] = priority[i];
        return cnt;
    }

    void up(int i) {
        siz[i] = siz[ls[i]] + siz[rs[i]] + 1;
    }

    void split(int l, int r, int i, int num) {
        if (i == 0) {
            rs[l] = ls[r] = 0;
        } else {
            i = copy(i);
            if (key[i] <= num) {
                rs[l] = i;
                split(i, r, rs[i], num);
            } else {
                ls[r] = i;
                split(l, i, ls[i], num);
            }
            up(i);
        }
    }

    int merge(int l, int r) {
        if (l == 0 || r == 0) {
            return l + r;
        }
        if (priority[l] >= priority[r]) {
            l = copy(l);
            rs[l] = merge(rs[l], r);
            up(l);
            return l;
        } else {
            r = copy(r);
            ls[r] = merge(l, ls[r]);
            up(r);
            return r;
        }
    }

    void add(int v, int i, int num) {
        split(0, 0, i, num);
        int l = rs[0];
        int r = ls[0];
        ls[0] = rs[0] = 0;
        ++cnt;
        key[cnt] = num;
        siz[cnt] = 1;
        priority[cnt] = (double)rand() / RAND_MAX;
        head[v] = merge(merge(l, cnt), r);
    }

    void remove(int v, int i, int num) {
        split(0, 0, i, num);
        int lm = rs[0];
        int r = ls[0];
        split(0, 0, lm, num - 1);
        int l = rs[0];
        int m = ls[0];
        ls[0] = rs[0] = 0;
        head[v] = merge(merge(l, merge(ls[m], rs[m])), r);
    }

    int small(int i, int num) {
        if (i == 0) {
            return 0;
        }
        if (key[i] >= num) {
            return small(ls[i], num);
        } else {
            return siz[ls[i]] + 1 + small(rs[i], num);
        }
    }

    int index(int i, int x) {
        if (siz[ls[i]] >= x) {
            return index(ls[i], x);
        } else if (siz[ls[i]] + 1 < x) {
            return index(rs[i], x - siz[ls[i]] - 1);
        } else {
            return key[i];
        }
    }

    int pre(int i, int num) {
        if (i == 0) {
            return INT_MIN + 1;
        }
        if (key[i] >= num) {
            return pre(ls[i], num);
        } else {
            return max(key[i], pre(rs[i], num));
        }
    }

    int post(int i, int num) {
        if (i == 0) {
            return INT_MAX;
        }
        if (key[i] <= num) {
            return post(rs[i], num);
        } else {
            return min(key[i], post(ls[i], num));
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        srand(time(0));
        int n;
        cin >> n;
        for (int i = 1; i <= n; i++) {
            int version, op, x;
            cin >> version >> op >> x;
            if (op == 1) {
                add(i, head[version], x);
            } else if (op == 2) {
                remove(i, head[version], x);
            } else {
                head[i] = head[version];
                if (op == 3) {
                    cout << small(head[i], x) + 1 << "\n";
                } else if (op == 4) {
                    cout << index(head[i], x) << "\n";
                } else if (op == 5) {
                    cout << pre(head[i], x) << "\n";
                } else {
                    cout << post(head[i], x) << "\n";
                }
            }
        }
        return 0;
    }
演示：版本链 v0 空树 → 插 3 → 插 1 → 插 2 → 再基于 v1 插 4（每帧显示该版本树与中序）

三、可持久化文艺平衡树 P5055：懒标记也要持久化

P5055 在可持久化平衡树的基础上加了两个东西：区间翻转的懒标记，和区间和查询。翻转区间时给中段根打 rev 标记；down 的时候麻烦来了：左右孩子可能是旧版本共享的节点，直接 swap 会破坏旧版本。所以 down 里先 copy 左右孩子再交换、再传标记，新版本的翻转才不碰旧版本。

每个节点多维护一个 sum（左子树和 + 右子树和 + 自己的 key），翻转只换左右孩子不换值，区间和不受影响。sum 用 long long，插入值能到 ±1e6、n 到 2e5，区间和会超 int。

这题强制在线：所有输入参数异或上一次查询的答案，lastAns 初始 0，查询操作把答案输出并写回 lastAns。操作 2 只读一个参数 x，其余操作还要再读一个 y 同样异或。

四、题6 完整实现（P5055 可持久化文艺平衡树）

插入是在第 x 个数后插入 y，所以按排名 split(head, x) 取前 x 个，新节点夹中间。查询区间和用两次 split 取出 m 段，输出 sum[m]。删除是取中段直接丢掉。所有参数读入后先异或 lastAns 再使用。

#include <iostream>
    #include <vector>
    #include <cstdlib>
    #include <ctime>
    using namespace std;

    const int MAXN = 200001;
    const int MAXM = MAXN * 100;

    int cnt = 0;
    int head[MAXN];
    int key[MAXM];
    int ls[MAXM];
    int rs[MAXM];
    int siz[MAXM];
    bool rev[MAXM];
    long long sum[MAXM];
    double priority[MAXM];

    int copy(int i) {
        key[++cnt] = key[i];
        ls[cnt] = ls[i];
        rs[cnt] = rs[i];
        siz[cnt] = siz[i];
        rev[cnt] = rev[i];
        sum[cnt] = sum[i];
        priority[cnt] = priority[i];
        return cnt;
    }

    void up(int i) {
        siz[i] = siz[ls[i]] + siz[rs[i]] + 1;
        sum[i] = sum[ls[i]] + sum[rs[i]] + key[i];
    }

    void down(int i) {
        if (rev[i]) {
            if (ls[i] != 0) {
                ls[i] = copy(ls[i]);
                rev[ls[i]] ^= 1;
            }
            if (rs[i] != 0) {
                rs[i] = copy(rs[i]);
                rev[rs[i]] ^= 1;
            }
            swap(ls[i], rs[i]);
            rev[i] = false;
        }
    }

    void split(int l, int r, int i, int rank) {
        if (i == 0) {
            rs[l] = ls[r] = 0;
        } else {
            i = copy(i);
            down(i);
            if (siz[ls[i]] + 1 <= rank) {
                rs[l] = i;
                split(i, r, rs[i], rank - siz[ls[i]] - 1);
            } else {
                ls[r] = i;
                split(l, i, ls[i], rank);
            }
            up(i);
        }
    }

    int merge(int l, int r) {
        if (l == 0 || r == 0) {
            return l + r;
        }
        if (priority[l] >= priority[r]) {
            l = copy(l);
            down(l);
            rs[l] = merge(rs[l], r);
            up(l);
            return l;
        } else {
            r = copy(r);
            down(r);
            ls[r] = merge(l, ls[r]);
            up(r);
            return r;
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        srand(time(0));
        int n;
        cin >> n;
        long long lastAns = 0;
        for (int i = 1; i <= n; i++) {
            int version, op;
            long long x, y = 0;
            cin >> version >> op >> x;
            x ^= lastAns;
            if (op != 2) {
                cin >> y;
                y ^= lastAns;
            }
            int l, m, lm, r;
            if (op == 1) {
                split(0, 0, head[version], x);
                l = rs[0];
                r = ls[0];
                ls[0] = rs[0] = 0;
                key[++cnt] = y;
                siz[cnt] = 1;
                sum[cnt] = y;
                priority[cnt] = (double)rand() / RAND_MAX;
                head[i] = merge(merge(l, cnt), r);
            } else if (op == 2) {
                split(0, 0, head[version], x);
                lm = rs[0];
                r = ls[0];
                split(0, 0, lm, x - 1);
                l = rs[0];
                m = ls[0];
                ls[0] = rs[0] = 0;
                head[i] = merge(l, r);
            } else if (op == 3) {
                split(0, 0, head[version], y);
                lm = rs[0];
                r = ls[0];
                split(0, 0, lm, x - 1);
                l = rs[0];
                m = ls[0];
                ls[0] = rs[0] = 0;
                rev[m] ^= 1;
                head[i] = merge(merge(l, m), r);
            } else {
                split(0, 0, head[version], y);
                lm = rs[0];
                r = ls[0];
                split(0, 0, lm, x - 1);
                l = rs[0];
                m = ls[0];
                ls[0] = rs[0] = 0;
                lastAns = sum[m];
                cout << lastAns << endl;
                head[i] = merge(merge(l, m), r);
            }
        }
        return 0;
    }

五、总结

可持久化 FHQ Treap 就两件事：split / merge 里先 copy 再改，保证旧版本节点不被触碰；head[v] 记版本根，0 版本恒为空树。加了懒标记之后，down 也要先 copy 孩子再换，否则翻转会写穿共享节点。可持久化有序表和可持久化线段树共用同一套「路径复制、子树共享」思想，学会一个另一个就是换皮。强制在线只是输入输出层的包装，异或 lastAns 读入解码即可。
