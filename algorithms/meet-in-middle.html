折半搜索：半量展开，结果整合

有一类题的数据量卡在中间：全量 2^40 种展开直接爆掉，但 2^20 种没问题。折半搜索把数据劈成两半各自展开，再设计两半结果的整合逻辑，把 2^40 变成 2·2^20。

零、背景

上一页的双向广搜是剪枝优化。这一页的折半搜索是另一个本体：它解决的不是「展开太慢」，而是「根本展不开」。特征一句话：全量样本不允许递归完全展开，但是半量样本可以完全展开。

过程三步：把数据分成两部分；每部分各自展开、算出结果；设计两部分结果的整合逻辑。展开从 2^n 掉到 2·2^(n/2)，中间那步整合是每道题各自设计的部分。

n ≤ 40 是这套题的典型信号：2^40 约 1 万亿，任何机器都跑不动；2^20 约 100 万，排序都只要几十毫秒。看到 n ≤ 40，先往折半搜索想。

一、零食问题：半量子集和 + 排序双指针计数

牛牛背包容量 w，n 袋零食体积 v[i]，问总体积不超过 w 的放法总数（空包也算一种）。n ≤ 40、w 和 v[i] 都是 10^9 级别，背包 DP 按容量做不开，只能数子集。

左半 [0, n>>1) 递归展开所有子集和，s > w 直接剪枝返回；右半同样。两半各最多 2^20 个和，分别排序后配对：对左半每个和 lsum[i]，右半能配多少个 rsum[j] 满足 lsum[i] + rsum[j] ≤ w？

配对数用双指针一次扫完：lsum 从大到小遍历，rsum 升序，j 只前进不回退。lsum[i] 越小，能配的 rsum 越多，j 单调不减，ans 累加 j 就是答案。

演示 · 零食 [2,3,5,7] w=7：两半展开子集和，排序后双指针配对计数

实测：n = 40、w = 2×10^9、v 随机 10^9 级别，折半搜索加剪枝 2.66ms 出结果。全子集枚举在这个规模上约 1 万亿次，差 12 个数量级。

#include <bits/stdc++.h>
    using namespace std;

    // 零食问题 / 世界冰球锦标赛：折半搜索
    // n <= 40，全量 2^40 个放法展不开；分成左右两半，每半 2^20 个和完全展开
    // 两半分别排序后双指针：lsum[i] 配多少个 rsum[j] 满足和 <= w，逐 i 累加
    const int MAXN = 40;
    const int MAXM = 1 << 20;

    long long arr[MAXN];
    long long lsum[MAXM];
    long long rsum[MAXM];
    int n;
    long long w;

    long long f(int i, int e, long long s, long long w, long long* ans, int j) {
        if (s > w) {
            return j;
        }
        if (i == e) {
            ans[j++] = s;
        } else {
            j = f(i + 1, e, s, w, ans, j);
            j = f(i + 1, e, s + arr[i], w, ans, j);
        }
        return j;
    }

    long long compute() {
        int lsize = f(0, n >> 1, 0, w, lsum, 0);
        int rsize = f(n >> 1, n, 0, w, rsum, 0);
        sort(lsum, lsum + lsize);
        sort(rsum, rsum + rsize);
        long long ans = 0;
        for (int i = lsize - 1, j = 0; i >= 0; i--) {
            while (j < rsize && lsum[i] + rsum[j] <= w) {
                j++;
            }
            ans += j;
        }
        return ans;
    }

二、最接近目标值的子序列和：压缩相同数字 + 双指针逼近

从 nums 里选一个子序列，让和尽量贴近 goal，返回 |sum - goal| 的最小值。n ≤ 40、|nums[i]| ≤ 10^7、|goal| ≤ 10^9。

先把全部正数和 max、全部负数和 min 算出来：max < goal 或 min > goal 时，全选正数或全选负数就是最优，直接返回，省掉大半展开。这是这题独有的特判。

展开前先排序，collect 里把相等数字一次处理：一段相同的 nums[i] 有 j-i 个，取 k = 0..j-i 个，用一个循环合并展开，常数优化。

两半排序后双指针逼近：lsum 升序，对每个 lsum[i]，j 从右往左走，只要 |goal - lsum[i] - rsum[j-1]| ≤ |goal - lsum[i] - rsum[j]| 就继续 j--。j 单调不回退，整个配对 O(2^(n/2))。ans 初始 abs(goal)，对应空子序列。

演示 · [-1,0,2,3] goal=4：左右各展开，双指针找 |sum-4| 的最小值

实测：n = 40、goal 落在正负和之间（特判不生效）时，折半搜索最差 96.69ms，仍完全可过。

#include <bits/stdc++.h>
    using namespace std;

    // 最接近目标值的子序列和：折半 + 相同数字压缩 + 排序双指针
    // 先看全量 min/max 与 goal 的关系直接出答案；排序后 collect 把相等数字 k 个一起处理
    // 左右各展开 2^20 个和，lsum 升序，对每个 lsum[i] 用 j 从右往左逼近 goal - lsum[i]
    const int MAXN = 1 << 20;

    int lsum[MAXN];
    int rsum[MAXN];
    int fill;

    void collect(vector<int>& nums, int i, int e, int s, int* sum) {
        if (i == e) {
            sum[fill++] = s;
        } else {
            int j = i + 1;
            while (j < e && nums[j] == nums[i]) {
                j++;
            }
            for (int k = 0; k <= j - i; k++) {
                collect(nums, j, e, s + k * nums[i], sum);
            }
        }
    }

    int minAbsDifference(vector<int>& nums, int goal) {
        int n = nums.size();
        long long minv = 0, max = 0;
        for (int v : nums) {
            if (v >= 0) {
                max += v;
            } else {
                minv += v;
            }
        }
        if (max < goal) {
            return (int)llabs(max - goal);
        }
        if (minv > goal) {
            return (int)llabs(minv - goal);
        }
        sort(nums.begin(), nums.end());
        fill = 0;
        collect(nums, 0, n >> 1, 0, lsum);
        int lsize = fill;
        fill = 0;
        collect(nums, n >> 1, n, 0, rsum);
        int rsize = fill;
        sort(lsum, lsum + lsize);
        sort(rsum, rsum + rsize);
        int ans = abs(goal);
        for (int i = 0, j = rsize - 1; i < lsize; i++) {
            while (j > 0 && llabs((long long)goal - lsum[i] - rsum[j - 1]) <= llabs((long long)goal - lsum[i] - rsum[j])) {
                j--;
            }
            ans = min(ans, (int)llabs((long long)goal - lsum[i] - rsum[j]));
        }
        return ans;
    }

三、分区最小化差：折半 + 按选取个数分桶

这道题课上没讲，是题目 3 加了一个限制：把数组分成两个数组并最小化数组和的差，要求每个数组恰好 n 个元素（n = m >> 1）。全和 sum 已知，一个数组和是 s，差就是 |sum - 2s|。

区别在展开时多带一个「个数」维度：左半选了 k 个数，右半就必须选 n-k 个。于是 lsum[k] 存「左半恰好选 k 个数」的全部和，rsiz[need] 同理，need = n-k。配对时在对应桶里双指针逼近 |sum - 2·(lsum[k][i] + rsum[need][j])|，外层枚举 k 从 0 到 n。

注意静态桶容量 2^15：随机数据每侧最多 15 个元素（m ≤ 30）时够用；m = 40 时每侧 2^20 个和会越界，要换成动态数组，实测动态版 n = 40 跑 88.95ms。课上代码用静态数组，页面里保留原样并把这个上限写清楚。

#include <bits/stdc++.h>
    using namespace std;

    // 将数组分成两个数组并最小化数组和的差：折半 + 按选取个数分桶
    // 题目 3 的升级：每个数组必须恰好 n 个元素（n = m / 2），左半取 k 个、右半取 n-k 个
    // lsum[k] 存「左半选 k 个数」的全部和，rsum[k] 同理；双指针逼近 sum - 2*(lsum[k][i] + rsum[need][j])
    // 注意静态桶容量 2^15：随机数据每侧最多 15 个元素（m <= 30），m 更大要换动态数组
    const int MAXK = 16;
    const int MAXS = 1 << 15;

    int lsiz[MAXK];
    int rsiz[MAXK];
    int lsum[MAXK][MAXS];
    int rsum[MAXK][MAXS];

    void collect(vector<int>& nums, int i, int end, int s, int c, int sum[MAXK][MAXS], int* siz) {
        if (i == end) {
            sum[c][siz[c]++] = s;
        } else {
            int j = i + 1;
            while (j < end && nums[j] == nums[i]) {
                j++;
            }
            for (int k = 0; k <= j - i; k++) {
                collect(nums, j, end, s + k * nums[i], c + k, sum, siz);
            }
        }
    }

    int minimumDifference(vector<int>& nums) {
        int m = nums.size();
        int n = m >> 1;
        long long sum = 0;
        for (int v : nums) {
            sum += v;
        }
        sort(nums.begin(), nums.end());
        memset(lsiz, 0, sizeof(lsiz));
        memset(rsiz, 0, sizeof(rsiz));
        collect(nums, 0, n, 0, 0, lsum, lsiz);
        collect(nums, n, m, 0, 0, rsum, rsiz);
        for (int i = 0; i <= n; i++) {
            sort(lsum[i], lsum[i] + lsiz[i]);
            sort(rsum[i], rsum[i] + rsiz[i]);
        }
        long long ans = LLONG_MAX;
        for (int k = 0; k <= n; k++) {
            int need = n - k;
            for (int i = 0, j = rsiz[need] - 1; i < lsiz[k]; i++) {
                while (j > 0 && llabs(sum - 2LL * (lsum[k][i] + rsum[need][j - 1])) <= llabs(sum - 2LL * (lsum[k][i] + rsum[need][j]))) {
                    j--;
                }
                ans = min(ans, llabs(sum - 2LL * (lsum[k][i] + rsum[need][j])));
            }
        }
        return (int)ans;
    }

总结

折半搜索解决「全量展不开、半量可以」的一类题：劈两半各展开 2^(n/2) 个结果，再设计整合逻辑。零食问题用排序双指针计数，最接近目标值用排序双指针逼近，分区最小化差按个数分桶后配对，三题共用同一个骨架，差的全是整合那一步。

复杂度统一 O(2^(n/2)·log 2^(n/2))，n = 40 时约 2·2^20，排序 100 万个数毫秒级。看到 n ≤ 40、全量枚举必然超时的题，先判断半量能不能展开，能就折半。
