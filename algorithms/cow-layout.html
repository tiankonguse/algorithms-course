布局奶牛：负环判定的边界

布局奶牛和倍杀测量者，都是上一页那套「差分约束 + spfa 判负环」的延伸，但各多一个坎：奶牛题要补一条相邻链再跑两遍 spfa；倍杀题要把乘法关系取对数变回加法，还要给确定得分的选手安排一个专门的源点。

零、题3 建图：好友、情敌、相邻链

奶牛从 1 到 n 从左往右站，位置记为 x1..xn。好友 u v w（输入保证 u<v）希望 xv − xu ≤ w，这就是一条边 u→v 权 w；情敌 u v w 希望 xv − xu ≥ w，乘 −1 后是 xu − xv ≤ −w，建边 v→u 权 −w。

还有一条很容易漏：奶牛从左到右站，位置必须 x1 ≤ x2 ≤ ... ≤ xn，也就是 xi+1 − xi ≥ 0，写成 xi − xi+1 ≤ 0，建边 i+1→i 权 0。这条链把「顺序」也写进约束，漏了它，算出来的布局可能让奶牛倒着站。

一、两段 spfa

先跑 spfa(0)，0 号连通源点把所有点拉进来：判到负环说明约束矛盾，输出 −1。没负环再跑 spfa(1)，dist[n] 就是 1 号到 n 号的最大距离：差分约束的最短路长度正好是「最大可行差值」。

dist[n] 还是 INF（1 号到 n 号之间没有有限上界）就输出 −2。距离可以无穷远，也算一种合法方案，只是题目要单独区分。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;
    const int MAXN = 1005, MAXM = 20005, MAXQ = 1000005;
    int head[MAXN], nxt[MAXM], to[MAXM], w[MAXM], cnt;
    int dist[MAXN], upd[MAXN], que[MAXQ], h, t;
    bool enter[MAXN];
    int n, m1, m2;

    void add(int u, int v, int ww) {
      nxt[cnt] = head[u]; to[cnt] = v; w[cnt] = ww; head[u] = cnt++;
    }

    int spfa(int s) {
      h = t = 0;
      fill(dist, dist + n + 1, INT_MAX);
      fill(upd, upd + n + 1, 0);
      fill(enter, enter + n + 1, false);
      dist[s] = 0; upd[s] = 1;
      que[t++] = s; enter[s] = true;
      while (h < t) {
        int u = que[h++]; enter[u] = false;
        for (int ei = head[u]; ei > 0; ei = nxt[ei]) {
          int v = to[ei], ww = w[ei];
          if (dist[v] > dist[u] + ww) {
            dist[v] = dist[u] + ww;
            if (!enter[v]) {
              if (++upd[v] > n) return -1;
              que[t++] = v; enter[v] = true;
            }
          }
        }
      }
      if (dist[n] == INT_MAX) return -2;
      return dist[n];
    }

    int main() {
      scanf("%d%d%d", &n, &m1, &m2);
      cnt = 1;
      fill(head, head + n + 1, 0);
      for (int i = 1; i <= n; i++) add(0, i, 0);
      for (int i = 1, u, v, ww; i <= m1; i++) {
        scanf("%d%d%d", &u, &v, &ww);
        add(u, v, ww);
      }
      for (int i = 1, u, v, ww; i <= m2; i++) {
        scanf("%d%d%d", &u, &v, &ww);
        add(v, u, -ww);
      }
      for (int i = 1; i < n; i++) add(i + 1, i, 0);
      int ans = spfa(0);
      if (ans == -1) {
        printf("-1\n");
      } else {
        ans = spfa(1);
        printf("%d\n", ans);
      }
      return 0;
    }

实测（对拍）：3000 组随机，固定 x1=0、枚举 x2..xn ∈ [−15,15] 过滤好友/情敌/相邻链，三种返回值与课程完全一致。对拍器自己踩过一个坑：u==v 的输入用 continue 跳过会让写进输入文件的条数少于声明的条数，改成 do-while 重采样后才全过。

二、若干变量确定：两个源点必须分开

倍杀题里有「选手 u 得了 w 分」这种确定值。差分约束要表达 xu = w，做法是加一个限制源点 n+1，连两条边：n+1→u 权 ln w、u→n+1 权 −ln w。取对数后 du = ln w 被钉死，跑完 spfa，u 的 dist 恒等于 ln w。

这里有个细节：连通源点 0 和限制源点 n+1 必须分开。如果共用，0 号的 0 权边会把固定选手的 ln w 顺着 u→0、0→其它点一路传出去，自由选手全被框死。分开了，限制只作用于有得分记录的选手。

三、题4 倍杀测量者：二分 + 对数

「A k 倍杀 B」定义为 xA ≥ xB·k。取对数：dA ≥ dB + ln k，乘法变成加法，正好落在差分约束里。二分 ans，把类型1 的比例从 k 调成 k−ans、类型2 从 k 调成 k+ans，check 一次就是「ans 取这个值时还有没有人穿女装」。

类型1 誓言「u 没 k 倍杀 v」：不穿女装需要 xu ≥ xv·(k−ans)。k−ans ≥ 0 时建边 u→v 权 −ln(k−ans)；k−ans < 0 时右边非正，恒真，不连边。

类型2 誓言「u 被 v k 倍杀」：不穿需要 xv < xu·(k+ans)。严格小于不能直接进差分约束，用 ln(k+ans−1e-6) 平移成 ≤。

得分记录：选手 u 得 w 分 → n+1→u 权 ln w、u→n+1 权 −ln w。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;
    const int MAXN = 1005, MAXM = 3005, MAXQ = 1000005;
    const double INF = 1e10, sml = 1e-6;
    int n, m1, m2;
    int vow[MAXN][4], score[MAXN][2];
    int head[MAXN], nxt[MAXM], to[MAXM], cnt;
    double w[MAXM];
    double dist[MAXN];
    int upd[MAXN], que[MAXQ], h, t;
    bool enter[MAXN];

    void add(int u, int v, double ww) {
      nxt[cnt] = head[u]; to[cnt] = v; w[cnt] = ww; head[u] = cnt++;
    }

    void prepare() {
      cnt = 1; h = t = 0;
      fill(head, head + n + 2, 0);
      fill(dist, dist + n + 2, INF);
      fill(upd, upd + n + 2, 0);
      fill(enter, enter + n + 2, false);
    }

    bool spfa(int s) {
      dist[s] = 0; upd[s] = 1;
      que[t++] = s; enter[s] = true;
      while (h < t) {
        int u = que[h++]; enter[u] = false;
        for (int ei = head[u]; ei > 0; ei = nxt[ei]) {
          int v = to[ei];
          if (dist[v] > dist[u] + w[ei]) {
            dist[v] = dist[u] + w[ei];
            if (!enter[v]) {
              if (++upd[v] > n + 1) return true;
              que[t++] = v; enter[v] = true;
            }
          }
        }
      }
      return false;
    }

    bool check(double limit) {
      prepare();
      for (int i = 1; i <= n; i++) add(0, i, 0);
      for (int i = 1; i <= m1; i++) {
        if (vow[i][0] == 1) {
          if (-limit + vow[i][3] >= 0) {
            add(vow[i][1], vow[i][2], -log(-limit + vow[i][3]));
          }
        } else {
          add(vow[i][1], vow[i][2], log(limit + vow[i][3] - sml));
        }
      }
      for (int i = 1; i <= m2; i++) {
        add(n + 1, score[i][0], log(score[i][1]));
        add(score[i][0], n + 1, -log(score[i][1]));
      }
      return spfa(0);
    }

    double compute() {
      double l = 0, r = INF, m, ans = 0;
      while (r - l >= sml) {
        m = (l + r) / 2;
        if (check(m)) {
          ans = m;
          l = m + sml;
        } else {
          r = m - sml;
        }
      }
      return ans;
    }

    int main() {
      scanf("%d%d%d", &n, &m1, &m2);
      for (int i = 1; i <= m1; i++) {
        scanf("%d%d%d%d", &vow[i][0], &vow[i][1], &vow[i][2], &vow[i][3]);
      }
      for (int i = 1; i <= m2; i++) {
        scanf("%d%d", &score[i][0], &score[i][1]);
      }
      double ans = compute();
      if (ans == 0) {
        printf("-1\n");
      } else {
        printf("%.4f\n", ans);
      }
      return 0;
    }

「有人穿女装」= 差分约束无解 = spfa 判到负环。check(m) 返回 true 就是「ans=m 时还有人穿」，二分在 [0, 1e10] 上找最大的 m 让 check 为 true。

四、二分口径与边界

二分收敛后 ans == 0 输出 −1（不干预也没人穿）；否则保留 4 位小数。ans 是「还有人穿」区间的下界，不是临界点本身：实测一个 3 选手例（誓言 2 1-2 3、1 3-1 4、1 1-2 2，得分 1=4、2=1），check(1.7) true、check(1.75) false，二分落在 1.749999，%.4f 输出 1.7500。

实测（对拍）：3000 组随机，用独立的线性可行性判定做暴力参考，含 0 边界容差，与课程完全一致。对拍器自己修掉过三个 bug：类型2 约束方向写反、单变量界上下颠倒、浮点候选点取到 1e18 丢精度。

总结

奶牛题：好友一条边、情敌一条边，再加相邻链 i+1→i 权 0；先 spfa(0) 判无解 −1，再 spfa(1) 求最大距离，跑不到就是 −2。

确定值的差分约束：限制源点 n+1 用双向 ±ln w 边把 du 钉死，与连通源点 0 分离，0 权边才不会把固定值污染到自由变量。

倍杀题：乘法取对数变加法；类型1 在 k−ans<0 时恒真不连边，类型2 的严格小于用 sml 平移；二分最大 check true，ans==0 输出 −1。

三题共用同一个 spfa 判负环，变的只是建图：边方向、边权、源点个数。
