分层图最短路：状态当点

不把实际位置看成图上的点，而是把「实际位置 + 状态」的组合看成点。位置没变、状态变了，也是另一个点。状态是什么、怎么转移、代价怎么算，每道题自己设计，BFS 或 Dijkstra 的过程不变。

零、扩点：位置加状态才是点

上一页的 Dijkstra 在「位置图」上跑。分层图的第一步是扩点：一个实际位置带着 k 个可能状态，拆成 k+1 个点（位置, 状态）。转移是「改位置或改状态的动作」，边权是动作的代价。BFS 或 Dijkstra 照常跑，复杂度按点数和边数算：点数 = 位置数 × 状态数。

三题的状态各不相同：钥匙题状态是「收集了哪些钥匙」（位图），电动车题是「剩余电量」，飞行路线是「已经免费了几次」。扩点方式每题都不同，这正是分层图的全部设计空间。

一、钥匙题：钥匙集合当状态

获取所有钥匙的最短路径（LC864）：网格里小写字母是钥匙、大写字母是对应锁，没有钥匙过不去锁，问拿到所有钥匙最少走几步。k ≤ 6，钥匙集合正好一个 int 的 6 个位。

状态是 (行, 列, 掩码)：掩码记录已经捡到的钥匙。走进一个格子的动作是 O(1) 的状态转移：越界或墙跳过；是锁且掩码没有对应位，跳过；是钥匙就或上对应位；掩码满即完成。边权全是 1，用 BFS 逐层跑，visited 是三维的 (行, 列, 掩码)。

演示 · 3×3 网格，钥匙 a、b，逐层 BFS，捡起最后一把钥匙即返回

同一格 (0,0) 带掩码 00 和掩码 01 是两个不同状态，都要进队列：先空手出来，再带钥匙回来。这就是「位置 + 状态才是点」的直接体现。实测 30×30、k = 6 随机网格 0.94ms，状态数 900 × 64 = 5.76 万。

#include <bits/stdc++.h>
    using namespace std;

    // 获取所有钥匙的最短路径（LC864）：状态 = (行, 列, 钥匙掩码)，BFS 逐层
    int shortestPathAllKeys(vector<string>& grid) {
        int n = grid.size(), m = grid[0].size();
        int key = 0, sx = -1, sy = -1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == '@') {
                    sx = i;
                    sy = j;
                }
                if (grid[i][j] >= 'a' && grid[i][j] <= 'f') {
                    key |= 1 << (grid[i][j] - 'a');
                }
            }
        }
        vector<vector<vector<bool>>> visited(
            n, vector<vector<bool>>(m, vector<bool>(1 << 6, false)));
        deque<tuple<int, int, int>> q;
        q.push_back({sx, sy, 0});
        visited[sx][sy][0] = true;
        int level = 1;
        int move[5] = {-1, 0, 1, 0, -1};
        while (!q.empty()) {
            int size = q.size();
            for (int k = 0; k < size; k++) {
                auto [x, y, s] = q.front();
                q.pop_front();
                for (int i = 0; i < 4; i++) {
                    int nx = x + move[i], ny = y + move[i + 1];
                    int ns = s;
                    if (nx < 0 || nx == n || ny < 0 || ny == m || grid[nx][ny] == '#') {
                        continue;
                    }
                    if (grid[nx][ny] >= 'A' && grid[nx][ny] <= 'F' &&
                        ((ns & (1 << (grid[nx][ny] - 'A'))) == 0)) {
                        continue;
                    }
                    if (grid[nx][ny] >= 'a' && grid[nx][ny] <= 'f') {
                        ns |= 1 << (grid[nx][ny] - 'a');
                    }
                    if (ns == key) {
                        return level;
                    }
                    if (!visited[nx][ny][ns]) {
                        visited[nx][ny][ns] = true;
                        q.push_back({nx, ny, ns});
                    }
                }
            }
            level++;
        }
        return -1;
    }

二、电动车：电量当状态

电动车游城市（LCP 35）：从 start 到 end，电量上限 cnt，每走 1 距离耗 1 电、花 1 时间，每个城市可以充电，charge[i] 是充 1 电花的单位时间，求最少时间。状态是 (城市, 电量)。

两类转移：原地充一格电，代价 charge[cur]，电量 +1；不充电走一条边，代价边权 w，电量 -w。都是非负代价，Dijkstra 按时间组织堆。cnt ≤ 100、n ≤ 100，状态数最多一万，直接开二维 distance 数组。

官方样例验证：样例 1 输出 43（1 充 3 电到 3，3 充 5 电到 0，充电 35 + 行驶 8），样例 2 输出 38，本页实现均吻合。实测 n = 100、cnt = 100、约 1500 条边 0.085ms。

#include <bits/stdc++.h>
    using namespace std;

    // 电动车游城市（LCP 35）：状态 = (城市, 电量)，边 = 充一格电 / 不充电走一条边
    int electricCarPlan(vector<vector<int>>& paths, int cnt, int start, int end,
                        vector<int>& charge) {
        int n = charge.size();
        vector<vector<pair<int, int>>> graph(n);
        for (auto& p : paths) {
            graph[p[0]].push_back({p[1], p[2]});
            graph[p[1]].push_back({p[0], p[2]});
        }
        vector<vector<int>> distance(n, vector<int>(cnt + 1, INT_MAX));
        distance[start][0] = 0;
        vector<vector<bool>> visited(n, vector<bool>(cnt + 1, false));
        priority_queue<tuple<int, int, int>,
                       vector<tuple<int, int, int>>,
                       greater<tuple<int, int, int>>> heap;
        heap.push({0, start, 0});
        while (!heap.empty()) {
            auto [cost, cur, power] = heap.top();
            heap.pop();
            if (visited[cur][power]) {
                continue;
            }
            if (cur == end) {
                return cost;
            }
            visited[cur][power] = true;
            if (power < cnt) {
                if (!visited[cur][power + 1] &&
                    cost + charge[cur] < distance[cur][power + 1]) {
                    distance[cur][power + 1] = cost + charge[cur];
                    heap.push({cost + charge[cur], cur, power + 1});
                }
            }
            for (auto [nextCity, w] : graph[cur]) {
                int restPower = power - w;
                int nextCost = cost + w;
                if (restPower >= 0 && !visited[nextCity][restPower] &&
                    nextCost < distance[nextCity][restPower]) {
                    distance[nextCity][restPower] = nextCost;
                    heap.push({nextCost, nextCity, restPower});
                }
            }
        }
        return -1;
    }

三、飞行路线：免费次数当层

飞行路线（洛谷 P4568）：最多 k 条航线免费，问最少花费。状态是 (城市, 已用免费次数)，免费次数就是层：第 use 层表示「已经免费用了 use 次」。

转移两条：用免费，边权按 0 算，use + 1；不用免费，边权照付，use 不变。答案在终点任意一层里取最小。层数 k + 1，点数 (k + 1) × n；手写堆版本把二维状态压成一维堆下标，注释里说清楚了为什么不用反向索引堆：一个 (城市, 次数) 组合才是一个堆元素，反向索引堆要按组合维护 where，不如普通堆直接。

演示 · 4 点图，k = 2：免费边 0 权跨层，付费边同层照付

实测：n = 10^4、m = 5×10^4、k = 10 的随机图 4.6ms；同图 k = 0（退化成普通 Dijkstra）0.11ms。分层的代价是状态数乘 k + 1 倍，收益是免费次数买到的更短路：链 1000 条边每条 10、加一条 0→999 直连 900，k = 0 走直连花 900，k = 1 免费掉直连花 0。

#include <bits/stdc++.h>
    using namespace std;

    // 飞行路线（洛谷 P4568）：分层图最短路，状态 = (城市, 已用免费次数)
    const int MAXN = 10001;
    const int MAXM = 100001;
    const int MAXK = 11;

    int head[MAXN], nxt[MAXM], to[MAXM], weight[MAXM], cnt;
    int distanceArr[MAXN][MAXK];
    bool visited[MAXN][MAXK];
    // 手写普通堆，静态结构（二维点压成堆下标，反向索引堆不好写）
    int heapArr[MAXM * MAXK][3];
    int heapSize;
    int n, m, k, s, t;

    void build() {
        cnt = 1;
        heapSize = 0;
        for (int i = 0; i < n; i++) {
            head[i] = 0;
            for (int j = 0; j <= k; j++) {
                distanceArr[i][j] = INT_MAX;
                visited[i][j] = false;
            }
        }
    }

    void addEdge(int u, int v, int w) {
        nxt[cnt] = head[u];
        to[cnt] = v;
        weight[cnt] = w;
        head[u] = cnt++;
    }

    void push(int u, int use, int c) {
        heapArr[heapSize][0] = u;
        heapArr[heapSize][1] = use;
        heapArr[heapSize][2] = c;
        int i = heapSize++;
        while (heapArr[i][2] < heapArr[(i - 1) / 2][2]) {
            swap(heapArr[i], heapArr[(i - 1) / 2]);
            i = (i - 1) / 2;
        }
    }

    void pop(int& u, int& use, int& cost) {
        u = heapArr[0][0];
        use = heapArr[0][1];
        cost = heapArr[0][2];
        swap(heapArr[0], heapArr[--heapSize]);
        int i = 0, l = 1;
        while (l < heapSize) {
            int best = l + 1 < heapSize && heapArr[l + 1][2] < heapArr[l][2] ? l + 1 : l;
            best = heapArr[best][2] < heapArr[i][2] ? best : i;
            if (best == i) {
                break;
            }
            swap(heapArr[best], heapArr[i]);
            i = best;
            l = i * 2 + 1;
        }
    }

    int dijkstra() {
        distanceArr[s][0] = 0;
        push(s, 0, 0);
        while (heapSize > 0) {
            int u, use, cost;
            pop(u, use, cost);
            if (visited[u][use]) {
                continue;
            }
            visited[u][use] = true;
            if (u == t) {
                return cost;
            }
            for (int ei = head[u]; ei > 0; ei = nxt[ei]) {
                int v = to[ei], w = weight[ei];
                if (use < k && distanceArr[v][use + 1] > distanceArr[u][use]) {
                    // 使用免费：边权当作 0，已用次数 +1
                    distanceArr[v][use + 1] = distanceArr[u][use];
                    push(v, use + 1, distanceArr[v][use + 1]);
                }
                if (distanceArr[v][use] > distanceArr[u][use] + w) {
                    // 不用免费：照常付钱
                    distanceArr[v][use] = distanceArr[u][use] + w;
                    push(v, use, distanceArr[v][use]);
                }
            }
        }
        return -1;
    }

    int main() {
        cin >> n >> m >> k >> s >> t;
        build();
        for (int i = 0, a, b, c; i < m; i++) {
            cin >> a >> b >> c;
            addEdge(a, b, c);
            addEdge(b, a, c);
        }
        cout << dijkstra() << endl;
        return 0;
    }

总结

分层图 = 扩点：位置不变、状态变，也是新点。三题三个状态维度：钥匙掩码、剩余电量、免费次数；转移动作分别是一次移动、充一格电、免一条边。BFS 或 Dijkstra 的骨架一字不改，改的只是状态定义和转移代价。状态数就是位置数乘状态数，n 和 k 的上限决定能不能开下这张表。

这一页三题的状态位数分别是 6、100、11，都在「一张二维表装得下」的范围；状态维度再大（比如 2^30 个状态），就得换思路了。
