边分树的合并

上一节建好了信息树，这一节让它发力。两道题都是同一副骨架：在 t1 上建边分树，沿 t2 自底向上合并，在合并的节点上枚举跨重心边的组合。暴力写挂是两棵树，通道是三棵树，后者的第三棵树正好用上前面 dfn 序求 LCA 的那套。

零、背景

上一节的可持久化是「链并入树」：前缀版本一个个叠上去，旧的全部保留。这一节的合并方向不同：沿着一棵树的 dfs 自底向上，把儿子代表的点集并进父亲，合并完就丢，不保留版本。

合并发生在 t2 的每个点 u 上。参与合并的两份信息来自 u 的不同子树，它们里面的点对的 lca2 恰好是 u，这一点会在式子里变成一个固定的修正量。

一、题目4：暴力写挂

两棵树 t1、t2 都有 n 个点，都以 1 为根，边权可能为负。dis1(x) 是 t1 中 x 到根的距离，lca1、lca2 分别是两棵树上的最近公共祖先。点对的指标 = dis1(x) + dis1(y) − dis1(lca1(x,y)) − dis2(lca2(x,y))，x ≤ y，打印最大值。1 ≤ n ≤ 366666。

先把式子拆开。t1 上的距离是 dis1(x) + dis1(y) − 2 × dis1(lca1)，代回去，指标乘 2 恰好变成三段：

于是在 t1 上建边分树（要三度化），每个实点在每一层记 lmax = dis1 + path 的最大值，左侧一份、右侧一份。沿 t2 自底向上合并：合并在 u 进行时，参与合并的点对 lca2 都是 u，add = −2 × dist2(u)，u 到 t2 根的距离。

合并到同一个节点时，左配右、右配左两种跨向组合更新答案，然后两侧各自取 max 存下来。x = y 的情况在合并前单独算：指标(x,x) = dis1(x) − dis2(x)。最后答案除以 2。

两个细节。t1 的边权可能为负，dis1 要用 dfs 递推，不能按层算。lmax、rmax 的初值是负无穷，建树前先整体赋值。

完整代码如下。

    
#include <bits/stdc++.h>

    using namespace std;

    using ll = long long;

    const int MAXN = 366667;
    const int MAXM = MAXN << 1;
    const int MAXT = 10000001;
    const ll INF = 1LL << 50;

    int n, cntn;

    int head0[MAXN];
    int next0[MAXN << 1];
    int to0[MAXN << 1];
    int weight0[MAXN << 1];
    int cnt0;

    int head1[MAXM];
    int next1[MAXM << 1];
    int to1[MAXM << 1];
    int weight1[MAXM << 1];
    int cnt1;

    int head2[MAXN];
    int next2[MAXN << 1];
    int to2[MAXN << 1];
    int weight2[MAXN << 1];
    int cnt2;

    ll dis1[MAXN];

    bool vis[MAXM];
    int siz[MAXM];

    int up[MAXN];
    int root[MAXN];
    int ls[MAXT];
    int rs[MAXT];
    ll lmax[MAXT];
    ll rmax[MAXT];
    int cntt;

    ll ans;

    void addEdge0(int u, int v, int w) {
        next0[++cnt0] = head0[u];
        to0[cnt0] = v;
        weight0[cnt0] = w;
        head0[u] = cnt0;
    }

    void addEdge1(int u, int v, int w) {
        next1[++cnt1] = head1[u];
        to1[cnt1] = v;
        weight1[cnt1] = w;
        head1[u] = cnt1;
    }

    void addEdge2(int u, int v, int w) {
        next2[++cnt2] = head2[u];
        to2[cnt2] = v;
        weight2[cnt2] = w;
        head2[u] = cnt2;
    }

    void getDist(int u, int fa, ll dist1) {
        dis1[u] = dist1;
        for (int e = head0[u]; e > 0; e = next0[e]) {
            int v = to0[e];
            if (v != fa) {
                getDist(v, u, dist1 + weight0[e]);
            }
        }
    }

    void rebuild(int u, int fa) {
        int last = 0;
        for (int e = head0[u]; e > 0; e = next0[e]) {
            int v = to0[e];
            int w = weight0[e];
            if (v != fa) {
                if (last == 0) {
                    last = u;
                    addEdge1(u, v, w);
                    addEdge1(v, u, w);
                } else {
                    int add = ++cntn;
                    addEdge1(last, add, 0);
                    addEdge1(add, last, 0);
                    addEdge1(add, v, w);
                    addEdge1(v, add, w);
                    last = add;
                }
                rebuild(v, u);
            }
        }
    }

    void getSize(int u, int fa) {
        siz[u] = 1;
        for (int e = head1[u]; e > 0; e = next1[e]) {
            int v = to1[e];
            if (v != fa && !vis[e >> 1]) {
                getSize(v, u);
                siz[u] += siz[v];
            }
        }
    }

    int getCentroidEdge(int u, int fa) {
        getSize(u, fa);
        int total = siz[u];
        int half = total >> 1;
        bool find = false;
        while (!find) {
            find = true;
            for (int e = head1[u]; e > 0; e = next1[e]) {
                int v = to1[e];
                if (v != fa && !vis[e >> 1] && siz[v] > half) {
                    fa = u;
                    u = v;
                    find = false;
                    break;
                }
            }
        }
        int best = 0, edge = 0;
        for (int e = head1[u]; e > 0; e = next1[e]) {
            if (!vis[e >> 1]) {
                int v = to1[e];
                int sub = (v == fa ? (total - siz[u]) : siz[v]);
                if (sub > best) {
                    best = sub;
                    edge = e;
                }
            }
        }
        return edge;
    }

    void dfs(int u, int fa, ll path, int op) {
        if (u <= n) {
            if (up[u] == 0) {
                up[u] = ++cntt;
                root[u] = cntt;
            }
            int cur = up[u];
            int nxt = ++cntt;
            if (op == 0) {
                ls[cur] = nxt;
                lmax[cur] = dis1[u] + path;
            } else {
                rs[cur] = nxt;
                rmax[cur] = dis1[u] + path;
            }
            up[u] = nxt;
        }
        for (int e = head1[u]; e > 0; e = next1[e]) {
            int v = to1[e];
            if (v != fa && !vis[e >> 1]) {
                dfs(v, u, path + weight1[e], op);
            }
        }
    }

    void solve(int u) {
        int edge = getCentroidEdge(u, 0);
        if (edge > 0) {
            vis[edge >> 1] = true;
            int v1 = to1[edge];
            int v2 = to1[edge ^ 1];
            dfs(v1, 0, 0, 0);
            dfs(v2, 0, weight1[edge], 1);
            solve(v1);
            solve(v2);
        }
    }

    int mergeTree(int x, int y, ll add) {
        if (x == 0 || y == 0) {
            return x + y;
        }
        ans = max(ans, max(lmax[x] + rmax[y], lmax[y] + rmax[x]) + add);
        lmax[x] = max(lmax[x], lmax[y]);
        rmax[x] = max(rmax[x], rmax[y]);
        ls[x] = mergeTree(ls[x], ls[y], add);
        rs[x] = mergeTree(rs[x], rs[y], add);
        return x;
    }

    void compute(int u, int fa, ll dist2) {
        ans = max(ans, (dis1[u] - dist2) * 2);
        for (int e = head2[u]; e > 0; e = next2[e]) {
            int v = to2[e];
            if (v != fa) {
                compute(v, u, dist2 + weight2[e]);
                root[u] = mergeTree(root[u], root[v], -dist2 * 2);
            }
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n;
        for (int i = 1, u, v, w; i < n; i++) {
            cin >> u >> v >> w;
            addEdge0(u, v, w);
            addEdge0(v, u, w);
        }
        for (int i = 1, u, v, w; i < n; i++) {
            cin >> u >> v >> w;
            addEdge2(u, v, w);
            addEdge2(v, u, w);
        }
        for (int i = 1; i < MAXT; i++) {
            lmax[i] = rmax[i] = -INF;
        }
        cntn = n;
        cnt1 = 1;
        ans = -INF;
        getDist(1, 0, 0);
        rebuild(1, 0);
        solve(1);
        compute(1, 0, 0);
        cout << (ans >> 1) << '\n';
        return 0;
    }

二、题目5：通道

三棵树 t1、t2、t3 都有 n 个点，边权非负。点对 (x, y) 要求两点不同，在树上的距离记 dis1(x,y)、dis2(x,y)、dis3(x,y)，打印三者之和的最大值。2 ≤ n ≤ 10⁵，边权到 10¹²。

三棵树各出一件事：t1 建边分树并三度化；t2 决定合并的顺序和修正量；t3 提供两点距离，用 dfn 序加倍增表求 LCA，就是前面「dfn 序求 LCA」那页的做法。

信息和上一题不同：每侧存的不再是一个数，而是一对点。lx、ly 是左侧最优对的两个端点，lxv 是它的值 = path1 + dep2 的和；右侧对称。为什么存「一对点」？因为 dist3(x,y) 不是一个能拆开的量，必须知道端点是谁才能查。

合并两个点集时，新的最优对不用重新枚举所有点对，靠的是树的直径那个结论：合并两个点集，新集合的最优对一定出自原来两个最优对的四个端点的两两组合。六种组合都试一遍，取最优。道理和直径合并一样，把一个端点换成最优对的端点之一，值不会变小。

直径结论：四个端点，六种组合

一条链上四个点，组合的值 = W(a) + W(b) + 两点距离。集合 A 的最优对是 (a1, b1)，集合 B 的最优对是 (a2, b2)。点「下一步」把六种组合都算一遍，看新最优对落在哪里。

集合 A 的端点 集合 B 的端点 当前组合

合并到边分树的同一个节点时，只有「左配右、右配左」两种跨向组合跨越当前重心边，直接更新答案；同侧的组合不跨这条边，它们会在更深层自己的分离边上被枚举，这里不用管。

还有一个省事的地方：点对的六种组合里允许暂时存下 (u, u) 这样的退化对，它单独算值时按 0 处理，不会真的参与竞争；真正的组合都在不同子树之间，端点不会重合。

        to0[cnt0] = v;
        weight0[cnt0] = w;
        head0[u] = cnt0;

    void addEdge1(int u, int v, ll w) {
        next1[++cnt1] = head1[u];
        to1[cnt1] = v;
        weight1[cnt1] = w;
        head1[u] = cnt1;

    void addEdge2(int u, int v, ll w) {
        next2[++cnt2] = head2[u];
        to2[cnt2] = v;
        weight2[cnt2] = w;
        head2[u] = cnt2;
    }

    void addEdge3(int u, int v, ll w) {
        next3[++cnt3] = head3[u];
        to3[cnt3] = v;
        weight3[cnt3] = w;
        head3[u] = cnt3;

    void dfsTree2(int u, int fa, ll dep) {
        dep2[u] = dep;
        for (int e = head2[u]; e > 0; e = next2[e]) {
            int v = to2[e];
            if (v != fa) {
                dfsTree2(v, u, dep + weight2[e]);
            }
        }
    }

    void rebuildTree1(int u, int fa) {
        int last = 0;
        for (int e = head0[u]; e > 0; e = next0[e]) {
            int v = to0[e];
            ll w = weight0[e];
            if (v != fa) {
                if (last == 0) {
                    last = u;
                    addEdge1(u, v, w);
                    addEdge1(v, u, w);
                } else {
                    int add = ++cntn;
                    addEdge1(last, add, 0);
                    addEdge1(add, last, 0);
                    addEdge1(add, v, w);
                    addEdge1(v, add, w);
                    last = add;
                }
                rebuildTree1(v, u);
            }
        }
    }

    void getSize(int u, int fa) {
        siz[u] = 1;
        for (int e = head1[u]; e > 0; e = next1[e]) {
            int v = to1[e];
            if (v != fa && !vis[e >> 1]) {
                getSize(v, u);
                siz[u] += siz[v];
            }
        }
    }

    int getCentroidEdge(int u, int fa) {
        getSize(u, fa);
        int total = siz[u];
        int half = total >> 1;
        bool find = false;
        while (!find) {
            find = true;
            for (int e = head1[u]; e > 0; e = next1[e]) {
                int v = to1[e];
                if (v != fa && !vis[e >> 1] && siz[v] > half) {
                    fa = u;
                    u = v;
                    find = false;
                    break;
                }
            }
        }
        int best = 0, edge = 0;
        for (int e = head1[u]; e > 0; e = next1[e]) {
            if (!vis[e >> 1]) {
                int v = to1[e];
                int sub = (v == fa ? (total - siz[u]) : siz[v]);
                if (sub > best) {
                    best = sub;
                    edge = e;
                }
            }
        }
        return edge;
    }

    void dfsTree1(int u, int fa, ll path, int op) {
        if (u <= n) {
            if (up[u] == 0) {
                up[u] = ++cntt;
                root[u] = cntt;
            }
            int cur = up[u];
            int nxt = ++cntt;
            ll val = path + dep2[u];
            if (op == 0) {
                ls[cur] = nxt;
                lx[cur] = ly[cur] = u;
                lxv[cur] = lyv[cur] = val;
            } else {
                rs[cur] = nxt;
                rx[cur] = ry[cur] = u;
                rxv[cur] = ryv[cur] = val;
            }
            up[u] = nxt;
        }
        for (int e = head1[u]; e > 0; e = next1[e]) {
            int v = to1[e];
            if (v != fa && !vis[e >> 1]) {
                dfsTree1(v, u, path + weight1[e], op);
            }
        }
    }

    void solveTree1(int u) {
        int edge = getCentroidEdge(u, 0);
        if (edge > 0) {
            vis[edge >> 1] = true;
            int v1 = to1[edge];
            int v2 = to1[edge ^ 1];
            dfsTree1(v1, 0, 0, 0);
            dfsTree1(v2, 0, weight1[edge], 1);
            solveTree1(v1);
            solveTree1(v2);
        }
    }

    int getUp(int x, int y) {
        return dfn[x] < dfn[y] ? x : y;
    }

    void dfsTree3(int u, int fa, ll dist) {
        dfn[u] = ++cntd;
        rmq[dfn[u]][0] = fa;
        dist3[u] = dist;
        for (int e = head3[u]; e > 0; e = next3[e]) {
            int v = to3[e];
            if (v != fa) {
                dfsTree3(v, u, dist + weight3[e]);
            }
        }
    }

    void rmqTree3() {
        dfsTree3(1, 0, 0);
        for (int i = 2; i <= n; i++) {
            lg2[i] = lg2[i >> 1] + 1;
        }
        for (int pre = 0, cur = 1; cur <= lg2[n]; pre++, cur++) {
            for (int i = 1; i + (1 << cur) - 1 <= n; i++) {
                rmq[i][cur] = getUp(rmq[i][pre], rmq[i + (1 << pre)][pre]);
            }
        }
    }

    int lcaTree3(int x, int y) {
        if (x == y) {
            return x;
        }
        x = dfn[x];
        y = dfn[y];
        if (x > y) {
            swap(x, y);
        }
        x++;
        int k = lg2[y - x + 1];
        return getUp(rmq[x][k], rmq[y - (1 << k) + 1][k]);
    }

    ll getDist3(int x, int y) {
        int l = lcaTree3(x, y);
        return dist3[x] + dist3[y] - dist3[l] * 2;
    }

    ll getDist(int x, int y, ll xv, ll yv) {
        if (x == y) {
            return 0;
        }
        return getDist3(x, y) + xv + yv;
    }

    int a, b, c, d;
    ll av, bv, cv, dv;

    void getInfo(int i, int j, int iop, int jop) {
        if (iop == 0) {
            a = lx[i]; b = ly[i]; av = lxv[i]; bv = lyv[i];
        } else {
            a = rx[i]; b = ry[i]; av = rxv[i]; bv = ryv[i];
        }
        if (jop == 0) {
            c = lx[j]; d = ly[j]; cv = lxv[j]; dv = lyv[j];
        } else {
            c = rx[j]; d = ry[j]; cv = rxv[j]; dv = ryv[j];
        }
    }

    ll bestCross(int i, int j, int iop, int jop) {
        getInfo(i, j, iop, jop);
        ll p1 = getDist(a, c, av, cv);
        ll p2 = getDist(a, d, av, dv);
        ll p3 = getDist(b, c, bv, cv);
        ll p4 = getDist(b, d, bv, dv);
        return max(max(p1, p2), max(p3, p4));
    }

    int x, y;
    ll xv, yv;
    ll bestDist;

    void better(int curx, int cury, ll curxv, ll curyv) {
        if (curx == 0 || cury == 0) {
            return;
        }
        ll curDist = getDist(curx, cury, curxv, curyv);
        if (curDist > bestDist) {
            x = curx; y = cury; xv = curxv; yv = curyv;
            bestDist = curDist;
        }
    }

    void mergeInfo(int i, int j, int op) {
        getInfo(i, j, op, op);
        x = y = 0;
        xv = yv = 0;
        bestDist = -INF;
        better(a, b, av, bv);
        better(c, d, cv, dv);
        better(a, c, av, cv);
        better(a, d, av, dv);
        better(b, c, bv, cv);
        better(b, d, bv, dv);
        if (op == 0) {
            lx[i] = x; ly[i] = y; lxv[i] = xv; lyv[i] = yv;
        } else {
            rx[i] = x; ry[i] = y; rxv[i] = xv; ryv[i] = yv;
        }
    }

    int mergeTree(int i, int j, ll add) {
        if (i == 0 || j == 0) {
            return i + j;
        }
        if (ls[i] > 0 && rs[j] > 0) {
            ans = max(ans, bestCross(i, j, 0, 1) + add);
        }
        if (rs[i] > 0 && ls[j] > 0) {
            ans = max(ans, bestCross(i, j, 1, 0) + add);
        }
        mergeInfo(i, j, 0);
        mergeInfo(i, j, 1);
        ls[i] = mergeTree(ls[i], ls[j], add);
        rs[i] = mergeTree(rs[i], rs[j], add);
        return i;
    }

    void compute(int u, int fa) {
        for (int e = head2[u]; e > 0; e = next2[e]) {
            int v = to2[e];
            if (v != fa) {
                compute(v, u);
                root[u] = mergeTree(root[u], root[v], -dep2[u] * 2);
            }
        }
    }

            cin >> u >> v >> w;
            addEdge0(u, v, w);
            addEdge0(v, u, w);
        }
        for (int i = 1; i < n; i++) {
            cin >> u >> v >> w;
            addEdge2(u, v, w);
            addEdge2(v, u, w);
        }
        for (int i = 1; i < n; i++) {
            cin >> u >> v >> w;
            addEdge3(u, v, w);
            addEdge3(v, u, w);
        }
        cntn = n;
        cnt1 = 1;
        ans = -INF;
        dfsTree2(1, 0, 0);
        rebuildTree1(1, 0);
        solveTree1(1);
        rmqTree3();
        compute(1, 0);
        cout << ans << '\n';
        return 0;
    }

两道题的主导都是信息树节点的总数，每层每个点一个节点，一共 n log n 个。

四、最后

边分树的合并把「分治时统计答案」变成了「数据结构上拼信息」：跨重心边的点对在合并的节点上被枚举，同侧的往下沉。两棵树、三棵树的距离题都吃这一套。
