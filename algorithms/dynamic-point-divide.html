点分树与动态点分治（二）：按重心移动、年龄统计与树上开关

上一页的三道题都是「点分树加开点线段树」的固定套件。这一页的四道题各变一点花样：幻想乡把查询变成沿树移动找最优点，开店给每个点带上年龄做区间统计，成都七中干脆不建点分树、只借它的分析框架，捉迷藏则把线段树换成对顶堆维护树上开关。

零、背景

动态点分治的题，读完后先问两个问题：答案是「以某个点为中心的统计」还是「树上全局的最优结构」，修改影响的是点权还是点的状态。前者沿用震波那套桶加容斥，后者往往要在点分树上移动或者换数据结构。

这一页的四个题正好各占一种变化。共同点只有一个：都靠「x 的祖先重心只有 O(log n) 个、每个祖先的统计规模一次到位」吃饭。

一、题目4：幻想乡战略游戏

n 个点的带边权的树，初始点权全是 0。指挥点 x 指挥 y 的花费是 x 到 y 的路径权值和乘 y 的点权，最小指挥总花费就是让这个总花费最小的那个 x 的花费。m 条操作 x v：先把 x 的点权加 v，再打印此时的最小指挥总花费，v 可以为负但保证点权非负。n、m 到 10⁵，边权到 1000。测试链接 P3345。

点分治那一讲的题目3「快递员」是同一个模型。这类问题的目标点叫加权重心，它有一个代数性质：对相邻的 u、v，设去掉边 (u,v) 后 v 那一侧的点权和是 side，全树点权和是 S，那么 cost(v) − cost(u) = (S − 2·side) × w(u,v)。

因此只要某个邻居那一侧的点权和超过总和的一半，走过去花费一定降；两侧都不超过，当前点就是全局最优。花费沿着树「下坡」，不会陷进局部最优。

维护还是震波那套：sum[c] 记 c 所在连通块的点权和，addCost[c] 记块内每个点到 c 的「距离乘点权」之和，minusCost[cur] 记 cur 块内每个点到上级重心的同样统计。查询某点 x 的花费时，沿祖先链每级做加 addCost[fa]、减 minusCost[cur]、再补 (sum[fa] − sum[cur])·dist(x, fa)，最后加上自己的 addCost[x]。

修改 x 点权：沿链每级更新三个数组，O(log n)。找最优点时从根重心出发，在当前点 u 检查所有邻居 v 的花费，发现更小就往那个方向跳。跳的目标不是 v 本身，而是 v 所在连通块的重心，这个重心在建点分树时顺手记在边上（cent[e]）。每跳一次连通块至少减半，所以跳 O(log n) 次。

从点 1 出发，找最佳指挥点

六个点的树，边权标在边上，每个点旁边的小字是点权。点「下一步」看每一步：当前指挥点的花费、每个邻居的花费、往哪边走。所有花费都是现场按定义算的。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 100001;
    int n, m;

    int head[MAXN];
    int nxt[MAXN << 1];
    int to[MAXN << 1];
    int weight[MAXN << 1];
    int cent[MAXN << 1];
    int cntg;

    int fa[MAXN];
    int dep[MAXN];
    int siz[MAXN];
    int son[MAXN];
    int top[MAXN];
    int dist[MAXN];

    bool vis[MAXN];
    int centfa[MAXN];

    long long sum[MAXN];
    long long addCost[MAXN];
    long long minusCost[MAXN];

    void addEdge(int u, int v, int w) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        weight[cntg] = w;
        head[u] = cntg;
    }

    void dfs1(int u, int f, int dis) {
        fa[u] = f;
        dep[u] = dep[f] + 1;
        dist[u] = dis;
        siz[u] = 1;
        for (int e = head[u]; e; e = nxt[e]) {
            int v = to[e];
            int w = weight[e];
            if (v != f) {
                dfs1(v, u, dis + w);
            }
        }
        for (int ei = head[u], v; ei; ei = nxt[ei]) {
            v = to[ei];
            if (v != f) {
                siz[u] += siz[v];
                if (son[u] == 0 || siz[son[u]] < siz[v]) {
                    son[u] = v;
                }
            }
        }
    }

    void dfs2(int u, int t) {
        top[u] = t;
        if (son[u] == 0) {
            return;
        }
        dfs2(son[u], t);
        for (int e = head[u], v; e; e = nxt[e]) {
            v = to[e];
            if (v != fa[u] && v != son[u]) {
                dfs2(v, v);
            }
        }
    }

    int getLca(int a, int b) {
        while (top[a] != top[b]) {
            if (dep[top[a]] <= dep[top[b]]) {
                b = fa[top[b]];
            } else {
                a = fa[top[a]];
            }
        }
        return dep[a] <= dep[b] ? a : b;
    }

    int getDist(int x, int y) {
        return dist[x] + dist[y] - (dist[getLca(x, y)] << 1);
    }

    void getSize(int u, int f) {
        siz[u] = 1;
        for (int e = head[u]; e; e = nxt[e]) {
            int v = to[e];
            if (v != f && !vis[v]) {
                getSize(v, u);
                siz[u] += siz[v];
            }
        }
    }

    int getCentroid(int u, int f) {
        getSize(u, f);
        int half = siz[u] >> 1;
        bool find = false;
        while (!find) {
            find = true;
            for (int e = head[u]; e; e = nxt[e]) {
                int v = to[e];
                if (v != f && !vis[v] && siz[v] > half) {
                    f = u;
                    u = v;
                    find = false;
                    break;
                }
            }
        }
        return u;
    }

    void centroidTree(int u, int f) {
        centfa[u] = f;
        vis[u] = true;
        for (int e = head[u]; e; e = nxt[e]) {
            int v = to[e];
            if (!vis[v]) {
                int nextCent = getCentroid(v, u);
                cent[e] = nextCent;
                centroidTree(nextCent, u);
            }
        }
    }

    void add(int x, int v) {
        sum[x] += v;
        for (int cur = x, fa = centfa[cur]; fa > 0; cur = fa, fa = centfa[cur]) {
            int dist = getDist(x, fa);
            sum[fa] += v;
            addCost[fa] += 1LL * v * dist;
            minusCost[cur] += 1LL * v * dist;
        }
    }

    long long query(int x) {
        long long ans = addCost[x];
        for (int cur = x, fa = centfa[cur]; fa > 0; cur = fa, fa = centfa[cur]) {
            int dist = getDist(x, fa);
            ans += addCost[fa];
            ans -= minusCost[cur];
            ans += (sum[fa] - sum[cur]) * dist;
        }
        return ans;
    }

    long long compute(int u) {
        long long ans = query(u);
        for (int e = head[u]; e; e = nxt[e]) {
            int v = to[e];
            if (query(v) < ans) {
                return compute(cent[e]);
            }
        }
        return ans;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1, u, v, w; i < n; i++) {
            cin >> u >> v >> w;
            addEdge(u, v, w);
            addEdge(v, u, w);
        }
        dfs1(1, 0, 0);
        dfs2(1, 1);
        int root = getCentroid(1, 0);
        centroidTree(root, 0);
        for (int i = 1, x, v; i <= m; i++) {
            cin >> x >> v;
            add(x, v);
            cout << compute(root) << '\n';
        }
        return 0;
    }

二、题目5：开店

n 个人连成一棵树，每人有年龄，边有距离。m 条查询 u l r：年龄在 [l, r] 内的所有人到 u 号人的距离总和。n 到 1.5×10⁵，m 到 2×10⁵，年龄到 10⁹，距离到 1000。强制在线，l 和 r 用上次答案按 A 取模解出来。测试链接 P3241。

朴素做法是每次查询枚举年龄范围内的人，逐个累加距离，单次 O(n)，总量 O(nm) 超时。年龄是 10⁹ 级的值，线段树按年龄开点也能做，但开点线段树在这种静态信息上性价比不高。

这里的做法是给每个重心准备两个列表，一次性收集、排序、建前缀和。cur 列表放「c 所在连通块里每个人（年龄，到 c 的距离）」，fa 列表放「cur 块里每个人（年龄，到 cur 的上级重心的距离）」，都按年龄排序并做距离前缀和。查询时二分年龄界，前缀和相减得到个数和距离和。

查询 u 的答案：u 自己的 cur 列表直接给一段；每级祖先 f 再做「f 的 cur 列表 − u 方向儿子的 fa 列表」加上个数差乘 dist(u, f)。个数差那一项是补偿：f 块里的人到 f 的距离是记好的，但他们到 u 还要多走 dist(u, f)。

这里有个细节。fa 列表记的是「到上级重心的距离」而不是「到自己重心的距离」。往上级转移时，多走的那截对块内每人都一样，是常数 dist(u, f)，乘个数就能补；如果记的是到自己重心的距离，就没有统一的补偿量，转移算不动。所以总距离往上级转移容易，往下级转移难。

    struct Node {

    bool NodeCmp(Node x, Node y) {

                    break;

    void getList(int u, int f, int sum, int rt) {
        curArr[++cntc] = { age[u], sum };
        if (centfa[rt] > 0) {
            faArr[++cntf] = { age[u], getDist(u, centfa[rt]) };
        }
        for (int e = head[u]; e; e = nxt[e]) {
            int v = to[e];
            int w = weight[e];
            if (v != f && !vis[v]) {
                getList(v, u, sum + w, rt);
            }
        }
    }

    void centroidTree(int u, int f) {
        centfa[u] = f;
        vis[u] = true;
        curl[u] = cntc + 1;
        fal[u] = cntf + 1;
        getList(u, 0, 0, u);
        curr[u] = cntc;
        far[u] = cntf;
        for (int e = head[u]; e; e = nxt[e]) {
            int v = to[e];
            if (!vis[v]) {
                centroidTree(getCentroid(v, u), u);
            }
        }
    }

    int kth(Node arr[], int l, int r, int x) {
        int ans = r + 1;
        while (l <= r) {
            int mid = (l + r) >> 1;
            if (arr[mid].age >= x) {
                ans = mid;
                r = mid - 1;
            } else {
                l = mid + 1;
            }
        }
        return ans;
    }

    long long nodeCnt, pathSum;

    void query(Node arr[], int l, int r, int agel, int ager) {
        nodeCnt = pathSum = 0;
        if (l <= r) {
            int a = kth(arr, l, r, agel);
            int b = kth(arr, l, r, ager + 1) - 1;
            if (a <= b) {
                nodeCnt = b - a + 1;
                pathSum = arr[b].sum - (a == l ? 0 : arr[a - 1].sum);
            }
        }
    }

    long long compute(int u, int agel, int ager) {
        query(curArr, curl[u], curr[u], agel, ager);
        long long ans = pathSum;
        long long cnt1, sum1, cnt2, sum2;
        for (int cur = u, f = centfa[cur]; f > 0; cur = f, f = centfa[cur]) {
            query(curArr, curl[f], curr[f], agel, ager);
            cnt1 = nodeCnt;
            sum1 = pathSum;
            query(faArr, fal[cur], far[cur], agel, ager);
            cnt2 = nodeCnt;
            sum2 = pathSum;
            ans += sum1;
            ans -= sum2;
            ans += (cnt1 - cnt2) * getDist(u, f);
        }
        return ans;
    }

        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m >> A;
        for (int i = 1; i <= n; i++) {
            cin >> age[i];
        }
        for (int i = 1, u, v, w; i < n; i++) {
            cin >> u >> v >> w;
            addEdge(u, v, w);
            addEdge(v, u, w);
        }
        dfs1(1, 0, 0);
        dfs2(1, 1);
        centroidTree(getCentroid(1, 0), 0);
        for (int i = 1; i <= n; i++) {
            sort(curArr + curl[i], curArr + curr[i] + 1, NodeCmp);
            for (int j = curl[i] + 1; j <= curr[i]; j++) {
                curArr[j].sum += curArr[j - 1].sum;
            }
            sort(faArr + fal[i], faArr + far[i] + 1, NodeCmp);
            for (int j = fal[i] + 1; j <= far[i]; j++) {
                faArr[j].sum += faArr[j - 1].sum;
            }
        }
        long long lastAns = 0;
        for (int i = 1, u, l, r; i <= m; i++) {
            cin >> u >> l >> r;
            l = (int)((lastAns + l) % A);
            r = (int)((lastAns + r) % A);
            if (l > r) {
                swap(l, r);
            }
            lastAns = compute(u, l, r);
            cout << lastAns << '\n';
        }
        return 0;
    }

三、题目6：成都七中

n 个点的树，每个点有颜色。m 条查询 l r x：只保留编号在 [l, r] 的点，打印 x 所在连通块的颜色数量，查询互不影响。所有数据到 10⁵。测试链接 P5311。

只保留 [l, r] 时，y 和 x 连通当且仅当 x 到 y 的简单路径上所有编号都在 [l, r] 里。这个条件没法预先对每对点算，但点分治能把「经过某个重心」的路径归到一起。

这题不建点分树，只借分治框架做离线。把查询挂在 x 上。每一层取重心 c，从 c 出发 dfs 整个块，走到点 u 时顺手记下路径上的编号最小值 nl 和最大值 nr。dfs 走到 x 且 [nl, nr] 完全落在 [l, r] 里时，这个查询就被这一层捕获。

捕获层统计什么：块内每个点贡献一个 (nl, nr, 颜色)，颜色数就是「编号范围完全落在 [l, r] 内的点的颜色种类」。做法是把点和捕获的查询都按 nr 排序，扫描线推进：BIT 里每种颜色只记它目前最右的出现位置（nl 最大的那次），位置上加一。回答查询时数位置 ≥ l 的颜色数。节点范围 [nl, nr] ⊆ [l, r] 恰好拆成 nr ≤ r（扫描顺序保证）和 nl ≥ l（BIT 查询），不多不少。

为什么取所有层的最大值是对的。同一层统计的点 y 满足「c 到 y」和「c 到 x」的路径都保留，这两段在分岔点拼起来就是 x 到 y 的路径，也保留，所以每层数出的颜色都属于 x 的连通块，不会多。反过来，连通块里任何 y，取 x 和 y 在分治意义上的会合层（相当于它们在点分树上的 LCA 那层），那层一定捕获查询，y 的范围也完全落在界内，所以最高的捕获层把整个连通块数全，更深的层数的是子集，max 不受影响。

        }
    }

    int getCentroid(int u, int fa) {
        getSize(u, fa);
        int half = siz[u] >> 1;
        bool find = false;
        while (!find) {
            find = true;
            for (int e = headg[u]; e > 0; e = nxtg[e]) {
                int v = tog[e];
                if (v != fa && !vis[v] && siz[v] > half) {
                    fa = u;
                    u = v;
                    find = false;
                    break;
                }
            }
        }
        return u;
    }

    void dfs(int u, int fa, int nl, int nr) {
        nodeArr[++cntNode] = { nl, nr, color[u] };
        for (int e = headq[u]; e > 0; e = nxtq[e]) {
            if (ql[e] <= nl && nr <= qr[e]) {
                quesArr[++cntQues] = { ql[e], qr[e], qid[e] };
            }
        }
        for (int e = headg[u]; e > 0; e = nxtg[e]) {
            int v = tog[e];
            if (v != fa && !vis[v]) {
                dfs(v, u, min(v, nl), max(v, nr));
            }
        }
    }

    void calc(int u) {
        cntNode = 0;
        cntQues = 0;
        dfs(u, 0, u, u);
        sort(nodeArr + 1, nodeArr + cntNode + 1, NodeCmp);
        sort(quesArr + 1, quesArr + cntQues + 1, QuesCmp);
        for (int i = 1, j = 1; i <= cntQues; i++) {
            while (j <= cntNode && nodeArr[j].r <= quesArr[i].r) {
                if (nodeArr[j].l > pos[nodeArr[j].color]) {
                    add(pos[nodeArr[j].color], -1);
                    pos[nodeArr[j].color] = nodeArr[j].l;
                    add(pos[nodeArr[j].color], 1);
                }
                j++;
            }
            ans[quesArr[i].qid] = max(ans[quesArr[i].qid], query(quesArr[i].l, n));
        }
        for (int i = 1; i <= cntNode; i++) {
            add(pos[nodeArr[i].color], -1);
            pos[nodeArr[i].color] = 0;
        }
    }

    void solve(int u) {
        vis[u] = true;
        calc(u);
        for (int e = headg[u]; e > 0; e = nxtg[e]) {
            int v = tog[e];
            if (!vis[v]) {
                solve(getCentroid(v, u));
            }
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            cin >> color[i];
        }
        for (int i = 1, u, v; i < n; i++) {
            cin >> u >> v;
            addEdge(u, v);
            addEdge(v, u);
        }
        for (int i = 1, l, r, x; i <= m; i++) {
            cin >> l >> r >> x;
            addQuery(x, l, r, i);
        }
        solve(getCentroid(1, 0));
        for (int i = 1; i <= m; i++) {
            cout << ans[i] << '\n';
        }
        return 0;
    }

四、题目7：捉迷藏

n 个点的树，边权都是 1，初始每个点都是黑色。m 条操作：C x 把 x 的颜色取反（黑变白、白变黑）；G 打印最远的两个黑点的距离，只有一个黑点打印 0，没有黑点打印 −1。n 到 10⁵，m 到 5×10⁵。测试链接 P2056。

这是树上开关类问题：点有开关状态，查询要最大距离。可用的解法有点分树、线段树分治、树链剖分、LCT 等，点分树版本单次查询 O(log² n) 以内。

点分树的思路是把「最远黑点对」拆到会合层。点对 (a, b) 的距离在它们在点分树上的会合重心 c 处拆成 dist(a, c) + dist(b, c)。在每个重心 c 处只维护三样东西。

distFa[u] 是 u 所在块里所有黑点到「u 的上级重心」的距离集合，用对顶堆（一个加堆一个删堆）实现可删除。sonMax[c] 是 c 的每个儿子 u 贡献的 distFa[u] 的最大值，另外黑点 c 自己贡献一个 0。maxDist 收集所有点 sonMax 的前两名之和。

这样维护的道理：sonMax[c] 的前两名来自 c 的两个不同方向（或者一个是 c 自己的 0），拼起来恰好是一条经过 c 的真实路径，长度不虚报；反过来最远点对在会合层一定被这样配到。查询 G 时黑点数不大于 1 做特判，否则答案就是 maxDist 堆顶。切换 x 的颜色时，沿 x 的祖先链逐级更新 distFa、sonMax 和 maxDist，每级堆操作 O(log n)。

捉迷藏：切换颜色，看最远的两个黑点

点任何圆圈切换黑白。黑点的最远点对用橙色边连出来，下面的数字就是 G 的答案：黑点只剩一个时是 0，全白时是 −1。

完整代码：

#include <bits/stdc++.h>

    using namespace std;

    struct Set {
        priority_queue<int> addHeap;
        priority_queue<int> delHeap;

        void clean() {
            while (!delHeap.empty() && !addHeap.empty() && delHeap.top() == addHeap.top()) {
                addHeap.pop();
                delHeap.pop();
            }
        }

        int popHead() {
            int ans = addHeap.top();
            addHeap.pop();
            clean();
            return ans;
        }

        int size() {
            return (int)addHeap.size() - (int)delHeap.size();
        }

        void add(int v) {
            addHeap.push(v);
        }

        void del(int v) {
            delHeap.push(v);
            clean();
        }

        int first() {
            return addHeap.top();
        }

        int second() {
            int a = popHead();
            int b = first();
            add(a);
            return b;
        }
    };

    const int MAXN = 100001;
    int n, m;
    bool black[MAXN];

    int head[MAXN];
    int nxt[MAXN << 1];
    int to[MAXN << 1];
    int cntg;

    int fa[MAXN];
    int dep[MAXN];
    int siz[MAXN];
    int son[MAXN];
    int top[MAXN];

    bool vis[MAXN];
    int centfa[MAXN];

    Set distFa[MAXN];
    Set sonMax[MAXN];
    Set maxDist;

    void addEdge(int u, int v) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        head[u] = cntg;
    }

    void dfs1(int u, int f) {
        fa[u] = f;
        dep[u] = dep[f] + 1;
        siz[u] = 1;
        for (int e = head[u], v; e; e = nxt[e]) {
            v = to[e];
            if (v != f) {
                dfs1(v, u);
            }
        }
        for (int e = head[u], v; e; e = nxt[e]) {
            v = to[e];
            if (v != f) {
                siz[u] += siz[v];
                if (son[u] == 0 || siz[son[u]] < siz[v]) {
                    son[u] = v;
                }
            }
        }
    }

    void dfs2(int u, int t) {
        top[u] = t;
        if (son[u] == 0) {
            return;
        }
        dfs2(son[u], t);
        for (int e = head[u], v; e; e = nxt[e]) {
            v = to[e];
            if (v != fa[u] && v != son[u]) {
                dfs2(v, v);
            }
        }
    }

    void getSize(int u, int f) {
        siz[u] = 1;
        for (int e = head[u]; e; e = nxt[e]) {
            int v = to[e];
            if (v != f && !vis[v]) {
                getSize(v, u);
                siz[u] += siz[v];
            }
        }
    }

    int getLca(int a, int b) {
        while (top[a] != top[b]) {
            if (dep[top[a]] <= dep[top[b]]) {
                b = fa[top[b]];
            } else {
                a = fa[top[a]];
            }
        }
        return dep[a] <= dep[b] ? a : b;
    }

    int getDist(int x, int y) {
        return dep[x] + dep[y] - (dep[getLca(x, y)] << 1);
    }

    int getCentroid(int u, int f) {
        getSize(u, f);
        int half = siz[u] >> 1;
        bool find = false;
        while (!find) {
            find = true;
            for (int e = head[u]; e; e = nxt[e]) {
                int v = to[e];
                if (v != f && !vis[v] && siz[v] > half) {
                    f = u;
                    u = v;
                    find = false;
                    break;
                }
            }
        }
        return u;
    }

    void centroidTree(int u, int f) {
        centfa[u] = f;
        vis[u] = true;
        for (int e = head[u]; e; e = nxt[e]) {
            int v = to[e];
            if (!vis[v]) {
                centroidTree(getCentroid(v, u), u);
            }
        }
    }

    void addAns(int x) {
        if (sonMax[x].size() >= 2) {
            maxDist.add(sonMax[x].first() + sonMax[x].second());
        }
    }

    void delAns(int x) {
        if (sonMax[x].size() >= 2) {
            maxDist.del(sonMax[x].first() + sonMax[x].second());
        }
    }

    void on(int x) {
        delAns(x);
        sonMax[x].del(0);
        addAns(x);
        for (int u = x, f = centfa[u]; f > 0; u = f, f = centfa[u]) {
            delAns(f);
            sonMax[f].del(distFa[u].first());
            distFa[u].del(getDist(x, f));
            if (distFa[u].size() > 0) {
                sonMax[f].add(distFa[u].first());
            }
            addAns(f);
        }
    }

    void off(int x) {
        delAns(x);
        sonMax[x].add(0);
        addAns(x);
        for (int u = x, f = centfa[u]; f > 0; u = f, f = centfa[u]) {
            delAns(f);
            if (distFa[u].size() > 0) {
                sonMax[f].del(distFa[u].first());
            }
            distFa[u].add(getDist(x, f));
            sonMax[f].add(distFa[u].first());
            addAns(f);
        }
    }

    void prepare() {
        for (int i = 1; i <= n; i++) {
            black[i] = true;
        }
        for (int i = 1; i <= n; i++) {
            for (int u = i, f = centfa[u]; f > 0; u = f, f = centfa[u]) {
                distFa[u].add(getDist(i, f));
            }
        }
        for (int i = 1; i <= n; i++) {
            sonMax[i].add(0);
            if (centfa[i] > 0) {
                sonMax[centfa[i]].add(distFa[i].first());
            }
        }
        for (int i = 1; i <= n; i++) {
            addAns(i);
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n;
        for (int i = 1, u, v; i < n; i++) {
            cin >> u >> v;
            addEdge(u, v);
            addEdge(v, u);
        }
        dfs1(1, 0);
        dfs2(1, 1);
        centroidTree(getCentroid(1, 0), 0);
        prepare();
        cin >> m;
        int blackCnt = n;
        char op;
        for (int i = 1, x; i <= m; i++) {
            cin >> op;
            if (op == 'C') {
                cin >> x;
                black[x] = !black[x];
                if (black[x]) {
                    off(x);
                    blackCnt++;
                } else {
                    on(x);
                    blackCnt--;
                }
            } else {
                if (blackCnt <= 1) {
                    cout << (blackCnt - 1) << '\n';
                } else {
                    cout << maxDist.first() << '\n';
                }
            }
        }
        return 0;
    }

这题还有一个单次查询 O(log n) 的最优解：写出树的括号序，每个点贡献左括号、编号、右括号，黑点位置记 0、白点位置记负无穷，用线段树维护六个量（左端进入的最大延伸、右端进入的最大延伸、净括号数等），pushup 时左右两段的量按拼接公式合并。修改是单点更新，查询直接读根的「区间内最长合法配对距离」，一次 O(log n) 都不用，读根即可。

最优解代码：

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 100001;
    const int MAXT = MAXN * 12;
    const int INF = 1000000001;
    const int PAR = -1;
    const int PAL = -2;
    int n, m;
    bool black[MAXN];

    int head[MAXN];
    int nxt[MAXN << 1];
    int to[MAXN << 1];
    int cntg;

    int dfn[MAXN];
    int seg[MAXN * 3];
    int cntd;

    int pr[MAXT];
    int pl[MAXT];
    int ladd[MAXT];
    int lminus[MAXT];
    int radd[MAXT];
    int rminus[MAXT];
    int dist[MAXT];

    void addEdge(int u, int v) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        head[u] = cntg;
    }

    void dfs(int u, int fa) {
        seg[++cntd] = PAL;
        seg[++cntd] = u;
        dfn[u] = cntd;
        for (int e = head[u]; e > 0; e = nxt[e]) {
            int v = to[e];
            if (v != fa) {
                dfs(v, u);
            }
        }
        seg[++cntd] = PAR;
    }

    void up(int i) {
        int pre = i << 1;
        int post = i << 1 | 1;
        if (pl[pre] > pr[post]) {
            pr[i] = pr[pre];
            pl[i] = pl[pre] - pr[post] + pl[post];
        } else {
            pr[i] = pr[pre] + pr[post] - pl[pre];
            pl[i] = pl[post];
        }
        ladd[i] = max(ladd[pre], max(pr[pre] + ladd[post] - pl[pre], pr[pre] + pl[pre] + lminus[post]));
        lminus[i] = max(lminus[pre], pl[pre] - pr[pre] + lminus[post]);
        radd[i] = max(radd[post], max(radd[pre] - pr[post] + pl[post], rminus[pre] + pr[post] + pl[post]));
        rminus[i] = max(rminus[post], rminus[pre] + pr[post] - pl[post]);
        dist[i] = max(max(dist[pre], dist[post]), max(radd[pre] + lminus[post], rminus[pre] + ladd[post]));
    }

    void point(int i, int v) {
        pr[i] = pl[i] = 0;
        ladd[i] = lminus[i] = radd[i] = rminus[i] = dist[i] = -INF;
        if (v == PAR) {
            pr[i] = 1;
        } else if (v == PAL) {
            pl[i] = 1;
        } else if (black[v]) {
            ladd[i] = lminus[i] = radd[i] = rminus[i] = dist[i] = 0;
        }
    }

    void build(int l, int r, int i) {
        if (l == r) {
            point(i, seg[l]);
        } else {
            int mid = (l + r) >> 1;
            build(l, mid, i << 1);
            build(mid + 1, r, i << 1 | 1);
            up(i);
        }
    }

    void update(int jobi, int l, int r, int i) {
        if (l == r) {
            point(i, seg[l]);
        } else {
            int mid = (l + r) >> 1;
            if (jobi <= mid) {
                update(jobi, l, mid, i << 1);
            } else {
                update(jobi, mid + 1, r, i << 1 | 1);
            }
            up(i);
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n;
        for (int i = 1; i <= n; i++) {
            black[i] = true;
        }
        for (int i = 1, u, v; i < n; i++) {
            cin >> u >> v;
            addEdge(u, v);
            addEdge(v, u);
        }
        dfs(1, 0);
        build(1, cntd, 1);
        cin >> m;
        int blackCnt = n;
        char op;
        for (int i = 1, x; i <= m; i++) {
            cin >> op;
            if (op == 'C') {
                cin >> x;
                black[x] = !black[x];
                if (black[x]) {
                    blackCnt++;
                } else {
                    blackCnt--;
                }
                update(dfn[x], 1, cntd, 1);
            } else {
                if (blackCnt <= 1) {
                    cout << (blackCnt - 1) << '\n';
                } else {
                    cout << dist[1] << '\n';
                }
            }
        }
        return 0;
    }

五、复杂度分析

四道题的预处理都建立在点分树 O(n log n) 的骨架上，差别在查询侧的移动、二分和堆操作上。

操作时间复杂度空间复杂度
幻想乡：点权修改 / 单点花费查询O(log n) / O(log² n)O(n)
幻想乡：找最佳指挥点（移动 O(log n) 跳，每跳比较邻居）O(邻居数 × log² n) 每跳—
开店：预处理（收集、排序、前缀和）O(n log² n)O(n log n)（两个列表）
开店：单次查询O(log² n)（链上每级四次二分）—
成都七中：整体（分治 + 排序 + 扫描线）O((n + m) log² n)O(n + m)
捉迷藏点分树：切换颜色 / 查询O(log² n) / O(1)O(n log n)（堆总元素）
捉迷藏最优解：切换颜色 / 查询O(log n) / O(1)O(n)

幻想乡的移动每一跳到下一个分治层的重心，连通块至少减半，跳 O(log n) 次；每跳的花费比较要对当前点所有邻居做一次 O(log² n) 的链上计算，一般数据下总量近似 O(log³ n)。开店的两个列表每个点在每层各出现一次，总元素 O(n log n)。捉迷藏的 distFa 堆同样每条父子边一个集合、总元素 O(n log n)，对顶堆把删除摊成 O(log n)。

六、最后

点分树这一页收在四道形状各异的题上。它们的共同底座还是那个观察：祖先重心只有 O(log n) 个，任何「以某个点为中心」的统计都能拆成很少的几份，每份用一个适合的数据结构管住。统计要距离和就上桶，要按年龄筛就排序加前缀和，要最大值就堆，要全局最优就顺着性质移动。
