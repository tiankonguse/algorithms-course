扫描线：线段树维护覆盖

一次扫描动的是一整段，不再是一个点。要问这一整段里被覆盖的总长度，堆答不了，线段树正好：这个量由左右两段相加得到，合得起来。

零、背景

上一门的扫描线配的是堆，因为那里每个边界造成的改变是「一个数进集合、一个数出集合」。矩形问题里换了一幅样子：一条扫描线带的是一整段 y 区间，要么把 [y1, y2) 整段盖住，要么整段取消。

要维护的量也随之变成「当前被覆盖的总长度」。这个量和区间与、区间和一样，由左右两段相加得到，能合并，所以线段树能维护它。面积并和周长并两道题，用的是同一棵线段树、同一套扫描线。

一、矩形面积并

题意：n 个矩形，第 i 个是 (x1, y1) 到 (x2, y2)，求它们并集的面积。n 到 10⁵，坐标 0 到 10⁹，答案在 64 位整数里。

扫 x 轴。每个矩形拆成两条边：左边缘 (x1, y1, y2, +1)，右边缘 (x2, y1, y2, -1)，意思是「从这条边开始，[y1, y2) 这段多盖一层」或者「少盖一层」。全部 2n 条边按 x 排序。

相邻两条边之间，覆盖情况不变，这段时间贡献的面积就是「当前覆盖长度 × 两条边的横向距离」。所以整道题只需要维护一个数：y 轴上当前被覆盖的总长度。

二、线段树上挂三个量

先把 y 坐标离散化：收集全部 y1 和 y2，排序去重，相邻两个坐标之间是一段。设去重后有 m 个坐标，线段树就建在这 m 减 1 段上。

每个节点挂三个量。length 是这个节点所辖范围的总长度，建树时定死，之后不动；times 是这个节点被整段覆盖了几层；cover 是这个节点所辖范围里，被覆盖的长度合计。查询永远只问根，cover[1] 就是当前覆盖总长度。

合并规则只有三条：times 大于 0，说明这段被整段盖住，cover 等于 length；times 是 0 且这是叶子，这段什么都没盖，cover 是 0；否则 cover 是左右儿子的 cover 相加。

这里有个省事的地方：不需要懒更新标记。查询永远只问根，而每次加一层后续一定有等量的减一层把它抵消，任务根本不用往叶子传，up 一路上推就能保证根是对的。

看这张图。三条线处理完之后，五段叶子的 times 都是 1，每段 cover 等于 length，一路合并到根，cover[1] 是 6，y 轴上 0 到 6 全盖住。注意中间那些 times 是 0 的节点：它们没被任何一条线完整盖住，cover 是两个儿子加起来的结果。

#include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 200001;

    int ysort[MAXN];
    // 线段树某范围总长度
    int length[MAXN << 2];
    // 线段树某范围覆盖长度
    int cover[MAXN << 2];
    // 线段树某范围覆盖次数
    int times[MAXN << 2];

    int prepare(int n) {
        sort(ysort + 1, ysort + n + 1);
        int m = 1;
        for (int i = 2; i <= n; i++) {
            if (ysort[m] != ysort[i]) {
                ysort[++m] = ysort[i];
            }
        }
        ysort[m + 1] = ysort[m];
        return m;
    }

    int rankOf(int m, int num) {
        int ans = 0, l = 1, r = m, mid;
        while (l <= r) {
            mid = (l + r) >> 1;
            if (ysort[mid] >= num) {
                ans = mid;
                r = mid - 1;
            } else {
                l = mid + 1;
            }
        }
        return ans;
    }

    void build(int l, int r, int i) {
        if (l < r) {
            int mid = (l + r) >> 1;
            build(l, mid, i << 1);
            build(mid + 1, r, i << 1 | 1);
        }
        length[i] = ysort[r + 1] - ysort[l];
        times[i] = 0;
        cover[i] = 0;
    }

    void up(int l, int r, int i) {
        if (times[i] > 0) {
            cover[i] = length[i];
        } else if (l == r) {
            cover[i] = 0;
        } else {
            cover[i] = cover[i << 1] + cover[i << 1 | 1];
        }
    }

    void add(int jobl, int jobr, int jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            times[i] += jobv;
        } else {
            int mid = (l + r) >> 1;
            if (jobl <= mid) {
                add(jobl, jobr, jobv, l, mid, i << 1);
            }
            if (jobr > mid) {
                add(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
            }
        }
        up(l, r, i);
    }

    long long compute(int n, vector<vector<int>>& rec) {
        // 0 : x 1 : y1 2 : y2 3 : +1 / -1
        vector<vector<int>> line(2 * n + 1, vector<int>(4));
        for (int i = 1, j = 1 + n; i <= n; i++, j++) {
            int x1 = rec[i - 1][0], y1 = rec[i - 1][1], x2 = rec[i - 1][2], y2 = rec[i - 1][3];
            ysort[i] = y1;
            ysort[j] = y2;
            line[i] = {x1, y1, y2, 1};
            line[j] = {x2, y1, y2, -1};
        }
        n <<= 1;
        int m = prepare(n);
        build(1, m, 1);
        sort(line.begin() + 1, line.begin() + n + 1, [](const vector<int>& a, const vector<int>& b) {
            return a[0] < b[0];
        });
        long long ans = 0;
        for (int i = 1, pre = 0; i <= n; i++) {
            ans += (long long)cover[1] * (line[i][0] - pre);
            pre = line[i][0];
            add(rankOf(m, line[i][1]), rankOf(m, line[i][2]) - 1, line[i][3], 1, m, 1);
        }
        return ans;
    }

这段代码里有三处容易看漏。build 里 length[i] 用的是 ysort[r + 1] - ysort[l]，是真实坐标差，不是段数；离散化时往 ysort 末尾多写了一个 ysort[m + 1] = ysort[m]，是为了让最右边那段取到右端点。up 里叶子的特判不能省，叶子没有儿子，不能去读。

三、矩形周长并

题意：还是 n 个矩形，求并集轮廓的周长。周长分两半，垂直边和水平边。

垂直边按 x 扫，流程和面积并一样，边还是 (x, y1, y2, ±1)。区别在统计方式：每处理一条边，cover[1] 会从处理前的值变成处理后的值，这个变化量的绝对值，就是这条边新增或消失的覆盖长度。把所有边上的变化量累加起来，就是垂直边的总长。

举个例子：覆盖长度从 4 涨到 6，说明 y 轴上多出一段长 2 的覆盖，那一定对应着某条垂直边露出了 2 的长度；从 6 掉到 4 同理。一条边盖住一小段已经盖着的地方，cover 不变，这段边藏在并集内部，不该算。

水平边把坐标换个名字再扫一遍：边的形式变成 (y, x1, x2, ±1)，离散化收 x 坐标，其余一模一样。两次扫描的结果相加就是周长。

三个矩形拼成一个缺角的方形，轮廓一眼数得清：垂直边共 16（左侧 8、右下 4、右上侧的内缘 4），水平边共 16，周长 32。两次扫描各算一半，谁也别替谁。

#include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 200001;

    int vsort[MAXN];
    // 线段树某范围总长度
    int length[MAXN << 2];
    // 线段树某范围覆盖长度
    int cover[MAXN << 2];
    // 线段树某范围覆盖次数
    int times[MAXN << 2];
    // 0 : 扫描线位置 1 : 区间左端 2 : 区间右端 3 : +1 / -1
    vector<vector<int>> line;

    int prepare(int n) {
        sort(vsort + 1, vsort + n + 1);
        int m = 1;
        for (int i = 2; i <= n; i++) {
            if (vsort[m] != vsort[i]) {
                vsort[++m] = vsort[i];
            }
        }
        vsort[m + 1] = vsort[m];
        return m;
    }

    int rankOf(int m, int num) {
        int ans = 0, l = 1, r = m, mid;
        while (l <= r) {
            mid = (l + r) >> 1;
            if (vsort[mid] >= num) {
                ans = mid;
                r = mid - 1;
            } else {
                l = mid + 1;
            }
        }
        return ans;
    }

    void build(int l, int r, int i) {
        if (l < r) {
            int mid = (l + r) >> 1;
            build(l, mid, i << 1);
            build(mid + 1, r, i << 1 | 1);
        }
        length[i] = vsort[r + 1] - vsort[l];
        times[i] = 0;
        cover[i] = 0;
    }

    void up(int l, int r, int i) {
        if (times[i] > 0) {
            cover[i] = length[i];
        } else if (l == r) {
            cover[i] = 0;
        } else {
            cover[i] = cover[i << 1] + cover[i << 1 | 1];
        }
    }

    void add(int jobl, int jobr, int jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            times[i] += jobv;
        } else {
            int mid = (l + r) >> 1;
            if (jobl <= mid) {
                add(jobl, jobr, jobv, l, mid, i << 1);
            }
            if (jobr > mid) {
                add(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
            }
        }
        up(l, r, i);
    }

    long long scan(int n) {
        int m = prepare(n);
        build(1, m, 1);
        // 同一个位置有多条扫描线时，先处理+1再处理-1，投影长度才不会频繁变化
        sort(line.begin() + 1, line.begin() + n + 1, [](const vector<int>& a, const vector<int>& b) {
            return a[0] != b[0] ? a[0] < b[0] : a[3] > b[3];
        });
        long long ans = 0;
        for (int i = 1; i <= n; i++) {
            int pre = cover[1];
            add(rankOf(m, line[i][1]), rankOf(m, line[i][2]) - 1, line[i][3], 1, m, 1);
            ans += abs(cover[1] - pre);
        }
        return ans;
    }

    long long scanY(int n, vector<vector<int>>& rec) {
        line.assign(2 * n + 1, vector<int>(4));
        for (int i = 1, j = 1 + n; i <= n; i++, j++) {
            int x1 = rec[i - 1][0], y1 = rec[i - 1][1], x2 = rec[i - 1][2], y2 = rec[i - 1][3];
            vsort[i] = y1;
            vsort[j] = y2;
            line[i] = {x1, y1, y2, 1};
            line[j] = {x2, y1, y2, -1};
        }
        return scan(n << 1);
    }

    long long scanX(int n, vector<vector<int>>& rec) {
        line.assign(2 * n + 1, vector<int>(4));
        for (int i = 1, j = 1 + n; i <= n; i++, j++) {
            int x1 = rec[i - 1][0], y1 = rec[i - 1][1], x2 = rec[i - 1][2], y2 = rec[i - 1][3];
            vsort[i] = x1;
            vsort[j] = x2;
            line[i] = {y1, x1, x2, 1};
            line[j] = {y2, x1, x2, -1};
        }
        return scan(n << 1);
    }

    long long compute(int n, vector<vector<int>>& rec) {
        return scanY(n, rec) + scanX(n, rec);
    }

排序那里有个坑。同一个 x 上可能同时压着 +1 和 -1 的边，比如两个矩形共享一条竖边。必须先处理 +1 再处理 -1：先让新矩形盖上去，覆盖长度不变；要是先减，覆盖长度先掉一个坑再涨回来，变化量白算一倍。测试数据不一定安排这种情况，但排序里写上只要几个字符。

演示：矩形面积并的扫描过程

点「下一步」逐条处理扫描线，看每条的 ±1 怎么改变 y 轴各段的覆盖层数、cover[1] 怎么变、累计面积怎么涨。三组矩形可切换。

四、复杂度分析

项面积并周长并
离散化O(n log n)O(n log n) × 2
扫描过程O(n log n)O(n log n)
全程O(n log n)O(n log n)
空间O(n)O(n)

2n 条边，每条边在线段树上走一遍是 O(log n)，没有懒标记的下发，常数很小。空间上四个数组按 2n 开，面积并按 2 × 10⁵ 开，约几 MB。

实测 n 是 10⁵、坐标 10⁹：面积并 0.134 秒，周长并 0.176 秒（本机 clang -O2）。

五、总结

扫描线配线段树的骨架：矩形拆边、坐标离散化、边排序、逐条 add，问根上的 cover[1]。面积并问的是「两条边之间覆盖长度是多少」，周长并问的是「一条边让覆盖长度变了多少」。

不需要懒标记是这两道题的特例，来源是「只问根」加「加减成对」；线段树本身并没有因此变强。

最后

一是 length 用真实坐标差，格子数不等于长度，最后一段要靠 ysort[m + 1] 兜住。

二是 up 的叶子特判、times 大于 0 直接给 length，这两句是合并规则的全部，写反一处整棵树的 cover 都是错的。

三是周长并记得扫两遍，x 管垂直边、y 管水平边；同位置的边先加后减。
