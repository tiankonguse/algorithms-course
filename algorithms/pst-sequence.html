可持久化线段树：版本里记位置信息

上一讲的可持久化线段树，版本里记的是「每个值域上有几个数」。这一讲的版本里换着记别的东西：第一次出现的位置、最后出现的位置。记的信息不同，能回答的问题就完全不同。

一、从右往左建版本：每种数第一次出现的位置

题目要求：给定 arr[1..n]，每次查询 [l, r]，把区间里每种数第一次出现的位置按从小到大排成一个序列，打印第 ceil(s/2) 个位置，s 是区间里不同数的个数。m、n ≤ 2e5，还带强制在线。

直接做是每个查询从左往右扫一遍，O(nm) 直接超时。换一个建版本的顺序：从右往左。root[i] 表示处理完 arr[i..n] 之后，「每种数第一次出现（也就是最靠左那次）的位置集合」。从 i+1 推到 i 时：

1）arr[i] 之前没出现过，直接把位置 i 标记上；
2）arr[i] 出现过（更靠右的位置 j 还标着），现在 i 更靠左，它才是「第一次出现」，所以取消位置 j、标记位置 i。

用一个 pos 数组记每个值当前标在哪个位置，root[i] 从 root[i+1] 继承，最多做两次单点修改就得到新版本。注意线段树建在位置 1..n 上，不是值域上：每个叶子是一个位置，值域是 arr 的取值。

查询 [l, r] 时，在 root[l] 上数 [l, r] 范围内的标记数 s。为什么能这么数：root[l] 只含 l..n 里每种数的首次位置，比 l 更左的位置不在版本里，比 r 更右的位置不落在查询范围，所以 [l, r] 内的标记恰好是区间里每种数第一次出现的位置。第 ceil(s/2) 个位置用 queryKth 取。两趟 O(log n)，一趟数个数、一趟取第 k 个。

第一次出现位置的序列（HDU5919）

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 200002;
    const int MAXT = MAXN * 37;
    int cases, n, m;
    int arr[MAXN];
    int pos[MAXN];
    int root[MAXN];
    int ls[MAXT];
    int rs[MAXT];
    int firstSize[MAXT];
    int cnt;

    int build(int l, int r) {
        int rt = ++cnt;
        if (l < r) {
            int mid = (l + r) / 2;
            ls[rt] = build(l, mid);
            rs[rt] = build(mid + 1, r);
        }
        firstSize[rt] = 0;
        return rt;
    }

    int update(int jobi, int jobv, int l, int r, int i) {
        int rt = ++cnt;
        ls[rt] = ls[i];
        rs[rt] = rs[i];
        firstSize[rt] = firstSize[i] + jobv;
        if (l == r) {
            return rt;
        }
        int mid = (l + r) / 2;
        if (jobi <= mid) {
            ls[rt] = update(jobi, jobv, l, mid, ls[rt]);
        } else {
            rs[rt] = update(jobi, jobv, mid + 1, r, rs[rt]);
        }
        return rt;
    }

    int querySize(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return firstSize[i];
        }
        int mid = (l + r) / 2;
        int ans = 0;
        if (jobl <= mid) {
            ans += querySize(jobl, jobr, l, mid, ls[i]);
        }
        if (jobr > mid) {
            ans += querySize(jobl, jobr, mid + 1, r, rs[i]);
        }
        return ans;
    }

    int queryKth(int jobk, int l, int r, int i) {
        if (l == r) {
            return l;
        }
        int mid = (l + r) / 2;
        int lsize = firstSize[ls[i]];
        if (lsize >= jobk) {
            return queryKth(jobk, l, mid, ls[i]);
        } else {
            return queryKth(jobk - lsize, mid + 1, r, rs[i]);
        }
    }

    void prepare() {
        cnt = 0;
        memset(pos, 0, sizeof(pos));
        root[n + 1] = build(1, n);
        for (int i = n; i >= 1; i--) {
            if (pos[arr[i]] == 0) {
                root[i] = update(i, 1, 1, n, root[i + 1]);
            } else {
                root[i] = update(pos[arr[i]], -1, 1, n, root[i + 1]);
                root[i] = update(i, 1, 1, n, root[i]);
            }
            pos[arr[i]] = i;
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> cases;
        for (int t = 1; t <= cases; t++) {
            cin >> n >> m;
            for (int i = 1; i <= n; i++) {
                cin >> arr[i];
            }
            prepare();
            cout << "Case #" << t << ":";
            for (int i = 1, a, b, l, r, k, lastAns = 0; i <= m; i++) {
                cin >> l >> r;
                a = (l + lastAns) % n + 1;
                b = (r + lastAns) % n + 1;
                l = min(a, b);
                r = max(a, b);
                k = (querySize(l, r, 1, n, root[l]) + 1) / 2;
                lastAns = queryKth(k, 1, n, root[l]);
                cout << " " << lastAns;
            }
            cout << "\n";
        }
        return 0;
    }

二、值域树记「最后出现位置的最小值」：最小自然数

题目要求：查询 [l, r] 内没出现过的最小自然数，0 也算自然数。n、m ≤ 2e5。这题版本里记的是每个值「最后一次出现的位置」。

先定值域：[0, n]。数组只有 n 个数，值域却有 n+1 个值，鸽子笼保证 [0, n] 里至少有一个没出现，答案一定在这个范围内。

root[i] 处理完 arr[1..i]：叶子 v 存 v 最后出现的位置，没出现过的值保持 0。节点的 lateLeft 存「子树里所有值最后出现位置的最小值」。查询 [l, r]：值 v 在区间里没出现过，等价于 v 的最后出现位置 < l。于是在 root[r] 上找最小的 v，满足 v 的 last < l：优先下左子树，左子树的 lateLeft ≥ l 说明左半边所有值最后都出现在区间里，答案只能去右子树。

值域从 0 开始，build 也建 [0, n]。arr[i] 超过 n 或小于 0 的值直接忽略：它们不可能成为 [0, n] 里的缺失值，记了反而把 lateLeft 弄乱。

区间内没有出现的最小自然数（P4137）

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 200001;
    const int MAXT = MAXN * 22;
    int n, m;
    int arr[MAXN];
    int root[MAXN];
    int ls[MAXT];
    int rs[MAXT];
    int lateLeft[MAXT];
    int cnt;

    int build(int l, int r) {
        int rt = ++cnt;
        lateLeft[rt] = 0;
        if (l < r) {
            int mid = (l + r) / 2;
            ls[rt] = build(l, mid);
            rs[rt] = build(mid + 1, r);
        }
        return rt;
    }

    int update(int jobi, int jobv, int l, int r, int i) {
        int rt = ++cnt;
        ls[rt] = ls[i];
        rs[rt] = rs[i];
        lateLeft[rt] = lateLeft[i];
        if (l == r) {
            lateLeft[rt] = jobv;
        } else {
            int mid = (l + r) / 2;
            if (jobi <= mid) {
                ls[rt] = update(jobi, jobv, l, mid, ls[rt]);
            } else {
                rs[rt] = update(jobi, jobv, mid + 1, r, rs[rt]);
            }
            lateLeft[rt] = min(lateLeft[ls[rt]], lateLeft[rs[rt]]);
        }
        return rt;
    }

    int query(int pos, int l, int r, int i) {
        if (l == r) {
            return l;
        }
        int mid = (l + r) / 2;
        if (lateLeft[ls[i]] < pos) {
            return query(pos, l, mid, ls[i]);
        } else {
            return query(pos, mid + 1, r, rs[i]);
        }
    }

    void prepare() {
        cnt = 0;
        root[0] = build(0, n);
        for (int i = 1; i <= n; i++) {
            if (arr[i] > n || arr[i] < 0) {
                root[i] = root[i - 1];
            } else {
                root[i] = update(arr[i], i, 0, n, root[i - 1]);
            }
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
        }
        prepare();
        for (int i = 1, l, r; i <= m; i++) {
            cin >> l >> r;
            cout << query(l, 0, n, root[r]) << "\n";
        }
        return 0;
    }

四、总结

这一讲的两个题共用一棵可持久化线段树，差别只在版本里记什么：题1 在位置域上记「首次出现」，从右往左建版本；题2 在值域上记「最后出现位置的最小值」，查询就是顺着 lateLeft 下钻找缺口。共同点是版本差：任何一个区间问题，只要能把信息拆成前缀可累积的形式，就能用两个版本相减回答。树上也有同样的玩法，下一讲把版本差搬到树上。
