扩展中国剩余定理：迭代合并同余方程

模数不互质时，中国剩余定理的构造直接失效：两个条件可能互相矛盾，也可能各自重复。扩展中国剩余定理换一条路，把同余方程一个一个「合并」成新的同余式，最后剩下的那一个就是答案。

零、背景

先复习上节课的一个公式。ax + by = d（d = gcd(a,b)）有一组特解 (x0, y0)，那么 ax + by = c（c 是 d 的倍数）的通解是 x = x0' + (b/d)·n、y = y0' − (a/d)·n，其中 x0' = x0·(c/d)、y0' = y0·(c/d)。通解里 x 每加一步 b/d 就换一个解，所以最小非负特解就是 x0' 对 b/d 取非负余数。这一页全程用这个公式。

回到问题：n 个同余方程 x ≡ ri (mod mi)，mi 不一定互质。两个模数有公因子时，条件可能重叠也可能冲突，中国剩余定理那种「每个 ci 各管各的」的构造站不住。办法是把方程按顺序两两合并，每次把已经满足的条件浓缩成一条新的同余式。

一、迭代合并：维护 ans = lcm·x + tail

一开始没有约束，写成 ans = 1·x + 0，即 lcm = 1、tail = 0。当前来到第 i 个方程 ans ≡ ri (mod mi)，写成 ans = mi·y + ri。两个式子相减：lcm·x + mi·y = ri − tail，这就是 ax + by = c 的样子（a = lcm，b = mi，c = ri − tail），用扩展欧几里得解。

c 对 d = gcd(lcm, mi) 取模不为 0 时方程无解，整个系统无解，返回 -1。有解时先求最小非负特解 x0，把 x = x0 + (b/d)·n 代回 ans = lcm·x + tail，整理出新的通解形式 ans = lcm'·x + tail'，其中 lcm' = lcm·(b/d)，tail' = (lcm·x0 + tail) mod lcm'。这个 (lcm', tail') 就是下一轮的状态。

每合并一个方程，lcm 变大、tail 更新，但形式永远是 ans = lcm·x + tail。全部方程处理完，tail 就是答案。

二、例子：两个方程合并

x ≡ 4 (mod 6)、x ≡ 8 (mod 10)。gcd(6,10) = 2，两个模数不互质。初始 lcm=1、tail=0。第一个方程：c = (4−0) mod 6 = 4，exgcd(1,6) 得 d=1、特解 x=1，最小非负特解 x0 = 4，于是 lcm' = 6、tail' = (1·4+0) mod 6 = 4，合并结果是 ans ≡ 4 (mod 6)。

第二个方程：c = (8−4) mod 10 = 4，exgcd(6,10) 得 d=2、特解 x=2（6·2 + 10·(−1) = 2），c mod d = 0 有解。x0 = (2·(4/2)) mod (10/2) = 4。lcm' = 6·(10/2) = 30，tail' = (4 + 4·6) mod 30 = 28。答案 28，检验 28 mod 6 = 4、28 mod 10 = 8。通解是 28 + 30k。

如果把第二个方程换成 x ≡ 7 (mod 10)：c = (7−4) mod 10 = 3，exgcd(6,10) 得 d=2，3 mod 2 = 1，不为 0，直接返回 -1。x ≡ 4 (mod 6) 给出 x = 4、10、16、22、……，x ≡ 7 (mod 10) 给出 x = 7、17、27、……，两个序列确实没有交集。

三、模板实现

代码就是上面的流程照抄：每次 exgcd 求出 d 和特解 x，c 取模成非负，x0 用龟速乘取最小非负特解，最后用龟速乘更新 tail。每一步乘法都可能越过 long long，不能省。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;
    ll d, x, y, px, py;
    void exgcd(ll a, ll b) {
      if (b == 0) {
        d = a;
        x = 1;
        y = 0;
      } else {
        exgcd(b, a % b);
        px = x;
        py = y;
        x = py;
        y = px - py * (a / b);
      }
    }
    ll multiply(ll a, ll b, ll mod) {
      a = (a % mod + mod) % mod;
      b = (b % mod + mod) % mod;
      ll ans = 0;
      while (b != 0) {
        if (b & 1) ans = (ans + a) % mod;
        a = (a + a) % mod;
        b >>= 1;
      }
      return ans;
    }
    ll excrt(int n, ll m[], ll r[]) {
      ll tail = 0, lcm = 1, tmp, b, c, x0;
      for (int i = 1; i <= n; i++) {
        b = m[i];
        c = ((r[i] - tail) % b + b) % b;
        exgcd(lcm, b);
        if (c % d != 0) return -1;
        x0 = multiply(x, c / d, b / d);
        tmp = lcm * (b / d);
        tail = (tail + multiply(x0, lcm, tmp)) % tmp;
        lcm = tmp;
      }
      return tail;
    }
    int main() {
      int n;
      cin >> n;
      ll m[100005], r[100005];
      for (int i = 1; i <= n; i++) cin >> m[i] >> r[i];
      cout << excrt(n, m, r) << endl;
      return 0;
    }

四、猜数字：负余数先转正

洛谷 P3868 的输入余数可以是负数，mi | (x − ri) 里的 ri 在 [−10^9, 10^9]。同余式两边同时加减模数的整数倍不改变条件，所以每个 ri 先转成非负余数再喂给 excrt。输入顺序注意是先给 r 数组再给 m 数组，与 P1495 的 m、r 顺序相反。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;
    ll d, x, y, px, py;
    void exgcd(ll a, ll b) {
      if (b == 0) {
        d = a;
        x = 1;
        y = 0;
      } else {
        exgcd(b, a % b);
        px = x;
        py = y;
        x = py;
        y = px - py * (a / b);
      }
    }
    ll multiply(ll a, ll b, ll mod) {
      a = (a % mod + mod) % mod;
      b = (b % mod + mod) % mod;
      ll ans = 0;
      while (b != 0) {
        if (b & 1) ans = (ans + a) % mod;
        a = (a + a) % mod;
        b >>= 1;
      }
      return ans;
    }
    ll excrt(int n, ll m[], ll r[]) {
      ll tail = 0, lcm = 1, tmp, b, c, x0;
      for (int i = 1; i <= n; i++) {
        b = m[i];
        c = ((r[i] - tail) % b + b) % b;
        exgcd(lcm, b);
        if (c % d != 0) return -1;
        x0 = multiply(x, c / d, b / d);
        tmp = lcm * (b / d);
        tail = (tail + multiply(x0, lcm, tmp)) % tmp;
        lcm = tmp;
      }
      return tail;
    }
    int main() {
      int n;
      cin >> n;
      ll r[15], m[15];
      for (int i = 1; i <= n; i++) cin >> r[i];
      for (int i = 1; i <= n; i++) cin >> m[i];
      for (int i = 1; i <= n; i++) r[i] = (r[i] % m[i] + m[i]) % m[i];
      cout << excrt(n, m, r) << endl;
      return 0;
    }

总结

扩展中国剩余定理把 n 个方程变成 n 次合并：每步解一个 lcm·x + mi·y = ri − tail 的不定方程，c 不被 gcd 整除就无解，有解就更新 lcm' = lcm·(mi/d)、tail' = (lcm·x0 + tail) mod lcm'。形式从头到尾是 ans = lcm·x + tail，处理完最后一个方程 tail 就是最小非负解。互质的模数是它的特例，输入余数为负就先转正，两条边界都在这一个模板里收掉。
