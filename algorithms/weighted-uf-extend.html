带权并查集：分类问题与异或关系

有的关系根本不关心距离，只问「敌对还是同类」，一个 enemy 数组就能表达，连 dist 都不用维护。异或关系更特别：单点赋值得先造一个值为 0 的虚点，才能写进并查集。

零、enemy 数组做二分类（P1525 关押罪犯）

n 个罪犯分到两个监狱，m 对仇恨值，要最大仇恨最小。贪心的方向是确定的：仇恨从大到小处理，每对都要求分到不同监狱，能分就分，分不下（两人已经被迫同狱）时当前仇恨值就是答案，因为前面的更大仇恨全都被安排开了。

实现上并查集合并的是「必须同狱」的人。处理一对 (a, b, w)：如果 find(a) == find(b)，说明前面的事实已经把它们绑进同一个监狱，现在又要求分开放，冲突，输出 w 收工。否则 a 的旧敌人和 b 没有直接仇恨，可以同狱，于是 union(enemy[a], b)；同时对称地 union(enemy[b], a)。enemy 数组记「当前已知的、和 i 有直接仇恨的一个人」，用一个就够，因为旧敌人和新敌人之间会通过 union 串起来。

这里有个细节：仇恨值最大 1e9，但程序不需要按仇恨建任何表，排个序就行。排序用结构体数组按 v 降序，别用裸数组，C++ 的 sort 对裸数组按自定义规则排容易写出歧义。

关押罪犯

// 关押罪犯
    // n 个犯人分配到两个监狱，m 条仇恨记录 l r v
    // 每个监狱的暴力值 = 该监狱内最深的仇恨，冲突值 = 两监狱暴力值最大值
    // 最小化冲突值，返回最小值
    // 贪心：按仇恨值降序，两人必须分到不同监狱（enemy 数组维护），
    // 出现 must 同监狱的两人时，当前仇恨值就是答案
    // 1 <= n <= 2e4，1 <= m <= 1e5，1 <= 仇恨值 <= 1e9
    // 测试链接 : https://www.luogu.com.cn/problem/P1525
    #include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 20002;
    const int MAXM = 100001;

    int n, m;
    int father[MAXN];
    int enemyArr[MAXN];
    struct Edge { int l, r, v; };
    Edge arr[MAXM];

    void prepare() {
        for (int i = 1; i <= n; i++) {
            father[i] = i;
            enemyArr[i] = 0;
        }
    }

    int find(int i) {
        father[i] = father[i] == i ? i : find(father[i]);
        return father[i];
    }

    void unionSet(int l, int r) {
        father[find(l)] = find(r);
    }

    bool same(int l, int r) {
        return find(l) == find(r);
    }

    int compute() {
        sort(arr + 1, arr + m + 1, [](const Edge& a, const Edge& b) {
            return a.v > b.v;
        });
        int ans = 0;
        for (int i = 1; i <= m; i++) {
            int l = arr[i].l, r = arr[i].r, v = arr[i].v;
            if (same(l, r)) {
                ans = v;
                break;
            } else {
                if (enemyArr[l] == 0) {
                    enemyArr[l] = r;
                } else {
                    unionSet(enemyArr[l], r);
                }
                if (enemyArr[r] == 0) {
                    enemyArr[r] = l;
                } else {
                    unionSet(l, enemyArr[r]);
                }
            }
        }
        return ans;
    }

    int main() {
        scanf("%d%d", &n, &m);
        prepare();
        for (int i = 1; i <= m; i++) {
            scanf("%d%d%d", &arr[i].l, &arr[i].r, &arr[i].v);
        }
        printf("%d\n", compute());
        return 0;
    }

一、敌人的敌人是朋友（P1892 团伙）

n 个人、两个黑帮，m 条事实：F a b 是朋友，E a b 是敌人。朋友同帮、敌人异帮、敌人的敌人是朋友、无法推断朋友的自己一帮。这题和关押罪犯的处理一字不差，区别在答案：关押罪犯找第一个冲突，团伙数事实全部可信，直接数集合。

洛谷这题的描述有点毛病，样例里混了 F 和 E 之外的格式，按课件语义做：F 只 union 同帮，E 只做敌人传播。最后数根：i == find(i) 的个数就是帮派数，无法推断朋友的节点自己成一个帮，正好算一个集合。

团伙

// 团伙
    // n 个黑帮成员，m 条事实：F l r 朋友、E l r 敌人
    // 敌人的敌人是朋友，朋友同黑帮，敌人不同黑帮，计算黑帮个数
    // 经典并查集分类处理：E 时互挂 enemy 数组，无法推断的成员自己一个黑帮
    // 1 <= n <= 1000，1 <= m <= 5000
    // 测试链接 : https://www.luogu.com.cn/problem/P1892
    #include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 1001;

    int n, m;
    int father[MAXN];
    int enemyArr[MAXN];

    void prepare() {
        for (int i = 1; i <= n; i++) {
            father[i] = i;
            enemyArr[i] = 0;
        }
    }

    int find(int i) {
        father[i] = father[i] == i ? i : find(father[i]);
        return father[i];
    }

    void unionSet(int l, int r) {
        father[find(l)] = find(r);
    }

    int main() {
        scanf("%d%d", &n, &m);
        prepare();
        char op;
        int l, r;
        for (int i = 1; i <= m; i++) {
            scanf(" %c%d%d", &op, &l, &r);
            if (op == 'F') {
                unionSet(l, r);
            } else {
                if (enemyArr[l] == 0) {
                    enemyArr[l] = r;
                } else {
                    unionSet(enemyArr[l], r);
                }
                if (enemyArr[r] == 0) {
                    enemyArr[r] = l;
                } else {
                    unionSet(l, enemyArr[r]);
                }
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (i == father[i]) {
                ans++;
            }
        }
        printf("%d\n", ans);
        return 0;
    }

二、异或关系（HDU 3234）

n 个非负整数，几类操作：I p v 说 xp = v，I p q v 说 xp ^ xq = v，Q p q 问 xp ^ xq，Q p1 p2 ... pk 问一堆数的异或和。值是 20 bit 内的，所以异或结果最大 2^20 - 1。exclu[i] 存 i 到根 i 的异或，路径压缩的累加换成累异或：exclu[i] ^= exclu[tmp]。

I p q v 就是 union(p, q, v)，exclu[lf] = exclu[r] ^ exclu[l] ^ v。I p v 先造虚点：把 p 和虚点 n 的异或设成 v。冲突判据：find 同根时 exclu[l] ^ exclu[r] != v。一旦冲突，本题剩余所有操作（I 和 Q）都不处理，只在输入结束时输出一条冲突提示。

Q p q：同根就答 exclu[l] ^ exclu[r]，不同根答 I don't know。Q 多个数：先把查询节点按集合分组，含虚点的块所有值确定，把块内每个查询节点的 exclu 值异或进答案；不含虚点的自由块，块内节点值会随自由变量整体翻转，只有自由块在查询里出现奇数次时答案才受影响，这时也答 I don't know。

异或关系

// 异或关系（HDU3234 / UVA12232）
    // n 个数编号 0~n-1，m 次操作：
    //   I x v     声明 第x个数 = v
    //   I x y v   声明 第x个数 ^ 第y个数 = v
    //   Q k a1..ak 查询这些数字异或起来的值
    // 带权并查集维护异或关系 + 虚点 n（值为 0 的基准）
    // 矛盾后不再处理任何操作；只有声明（I）计数，Q 不计数
    // 1 <= n <= 2e4，1 <= m <= 4e4，1 <= k <= 15，多测，n=m=0 结束
    // 测试链接 : https://acm.hdu.edu.cn/showproblem.php?pid=3234
    #include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 20002;
    const int MAXK = 21;

    int n, m;
    bool conflict;
    int cnti;
    int father[MAXN];
    int exclu[MAXN];
    int nums[MAXK];
    int fas[MAXK];

    void prepare() {
        conflict = false;
        cnti = 0;
        for (int i = 0; i <= n; i++) {
            father[i] = i;
            exclu[i] = 0;
        }
    }

    int find(int i) {
        if (i != father[i]) {
            int tmp = father[i];
            father[i] = find(tmp);
            exclu[i] ^= exclu[tmp];
        }
        return father[i];
    }

    bool opi(int l, int r, int v) {
        cnti++;
        int lf = find(l), rf = find(r);
        if (lf == rf) {
            if ((exclu[l] ^ exclu[r]) != v) {
                conflict = true;
                return false;
            }
        } else {
            if (lf == n) {
                lf = rf;
                rf = n;
            }
            father[lf] = rf;
            exclu[lf] = exclu[r] ^ exclu[l] ^ v;
        }
        return true;
    }

    int opq(int k) {
        int ans = 0;
        for (int i = 1; i <= k; i++) {
            int fa = find(nums[i]);
            ans ^= exclu[nums[i]];
            fas[i] = fa;
        }
        sort(fas + 1, fas + k + 1);
        for (int l = 1, r = 1; l <= k; l = ++r) {
            while (r + 1 <= k && fas[r + 1] == fas[l]) {
                r++;
            }
            if ((r - l + 1) % 2 != 0 && fas[l] != n) {
                return -1;
            }
        }
        return ans;
    }

    // I 行的数据解析：一行里可能是 2 个数（x v）或 3 个数（x y v）
    int parseLine(const string& line, int& a, int& b, int& c) {
        stringstream ss(line);
        vector<int> tmp;
        int x;
        while (ss >> x) {
            tmp.push_back(x);
        }
        a = tmp.size() >= 1 ? tmp[0] : 0;
        b = tmp.size() >= 2 ? tmp[1] : 0;
        c = tmp.size() >= 3 ? tmp[2] : 0;
        return (int)tmp.size();
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        int caseNo = 0;
        cin >> n >> m;
        string line;
        while (!(n == 0 && m == 0)) {
            prepare();
            cout << "Case " << ++caseNo << ":" << endl;
            for (int i = 1; i <= m; i++) {
                string op;
                cin >> op;
                if (op == "I") {
                    getline(cin, line);
                    int a, b, c;
                    int sz = parseLine(line, a, b, c);
                    if (!conflict) {
                        int l, r, v;
                        if (sz == 2) {
                            l = a;
                            r = n;
                            v = b;
                        } else {
                            l = a;
                            r = b;
                            v = c;
                        }
                        if (!opi(l, r, v)) {
                            cout << "The first " << cnti << " facts are conflicting." << endl;
                        }
                    }
                } else {
                    int k;
                    cin >> k;
                    for (int j = 1; j <= k; j++) {
                        cin >> nums[j];
                    }
                    if (!conflict) {
                        int ans = opq(k);
                        if (ans == -1) {
                            cout << "I don't know." << endl;
                        } else {
                            cout << ans << endl;
                        }
                    }
                }
            }
            cout << endl;
            cin >> n >> m;
        }
        return 0;
    }

三、总结

三种关系三种形状：二分类用一个 enemy 数组，union(enemy[a], b) 一句就把「敌人的敌人」串起来，关押罪犯找第一个分不下的点，团伙直接数集合；异或关系把 dist 换成 exclu，累加换异或，单点赋值靠虚点 n 变成普通合并，冲突之后整题停摆。

带权并查集的 dist 一路换了四种运算（差值、倍数、%3、异或），公式跟着变，骨架不动。遇到「给定两个元素的关系、再查询任意两个元素的关系」的题，先想能不能压成数轴上的距离。

异或关系合并演示

4 个值 x0..x3 加虚点 x4 = 0。依次 I 2 1 1（x2^x1=1）、I 1 4（x1^x4=4）、I 3 2（x3^x2=2），最后 Q 1 3 得 x1^x3。
