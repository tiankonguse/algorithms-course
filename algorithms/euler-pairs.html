首尾相接的欧拉路径题型

这一页三道题：数对要首尾相接、单词要接龙、字母对要相邻。表面完全不同，其实都是同一件事：每个素材是一条边，要求全用一次，就是欧拉路径。差别只在值域、边权和有向还是无向。

零、背景

识别这类题有一套固定的眼光。素材有「头」和「尾」，拼接规则是前一个的尾等于后一个的头，每个素材恰好用一次。把这话说成图论：素材是边，头尾是点，拼接就是沿边走。

三道题各自的坑。数对的值域到 10⁹，点要先离散化；单词的边权是整个字符串，比较时首字符和单词本身都要参与排序；字母对是无向边，还要在 52 个字符的点集上判连通和奇度。

一、题目3：合法排列数对

给定 n 个数对 (a, b)，a 不等于 b，没有重复数对。希望所有数对形成首尾相接的样子，比如 [5,1] [4,5] [11,9] [9,4] 排成 [11,9] [9,4] [4,5] [5,1]。每个数对用且只用一次，数据保证存在合法排列，返回任何一个。n 到 10⁵，数字 0 到 10⁹。测试链接：LeetCode 2103 类型的 valid-arrangement-of-pairs。

数对 (a, b) 就是有向边 a→b。合法排列就是一条欧拉路径。值域大，把出现过的端点收集起来排序去重，二分找编号，2n 个值离散化成 2n 个点。

数据保证有解，但代码仍然把判定写全了：度数找起点，走完检查 cntp。排列的还原有个小技巧：path 数组里相邻两个点倒着取，前一个是边头、后一个是边尾。

LeetCode 版的完整代码。离散化用排序去重加二分，建图时同样从大到小插入保证字典序，数据保证有解所以判定只是防御。

class Solution {
    public:
        static const int MAXN = 200001;
        int n, m;
        vector<vector<int>> *pair;
        int sortv[MAXN];

        int head[MAXN];
        int nxt[MAXN];
        int to[MAXN];
        int cntg;

        int cur[MAXN];
        int outDeg[MAXN];
        int inDeg[MAXN];

        int path[MAXN];
        int cntp;

        void addEdge(int u, int v) {
            nxt[++cntg] = head[u];
            to[cntg] = v;
            head[u] = cntg;
        }

        int kth(int num) {
            int l = 1, r = n, ans = 0;
            while (l <= r) {
                int mid = (l + r) >> 1;
                if (sortv[mid] >= num) {
                    ans = mid;
                    r = mid - 1;
                } else {
                    l = mid + 1;
                }
            }
            return ans;
        }

        void prepare() {
            int len = 0;
            for (int i = 0; i < m; i++) {
                sortv[++len] = (*pair)[i][0];
                sortv[++len] = (*pair)[i][1];
            }
            sort(sortv + 1, sortv + len + 1);
            n = 1;
            for (int i = 2; i <= len; i++) {
                if (sortv[n] != sortv[i]) {
                    sortv[++n] = sortv[i];
                }
            }
            cntg = cntp = 0;
            for (int i = 1; i <= n; i++) {
                head[i] = outDeg[i] = inDeg[i] = 0;
            }
        }

        void connect() {
            for (int i = 0, u, v; i < m; i++) {
                u = kth((*pair)[i][0]);
                v = kth((*pair)[i][1]);
                outDeg[u]++;
                inDeg[v]++;
                addEdge(u, v);
            }
            for (int i = 1; i <= n; i++) {
                cur[i] = head[i];
            }
        }

        int directedStart() {
            int start = -1, end = -1;
            for (int i = 1; i <= n; i++) {
                int v = outDeg[i] - inDeg[i];
                if (v < -1 || v > 1 || (v == 1 && start != -1) || (v == -1 && end != -1)) {
                    return -1;
                }
                if (v == 1) {
                    start = i;
                }
                if (v == -1) {
                    end = i;
                }
            }
            if ((start == -1) ^ (end == -1)) {
                return -1;
            }
            if (start != -1) {
                return start;
            }
            for (int i = 1; i <= n; i++) {
                if (outDeg[i] > 0) {
                    return i;
                }
            }
            return -1;
        }

        void euler(int u) {
            for (int e = cur[u]; e > 0; e = cur[u]) {
                cur[u] = nxt[e];
                euler(to[e]);
            }
            path[++cntp] = u;
        }

        vector<vector<int>> validArrangement(vector<vector<int>>& pairs) {
            m = (int)pairs.size();
            pair = &pairs;
            prepare();
            connect();
            int start = directedStart();
            if (start == -1) {
                return {};
            }
            euler(start);
            if (cntp != m + 1) {
                return {};
            }
            vector<vector<int>> ans(m, vector<int>(2));
            for (int i = 0, j = cntp; i < m; i++, j--) {
                ans[i][0] = sortv[path[j]];
                ans[i][1] = sortv[path[j - 1]];
            }
            return ans;
        }

    };

二、题目4：词链

给定 n 个小写单词，想把它们串成一个词链，要求前一个单词的结尾字符等于后一个单词的开头字符。每个单词恰好用一次，重复出现 k 次就用 k 次。返回字典序最小的结果，注意点号小于任何小写字母；不存在输出三个星号。n 到 1000。测试链接：Luogu P1127。

单词是一条边：头是首字符，尾是尾字符，边权是整个单词。aloha 是 a 到 a 的边，arachnid 是 a 到 d 的边。词链 aloha.arachnid.dog.gopher.rat.tiger 对应路径 a→a→d→g→r→t→r，六条边恰好用一次。

与前一题最大的不同：欧拉路径收集的是边不是点。递归函数多带一个边权参数，path 数组存字符串。输出时 path 末尾多一个空串（起点的边权），从倒数第二个开始用点号连接。

字典序的排序键是（首字符，单词本身）。相同首字符的单词里，字典序小的先出。建图时对边编号数组排序再从大到小插入，和第一页的套路一致。

完整代码。注意 euler 函数带上了边权参数，path 存的是字符串。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 27;
    const int MAXM = 1002;
    int n = 26, m;

    string str[MAXM];
    int a[MAXM];
    int b[MAXM];
    int eidArr[MAXM];

    int head[MAXN];
    int nxt[MAXM];
    int to[MAXM];
    string weight[MAXM];
    int cntg;

    int cur[MAXN];
    int outDeg[MAXN];
    int inDeg[MAXN];

    string path[MAXM];
    int cntp;

    bool EdgeCmp(int i, int j) {
        if (a[i] != a[j]) {
            return a[i] < a[j];
        }
        return str[i] < str[j];
    }

    void addEdge(int u, int v, string w) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        weight[cntg] = w;
        head[u] = cntg;
    }

    int startNode(const string &str) {
        return (int)(str[0] - 'a' + 1);
    }

    int endNode(const string &str) {
        return (int)(str[(int)str.size() - 1] - 'a' + 1);
    }

    void connect() {
        sort(eidArr + 1, eidArr + m + 1, EdgeCmp);
        for (int l = 1, r = 1; l <= m; l = ++r) {
            while (r + 1 <= m && a[eidArr[l]] == a[eidArr[r + 1]]) {
                r++;
            }
            for (int i = r, u, v; i >= l; i--) {
                u = a[eidArr[i]];
                v = b[eidArr[i]];
                outDeg[u]++;
                inDeg[v]++;
                addEdge(u, v, str[eidArr[i]]);
            }
        }
        for (int i = 1; i <= n; i++) {
            cur[i] = head[i];
        }
    }

    int directedStart() {
        int start = -1, end = -1;
        for (int i = 1; i <= n; i++) {
            int v = outDeg[i] - inDeg[i];
            if (v < -1 || v > 1 || (v == 1 && start != -1) || (v == -1 && end != -1)) {
                return -1;
            }
            if (v == 1) {
                start = i;
            }
            if (v == -1) {
                end = i;
            }
        }
        if ((start == -1) ^ (end == -1)) {
            return -1;
        }
        if (start != -1) {
            return start;
        }
        for (int i = 1; i <= n; i++) {
            if (outDeg[i] > 0) {
                return i;
            }
        }
        return -1;
    }

    void euler(int u, string w) {
        for (int e = cur[u]; e > 0; e = cur[u]) {
            cur[u] = nxt[e];
            euler(to[e], weight[e]);
        }
        path[++cntp] = w;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> m;
        for (int i = 1; i <= m; i++) {
            cin >> str[i];
            a[i] = startNode(str[i]);
            b[i] = endNode(str[i]);
            eidArr[i] = i;
        }
        connect();
        int start = directedStart();
        if (start == -1) {
            cout << "***\n";
        } else {
            euler(start, "");
            if (cntp != m + 1) {
                cout << "***\n";
            } else {
                cout << path[cntp - 1];
                for (int i = cntp - 2; i >= 1; i--) {
                    cout << "." << path[i];
                }
                cout << "\n";
            }
        }
        return 0;
    }

三、题目5：无序字母对

给定 n 个字母对，字母是 A 到 Z 和 a 到 z，对内不区分顺序，ab 和 ba 相同，但区分大小写。没有重复字母对。构造长度 n+1 的字符串，让每个字母对的两個字母在串中相邻出现，输出字典序最小的方案；无解打印 No Solution。n 到 1326。测试链接：Luogu P1341。

字母对是无向边，52 个字符是点。相邻出现恰好一次，就是无向图的欧拉路径：奇度点 0 个是回路、2 个是路径，其余无解。三个字母对 ab、bc、ca 的答案是 abca，a 和 c 度数为 2，b 度数为 2，回路从 a 出发。

点只有 52 个、边一千多条，直接上邻接矩阵 graph[u][v] 存剩余边数，多重边累加计数。当前弧优化变成列指针 cur[u]：扫到 v 就加一，可用就递归、把两个方向的计数都减一。

字典序来自扫描顺序：列编号从小到大，正好是字符的先后。大写字母排在小写前面，这也是题目要的顺序。

完整代码。矩阵版建图省掉了排序，因为扫描顺序本身就是排序结果。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 53;
    const int MAXM = 2001;
    int n = 52, m;
    char arr[MAXM][2];

    int graph[MAXN][MAXN];
    int deg[MAXN];
    int cur[MAXN];

    int path[MAXM];
    int cntp;

    int getInt(char c) {
        return c <= 'Z' ? (c - 'A' + 1) : (c - 'a' + 27);
    }

    char getChar(int v) {
        return (char)(v <= 26 ? ('A' + v - 1) : ('a' + v - 27));
    }

    void connect() {
        for (int i = 1, u, v; i <= m; i++) {
            u = getInt(arr[i][0]);
            v = getInt(arr[i][1]);
            graph[u][v]++;
            graph[v][u]++;
            deg[u]++;
            deg[v]++;
        }
        for (int i = 1; i <= n; i++) {
            cur[i] = 1;
        }
    }

    int undirectedStart() {
        int odd = 0;
        for (int i = 1; i <= n; i++) {
            if ((deg[i] & 1) == 1) {
                odd++;
            }
        }
        if (odd != 0 && odd != 2) {
            return -1;
        }
        for (int i = 1; i <= n; i++) {
            if (odd == 0 && deg[i] > 0) {
                return i;
            }
            if (odd == 2 && (deg[i] & 1) == 1) {
                return i;
            }
        }
        return -1;
    }

    void euler(int u) {
        for (int v = cur[u]; v <= n; v = cur[u]) {
            cur[u]++;
            if (graph[u][v] > 0) {
                graph[u][v]--;
                graph[v][u]--;
                euler(v);
            }
        }
        path[++cntp] = u;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> m;
        for (int i = 1; i <= m; i++) {
            cin >> arr[i][0] >> arr[i][1];
        }
        connect();
        int start = undirectedStart();
        if (start == -1) {
            cout << "No Solution\n";
        } else {
            euler(start);
            if (cntp != m + 1) {
                cout << "No Solution\n";
            } else {
                for (int i = cntp; i >= 1; i--) {
                    cout << getChar(path[i]);
                }
                cout << "\n";
            }
        }
        return 0;
    }

四、复杂度分析

三道题的骨架相同：离散化或映射是线性或线性对数，建图和生成是线性。

操作时间复杂度空间复杂度
题目3：离散化 + 排序建图 + 欧拉路径O(n log n)O(n)
题目4：单词排序 + 建图 + 欧拉路径O(n log n + L)O(n + L)（L 为单词总长）
题目5：邻接矩阵建图 + 欧拉路径O(52² + m)O(52² + m)

题目5 的矩阵是 52 乘 52，可以当成常数；题目4 排序时比较字符串要算上长度。三题的生成阶段都严格线性，瓶颈全在排序和读入。

五、最后

识别信号就一句话：素材有头有尾、要求全用一次、首尾相接。剩下的都是模版：有向看度数差，无向看奇度点，字典序靠排序加头插或者扫描顺序。

下一节换一类玩法：不给素材，要自己造一个最短的序列，把所有可能性都包进去，也就是 de Bruijn 序列。
