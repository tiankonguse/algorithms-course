二分答案 + 可持久化：浮动区间的最大上中位数

这一讲的问题长这样：左端点落在 [a, b] 里、右端点落在 [c, d] 里（a<b<c<d），问这些区间里哪个的上中位数最大。上中位数本身不可持久化，可持久化的是「判定」：二分一个值 v，检查存不存在这样的区间让上中位数达到 v。

一、判定：区间上中位数 ≥ v

单个区间 [L, R] 的上中位数 ≥ v，怎么判断？把数组转成符号：arr[i] ≥ v 记 +1，arr[i] < v 记 -1。区间和 ≥ 0，说明区间里 ≥ v 的数不少于一半，上中位数就 ≥ v。

反过来也成立：上中位数 ≥ v 的时候，≥ v 的数必然占一半以上，区间和 ≥ 0。所以「区间和 ≥ 0」和「上中位数 ≥ v」是同一个判定，符号化之后把「中位数」这种排序问题变成了求和问题，线段树就能接了。

二、浮动区间：三段拆开各取最大

左端点在 [a, b] 里、右端点在 [c, d] 里，区间 [L, R] 必然包含整个 [b+1, c-1]。于是和拆成三段：[L, b] + [b+1, c-1] + [c, R]。

要让总和最大：第一段取 [a, b] 里最大后缀和，第二段是固定总和，第三段取 [c, d] 里最大前缀和。这三样正好是线段树节点能合并的三个信息：pre 最大前缀和、suf 最大后缀和、sum 总和。

query 按中序遍历从左往右合并区间时，用三个变量滚动维护：当前累计和、累计的最大前缀、累计的最大后缀。这样一趟 query 拿到任意区间的 pre/suf/sum，check 里三段的组合就是 best。

三、按值建版本 + 二分答案

每个 v 都要一棵符号树，n 个 v 就是 n 个版本。建法很省：先把 arr 按值排序，root[1] 全 +1；root[i] 从 root[i-1] 继承，只把「排序后第 i-1 小的那个位置」改成 -1。于是 root[v] 里，值 ≥ arr[v] 的位置全是 +1，值更小的全是 -1。

check(a, b, c, d, v) 用 root[v] 查三段、拼 best；二分 v 的排名，check 为真的最大 v 就是答案（输出对应的值）。强制在线用 lastAns 把每个参数解码，查完更新 lastAns。

浮动区间的最大上中位数（P2839）

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 20001;
    const int MAXT = MAXN * 20;
    const int INF = 10000001;
    int n, m;
    vector<pair<int, int>> arr;
    int root[MAXN];
    int ls[MAXT];
    int rs[MAXT];
    int pre[MAXT];
    int suf[MAXT];
    int sum[MAXT];
    int cnt;
    int ques[4], info[3];

    int build(int l, int r) {
        int rt = ++cnt;
        pre[rt] = suf[rt] = sum[rt] = r - l + 1;
        if (l < r) {
            int mid = (l + r) / 2;
            ls[rt] = build(l, mid);
            rs[rt] = build(mid + 1, r);
        }
        return rt;
    }

    void up(int i) {
        pre[i] = max(pre[ls[i]], sum[ls[i]] + pre[rs[i]]);
        suf[i] = max(suf[rs[i]], suf[ls[i]] + sum[rs[i]]);
        sum[i] = sum[ls[i]] + sum[rs[i]];
    }

    int update(int jobi, int l, int r, int i) {
        int rt = ++cnt;
        ls[rt] = ls[i];
        rs[rt] = rs[i];
        pre[rt] = pre[i];
        suf[rt] = suf[i];
        sum[rt] = sum[i];
        if (l == r) {
            pre[rt] = suf[rt] = sum[rt] = -1;
        } else {
            int mid = (l + r) / 2;
            if (jobi <= mid) {
                ls[rt] = update(jobi, l, mid, ls[rt]);
            } else {
                rs[rt] = update(jobi, mid + 1, r, rs[rt]);
            }
            up(rt);
        }
        return rt;
    }

    void initInfo() {
        info[0] = info[1] = -INF;
        info[2] = 0;
    }

    void mergeRight(int r) {
        info[0] = max(info[0], info[2] + pre[r]);
        info[1] = max(suf[r], info[1] + sum[r]);
        info[2] += sum[r];
    }

    void query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            mergeRight(i);
        } else {
            int mid = (l + r) / 2;
            if (jobl <= mid) {
                query(jobl, jobr, l, mid, ls[i]);
            }
            if (jobr > mid) {
                query(jobl, jobr, mid + 1, r, rs[i]);
            }
        }
    }

    void prepare() {
        sort(arr.begin() + 1, arr.end(), [](const pair<int, int>& a, const pair<int, int>& b) {
            return a.second < b.second;
        });
        cnt = 0;
        root[1] = build(1, n);
        for (int i = 2; i <= n; i++) {
            root[i] = update(arr[i - 1].first, 1, n, root[i - 1]);
        }
    }

    bool check(int a, int b, int c, int d, int v) {
        initInfo();
        query(a, b, 1, n, root[v]);
        int best = info[1];
        initInfo();
        query(c, d, 1, n, root[v]);
        best += info[0];
        if (b + 1 <= c - 1) {
            initInfo();
            query(b + 1, c - 1, 1, n, root[v]);
            best += info[2];
        }
        return best >= 0;
    }

    int compute(int a, int b, int c, int d) {
        int left = 1, right = n, mid, ans = 0;
        while (left <= right) {
            mid = (left + right) / 2;
            if (check(a, b, c, d, mid)) {
                ans = arr[mid].second;
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        return ans;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n;
        arr.resize(n + 1);
        for (int i = 1; i <= n; i++) {
            arr[i].first = i;
            cin >> arr[i].second;
        }
        prepare();
        cin >> m;
        for (int i = 1, lastAns = 0; i <= m; i++) {
            for (int j = 0; j < 4; j++) {
                cin >> ques[j];
                ques[j] = (ques[j] + lastAns) % n + 1;
            }
            sort(ques, ques + 4);
            lastAns = compute(ques[0], ques[1], ques[2], ques[3]);
            cout << lastAns << '\n';
        }
        return 0;
    }

五、总结

这题的关键是把「中位数 ≥ v」翻译成「符号区间和 ≥ 0」，排序问题就变成求和问题。浮动区间的约束拆三段取最值，每个节点存 pre/suf/sum。可持久化的角色是给每个 v 准备一棵符号树，代价是 O((n+m) log² n)。这类「二分答案 + 版本树」的组合在可持久化题目里很常见，难点总在怎么把判定条件转成可合并的区间信息。
