整体二分：矩阵内第 k 小

上一页的整体二分用在了一维区间上，这一页把它搬到矩阵里。答案范围还是排序下标，判定的工具从一维树状数组换成二维树状数组，递归框架一个字都不用改。

零、背景

题目 2 是「矩阵内第 k 小」，也叫矩阵 kth。它的形状跟题目 1 完全一样：一堆查询，每个查询要一个矩形里的第 k 小。差别只在判定时要数的是矩形，不是区间。

一、题目2：矩阵内第 k 小

给定一个 n × n 的矩阵，接下来有 q 条查询，格式如下。

  • 查询 a b c d k：左上角 (a, b)，右下角 (c, d)，打印该区域中第 k 小的数。

1 ≤ n ≤ 500，1 ≤ q ≤ 6 × 10⁴，0 ≤ 矩阵中的数字 ≤ 10⁹。测试链接：洛谷 P1527。

二、套进整体二分

把矩阵里 n² 个数连同坐标一起按值排序，记成 xyv[1..n²]。答案范围就是这个排序下标 [1..n²]，第 i 个位置背后的操作是「在 (x, y) 上加 1」。

执行 [vl..mid] 上的操作，就是把这一段数对应的格子在二维树状数组里各加 1。判定时数矩形里加了多少个：satisfy = query(a, b, c, d)。之后照旧，达标去左组，不达标 k 减掉 satisfy 去右组。

可以发现这一页的代码和上一页几乎是同一份：把 arr[i].i 换成 xyv[i].x 和 xyv[i].y，把 query(l, r) 换成 query(a, b, c, d)，就换完了。

三、写法一：范围操作 + 撤销

    
#include <bits/stdc++.h>

    using namespace std;

    struct Number {
        int x, y, v;
    };

    bool NumberCmp(Number a, Number b) {
        return a.v < b.v;
    }

    const int MAXN = 501;
    const int MAXQ = 1000001;
    int n, q;

    Number xyv[MAXN * MAXN];
    int cntv = 0;

    int qid[MAXQ];
    int a[MAXQ];
    int b[MAXQ];
    int c[MAXQ];
    int d[MAXQ];
    int k[MAXQ];

    int tree[MAXN][MAXN];

    int lset[MAXQ];
    int rset[MAXQ];

    int ans[MAXQ];

    int lowbit(int i) {
        return i & -i;
    }

    void add(int x, int y, int v) {
        for (int i = x; i <= n; i += lowbit(i)) {
            for (int j = y; j <= n; j += lowbit(j)) {
                tree[i][j] += v;
            }
        }
    }

    int sum(int x, int y) {
        int ret = 0;
        for (int i = x; i > 0; i -= lowbit(i)) {
            for (int j = y; j > 0; j -= lowbit(j)) {
                ret += tree[i][j];
            }
        }
        return ret;
    }

    int query(int a, int b, int c, int d) {
        return sum(c, d) - sum(a - 1, d) - sum(c, b - 1) + sum(a - 1, b - 1);
    }

    void compute(int ql, int qr, int vl, int vr) {
        if (ql > qr) {
            return;
        }
        if (vl == vr) {
            for (int i = ql; i <= qr; i++) {
                ans[qid[i]] = xyv[vl].v;
            }
        } else {
            int mid = (vl + vr) >> 1;
            for (int i = vl; i <= mid; i++) {
                add(xyv[i].x, xyv[i].y, 1);
            }
            int lsiz = 0, rsiz = 0;
            for (int i = ql; i <= qr; i++) {
                int id = qid[i];
                int satisfy = query(a[id], b[id], c[id], d[id]);
                if (satisfy >= k[id]) {
                    lset[++lsiz] = id;
                } else {
                    k[id] -= satisfy;
                    rset[++rsiz] = id;
                }
            }
            for (int i = 1; i <= lsiz; i++) {
                qid[ql + i - 1] = lset[i];
            }
            for (int i = 1; i <= rsiz; i++) {
                qid[ql + lsiz + i - 1] = rset[i];
            }
            for (int i = vl; i <= mid; i++) {
                add(xyv[i].x, xyv[i].y, -1);
            }
            compute(ql, ql + lsiz - 1, vl, mid);
            compute(ql + lsiz, qr, mid + 1, vr);
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> q;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                xyv[++cntv].x = i;
                xyv[cntv].y = j;
                cin >> xyv[cntv].v;
            }
        }
        for (int i = 1; i <= q; i++) {
            qid[i] = i;
            cin >> a[i] >> b[i] >> c[i] >> d[i] >> k[i];
        }
        sort(xyv + 1, xyv + cntv + 1, NumberCmp);
        compute(1, q, 1, cntv);
        for (int i = 1; i <= q; i++) {
            cout << ans[i] << '\n';
        }
        return 0;
    }

n² 个数排序是 O(n² log n²)，n = 500 时是 2.5 × 10⁵ 个数，排序和树状数组都开得下。二维树状数组直接开成 tree[501][501]，一百万个 int，四个字节一百万就是 4 MB。

四、写法二：指针滚动

和上一页一样，把每层的「加一遍、撤一遍」换成全局指针 used 挪到 mid。树状数组里始终是排序后的前 used 个数，计数是前缀贡献，单调不减，k 不用减。

    
int used = 0;

    void compute(int ql, int qr, int vl, int vr) {
        if (ql > qr) {
            return;
        }
        if (vl == vr) {
            for (int i = ql; i <= qr; i++) {
                ans[qid[i]] = xyv[vl].v;
            }
        } else {
            int mid = (vl + vr) >> 1;
            int lsiz = 0, rsiz = 0;
            while (used < mid) {
                used++;
                add(xyv[used].x, xyv[used].y, 1);
            }
            while (used > mid) {
                add(xyv[used].x, xyv[used].y, -1);
                used--;
            }
            for (int i = ql; i <= qr; i++) {
                int id = qid[i];
                int satisfy = query(a[id], b[id], c[id], d[id]);
                if (satisfy >= k[id]) {
                    lset[++lsiz] = id;
                } else {
                    rset[++rsiz] = id;
                }
            }
            for (int i = 1; i <= lsiz; i++) {
                qid[ql + i - 1] = lset[i];
            }
            for (int i = 1; i <= rsiz; i++) {
                qid[ql + lsiz + i - 1] = rset[i];
            }
            compute(ql, ql + lsiz - 1, vl, mid);
            compute(ql + lsiz, qr, mid + 1, vr);
        }
    }

两种写法在这道题上的取舍和题目 1 一致。n² 是 2.5 × 10⁵，比 m 大一个量级，写法二省掉的那部分加法是本体的主要常数开销，实测更容易通过。

六、复杂度分析

记 N = n² 为矩阵里的数字个数，q 为查询条数。二维树状数组上单点修改和矩形求和都是 O(log²n)，这就是判定代价 T。递归层数 log N = 2 log n。

操作时间复杂度空间复杂度
n² 个数按值排序O(N log N) = O(n² log n)O(n² + q)：排序数组 O(n²)，查询参数与左右组暂存 O(q)，二维树状数组 O(n²)
一层的全部修改（写法一加一遍撤一遍 / 写法二挪指针）O(N log²n)
一层的全部查询检查（每条查询恰好一次，四个前缀和）O(q log²n)
递归层数O(log N) = O(log n)
题目2 总代价O((N + q) log N log²n)
朴素做法：每条查询取矩形排序，或每条各自二分答案O(q · n² log n)O(n²)

N = 2.5 × 10⁵、q = 6 × 10⁴ 时，整体二分大约是 3 × 10⁵ × 18 × 81 ≈ 4 × 10⁸ 次基本操作，二维树状数组常数小，可以通过。写法二把「加一遍撤一遍」省成一次挪动，实际更快。

七、最后

矩阵这一题说明的是：整体二分的框架跟判定的数据结构无关。一维区间换成二维矩形，只动 add 和 query 两个函数，递归部分照抄。
