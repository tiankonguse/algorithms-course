卡特兰数：路径与组合模型

进出栈序列画成折线，就是一条不越过对角线的路径。路径计数模型把 n 个 1 和 n 个 −1 变成 (0,0) 到 (n,n) 的走法，于是圆上连线、红黑前缀、生成字符串三题全归到同一套组合数上。这一页把三题逐个落地，顺带处理两个取模陷阱：1e8+7 和 100。

零、路径计数模型

正方形网格里从 (0,0) 走到 (n,n)，每一步只能向上或向右。把向右记成红、向上记成黑，任意前缀红数 ≥ 黑数就是路径不越过对角线（可以碰）。这类路径数正好是公式1：C(2n,n) − C(2n,n−1)。

换一个约束：除了起点终点外完全不碰对角线，路径数变成 2·C(2n−2,n−1) − 2·C(2n−2,n)，推导方法和不越线版本完全同套路，只是把第一段固定为向某个方向，剩下的数一个「不越过偏移后对角线」的问题。掌握路径模型的分析方法，题 6 直接套。

一、题目6：生成字符串

洛谷 P1641。有 n 个 1 和 m 个 0，组成 n+m 长度数列，任意前缀 1 的数量 ≥ 0 的数量。m ≤ n，n+m ≤ 2×10⁶，答案对 20100403 取模。这就是路径模型的不对称版本：从 (0,0) 走到 (n,m)，不越过对角线，答案是 C(n+m,m) − C(n+m,m−1)，把公式1 里的 n 换成 m 就行。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;

    // 生成字符串：有 n 个 1 和 m 个 0，任意前缀 1 的数量 >= 0 的数量
    // 答案对 20100403 取模，1 <= m <= n <= 10^6
    // 从 (0,0) 走到 (n,m)，不越过对角线的路径数 = C(n+m,m) - C(n+m,m-1)
    const int MOD = 20100403;
    const int MAXN = 2000001;

    ll fac[MAXN];
    ll inv[MAXN];

    ll power(ll x, ll p) {
    	ll ans = 1;
    	while (p > 0) {
    		if (p & 1) {
    			ans = ans * x % MOD;
    		}
    		x = x * x % MOD;
    		p >>= 1;
    	}
    	return ans;
    }

    void build(int n) {
    	fac[0] = inv[0] = 1;
    	fac[1] = 1;
    	for (int i = 2; i <= n; i++) {
    		fac[i] = (ll)i * fac[i - 1] % MOD;
    	}
    	inv[n] = power(fac[n], MOD - 2);
    	for (int i = n - 1; i >= 1; i--) {
    		inv[i] = (ll)(i + 1) * inv[i + 1] % MOD;
    	}
    }


    ll c(int n, int k) {
    	return fac[n] * inv[k] % MOD * inv[n - k] % MOD;
    }

    ll compute(int n, int m) {
    	build(n + m);
    	return (c(n + m, m) - c(n + m, m - 1) + MOD) % MOD;
    }

    int main() {
    	int n, m;
    	cin >> n >> m;
    	cout << compute(n, m) << endl;
    	return 0;
    }

二、题目2：圆上连线

洛谷 P1976。圆上有 2n 个点，两两成对连成 n 条线段，任意两条线段不能相交。按点的顺序做括号匹配：一条线段连 (i,j) 当且仅当 i 是左括号、j 是右括号，不相交 ⟺ 括号序列合法。于是连接方法数 = 卡特兰数，用公式1。

取模这里是个陷阱：题目对 10⁸+7 取模，不是常见的 10⁹+7。公式代码里 MOD 常量写错一个 0，整份答案全错还很难看出来。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;

    // 圆上连线：圆上有 2n 个点，成对连成 n 条线段且两两不相交
    // 答案对 10^8 + 7 取模（不是 10^9 + 7），1 <= n <= 2999
    const int MOD = 100000007;
    const int MAXN = 2000001;

    ll fac[MAXN];
    ll inv[MAXN];

    ll power(ll x, ll p) {
    	ll ans = 1;
    	while (p > 0) {
    		if (p & 1) {
    			ans = ans * x % MOD;
    		}
    		x = x * x % MOD;
    		p >>= 1;
    	}
    	return ans;
    }

    void build(int n) {
    	fac[0] = inv[0] = 1;
    	fac[1] = 1;
    	for (int i = 2; i <= n; i++) {
    		fac[i] = (ll)i * fac[i - 1] % MOD;
    	}
    	inv[n] = power(fac[n], MOD - 2);
    	for (int i = n - 1; i >= 1; i--) {
    		inv[i] = (ll)(i + 1) * inv[i + 1] % MOD;
    	}
    }


    ll c(int n, int k) {
    	return fac[n] * inv[k] % MOD * inv[n - k] % MOD;
    }

    // 公式1
    ll compute(int n) {
    	build(2 * n);
    	return (c(2 * n, n) - c(2 * n, n - 1) + MOD) % MOD;
    }

    int main() {
    	int n;
    	cin >> n;
    	cout << compute(n) << endl;
    	return 0;
    }

三、题目3：任意前缀上红多于黑

洛谷 P1722。n 个红 n 个黑组成 2n 长度数列，任意前缀红数 ≥ 黑数。这就是路径模型原版，答案对 100 取模。100 含很多因子，费马小定理和扩展欧几里得都求不出逆元，公式1、2、3 全废，直接用公式4。n ≤ 100，O(n²) 随便跑。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;

    // 任意前缀上红多于黑：n 个红 n 个黑组成 2n 长度数列，任意前缀红数 >= 黑数
    // 答案对 100 取模，1 <= n <= 100
    // 取模数字含有很多因子，无法用费马小定理或扩展欧几里得求逆元，直接用公式4
    ll compute(int n) {
    	ll f[101] = {};
    	f[0] = f[1] = 1;
    	for (int i = 2; i <= n; i++) {
    		for (int l = 0, r = i - 1; l < i; l++, r--) {
    			f[i] = (f[i] + f[l] * f[r] % 100) % 100;
    		}
    	}
    	return f[n];
    }

    int main() {
    	int n;
    	cin >> n;
    	cout << compute(n) << endl;
    	return 0;
    }

总结

路径计数模型把「任意前缀 1 比 0 多」翻译成「不越过对角线的路径」，答案统一是 C(2n,n) − C(2n,n−1)（对称版）或 C(n+m,m) − C(n+m,m−1)（不对称版）。圆上连线用括号匹配归到同一公式；红黑前缀被合数模数 100 逼到公式4。两个取模细节容易翻车：10⁸+7 少一个 0、合数模数求不了逆元。三题对拍全部通过，公式与题目语义一致。下一页处理更硬的取模场景：p 可能不是质数，n 到 10⁶，只能用因子计数法。
