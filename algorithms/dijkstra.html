Dijkstra 算法：普通堆与反向索引堆

单源最短路径，最常用的是普通堆实现的 Dijkstra：小根堆按距离组织，谁近先弹谁，弹过的点距离定死。要求边权非负，有向无向都行。反向索引堆是极致版，堆里每个点只占一条记录。

零、为什么边权不能是负数

Dijkstra 的合法性建立在一条性质上：从堆里弹出记录 (u, d) 时，d 就是源点到 u 的最短距离，之后不会再变小。理由：堆里所有记录的 d 都比 d 大，任何没弹出的路径想绕到 u，都要先经过某个记录（距离 ≥ d），再加非负边权，只会更大。

边权是负数这条性质就塌了：已经弹过的 u，可能经一条负边回头把某个距离再变小，而 u 已经标记过、不会再处理。负环更没救，绕一圈距离越来越小。所以负权单源最短路要交给别的算法（SPFA、Bellman-Ford），Dijkstra 只接非负权。

注意 0 边权没问题：多走一条 0 权边不会让距离变小，弹出即定死的性质还在。

一、普通堆实现的 Dijkstra

流程五步：1）distance[i] 记源点到 i 的最短距离，visited[i] 记 i 是否弹出过；2）小根堆放记录 (点, 源点到该点距离)，按距离组织；3）distance[源点] = 0，(源点, 0) 入堆；4）弹出 (u, 源点到 u 的距离)：u 弹过就跳过；没弹过就标记 u，考察 u 的每条边 (v, w)，如果 v 没弹过且 distance[u] + w < distance[v]，更新 distance[v] 并把 (v, distance[u] + w) 入堆；5）堆空结束，distance 就是答案。

冗余记录是普通堆的特点：同一个点可能先后以不同距离多次入堆，弹出来发现 visited 已标记就直接跳过。堆里最多同时躺着 O(m) 条记录，每次操作 O(log m)，总复杂度 O(m log m)。

演示 · 普通堆逐帧：5 点 6 边，源点 1，弹出顺序与距离表

网络延迟时间（LC743）就是模版题：信号从 s 出发沿有向边传播，问全部节点收到信号要多久，就是 distance 的最大值；有节点不可达返回 -1。实现照上面的流程，一个 priority_queue 的事。

#include <bits/stdc++.h>
    using namespace std;

    // 普通堆实现的 Dijkstra（LC743 网络延迟时间）
    int networkDelayTime(vector<vector<int>>& times, int n, int s) {
        vector<vector<pair<int, int>>> graph(n + 1);
        for (auto& e : times) {
            graph[e[0]].push_back({e[1], e[2]});
        }
        vector<int> distance(n + 1, INT_MAX);
        vector<bool> visited(n + 1, false);
        distance[s] = 0;
        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> heap;
        heap.push({0, s});
        while (!heap.empty()) {
            int u = heap.top().second;
            heap.pop();
            if (visited[u]) {
                continue;
            }
            visited[u] = true;
            for (auto [v, w] : graph[u]) {
                if (!visited[v] && distance[u] + w < distance[v]) {
                    distance[v] = distance[u] + w;
                    heap.push({distance[v], v});
                }
            }
        }
        int ans = INT_MIN;
        for (int i = 1; i <= n; i++) {
            if (distance[i] == INT_MAX) {
                return -1;
            }
            ans = max(ans, distance[i]);
        }
        return ans;
    }

二、反向索引堆：堆里每个点只有一条记录

冗余记录能不能去掉？普通堆里 (v, 5) 和 (v, 7) 同时存在，其中一条注定被丢弃。反向索引堆让每个点永远只占一个堆位置：where[v] 记 v 在堆上的下标，距离变小就原地更新、上滤，不再新增记录。

where[v] 三个值：-1 表示 v 从没进过堆；-2 表示 v 已经弹出过、以后忽略；i（≥ 0）表示 v 在堆的 i 位置。弹出堆顶后 where[弹出的点] 置 -2。松弛边 (v, w) 时分三种情况：v 没进过堆，新增记录入堆；v 弹出过，忽略；v 在堆里，看看 distance[v] 能不能更小，能就原地改值上滤，不能就忽略。

堆规模恒为 n，每次操作 O(log n)，总复杂度 O(m log n)，比普通堆的 O(m log m) 理论上限好。代价是每一处都要维护 where 数组的同步：swap 时两个点的 where 一起换，弹完置 -2。

实测（本机 clang，n = 10^5、m = 2×10^5 强连通随机图）：-O0 下反向索引堆 22.3ms，普通堆 52.8ms，快约 2.4 倍；稀疏图 m = n 时 3.7ms 对 13.0ms，约 3.6 倍。但 -O2 下普通堆反而 9.6ms 对 19.6ms：STL 的 priority_queue 被编译器优化得很彻底，而手写数组堆的 where 维护开销省不掉。反向索引堆的价值在堆规模恒 n 这个结构性优势，常数上别指望稳赢。

洛谷 P4779 用链式前向星建图 + 反向索引堆跑单源最短路，是这版实现的完整提交形态。

#include <bits/stdc++.h>
    using namespace std;

    // 链式前向星 + 反向索引堆实现的 Dijkstra（洛谷 P4779）
    const int MAXN = 100001;
    const int MAXM = 200001;

    int head[MAXN], nxt[MAXM], to[MAXM], weight[MAXM], cnt;
    int heapArr[MAXN], whereArr[MAXN], heapSize;
    int distanceArr[MAXN];
    int n, m, s;

    void build() {
        cnt = 1;
        heapSize = 0;
        fill(head, head + n + 1, 0);
        fill(whereArr, whereArr + n + 1, -1);
        fill(distanceArr, distanceArr + n + 1, INT_MAX);
    }

    void addEdge(int u, int v, int w) {
        nxt[cnt] = head[u];
        to[cnt] = v;
        weight[cnt] = w;
        head[u] = cnt++;
    }

    void swapNode(int i, int j) {
        int tmp = heapArr[i];
        heapArr[i] = heapArr[j];
        heapArr[j] = tmp;
        whereArr[heapArr[i]] = i;
        whereArr[heapArr[j]] = j;
    }

    // whereArr[v] = -1 从未进过堆，-2 已弹出，i 在堆上 i 位置
    void addOrUpdateOrIgnore(int v, int c) {
        if (whereArr[v] == -1) {
            heapArr[heapSize] = v;
            whereArr[v] = heapSize++;
            distanceArr[v] = c;
            heapInsert(whereArr[v]);
        } else if (whereArr[v] >= 0) {
            distanceArr[v] = min(distanceArr[v], c);
            heapInsert(whereArr[v]);
        }
    }

    void heapInsert(int i) {
        while (distanceArr[heapArr[i]] < distanceArr[heapArr[(i - 1) / 2]]) {
            swapNode(i, (i - 1) / 2);
            i = (i - 1) / 2;
        }
    }

    int pop() {
        int ans = heapArr[0];
        swapNode(0, --heapSize);
        heapify(0);
        whereArr[ans] = -2;
        return ans;
    }

    void heapify(int i) {
        int l = i * 2 + 1;
        while (l < heapSize) {
            int best = l + 1 < heapSize && distanceArr[heapArr[l + 1]] < distanceArr[heapArr[l]] ? l + 1 : l;
            best = distanceArr[heapArr[best]] < distanceArr[heapArr[i]] ? best : i;
            if (best == i) {
                break;
            }
            swapNode(best, i);
            i = best;
            l = i * 2 + 1;
        }
    }

    bool isEmpty() {
        return heapSize == 0;
    }

    int main() {
        cin >> n >> m >> s;
        build();
        for (int i = 0, u, v, w; i < m; i++) {
            cin >> u >> v >> w;
            addEdge(u, v, w);
        }
        addOrUpdateOrIgnore(s, 0);
        while (!isEmpty()) {
            int u = pop();
            for (int ei = head[u]; ei > 0; ei = nxt[ei]) {
                addOrUpdateOrIgnore(to[ei], distanceArr[u] + weight[ei]);
            }
        }
        for (int i = 1; i <= n; i++) {
            cout << distanceArr[i] << " ";
        }
        return 0;
    }

三、二维格子上的 Dijkstra：代价取 max（最小体力）

最小体力消耗路径（LC1631）：从左上到右下，代价是路径上相邻格子高度差绝对值的最大值，求最小代价。这题的边权不是「一步的差」，而是「整条路径上的最大差」，但 Dijkstra 照跑：distance[x][y] 记到 (x, y) 的最优路径上最大差，转移时新代价 = max(当前代价, 这一步的高度差)。

max 型代价同样满足「弹出即定死」：已经弹出的 (x, y, c)，别的路径想以更小代价到 (x, y)，必须经过某个代价 ≥ c 的格子，max 只会更大。终点弹出直接返回，不用等全部跑完。

演示 · 3×3 高度网格，max 代价传播，终点弹出返回 2

实测 200×200 随机高度网格 3.63ms，100×100 是 1.15ms。二维格子题把每个格子当图上的点、四邻当边，Dijkstra 直接落上去，区别只在代价怎么定义。

#include <bits/stdc++.h>
    using namespace std;

    // 最小体力消耗路径（LC1631）：代价 = 路径上相邻高度差绝对值的最大值
    int minimumEffortPath(vector<vector<int>>& heights) {
        int n = heights.size(), m = heights[0].size();
        vector<vector<int>> distance(n, vector<int>(m, INT_MAX));
        distance[0][0] = 0;
        vector<vector<bool>> visited(n, vector<bool>(m, false));
        priority_queue<tuple<int, int, int>,
                       vector<tuple<int, int, int>>,
                       greater<tuple<int, int, int>>> heap;
        heap.push({0, 0, 0});
        int move[5] = {-1, 0, 1, 0, -1};
        while (!heap.empty()) {
            auto [c, x, y] = heap.top();
            heap.pop();
            if (visited[x][y]) {
                continue;
            }
            if (x == n - 1 && y == m - 1) {
                return c;
            }
            visited[x][y] = true;
            for (int i = 0; i < 4; i++) {
                int nx = x + move[i], ny = y + move[i + 1];
                if (nx >= 0 && nx < n && ny >= 0 && ny < m && !visited[nx][ny]) {
                    int nc = max(c, abs(heights[x][y] - heights[nx][ny]));
                    if (nc < distance[nx][ny]) {
                        distance[nx][ny] = nc;
                        heap.push({nc, nx, ny});
                    }
                }
            }
        }
        return -1;
    }

四、水位上升的泳池：把「时间」当边权

水位上升的泳池中游泳（LC778）：水位 t 时，高度 ≤ t 的平台才能站。最少时间 = 从左上到右下某条路径上所有平台高度的最大值（起点高度也要算进去）。这题没有边权，代价就压在点上：distance[0][0] 直接是 grid[0][0]，转移时新代价 = max(当前代价, 新平台高度)。

和上一题只差一个字：上一题 max 的是「高度差」，这题 max 的是「平台高度本身」。同样终点弹出直接返回。

实测 200×200 随机网格 3.32ms，100×100 是 0.65ms。两道二维题合起来的结论：Dijkstra 的「距离」可以是任何非负、可累积的量，max 型代价照样成立。

#include <bits/stdc++.h>
    using namespace std;

    // 水位上升的泳池中游泳（LC778）：代价 = 途经平台高度的最大值
    int swimInWater(vector<vector<int>>& grid) {
        int n = grid.size(), m = grid[0].size();
        vector<vector<int>> distance(n, vector<int>(m, INT_MAX));
        distance[0][0] = grid[0][0];
        vector<vector<bool>> visited(n, vector<bool>(m, false));
        priority_queue<tuple<int, int, int>,
                       vector<tuple<int, int, int>>,
                       greater<tuple<int, int, int>>> heap;
        heap.push({grid[0][0], 0, 0});
        int move[5] = {-1, 0, 1, 0, -1};
        while (!heap.empty()) {
            auto [c, x, y] = heap.top();
            heap.pop();
            if (visited[x][y]) {
                continue;
            }
            visited[x][y] = true;
            if (x == n - 1 && y == m - 1) {
                return c;
            }
            for (int i = 0; i < 4; i++) {
                int nx = x + move[i], ny = y + move[i + 1];
                if (nx >= 0 && nx < n && ny >= 0 && ny < m && !visited[nx][ny]) {
                    int nc = max(c, grid[nx][ny]);
                    if (nc < distance[nx][ny]) {
                        distance[nx][ny] = nc;
                        heap.push({nc, nx, ny});
                    }
                }
            }
        }
        return -1;
    }

总结

Dijkstra 三件事：堆按距离组织、弹出即定死、只接非负权。普通堆实现 O(m log m)，堆里点会重复；反向索引堆 O(m log n)，where 三态保证每点一条记录；二维题把格子和四邻当图，代价换成 max 型照跑。本机实测：普通堆与反向索引堆在 -O0 下差 2.4~3.6 倍，-O2 下 STL 堆常数反超。
