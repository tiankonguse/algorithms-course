优化建图题目

上一页讲完了线段树优化建图的原理。这一页把本课剩下的题过一遍，看每种约束怎么翻译成图上的边，以及建完图之后接着跑什么算法。

零、背景

本课的题分两条线。题目 2 到 5 在数轴上做区间连边，建完图接着跑 0-1 BFS、分层最短路、拓扑排序、强连通分量。

题目 6 到 9 把「区间」换成树上的一段路径。路径连线不能直接套线段树，得先用树链剖分或倍增把它拆成 O(log n) 个连续段，再让每段落到线段树的节点上。

可以发现，建图方式变了，建完之后的算法没变。优化建图做的只是把图变小，让后面的算法跑得动。

一、题目2：旅程

n 个点，给定起点 p，m 次操作。每次给出 a b c d，表示 [a, b] 里每个点与 [c, d] 里每个点之间都加一条无向边。求 p 到每个点走过的最少边数，题目保证整张图连通。n 到 5 × 10⁵，m 到 10⁵。

约束直接就是区间连区间。拿两个辅助点 u、v 中转：出树上 [a, b] 的节点连到 u，u 连 v，v 连到入树上 [c, d] 的节点。因为是无向边，正向连一遍，再把 c、d 和 a、b 调换位置连一遍。

树内的边权都是 0，只有 u→v 那条边权是 1。整张图的边权只有 0 和 1，最短路用 0-1 BFS：双端队列，走 0 边压队首，走 1 边压队尾。

点数 O(n)，边数 O(m log n)，0-1 BFS 一趟 O(n + m log n)。

完整代码：

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 500001;
    const int MAXT = MAXN * 10;
    const int MAXE = MAXN * 20;
    const int INF = 1 << 30;
    int n, m, p;

    int head[MAXT];
    int nxt[MAXE];
    int to[MAXE];
    int weight[MAXE];
    int cntg;

    int ls[MAXT];
    int rs[MAXT];
    int rootOut, rootIn;
    int cntt;

    int dist[MAXT];
    deque<int> deq;

    void addEdge(int u, int v, int w) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        weight[cntg] = w;
        head[u] = cntg;
    }

    int buildOut(int l, int r) {
        int rt;
        if (l == r) {
            rt = l;
        } else {
            rt = ++cntt;
            int mid = (l + r) >> 1;
            ls[rt] = buildOut(l, mid);
            rs[rt] = buildOut(mid + 1, r);
            addEdge(ls[rt], rt, 0);
            addEdge(rs[rt], rt, 0);
        }
        return rt;
    }

    int buildIn(int l, int r) {
        int rt;
        if (l == r) {
            rt = l;
        } else {
            rt = ++cntt;
            int mid = (l + r) >> 1;
            ls[rt] = buildIn(l, mid);
            rs[rt] = buildIn(mid + 1, r);
            addEdge(rt, ls[rt], 0);
            addEdge(rt, rs[rt], 0);
        }
        return rt;
    }

    void xToRange(int jobx, int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            addEdge(jobx, i, 0);
        } else {
            int mid = (l + r) >> 1;
            if (jobl <= mid) {
                xToRange(jobx, jobl, jobr, l, mid, ls[i]);
            }
            if (jobr > mid) {
                xToRange(jobx, jobl, jobr, mid + 1, r, rs[i]);
            }
        }
    }

    void rangeToX(int jobl, int jobr, int jobx, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            addEdge(i, jobx, 0);
        } else {
            int mid = (l + r) >> 1;
            if (jobl <= mid) {
                rangeToX(jobl, jobr, jobx, l, mid, ls[i]);
            }
            if (jobr > mid) {
                rangeToX(jobl, jobr, jobx, mid + 1, r, rs[i]);
            }
        }
    }

    void rangeToRange(int a, int b, int c, int d) {
        int x = ++cntt;
        int y = ++cntt;
        rangeToX(a, b, x, 1, n, rootOut);
        xToRange(y, c, d, 1, n, rootIn);
        addEdge(x, y, 1);
    }

    void bfs01() {
        for (int i = 1; i <= cntt; i++) {
            dist[i] = INF;
        }
        dist[p] = 0;
        deq.clear();
        deq.push_front(p);
        while (!deq.empty()) {
            int u = deq.front();
            deq.pop_front();
            for (int e = head[u]; e > 0; e = nxt[e]) {
                int v = to[e];
                int w = weight[e];
                if (dist[v] > dist[u] + w) {
                    dist[v] = dist[u] + w;
                    if (w == 0) {
                        deq.push_front(v);
                    } else {
                        deq.push_back(v);
                    }
                }
            }
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m >> p;
        cntt = n;
        rootOut = buildOut(1, n);
        rootIn = buildIn(1, n);
        for (int i = 1, a, b, c, d; i <= m; i++) {
            cin >> a >> b >> c >> d;
            rangeToRange(a, b, c, d);
            rangeToRange(c, d, a, b);
        }
        bfs01();
        for (int i = 1; i <= n; i++) {
            cout << dist[i] << "\n";
        }
        return 0;
    }

二、题目3：道路

n 到 5 × 10⁴，m 到 10⁵，另外给一个 k，1 ≤ k ≤ 10。每次操作 a b c d w，[a, b] 里每个点与 [c, d] 里每个点之间加一条边权 w 的无向边。你有 k 次机会，每次经过一条边时可以不出这条边的代价。求 1 号点到 n 号点的最低代价。

区间连区间的部分和上一题一样，还是 u、v 中转，只是边权换成 w。真正新增的是「免费」这件事。

k 很小，所以把已经用掉的免费次数记成状态的一维。用 dist[v][time] 表示走到 v、已经免了 time 次的最小代价。

转移有两条。正常走一条边 (u, v)，用 dist[u][time] + w 更新 dist[v][time]；用一次免费，就是 time < k 时用 dist[u][time] 直接更新 dist[v][time + 1]，代价不加 w。

状态总数 O(n k)，在状态图上跑一趟 Dijkstra，复杂度 O(m k log(n k))。

分层 Dijkstra 的核心转移：

int dijkstra(int start, int target) {
        while (!heap.empty()) {
            heap.pop();
        }
        for (int i = 1; i <= cntt; i++) {
            for (int j = 0; j <= k; j++) {
                dist[i][j] = INF;
                vis[i][j] = false;
            }
        }
        dist[start][0] = 0;
        heap.push({start, 0, 0});
        while (!heap.empty()) {
            Node cur = heap.top();
            heap.pop();
            int node = cur.node;
            int time = cur.time;
            int cost = cur.cost;
            if (!vis[node][time]) {
                vis[node][time] = true;
                if (node == target) {
                    return cost;
                }
                for (int e = head[node]; e > 0; e = nxt[e]) {
                    int v = to[e];
                    int w = weight[e];
                    if (!vis[v][time] && dist[v][time] > cost + w) {
                        dist[v][time] = cost + w;
                        heap.push({v, time, dist[v][time]});
                    }
                    if (time < k && !vis[v][time + 1] && dist[v][time + 1] > cost) {
                        dist[v][time + 1] = cost;
                        heap.push({v, time + 1, dist[v][time + 1]});
                    }
                }
            }
        }
        return -1;
    }

三、题目4：沙漠

n 个数字，每个都在 1 到 10⁹ 之间。先给 s 条设定，第 x 个数确定是 v。再给 m 条关系，格式是 l r k x1 x2 … xk：第 l 到第 r 个数里有 k 个特殊的数，位置是 x1 到 xk，它们中的每一个都严格大于区间里其余的数。要求找一组没有矛盾的赋值，任意一组合法方案都可以，无解输出 NIE。

这是「值的大小」上的约束，不是最短路。把约束建成一张有向图，一条边 u→v 带权 w 表示 value[v] ≤ value[u] + w，后面用拓扑排序推每个数的上界。

每条关系建一个辅助点 vnode。每个特殊位置 x 连一条 x→vnode 的 0 边，于是 vnode 的值不超过每个特殊位置的值。vnode 再向区间里除它们之外的每个数连一条边权 -1 的边，这些数的值都被压到 vnode 的值减 1 以下。

「vnode 连向区间里的其余数」是区间连点，用线段树下降完成；其余部分被 x1 到 xk 切成前后几段，都是同样的 xToRange。这样每条关系加 O(k) 条边乘上 log n。

剩下的交给拓扑排序。按拓扑序推上界，atMost[v] 取所有前驱 atMost[u] + w 的最小值。给定定值的数当作初始上界；某个数被压到小于 1，或者小于它自己的定值，就无解。图上有环、拓扑排不完，同样无解。

排完之后 atMost[i] 就是一组合法赋值。值域 10⁹ 只当初始上界用，不参与建图规模。

完整代码：

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 100001;
    const int MAXT = MAXN * 5;
    const int MAXE = MAXN * 20;
    const int LIMIT = 1000000000;
    int n, s, m;

    int head[MAXT];
    int nxt[MAXE];
    int to[MAXE];
    int weight[MAXE];
    int cntg;

    int ls[MAXT];
    int rs[MAXT];
    int root;
    int cntt;

    int val[MAXT];
    int atMost[MAXT];
    int indegree[MAXT];
    int que[MAXT];

    void addEdge(int u, int v, int w) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        weight[cntg] = w;
        head[u] = cntg;
        indegree[v]++;
    }

    int build(int l, int r) {
        int rt;
        if (l == r) {
            rt = l;
        } else {
            rt = ++cntt;
            int mid = (l + r) >> 1;
            ls[rt] = build(l, mid);
            rs[rt] = build(mid + 1, r);
            addEdge(rt, ls[rt], 0);
            addEdge(rt, rs[rt], 0);
        }
        return rt;
    }

    void xToRange(int jobx, int jobl, int jobr, int jobw, int l, int r, int i) {
        if (jobl > jobr) {
            return;
        }
        if (jobl <= l && r <= jobr) {
            addEdge(jobx, i, jobw);
        } else {
            int mid = (l + r) >> 1;
            if (jobl <= mid) {
                xToRange(jobx, jobl, jobr, jobw, l, mid, ls[i]);
            }
            if (jobr > mid) {
                xToRange(jobx, jobl, jobr, jobw, mid + 1, r, rs[i]);
            }
        }
    }

    bool topo() {
        int qi = 1, qsiz = 0;
        for (int i = 1; i <= cntt; i++) {
            if (indegree[i] == 0) {
                que[++qsiz] = i;
            }
            atMost[i] = val[i] == 0 ? LIMIT : val[i];
        }
        while (qi <= qsiz) {
            int u = que[qi++];
            for (int e = head[u]; e > 0; e = nxt[e]) {
                int v = to[e];
                int w = weight[e];
                if (atMost[v] > atMost[u] + w) {
                    atMost[v] = atMost[u] + w;
                    if ((val[v] != 0 && atMost[v] < val[v]) || atMost[v] < 1) {
                        return false;
                    }
                }
                if (--indegree[v] == 0) {
                    que[++qsiz] = v;
                }
            }
        }
        return qsiz == cntt;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> s >> m;
        cntt = n;
        root = build(1, n);
        for (int i = 1; i <= s; i++) {
            int x, v;
            cin >> x >> v;
            val[x] = v;
        }
        for (int i = 1; i <= m; i++) {
            int l, r, k;
            cin >> l >> r >> k;
            int vnode = ++cntt;
            for (int j = 1; j <= k; j++) {
                int x;
                cin >> x;
                addEdge(x, vnode, 0);
                xToRange(vnode, l, x - 1, -1, 1, n, root);
                l = x + 1;
            }
            xToRange(vnode, l, r, -1, 1, n, root);
        }
        bool check = topo();
        if (check) {
            cout << "TAK" << "\n";
            for (int i = 1; i <= n; i++) {
                cout << atMost[i] << " ";
            }
            cout << "\n";
        } else {
            cout << "NIE" << "\n";
        }
        return 0;
    }

四、题目5：炸弹

n 到 5 × 10⁵ 个炸弹排在一条直线上，第 i 个坐标 xi、爆炸半径 ri，坐标严格递增。炸弹 A 引爆时，落在 A 影响范围内的炸弹 B 也会爆，连锁下去。记 query(i) 为 i 作为起爆点时最终引爆的炸弹数，求 Σ i × query(i) 对 10⁹ + 7 取余。

i 能直接引爆的是一段连续下标区间。用二分定位 [xi − ri, xi + ri] 覆盖的下标范围 [l, r]，再让 i 向 [l, r] 里每个点连边。这是单点连区间，用出树上的 rangeToX，一次 O(log n) 条边。

几个炸弹互相引爆会成环，环上任意一个都能到整环，所以先 Tarjan 缩点。

可以发现，缩点后每个 SCC 能到达的炸弹仍是一段连续下标区间，记成 mostl、mostr。在 DAG 上按逆拓扑序合并：mostl[v] 取 min、mostr[v] 取 max。Tarjan 给出的 SCC 编号本身就是逆拓扑序，直接从 sccCnt 倒着扫即可。

最后 query(i) = mostr[scc(i)] − mostl[scc(i)] + 1。代码里把原始边额外存了一份，缩点后重建 DAG 用，因为 Tarjan 已经把原图邻接表的 head 改掉了。

完整代码：

#include <bits/stdc++.h>

    using namespace std;

    using ll = long long;

    const int MAXN = 500001;
    const int MAXT = MAXN * 5;
    const int MAXE = MAXN * 20;
    const int INF = 1 << 30;
    const int MOD = 1000000007;
    int n;
    ll location[MAXN];
    ll radius[MAXN];

    int a[MAXE];
    int b[MAXE];
    int cnte;

    int head[MAXT];
    int nxt[MAXE];
    int to[MAXE];
    int cntg;

    int rangel[MAXT];
    int ranger[MAXT];
    int ls[MAXT];
    int rs[MAXT];
    int root;
    int cntt;

    int dfn[MAXT];
    int low[MAXT];
    int cntd;
    int sta[MAXT];
    int top;

    int belong[MAXT];
    int mostl[MAXT];
    int mostr[MAXT];
    int sccCnt;

    void addEdge(int u, int v) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        head[u] = cntg;
    }

    void saveEdge(int u, int v) {
        a[++cnte] = u;
        b[cnte] = v;
    }

    int lower(ll num) {
        int l = 1, r = n, mid, ans = n + 1;
        while (l <= r) {
            mid = (l + r) >> 1;
            if (location[mid] >= num) {
                ans = mid;
                r = mid - 1;
            } else {
                l = mid + 1;
            }
        }
        return ans;
    }

    int build(int l, int r) {
        int rt;
        if (l == r) {
            rt = l;
        } else {
            rt = ++cntt;
            int mid = (l + r) >> 1;
            ls[rt] = build(l, mid);
            rs[rt] = build(mid + 1, r);
            addEdge(ls[rt], rt);
            addEdge(rs[rt], rt);
            saveEdge(ls[rt], rt);
            saveEdge(rs[rt], rt);
        }
        rangel[rt] = l;
        ranger[rt] = r;
        return rt;
    }

    void rangeToX(int jobl, int jobr, int jobx, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            addEdge(i, jobx);
            saveEdge(i, jobx);
        } else {
            int mid = (l + r) >> 1;
            if (jobl <= mid) {
                rangeToX(jobl, jobr, jobx, l, mid, ls[i]);
            }
            if (jobr > mid) {
                rangeToX(jobl, jobr, jobx, mid + 1, r, rs[i]);
            }
        }
    }

    void tarjan(int u) {
        dfn[u] = low[u] = ++cntd;
        sta[++top] = u;
        for (int e = head[u]; e > 0; e = nxt[e]) {
            int v = to[e];
            if (dfn[v] == 0) {
                tarjan(v);
                low[u] = min(low[u], low[v]);
            } else {
                if (belong[v] == 0) {
                    low[u] = min(low[u], dfn[v]);
                }
            }
        }
        if (dfn[u] == low[u]) {
            sccCnt++;
            mostl[sccCnt] = INF;
            mostr[sccCnt] = -INF;
            int pop;
            do {
                pop = sta[top--];
                belong[pop] = sccCnt;
                mostl[sccCnt] = min(mostl[sccCnt], rangel[pop]);
                mostr[sccCnt] = max(mostr[sccCnt], ranger[pop]);
            } while (pop != u);
        }
    }

    void condense() {
        cntg = 0;
        for (int i = 1; i <= sccCnt; i++) {
            head[i] = 0;
        }
        for (int i = 1; i <= cnte; i++) {
            int scc1 = belong[a[i]];
            int scc2 = belong[b[i]];
            if (scc1 != scc2) {
                addEdge(scc1, scc2);
            }
        }
    }

    void dpOnDAG() {
        for (int u = sccCnt; u > 0; u--) {
            for (int e = head[u]; e > 0; e = nxt[e]) {
                int v = to[e];
                mostl[v] = min(mostl[v], mostl[u]);
                mostr[v] = max(mostr[v], mostr[u]);
            }
        }
    }

    int query(int u) {
        int scc = belong[u];
        int num = mostr[scc] - mostl[scc] + 1;
        return num;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n;
        cntt = n;
        for (int i = 1; i <= n; i++) {
            cin >> location[i] >> radius[i];
        }
        root = build(1, n);
        for (int i = 1; i <= n; i++) {
            int l = lower(location[i] - radius[i]);
            int r = lower(location[i] + radius[i] + 1) - 1;
            rangeToX(l, r, i, 1, n, root);
        }
        for (int i = 1; i <= cntt; i++) {
            if (dfn[i] == 0) {
                tarjan(i);
            }
        }
        condense();
        dpOnDAG();
        ll ans = 0;
        for (int i = 1; i <= n; i++) {
            ans = (ans + 1LL * query(i) * i) % MOD;
        }
        cout << ans << "\n";
        return 0;
    }

五、题目6：美丽的树（树剖优化建图）

n 个节点的一棵树，1 号是根。要给出 1 到 n 的一个排列赋给节点，满足 m 条关系。关系 1 a b c：a 到 b 路径上值最小的节点必须是 c，保证 c 在路径上；关系 2 a b c：路径上值最大的必须是 c。存在方案就输出方案，否则输出 -1。n、m 到 2 × 10⁵。

「路径上最小的必须是 c」拆成一组偏序：c 必须小于路径上其它每个点，也就是路径上其余各点都连向 c。「最大的必须是 c」反过来，c 连向其余各点。

又是路径连点，只是「区间」变成了树上的一段路径。树链剖分正好用在这里：一条路径被拆成 O(log n) 段，每段在 dfn 序上是一段连续区间。

于是把入树、出树的下标从「数组下标」换成「dfn 序号」，其余照搬。建树方式和上一页完全一致。

具体到代码，pathSet 处理一段 dfn 区间到单点、或单点到一段 dfn 区间的连边；link 沿重链往上跳，把整条路径拆成若干 dfn 区间交给 pathSet。这里有个细节：c 自己也在路径上，要把它从区间里挖掉、单独处理，所以 pathSet 里对 c 两侧各拆一次。

建完图跑拓扑排序，按出队顺序给每个真实节点编号 1 到 n，就是一组解；有环输出 -1。

路径拆段与连边的核心：

void pathSet(int op, int x, int y, int z) {
        if (op == 1) {
            if (x <= z && z <= y) {
                xToRange(z, x, z - 1, 1, n, rootIn);
                xToRange(z, z + 1, y, 1, n, rootIn);
            } else {
                xToRange(z, x, y, 1, n, rootIn);
            }
        } else {
            if (x <= z && z <= y) {
                rangeToX(x, z - 1, z, 1, n, rootOut);
                rangeToX(z + 1, y, z, 1, n, rootOut);
            } else {
                rangeToX(x, y, z, 1, n, rootOut);
            }
        }
    }

    void link(int op, int a, int b, int c) {
        while (top[a] != top[b]) {
            if (dep[top[a]] < dep[top[b]]) {
                swap(a, b);
            }
            pathSet(op, dfn[top[a]], dfn[a], dfn[c]);
            a = fa[top[a]];
        }
        pathSet(op, min(dfn[a], dfn[b]), max(dfn[a], dfn[b]), dfn[c]);
    }

六、题目7：美丽的树（倍增优化建图）

和上一题是同一道题，换一种拆路径的姿势。树剖把路径拆成 O(log n) 段，每段还要再靠入树、出树连边，单次操作是 O(log² n)。倍增表可以直接把路径拆成 O(log n) 个区间点，单次操作降到 O(log n)。

给每个节点开两张表。出表 stout[u][p] 代表从 u 往上连续 2^p 个节点这一段，用于区间连点；入表 stin[u][p] 代表同一段，用于点连区间。stout[u][0] 是 u 和它父亲两个点。

合并方式：出表由两个 2^(p−1) 的表点连到 2^p 的表点，入表反过来。最底层的单点就是真实节点，编号共用。

于是路径 a 到 b 被拆成 O(log n) 个长度为 2 的幂的向上区间，每个区间对应一个表点，连边 O(log n) 条。空间 O(n log n)，单次操作 O(log n)，常数比树剖大，但不用线段树。

完整代码：

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 200001;
    const int MAXT = MAXN * 41;
    const int MAXE = MAXN * 201;
    const int MAXP = 17;
    int n, m;

    int head1[MAXN];
    int next1[MAXN << 1];
    int to1[MAXN << 1];
    int cnt1;

    int indegree[MAXT];
    int head2[MAXT];
    int next2[MAXE];
    int to2[MAXE];
    int cnt2;

    int dep[MAXN];
    int dfn[MAXN];
    int siz[MAXN];
    int stjump[MAXN][MAXP];
    int cntd;

    int stout[MAXN][MAXP];
    int stin[MAXN][MAXP];
    int cntt;

    int que[MAXT];
    int ans[MAXN];

    void addEdge1(int u, int v) {
        next1[++cnt1] = head1[u];
        to1[cnt1] = v;
        head1[u] = cnt1;
    }

    void addEdge2(int u, int v) {
        indegree[v]++;
        next2[++cnt2] = head2[u];
        to2[cnt2] = v;
        head2[u] = cnt2;
    }

    void build(int u, int fa) {
        dep[u] = dep[fa] + 1;
        dfn[u] = ++cntd;
        siz[u] = 1;
        stjump[u][0] = fa;
        stout[u][0] = ++cntt;
        addEdge2(u, cntt);
        addEdge2(fa, cntt);
        stin[u][0] = ++cntt;
        addEdge2(cntt, u);
        addEdge2(cntt, fa);
        for (int p = 1; p < MAXP; p++) {
            stjump[u][p] = stjump[stjump[u][p - 1]][p - 1];
            stout[u][p] = ++cntt;
            addEdge2(stout[u][p - 1], cntt);
            addEdge2(stout[stjump[u][p - 1]][p - 1], cntt);
            stin[u][p] = ++cntt;
            addEdge2(cntt, stin[u][p - 1]);
            addEdge2(cntt, stin[stjump[u][p - 1]][p - 1]);
        }
        for (int e = head1[u]; e > 0; e = next1[e]) {
            int v = to1[e];
            if (v != fa) {
                build(v, u);
                siz[u] += siz[v];
            }
        }
    }

    bool isAncestor(int a, int b) {
        return dfn[a] <= dfn[b] && dfn[b] < dfn[a] + siz[a];
    }

    int kthAncestor(int x, int k) {
        for (int p = 0; p < MAXP; p++) {
            if (((k >> p) & 1) != 0) {
                x = stjump[x][p];
            }
        }
        return x;
    }

    int nearest(int x, int y) {
        if (isAncestor(y, x)) {
            return kthAncestor(x, dep[x] - dep[y] - 1);
        } else {
            return stjump[y][0];
        }
    }

    void pathOut(int x, int y, int c) {
        if (dep[x] < dep[y]) {
            swap(x, y);
        }
        addEdge2(y, c);
        for (int p = MAXP - 1; p >= 0; p--) {
            if (dep[stjump[x][p]] >= dep[y]) {
                addEdge2(stout[x][p], c);
                x = stjump[x][p];
            }
        }
        if (x == y) {
            return;
        }
        for (int p = MAXP - 1; p >= 0; p--) {
            if (stjump[x][p] != stjump[y][p]) {
                addEdge2(stout[x][p], c);
                addEdge2(stout[y][p], c);
                x = stjump[x][p];
                y = stjump[y][p];
            }
        }
        addEdge2(stout[x][0], c);
    }

    void pathIn(int x, int y, int c) {
        if (dep[x] < dep[y]) {
            swap(x, y);
        }
        addEdge2(c, y);
        for (int p = MAXP - 1; p >= 0; p--) {
            if (dep[stjump[x][p]] >= dep[y]) {
                addEdge2(c, stin[x][p]);
                x = stjump[x][p];
            }
        }
        if (x == y) {
            return;
        }
        for (int p = MAXP - 1; p >= 0; p--) {
            if (stjump[x][p] != stjump[y][p]) {
                addEdge2(c, stin[x][p]);
                addEdge2(c, stin[y][p]);
                x = stjump[x][p];
                y = stjump[y][p];
            }
        }
        addEdge2(c, stin[x][0]);
    }

    void pathMin(int a, int b, int c) {
        if (a != c) {
            pathIn(a, nearest(a, c), c);
        }
        if (b != c) {
            pathIn(b, nearest(b, c), c);
        }
    }

    void pathMax(int a, int b, int c) {
        if (a != c) {
            pathOut(a, nearest(a, c), c);
        }
        if (b != c) {
            pathOut(b, nearest(b, c), c);
        }
    }

    bool topo() {
        int qi = 1, qsiz = 0;
        for (int i = 1; i <= cntt; i++) {
            if (indegree[i] == 0) {
                que[++qsiz] = i;
            }
        }
        int val = 0;
        while (qi <= qsiz) {
            int u = que[qi++];
            if (u <= n) {
                ans[u] = ++val;
            }
            for (int e = head2[u]; e > 0; e = next2[e]) {
                int v = to2[e];
                if (--indegree[v] == 0) {
                    que[++qsiz] = v;
                }
            }
        }
        return qsiz == cntt;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        cntt = n;
        for (int i = 1, u, v; i < n; i++) {
            cin >> u >> v;
            addEdge1(u, v);
            addEdge1(v, u);
        }
        build(1, 1);
        for (int i = 1, op, a, b, c; i <= m; i++) {
            cin >> op >> a >> b >> c;
            if (op == 1) {
                pathMin(a, b, c);
            } else {
                pathMax(a, b, c);
            }
        }
        bool check = topo();
        if (check) {
            for (int i = 1; i <= n; i++) {
                cout << ans[i] << " ";
            }
        } else {
            cout << -1 << "\n";
        }
        return 0;
    }

七、题目8：逛森林

n 到 5 × 10⁴，m 到 10⁶。初始没有边，两种操作。操作 2 u v w：u、v 已经连通就忽略，否则加一条边权 w 的无向边。操作 1 a b c d w：如果 a、b 不连通，或者 c、d 不连通，忽略；否则 a 到 b 路径上任意一点，可以花 w 传送到 c 到 d 路径上任意一点。传送是单向的，只相当于加一条「从一段路径到另一段路径」的边，不改变连通性。求起点 s 到每个点的最小花费。

操作 1 是否生效，取决于执行到它时操作 2 已经连出的连通块。所以先读入所有操作、用并查集顺序扫一遍，把生效的操作 1 记下来，同时也把最终的森林建出来。

森林建好后，用倍增表把每棵树建成优化图。生效的操作 1 就是路径连路径：新建两个辅助点 u、v，出表把 a 到 b 这段路径收到 u，u 连 v 带权 w，入表把 v 散到 c 到 d 这段路径。

最后从 s 跑 Dijkstra。点 O(n log n)，边 O(m log n)。

完整代码：

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 50001;
    const int MAXM = 1000001;
    const int MAXT = 4000001;
    const int MAXE = 30000001;
    const int MAXP = 17;
    const int INF = 1 << 30;
    int n, m, s;

    int u1[MAXM];
    int v1[MAXM];
    int u2[MAXM];
    int v2[MAXM];
    int weight[MAXM];
    int cntq;

    int head1[MAXN];
    int next1[MAXN << 1];
    int to1[MAXN << 1];
    int cnt1;

    int head2[MAXT];
    int next2[MAXE];
    int to2[MAXE];
    int weight2[MAXE];
    int cnt2;

    int father[MAXN];

    int dep[MAXN];
    int stjump[MAXN][MAXP];
    int stout[MAXN][MAXP];
    int stin[MAXN][MAXP];
    int cntt;

    int dist[MAXT];
    bool vis[MAXT];

    struct Node {
        int u;
        int d;

        bool operator < (const Node &other) const {
            return d > other.d;
        }
    };

    priority_queue<Node> heap;

    void addEdge1(int u, int v) {
        next1[++cnt1] = head1[u];
        to1[cnt1] = v;
        head1[u] = cnt1;
    }

    void addEdge2(int u, int v, int w) {
        next2[++cnt2] = head2[u];
        to2[cnt2] = v;
        weight2[cnt2] = w;
        head2[u] = cnt2;
    }

    int find(int i) {
        if (i != father[i]) {
            father[i] = find(father[i]);
        }
        return father[i];
    }

    void build(int u, int fa) {
        dep[u] = dep[fa] + 1;
        stjump[u][0] = fa;
        stout[u][0] = ++cntt;
        addEdge2(u, cntt, 0);
        addEdge2(fa, cntt, 0);
        stin[u][0] = ++cntt;
        addEdge2(cntt, u, 0);
        addEdge2(cntt, fa, 0);
        for (int p = 1; p < MAXP; p++) {
            stjump[u][p] = stjump[stjump[u][p - 1]][p - 1];
            stout[u][p] = ++cntt;
            addEdge2(stout[u][p - 1], cntt, 0);
            addEdge2(stout[stjump[u][p - 1]][p - 1], cntt, 0);
            stin[u][p] = ++cntt;
            addEdge2(cntt, stin[u][p - 1], 0);
            addEdge2(cntt, stin[stjump[u][p - 1]][p - 1], 0);
        }
        for (int e = head1[u]; e > 0; e = next1[e]) {
            int v = to1[e];
            if (v != fa) {
                build(v, u);
            }
        }
    }

    void pathOut(int x, int y, int vnode) {
        if (dep[x] < dep[y]) {
            swap(x, y);
        }
        addEdge2(y, vnode, 0);
        for (int p = MAXP - 1; p >= 0; p--) {
            if (dep[stjump[x][p]] >= dep[y]) {
                addEdge2(stout[x][p], vnode, 0);
                x = stjump[x][p];
            }
        }
        if (x == y) {
            return;
        }
        for (int p = MAXP - 1; p >= 0; p--) {
            if (stjump[x][p] != stjump[y][p]) {
                addEdge2(stout[x][p], vnode, 0);
                addEdge2(stout[y][p], vnode, 0);
                x = stjump[x][p];
                y = stjump[y][p];
            }
        }
        addEdge2(stout[x][0], vnode, 0);
    }

    void pathIn(int x, int y, int vnode) {
        if (dep[x] < dep[y]) {
            swap(x, y);
        }
        addEdge2(vnode, y, 0);
        for (int p = MAXP - 1; p >= 0; p--) {
            if (dep[stjump[x][p]] >= dep[y]) {
                addEdge2(vnode, stin[x][p], 0);
                x = stjump[x][p];
            }
        }
        if (x == y) {
            return;
        }
        for (int p = MAXP - 1; p >= 0; p--) {
            if (stjump[x][p] != stjump[y][p]) {
                addEdge2(vnode, stin[x][p], 0);
                addEdge2(vnode, stin[y][p], 0);
                x = stjump[x][p];
                y = stjump[y][p];
            }
        }
        addEdge2(vnode, stin[x][0], 0);
    }

    void pathToPath(int a, int b, int c, int d, int w) {
        int x = ++cntt;
        int y = ++cntt;
        pathOut(a, b, x);
        pathIn(c, d, y);
        addEdge2(x, y, w);
    }

    void dijkstra() {
        for (int i = 1; i <= cntt; i++) {
            dist[i] = INF;
        }
        dist[s] = 0;
        heap.push({s, 0});
        while (!heap.empty()) {
            Node cur = heap.top();
            heap.pop();
            int u = cur.u;
            int d = cur.d;
            if (!vis[u]) {
                vis[u] = true;
                for (int e = head2[u]; e > 0; e = next2[e]) {
                    int v = to2[e];
                    int w = weight2[e];
                    if (!vis[v] && dist[v] > d + w) {
                        dist[v] = d + w;
                        heap.push({v, dist[v]});
                    }
                }
            }
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m >> s;
        cntt = n;
        for (int i = 1; i <= n; i++) {
            father[i] = i;
        }
        for (int i = 1, op, a, b, c, d, w, u, v; i <= m; i++) {
            cin >> op;
            if (op == 1) {
                cin >> a >> b >> c >> d >> w;
                if (find(a) == find(b) && find(c) == find(d)) {
                    u1[++cntq] = a;
                    v1[cntq] = b;
                    u2[cntq] = c;
                    v2[cntq] = d;
                    weight[cntq] = w;
                }
            } else {
                cin >> u >> v >> w;
                int ufa = find(u);
                int vfa = find(v);
                if (ufa != vfa) {
                    addEdge1(u, v);
                    addEdge1(v, u);
                    addEdge2(u, v, w);
                    addEdge2(v, u, w);
                    father[ufa] = vfa;
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            if (dep[i] == 0) {
                build(i, i);
            }
        }
        for (int i = 1; i <= cntq; i++) {
            pathToPath(u1[i], v1[i], u2[i], v2[i], weight[i]);
        }
        dijkstra();
        for (int i = 1; i <= n; i++) {
            cout << (dist[i] == INF ? -1 : dist[i]) << " ";
        }
        return 0;
    }

八、题目9：监狱

n 个房间组成一棵树，m 个囚犯各有卧室 x、工作室 y。卧室互不相同，工作室也互不相同，但一个房间可以是某人的卧室、同时是另一个人的工作室。要让所有囚犯各自只走最短路从卧室走到工作室，任一时刻同一房间里不能有两个人。问能否做到。n、m 到 1.2 × 10⁵。

关键不在建图，而在一个结论：能不能完成，等价于某些「先后顺序」的约束有没有环。把每个房间的经过顺序看成偏序，就能用拓扑排序判定。

把每个囚犯当成一个节点。若某个房间必须先被甲经过、再被乙经过，就在图上连一条有向边；一条路径上的房间被多个囚犯先后用到，也会产生一串顺序约束。

这些约束落在树上的一段段路径之间，单次要连的又是一段路径。仍用倍增表把路径拆成 O(log n) 个区间点：囚犯节点连到这些表点上，或从表点连到囚犯节点。最后跑拓扑排序判环，有环输出 No。

代码里每个房间拆成 startTag、endTag 两个状态点，表示进入和离开；link 处理一个囚犯的 x、y 两个端点，pathSet 处理中间那段路径的先后关系。

路径约束的连边：

void pathSet(int x, int y, int prisoner) {
        if (dep[x] < dep[y]) {
            swap(x, y);
        }
        addEdge2(startTag[y], prisoner);
        addEdge2(prisoner, endTag[y]);
        for (int p = MAXP - 1; p >= 0; p--) {
            if (dep[stjump[x][p]] >= dep[y]) {
                addEdge2(stout[x][p], prisoner);
                addEdge2(prisoner, stin[x][p]);
                x = stjump[x][p];
            }
        }
        if (x == y) {
            return;
        }
        for (int p = MAXP - 1; p >= 0; p--) {
            if (stjump[x][p] != stjump[y][p]) {
                addEdge2(stout[x][p], prisoner);
                addEdge2(stout[y][p], prisoner);
                addEdge2(prisoner, stin[x][p]);
                addEdge2(prisoner, stin[y][p]);
                x = stjump[x][p];
                y = stjump[y][p];
            }
        }
        addEdge2(stout[x][0], prisoner);
        addEdge2(prisoner, stin[x][0]);
    }

    void link(int x, int y) {
        int prisoner = ++cntt;
        addEdge2(prisoner, startTag[x]);
        addEdge2(prisoner, endTag[x]);
        addEdge2(startTag[y], prisoner);
        addEdge2(endTag[y], prisoner);
        if (stjump[x][0] != y && stjump[y][0] != x) {
            int a = nearest(y, x);
            int b = nearest(x, y);
            pathSet(a, b, prisoner);
        }
    }
