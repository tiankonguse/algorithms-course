线段树分治：整理边的有效时间段

模版题里时间段是白给的，更多时候题目只给一串操作，每条边什么时候生效要自己整理。这一页三道题，按点对整理、按边整理、按边整理再加上「判定失败要回退」。

零、从操作序列里找时间段

线段树分治的入口永远是同一句话：把每条修改变成「轴上的哪几段生效」。时间不给，就从操作序列里推。套路是按对象分组：同一对点、同一条边的操作排在一起，两次出现之间就是生效区间。

一、题目5：大融合

n 个点，q 条操作。A x y：给 x 和 y 连边，保证之前不连通；Q x y：打印边 x-y 的负载，保证这条边存在。负载定义为这条边两侧连通区大小的乘积。1 ≤ n、q ≤ 10⁵。测试链接：P4219。

查询时把边 x-y 摘掉，负载就是 x 那侧的块大小乘以 y 那侧的块大小。这是线段树分治最顺手的形式：对每对点 (x, y)，把它的所有操作按时间排序。第一次出现必然是 A（题目保证 A 之前不连通），之后全是 Q。边在两次相邻出现之间都存在，唯独 Q 的那个时刻要空出来，因为查询时这条边被摘掉了。

排序按 (x, y, t) 三元组做，同组内部按时间升序，然后线性扫一遍就能把区间全挂上去。叶子上做查询：ans = siz[find(x)] × siz[find(y)]。

#include <bits/stdc++.h>

    using namespace std;

    struct Event {
        int x, y, t;
    };

    bool EventCmp(Event a, Event b) {
        if (a.x != b.x) {
            return a.x < b.x;
        } else if (a.y != b.y) {
            return a.y < b.y;
        } else {
            return a.t < b.t;
        }
    }

    const int MAXN = 100001;
    const int MAXT = 3000001;
    int n, q;

    int op[MAXN];
    int u[MAXN];
    int v[MAXN];

    Event event[MAXN];

    int father[MAXN];
    int siz[MAXN];
    int rollback[MAXN][2];
    int opsize = 0;

    int head[MAXN << 2];
    int nxt[MAXT];
    int tox[MAXT];
    int toy[MAXT];
    int cnt = 0;

    long long ans[MAXN];

    void addEdge(int i, int x, int y) {
        nxt[++cnt] = head[i];
        tox[cnt] = x;
        toy[cnt] = y;
        head[i] = cnt;
    }

    int find(int i) {
        while (i != father[i]) {
            i = father[i];
        }
        return i;
    }

    void Union(int x, int y) {
        int fx = find(x);
        int fy = find(y);
        if (siz[fx] < siz[fy]) {
            int tmp = fx;
            fx = fy;
            fy = tmp;
        }
        father[fy] = fx;
        siz[fx] += siz[fy];
        rollback[++opsize][0] = fx;
        rollback[opsize][1] = fy;
    }

    void undo() {
        int fx = rollback[opsize][0];
        int fy = rollback[opsize--][1];
        father[fy] = fy;
        siz[fx] -= siz[fy];
    }

    void add(int jobl, int jobr, int jobx, int joby, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            addEdge(i, jobx, joby);
        } else {
            int mid = (l + r) >> 1;
            if (jobl <= mid) {
                add(jobl, jobr, jobx, joby, l, mid, i << 1);
            }
            if (jobr > mid) {
                add(jobl, jobr, jobx, joby, mid + 1, r, i << 1 | 1);
            }
        }
    }

    void dfs(int l, int r, int i) {
        int unionCnt = 0;
        for (int ei = head[i]; ei > 0; ei = nxt[ei]) {
            Union(tox[ei], toy[ei]);
            unionCnt++;
        }
        if (l == r) {
            if (op[l] == 2) {
                ans[l] = 1LL * siz[find(u[l])] * siz[find(v[l])];
            }
        } else {
            int mid = (l + r) >> 1;
            dfs(l, mid, i << 1);
            dfs(mid + 1, r, i << 1 | 1);
        }
        for (int k = 1; k <= unionCnt; k++) {
            undo();
        }
    }

    void prepare() {
        for (int i = 1; i <= n; i++) {
            father[i] = i;
            siz[i] = 1;
        }
        for (int i = 1; i <= q; i++) {
            event[i].x = u[i];
            event[i].y = v[i];
            event[i].t = i;
        }
        sort(event + 1, event + q + 1, EventCmp);
        for (int l = 1, r = 1; l <= q; l = ++r) {
            int x = event[l].x, y = event[l].y, t = event[l].t;
            while (r + 1 <= q && event[r + 1].x == x && event[r + 1].y == y) {
                r++;
            }
            for (int j = l + 1; j <= r; j++) {
                add(t, event[j].t - 1, x, y, 1, q, 1);
                t = event[j].t + 1;
            }
            if (t <= q) {
                add(t, q, x, y, 1, q, 1);
            }
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> q;
        char t;
        int x, y;
        for (int i = 1; i <= q; i++) {
            cin >> t >> x >> y;
            op[i] = (t == 'A') ? 1 : 2;
            u[i] = min(x, y);
            v[i] = max(x, y);
        }
        prepare();
        dfs(1, q, 1);
        for (int i = 1; i <= q; i++) {
            if (op[i] == 2) {
                cout << ans[i] << '\n';
            }
        }
        return 0;
    }

二、题目6：连通图

n 个点、m 条边的无向图，初始连通。k 条操作，每条操作挑出最多 4 条边，问假设删掉这些边图是否连通。每条操作独立，互不影响。1 ≤ n、k ≤ 10⁵，1 ≤ m ≤ 2 × 10⁵。测试链接：P5227。

也就是说不删任何边时图始终连通，某条操作下不连通，当且仅当「删掉的边里盖住了所有的割口」。对每条边整理它的失效时间点：操作 i 删了边 e，那么对 e 来说 i 是失效时刻，其余 k 个时刻它都生效。

做法和上一题一个模子：把 (边编号，操作时间) 的二元组按边分组排序。某条边在操作 t1、t2、… 依次被删，它生效的区间是 [1, t1-1]、[t1+1, t2-1]、…、[最后+1, k]。从没出现过的边全程生效，别忘了一起挂上。

叶子上判定全图是否连通。这里有个值得学的剪枝：进某个节点时，如果并完这个节点的任务后图已经全连通，那么它覆盖的每个时刻都连通（节点任务是叶子任务的子集），整段叶子直接标 Connected，不用再往下走。

#include <bits/stdc++.h>

    using namespace std;

    struct Event {
        int ei, t;
    };

    bool EventCmp(Event a, Event b) {
        if (a.ei != b.ei) {
            return a.ei < b.ei;
        } else {
            return a.t < b.t;
        }
    }

    const int MAXN = 100001;
    const int MAXM = 200001;
    const int MAXE = 400001;
    const int MAXT = 10000001;

    int n, m, k;

    int u[MAXM];
    int v[MAXM];

    Event event[MAXE];
    int ecnt = 0;
    bool vis[MAXM];

    int father[MAXN];
    int siz[MAXN];
    int rollback[MAXN][2];
    int opsize = 0;

    int head[MAXN << 2];
    int nxt[MAXT];
    int tox[MAXT];
    int toy[MAXT];
    int cnt = 0;

    bool ans[MAXN];

    void addEdge(int i, int x, int y) {
        nxt[++cnt] = head[i];
        tox[cnt] = x;
        toy[cnt] = y;
        head[i] = cnt;
    }

    int find(int i) {
        while (i != father[i]) {
            i = father[i];
        }
        return i;
    }

    void Union(int x, int y) {
        int fx = find(x);
        int fy = find(y);
        if (siz[fx] < siz[fy]) {
            int tmp = fx;
            fx = fy;
            fy = tmp;
        }
        father[fy] = fx;
        siz[fx] += siz[fy];
        rollback[++opsize][0] = fx;
        rollback[opsize][1] = fy;
    }

    void undo() {
        int fx = rollback[opsize][0];
        int fy = rollback[opsize--][1];
        father[fy] = fy;
        siz[fx] -= siz[fy];
    }

    void add(int jobl, int jobr, int jobx, int joby, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            addEdge(i, jobx, joby);
        } else {
            int mid = (l + r) >> 1;
            if (jobl <= mid) {
                add(jobl, jobr, jobx, joby, l, mid, i << 1);
            }
            if (jobr > mid) {
                add(jobl, jobr, jobx, joby, mid + 1, r, i << 1 | 1);
            }
        }
    }

    void dfs(int l, int r, int i) {
        bool check = false;
        int unionCnt = 0;
        for (int ei = head[i]; ei > 0; ei = nxt[ei]) {
            int x = tox[ei], y = toy[ei], fx = find(x), fy = find(y);
            if (fx != fy) {
                Union(fx, fy);
                unionCnt++;
            }
            if (siz[find(fx)] == n) {
                check = true;
                break;
            }
        }
        if (check) {
            for (int j = l; j <= r; j++) {
                ans[j] = true;
            }
        } else {
            if (l == r) {
                ans[l] = false;
            } else {
                int mid = (l + r) >> 1;
                dfs(l, mid, i << 1);
                dfs(mid + 1, r, i << 1 | 1);
            }
        }
        for (int j = 1; j <= unionCnt; j++) {
            undo();
        }
    }

    void prepare() {
        for (int i = 1; i <= n; i++) {
            father[i] = i;
            siz[i] = 1;
        }
        sort(event + 1, event + ecnt + 1, EventCmp);
        for (int l = 1, r = 1, eid; l <= ecnt; l = ++r) {
            eid = event[l].ei;
            vis[eid] = true;
            while (r + 1 <= ecnt && event[r + 1].ei == eid) {
                r++;
            }
            int t = 1;
            for (int i = l; i <= r; i++) {
                if (t <= event[i].t - 1) {
                    add(t, event[i].t - 1, u[eid], v[eid], 1, k, 1);
                }
                t = event[i].t + 1;
            }
            if (t <= k) {
                add(t, k, u[eid], v[eid], 1, k, 1);
            }
        }
        for (int i = 1; i <= m; i++) {
            if (!vis[i]) {
                add(1, k, u[i], v[i], 1, k, 1);
            }
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1; i <= m; i++) {
            cin >> u[i] >> v[i];
        }
        cin >> k;
        for (int i = 1, c; i <= k; i++) {
            cin >> c;
            for (int j = 1; j <= c; j++) {
                cin >> event[++ecnt].ei;
                event[ecnt].t = i;
            }
        }
        prepare();
        dfs(1, k, 1);
        for (int i = 1; i <= k; i++) {
            if (ans[i]) {
                cout << "Connected" << "\n";
            } else {
                cout << "Disconnected" << "\n";
            }
        }
        return 0;
    }

三、题目7：给边涂色

n 个点、m 条边，k 种颜色，初始无边色。合法状态：只保留任何一种颜色的边，剩下的图都是二分图。q 条操作 e c：把第 e 条边涂成颜色 c。如果涂完合法，执行并输出 YES；否则不执行，输出 NO。1 ≤ n、m、q ≤ 5 × 10⁵，1 ≤ k ≤ 50。测试链接：CF576E。

每个颜色一套扩充域可撤销并查集，k ≤ 50，50 套并查集开得下。涂色操作 e c 对颜色 c 的图来说是一条「在一段时间里存在的边」，问题又回到整理时间段。

整理的规则：把每条边的操作按时间排好。操作 i 给边 e 涂色，这次涂色决定的是 e 从 i+1 到下一次被碰之前的样子。所以生效区间是 [i+1, next(e, i)]，next 用倒序扫一遍就能填出来。时刻 i 本身留给判定用。

判定的时刻在叶子 i 上：任务里没带这条边，但它的两个端点此刻在颜色 c 的并查集里的位置是确定的。若 find(c, x) == find(c, y)，说明涂成 c 会出现奇环，操作失败。

失败之后麻烦来了：这条边保持原来的颜色，后面挂着引用操作 i 的那些任务，颜色全得跟着改。这里有个巧办法：给操作记一个 lastColor（这条边上一次成功涂上的颜色），失败时把 c[i] 改成 lastColor[e]。后面的任务引用的是操作下标，取颜色时取到的是改过的值，自动就对了。dfs 从左往右走，到任何叶子时它之前所有操作的成败都已定案，这个改法是安全的。

初始 lastColor 设为 0：失败的边「无色」，色 0 的并查集爱怎么并怎么并，反正没有任何判定会去查它。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 500001;
    const int MAXK = 51;
    const int MAXT = 10000001;
    int n, m, k, q;

    int u[MAXN];
    int v[MAXN];

    int e[MAXN];
    int c[MAXN];
    int post[MAXN];

    int father[MAXK][MAXN << 1];
    int siz[MAXK][MAXN << 1];
    int rollback[MAXN << 1][3];
    int opsize = 0;

    int head[MAXN << 2];
    int nxt[MAXT];
    int qid[MAXT];
    int cnt = 0;

    int lastColor[MAXN];

    bool ans[MAXN];

    void addEdge(int i, int qi) {
        nxt[++cnt] = head[i];
        qid[cnt] = qi;
        head[i] = cnt;
    }

    int find(int color, int i) {
        while (i != father[color][i]) {
            i = father[color][i];
        }
        return i;
    }

    void Union(int color, int x, int y) {
        int fx = find(color, x);
        int fy = find(color, y);
        if (siz[color][fx] < siz[color][fy]) {
            int tmp = fx;
            fx = fy;
            fy = tmp;
        }
        father[color][fy] = fx;
        siz[color][fx] += siz[color][fy];
        rollback[++opsize][0] = color;
        rollback[opsize][1] = fx;
        rollback[opsize][2] = fy;
    }

    void undo() {
        int color = rollback[opsize][0];
        int fx = rollback[opsize][1];
        int fy = rollback[opsize--][2];
        father[color][fy] = fy;
        siz[color][fx] -= siz[color][fy];
    }

    void add(int jobl, int jobr, int jobq, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            addEdge(i, jobq);
        } else {
            int mid = (l + r) >> 1;
            if (jobl <= mid) {
                add(jobl, jobr, jobq, l, mid, i << 1);
            }
            if (jobr > mid) {
                add(jobl, jobr, jobq, mid + 1, r, i << 1 | 1);
            }
        }
    }

    void dfs(int l, int r, int i) {
        int unionCnt = 0;
        int color, x, y, xn, yn, fx, fy, fxn, fyn;
        for (int ei = head[i]; ei > 0; ei = nxt[ei]) {
            color = c[qid[ei]];
            x = u[e[qid[ei]]];
            y = v[e[qid[ei]]];
            xn = x + n;
            yn = y + n;
            fx = find(color, x);
            fy = find(color, y);
            fxn = find(color, xn);
            fyn = find(color, yn);
            if (fx != fyn) {
                Union(color, fx, fyn);
                unionCnt++;
            }
            if (fy != fxn) {
                Union(color, fy, fxn);
                unionCnt++;
            }
        }
        if (l == r) {
            if (find(c[l], u[e[l]]) == find(c[l], v[e[l]])) {
                ans[l] = false;
                c[l] = lastColor[e[l]];
            } else {
                ans[l] = true;
                lastColor[e[l]] = c[l];
            }
        } else {
            int mid = (l + r) >> 1;
            dfs(l, mid, i << 1);
            dfs(mid + 1, r, i << 1 | 1);
        }
        for (int j = 1; j <= unionCnt; j++) {
            undo();
        }
    }

    void prepare() {
        for (int color = 1; color <= k; color++) {
            for (int i = 1; i <= n; i++) {
                father[color][i] = i;
                father[color][i + n] = i + n;
                siz[color][i] = 1;
                siz[color][i + n] = 1;
            }
        }
        for (int i = 1; i <= m; i++) {
            post[i] = q;
        }
        for (int i = q; i >= 1; i--) {
            if (i + 1 <= post[e[i]]) {
                add(i + 1, post[e[i]], i, 1, q, 1);
            }
            post[e[i]] = i;
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m >> k >> q;
        for (int i = 1; i <= m; i++) {
            cin >> u[i] >> v[i];
        }
        for (int i = 1; i <= q; i++) {
            cin >> e[i] >> c[i];
        }
        prepare();
        dfs(1, q, 1);
        for (int i = 1; i <= q; i++) {
            if (ans[i]) {
                cout << "YES" << "\n";
            } else {
                cout << "NO" << "\n";
            }
        }
        return 0;
    }

四、复杂度分析

三道题的挂任务都来自「分组排序 + 线性扫描」，任务数是操作数的 O(log) 倍。判定全部落在可撤销并查集上。

操作时间复杂度空间复杂度
题目 5：排序 + 挂任务 + 遍历O(q log q log q log n)O(q log q) 任务链表 + 并查集 O(n)
题目 6：排序 + 挂任务 + 遍历O((m + k) log k log n)O(k log k) 任务链表 + 并查集 O(n)
题目 7：挂任务 + 遍历（k 套并查集）O(q log q log n)O(kn + q log q)：k 套扩充域并查集 + 任务链表
一次 Union / undoO(log n)回滚栈 O(任务总数)

题目 7 的任务链表存的是操作下标而不是边，颜色取的是 c[操作下标]，回退改色能生效，靠的就是这一层间接。

五、最后

三道题把「整理时间段」的常见姿势凑齐了：按点对分组、按边分组、按边分组再加一个失败回退。加上前两页的模版和换轴，线段树分治能碰的题型基本就是这些，剩下的都是把题目翻译成「任务 + 生效区间」的功夫。

《完》
