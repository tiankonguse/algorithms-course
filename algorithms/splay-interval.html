Splay 区间操作

把区间整段切出来：find 两次 + splay 两次，剩下要处理的子树就是你要的区间。

零、背景

上一课 Splay 处理的是有序表：按值插、按值查。这一课换成按位置操作：给一个排名区间，翻转它、移动它、批量改值、求和。文艺平衡树、书架、维护数列全靠同一个手法：把区间对应的子树整棵切出来。

切法就三步：find(l−1) 提根，find(r+1) 挂到根的右孩子，于是根的右孩子的左子树正好是中序排名 [l, r] 的那一段。之后对这段子树做什么都行：打懒标记、整体回收、求和，代价都是 O(log n)。

后面的三题都是这个模板的变体。区别只在切出来的那段子树上面做什么。

一、P1486 郁闷的出纳员：偏移量代替逐个改

公司整体加工资、整体扣工资，还随时可能有一批人因为工资低于下限被开除。如果每次整体加减都去改树上每个节点的值，那就 O(n) 了。

这里的技巧是只维护一个偏移量 change：树里存的是相对工资 x − change，真实工资 = 树里的值 + change。整体加工资就是把 change 加一加，树一个节点都不用动。

整体扣工资时，真实工资低于 limit 的段要被砍掉。departure() 算 num = limit − change − 1（比 num 大的真实工资才合格），找第一个 key > num 的节点 splay 到根，把根左子树整段丢掉（head 指到根，左子树作废）。插入时只有真实工资 ≥ limit 的才进树，同时累计入职人数。

查询第 k 大的工资，先看树里人数够不够，不够输出 −1，够就按排名找节点、真实工资 = 树值 + change。最后统计离职人数：入职总数 − 当前树里的人数。

#include <iostream>
    #include <cstring>
    #include <algorithm>
    using namespace std;

    const int MAXN = 300001;

    int head = 0;
    int cnt = 0;
    int key[MAXN];
    int fa[MAXN];
    int ls[MAXN];
    int rs[MAXN];
    int siz[MAXN];
    int limit;
    int change = 0;
    int enter = 0;

    void up(int i) {
        siz[i] = siz[ls[i]] + siz[rs[i]] + 1;
    }

    int lr(int i) {
        return rs[fa[i]] == i ? 1 : 0;
    }

    void rotate(int i) {
        int f = fa[i], g = fa[f], soni = lr(i), sonf = lr(f);
        if (soni == 1) {
            rs[f] = ls[i];
            if (rs[f] != 0) {
                fa[rs[f]] = f;
            }
            ls[i] = f;
        } else {
            ls[f] = rs[i];
            if (ls[f] != 0) {
                fa[ls[f]] = f;
            }
            rs[i] = f;
        }
        if (g != 0) {
            if (sonf == 1) {
                rs[g] = i;
            } else {
                ls[g] = i;
            }
        }
        fa[f] = i;
        fa[i] = g;
        up(f);
        up(i);
    }

    void splay(int i, int goal) {
        int f = fa[i], g = fa[f];
        while (f != goal) {
            if (g != goal) {
                if (lr(i) == lr(f)) {
                    rotate(f);
                } else {
                    rotate(i);
                }
            }
            rotate(i);
            f = fa[i];
            g = fa[f];
        }
        if (goal == 0) {
            head = i;
        }
    }

    void add(int num) {
        key[++cnt] = num;
        siz[cnt] = 1;
        if (head == 0) {
            head = cnt;
        } else {
            int f = 0, i = head, son = 0;
            while (i != 0) {
                f = i;
                if (key[i] <= num) {
                    son = 1;
                    i = rs[i];
                } else {
                    son = 0;
                    i = ls[i];
                }
            }
            if (son == 1) {
                rs[f] = cnt;
            } else {
                ls[f] = cnt;
            }
            fa[cnt] = f;
            splay(cnt, 0);
        }
    }

    int index(int x) {
        int i = head, last = head;
        while (i != 0) {
            last = i;
            if (siz[ls[i]] >= x) {
                i = ls[i];
            } else if (siz[ls[i]] + 1 < x) {
                x -= siz[ls[i]] + 1;
                i = rs[i];
            } else {
                i = 0;
            }
        }
        splay(last, 0);
        return key[last];
    }

    void departure() {
        int num = limit - change - 1;
        int i = head, ans = 0;
        while (i != 0) {
            if (key[i] > num) {
                ans = i;
                i = ls[i];
            } else {
                i = rs[i];
            }
        }
        if (ans == 0) {
            head = 0;
        } else {
            splay(ans, 0);
            ls[head] = 0;
            up(head);
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        int n, x;
        char op;
        cin >> n >> limit;
        for (int i = 1; i <= n; i++) {
            cin >> op >> x;
            if (op == 'I') {
                if (x >= limit) {
                    enter++;
                    add(x - change);
                }
            } else if (op == 'A') {
                change += x;
            } else if (op == 'S') {
                change -= x;
                departure();
            } else if (op == 'F') {
                if (x > siz[head]) {
                    cout << -1 << endl;
                } else {
                    cout << index(siz[head] - x + 1) + change << endl;
                }
            }
        }
        cout << enter - siz[head] << endl;
        return 0;
    }

二、P3391 文艺平衡树：懒翻转

给一个 1 到 n 的序列，翻转若干区间，最后输出结果。直接逐个交换子树里的节点是 O(n)，所以翻转要打懒标记：在子树的根上记 rev，等以后真的需要走进这棵子树时再往下传。

头尾各放一个值为 0 的虚拟节点（排名 1 和 n+2），真实元素占 2 到 n+1。这样翻转区间 [l, r] 时，find(l) 和 find(r+2) 切出来的正好是真实元素那一段，不用处理边界特判。

down 传播时做两件事：rev 异或翻转，同时交换左右孩子。交换孩子这一步很关键：翻转整棵子树等价于交换左右子树，再给两边打上翻转标记，一直递归下去。中序遍历时沿途 down，输出跳过两个虚拟节点。

#include <iostream>

    using namespace std;

    const int MAXN = 100005;

    int head = 0;
    int cnt = 0;
    int num[MAXN];
    int fa[MAXN];
    int ls[MAXN];
    int rs[MAXN];
    int siz[MAXN];
    bool rev[MAXN];
    int sta[MAXN];
    int si;
    int ans[MAXN];
    int ai;

    void up(int i) {
        siz[i] = siz[ls[i]] + siz[rs[i]] + 1;
    }

    int lr(int i) {
        return rs[fa[i]] == i ? 1 : 0;
    }

    void rotate(int i) {
        int f = fa[i], g = fa[f], soni = lr(i), sonf = lr(f);
        if (soni == 1) {
            rs[f] = ls[i];
            if (rs[f] != 0) {
                fa[rs[f]] = f;
            }
            ls[i] = f;
        } else {
            ls[f] = rs[i];
            if (ls[f] != 0) {
                fa[ls[f]] = f;
            }
            rs[i] = f;
        }
        if (g != 0) {
            if (sonf == 1) {
                rs[g] = i;
            } else {
                ls[g] = i;
            }
        }
        fa[f] = i;
        fa[i] = g;
        up(f);
        up(i);
    }

    void splay(int i, int goal) {
        int f = fa[i], g = fa[f];
        while (f != goal) {
            if (g != goal) {
                if (lr(i) == lr(f)) {
                    rotate(f);
                } else {
                    rotate(i);
                }
            }
            rotate(i);
            f = fa[i];
            g = fa[f];
        }
        if (goal == 0) {
            head = i;
        }
    }

    void down(int i) {
        if (rev[i]) {
            rev[ls[i]] = !rev[ls[i]];
            rev[rs[i]] = !rev[rs[i]];
            int tmp = ls[i];
            ls[i] = rs[i];
            rs[i] = tmp;
            rev[i] = false;
        }
    }

    int find(int rank) {
        int i = head;
        while (i != 0) {
            down(i);
            if (siz[ls[i]] + 1 == rank) {
                return i;
            } else if (siz[ls[i]] >= rank) {
                i = ls[i];
            } else {
                rank -= siz[ls[i]] + 1;
                i = rs[i];
            }
        }
        return 0;
    }

    void add(int x) {
        num[++cnt] = x;
        siz[cnt] = 1;
        fa[cnt] = head;
        rs[head] = cnt;
        splay(cnt, 0);
    }

    void reverse(int l, int r) {
        int i = find(l - 1);
        int j = find(r + 1);
        splay(i, 0);
        splay(j, i);
        rev[ls[rs[head]]] = !rev[ls[rs[head]]];
    }

    void inorder(int i) {
        if (i != 0) {
            down(i);
            inorder(ls[i]);
            ans[++ai] = num[i];
            inorder(rs[i]);
        }
    }

    void inorder() {
        si = 0;
        int i = head;
        while (si != 0 || i != 0) {
            if (i != 0) {
                down(i);
                sta[++si] = i;
                i = ls[i];
            } else {
                i = sta[si--];
                ans[++ai] = num[i];
                i = rs[i];
            }
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        int n, m;
        cin >> n >> m;
        add(0);
        for (int i = 1; i <= n; i++) {
            add(i);
        }
        add(0);
        for (int i = 1, x, y; i <= m; i++) {
            cin >> x >> y;
            reverse(x + 1, y + 1);
        }
        ai = 0;
        // inorder(head);
        inorder();
        for (int i = 2; i < ai; i++) {
            cout << ans[i] << " ";
        }
        cout << endl;
        return 0;
    }

三、P2596 书架：按编号反查节点

书架题里书的编号是外部给的，操作都是按编号（Top、Bottom、Ask、Query），不是按排名。所以除了 splay 树本身，还要维护一个 pos[s] 数组：编号 s 的书对应的节点下标。这样给一个编号就能 O(log n) 定位到它在树里的位置。

Top s（把书 s 放最上面）就是 move(rank, 2)：把 s 从它现在的排名挪到排名 2（虚拟头后面）。Bottom s 是 move(rank, n−1)（虚拟尾前面）。Insert 是 move(rank, rank+t)，t 可正可负，正数往后挪、负数往前挪。

move 的实现把三棵树段拼起来：把 a 段从原位置切出来，删掉（但节点不回收，还要复用），再插到 b 位置前。每段操作都是提取子树 + 拼接，全程 O(log n)。Ask s 输出 s 的排名减 1（去掉虚拟头）。

#include <iostream>
    #include <string>

    using namespace std;

    const int MAXN = 80005;

    int head = 0;
    int cnt = 0;
    int num[MAXN];
    int fa[MAXN];
    int ls[MAXN];
    int rs[MAXN];
    int siz[MAXN];
    int pos[MAXN];

    void up(int i) {
        siz[i] = siz[ls[i]] + siz[rs[i]] + 1;
    }

    int lr(int i) {
        return rs[fa[i]] == i ? 1 : 0;
    }

    void rotate(int i) {
        int f = fa[i], g = fa[f], soni = lr(i), sonf = lr(f);
        if (soni == 1) {
            rs[f] = ls[i];
            if (rs[f] != 0) {
                fa[rs[f]] = f;
            }
            ls[i] = f;
        } else {
            ls[f] = rs[i];
            if (ls[f] != 0) {
                fa[ls[f]] = f;
            }
            rs[i] = f;
        }
        if (g != 0) {
            if (sonf == 1) {
                rs[g] = i;
            } else {
                ls[g] = i;
            }
        }
        fa[f] = i;
        fa[i] = g;
        up(f);
        up(i);
    }

    void splay(int i, int goal) {
        int f = fa[i], g = fa[f];
        while (f != goal) {
            if (g != goal) {
                if (lr(i) == lr(f)) {
                    rotate(f);
                } else {
                    rotate(i);
                }
            }
            rotate(i);
            f = fa[i];
            g = fa[f];
        }
        if (goal == 0) {
            head = i;
        }
    }

    int find(int rank) {
        int i = head;
        while (i != 0) {
            if (siz[ls[i]] + 1 == rank) {
                return i;
            } else if (siz[ls[i]] >= rank) {
                i = ls[i];
            } else {
                rank -= siz[ls[i]] + 1;
                i = rs[i];
            }
        }
        return 0;
    }

    void add(int s) {
        num[++cnt] = s;
        pos[s] = cnt;
        siz[cnt] = 1;
        fa[cnt] = head;
        rs[head] = cnt;
        splay(cnt, 0);
    }

    int ask(int s) {
        int i = pos[s];
        splay(i, 0);
        return siz[ls[i]];
    }

    int query(int s) {
        int i = find(s);
        splay(i, 0);
        return num[i];
    }

    void move(int a, int b) {
        int l = find(a - 1);
        int r = find(a + 1);
        splay(l, 0);
        splay(r, l);
        int i = ls[r];
        ls[r] = 0;
        up(r);
        up(l);
        l = find(b - 1);
        r = find(b);
        splay(l, 0);
        splay(r, l);
        ls[r] = i;
        fa[i] = r;
        up(r);
        up(l);
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        int n, m;
        cin >> n >> m;
        add(0);
        for (int i = 1, x; i <= n; i++) {
            cin >> x;
            add(x);
        }
        add(n + 1);
        n = n + 2;
        for (int i = 1, s, t, rank; i <= m; i++) {
            string op;
            cin >> op >> s;
            rank = ask(s) + 1;
            if (op == "Top") {
                move(rank, 2);
            } else if (op == "Bottom") {
                move(rank, n - 1);
            } else if (op == "Insert") {
                cin >> t;
                move(rank, rank + t);
            } else if (op == "Ask") {
                cout << rank - 2 << endl;
            } else {
                cout << query(s + 1) << endl;
            }
        }
        return 0;
    }

四、区间翻转演示

演示懒翻转标记怎么一层层传下去：中序序列保持不变的值在子树里，翻转标记打上后，只有真正访问到那棵子树才交换孩子、继续下传。

总结

区间操作三题，套路是一样的：

1）提取区间：find(l−1) 提根 + find(r+1) 挂右，ls[rs[head]] 就是要处理的段。三题全部复用这一个模板。

2）P1486 用 change 偏移量让整体加减变成 O(1)，树里存相对值；扣工资后按值域斩段。

3）P3391 的翻转是懒标记 rev 加上交换孩子；P2596 靠 pos[] 把编号映射到节点，move 用三段拼接实现。
