树屋阶梯与叶子期望

同一串卡特兰数，两题走出了两个极端：树屋阶梯要打出 n=500 时将近 300 位的完整真实答案，叶子期望要把期望公式推到 n=10⁹ 的浮点数。一个用大数精确乘，一个用闭式公式直接算。

零、题目8：树屋阶梯

洛谷 P2532。地面高度是 0，要搭建一个阶梯，每个台阶上升 1 的高度，最终到达高度 n。有无穷多任意规格的矩形材料，必须选 n 个矩形，问有多少种搭建方法。答案是卡特兰数，n ≤ 500，不取模打印真实答案。

为什么是卡特兰数：看最左下角的那个矩形，它决定了整个阶梯的划分。它的上方必须再搭一个高度 i 的阶梯，右侧必须搭一个高度 n−1−i 的阶梯，i 从 0 到 n−1。这就是划分左右相乘模型，方法数是公式4 的递推，和二叉树结构数同一个答案。

答案有多大：C(500) 有 297 位，long long 装不下。这一页的代码用公式2 的质因子计数法精确相乘，只做大数乘小整数，不碰大数除法：把 (2n)!/(n!(n+1)!) 分解成质数幂，乘一个质数幂就相当于大数连乘几次小整数。n=500 时整段跑完 0.01 秒级别。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;

    // 树屋阶梯：用 n 个矩形搭到高度 n 的阶梯，返回不同搭建方法数
    // 答案可能很大，不取模打印真实答案，1 <= n <= 500
    // 答案是卡特兰数，用公式2 的质因子计数法精确相乘，避免大数除法
    const int MAXN = 1001;

    int minpf[MAXN];
    int prime[MAXN];
    int cnt;
    int counts[MAXN];

    // 大数：十进制，低位在前
    struct Big {
    	vector<int> d;
    	Big(int v) {
    		if (v == 0) {
    			d.push_back(0);
    		} else {
    			while (v) {
    				d.push_back(v % 10);
    				v /= 10;
    			}
    		}
    	}
    	void mul(int x) {
    		ll carry = 0;
    		for (int i = 0; i < (int)d.size(); i++) {
    			ll cur = (ll)d[i] * x + carry;
    			d[i] = cur % 10;
    			carry = cur / 10;
    		}
    		while (carry) {
    			d.push_back(carry % 10);
    			carry /= 10;
    		}
    	}
    	string str() {
    		string s;
    		for (int i = (int)d.size() - 1; i >= 0; i--) {
    			s += char('0' + d[i]);
    		}
    		return s;
    	}
    };

    void euler(int n) {
    	memset(minpf, 0, sizeof(minpf));
    	cnt = 0;
    	for (int i = 2; i <= n; i++) {
    		if (minpf[i] == 0) {
    			prime[cnt++] = i;
    		}
    		for (int j = 0; j < cnt; j++) {
    			if ((ll)i * prime[j] > n) {
    				break;
    			}
    			minpf[i * prime[j]] = prime[j];
    			if (i % prime[j] == 0) {
    				break;
    			}
    		}
    	}
    }

    // 卡特兰数 C(n) = (2n)! / (n!(n+1)!) = 分子 n+2..2n 各 +1，分母 2..n 各 -1
    string compute(int n) {
    	euler(2 * n);
    	for (int i = 2; i <= n; i++) {
    		counts[i] = -1;
    	}
    	for (int i = n + 2; i <= 2 * n; i++) {
    		counts[i] = 1;
    	}
    	for (int i = 2 * n; i >= 2; i--) {
    		if (minpf[i] != 0) {
    			counts[minpf[i]] += counts[i];
    			counts[i / minpf[i]] += counts[i];
    			counts[i] = 0;
    		}
    	}
    	Big ans(1);
    	for (int i = 2; i <= 2 * n; i++) {
    		for (int j = 0; j < counts[i]; j++) {
    			ans.mul(i);
    		}
    	}
    	return ans.str();
    }

    int main() {
    	int n;
    	cin >> n;
    	cout << compute(n) << endl;
    	return 0;
    }

一、题目9：叶子节点数的期望

洛谷 P3978。n 个无差别节点能形成很多不同结构的二叉树，假设所有结构等概率出现一棵，问叶子节点的期望。n ≤ 10⁹，答案误差小于 10⁻⁹。期望是

E(n) = n(n+1) / (2(2n−1))

n=3：5 棵树里 1 棵有 2 个叶子、4 棵有 1 个叶子，期望 6/5 = 1.2，公式 3·4/(2·5) = 1.2 对上。n=10⁹ 时代入公式直接算 double，输入用 long long 读，乘法用 2.0 保浮点。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;

    // 叶子节点数的期望
    // n 个无差别节点能形成很多不同结构的二叉树，等概率出现一棵，返回叶子节点数的期望
    // 1 <= n <= 10^9，答案误差小于 10^-9
    // 期望 = n(n+1) / (2(2n-1))
    int main() {
    	ll n;
    	cin >> n;
    	printf("%.9f\n", (double)n * (n + 1) / ((2 * n - 1) * 2.0));
    	return 0;
    }

总结

树屋阶梯把卡特兰数推到「必须打印真实答案」的极端：n=500 答案 297 位，C++ 用质因子计数法只做大数乘小整数，避免手写大数除法。叶子期望走另一个极端：n=10⁹ 的期望有闭式 E(n)=n(n+1)/(2(2n−1))，一行公式解决。两题对拍全过：树屋阶梯 n=1..500 全部与 Python 大整数一致，期望枚举小树逐棵核对一致。到此卡特兰数的三种模型、四种公式、七种取模状况全部落地。
