完美洗牌：原地 O(1) 的下标循环怼

把数组的一段（偶数长度）原地交叉成「右半、左半、右半、左半…」，要求 O(n) 时间、O(1) 额外空间。直接开辅助数组太简单，原地做要处理下标映射成的一堆循环。特殊长度下循环个数有闭合公式，任意偶数长度都能从大到小拆成特殊长度，于是整段原地交叉。摇摆排序 II 的进阶解把中位数三分区和完美洗牌拼在一起。

零、下标映射与循环

给定范围 [l, r]，长度 n 偶数，左半 [l, m]、右半 [m+1, r]，m = (l+r)/2。目标：位置 l+j 放原右半第 j 个，位置 m+j 放原左半第 j 个。写成下标映射 to(i)：

to(i) = i + (i−l+1)（i 在左半）；to(i) = i − (r−i+1)（i 在右半）

左半的元素整体移到右半，右半的元素整体移到左半。这个映射是排列，可以拆成若干不相交的循环。沿循环把值逐个往后传（curv/nextv 滚动），一个循环 O(循环长)，所有循环加起来 O(n)。问题只剩：循环的起点在哪？

对一般长度，循环个数没有简单公式，必须试探。但对特殊长度 2、8、26、80…（3^k − 1），有个数学结论：下标循环的个数 = k，且第 j 个循环的起点是 l + start[j] − 1（相对块左端偏移 start[j]−1 = 0、2、8…）。起点齐了，循环怼就能逐个启动。

一、任意偶数拆成特殊长度

特殊长度序列 p = 2、8、26、80…（3^k − 1），从大到小把 n 拆掉：先取 ≤ n 的最大特殊长度，处理完这块再取剩下的，处理不了就换小的。每个块的处理分两步：

第一步三段旋转 rotate(l+s/2, m, m+s/2)，把块内左右两半先摆到各自该去的位置。第二步 circle 沿块内每个循环滚值。块处理完，l 前进 s、n 减 s，直到 n 为 0。

为什么能拆干净：任何偶数 n 都能表示成若干个特殊长度之和（2、8、26…类似进制的基底）。对拍里随机偶数长度 3000 组，原地结果和辅助数组版本逐位一致。

二、完美洗牌实现

shuffle(l, r) 是核心：build 生成特殊长度表，主循环按拆分处理每块。rotate 用三次 reverse 实现三段交换，circle 对块内每个循环滚动值，to 是下标映射。

#include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 1000005;

    int arr[MAXN];
    int start[MAXN];
    int split[MAXN];
    int size;

    void swap(int i, int j) {
    	int t = arr[i];
    	arr[i] = arr[j];
    	arr[j] = t;
    }

    void reverse(int l, int r) {
    	while (l < r) {
    		swap(l++, r--);
    	}
    }

    void rotate(int l, int m, int r) {
    	reverse(l, m);
    	reverse(m + 1, r);
    	reverse(l, r);
    }

    int to(int i, int l, int r) {
    	if (i <= (l + r) >> 1) {
    		return i + (i - l + 1);
    	} else {
    		return i - (r - i + 1);
    	}
    }

    void circle(int l, int r, int i) {
    	for (int j = 1, init, cur, next, curv, nextv; j <= i; j++) {
    		init = cur = l + start[j] - 1;
    		next = to(cur, l, r);
    		curv = arr[cur];
    		while (next != init) {
    			nextv = arr[next];
    			arr[next] = curv;
    			curv = nextv;
    			cur = next;
    			next = to(cur, l, r);
    		}
    		arr[init] = curv;
    	}
    }

    void build(int n) {
    	size = 0;
    	for (int s = 1, p = 2; p <= n; s *= 3, p = s * 3 - 1) {
    		start[++size] = s;
    		split[size] = p;
    	}
    }

    void shuffle(int l, int r) {
    	int n = r - l + 1;
    	build(n);
    	for (int i = size, m; n > 0;) {
    		if (split[i] <= n) {
    			m = (l + r) / 2;
    			rotate(l + split[i] / 2, m, m + split[i] / 2);
    			circle(l, l + split[i] - 1, i);
    			l += split[i];
    			n -= split[i];
    		} else {
    			i--;
    		}
    	}
    }

    int main() {
    	ios::sync_with_stdio(false);
    	cin.tie(nullptr);
    	int n;
    	cin >> n;
    	for (int i = 1; i <= n; i++) {
    		cin >> arr[i];
    	}
    	shuffle(1, n);
    	for (int i = 1; i <= n; i++) {
    		cout << arr[i] << " ";
    	}
    	cout << endl;
    	return 0;
    }

三、摇摆排序 II：拼上中位数

题目要 arr[0] < arr[1] > arr[2] < arr[3] > …，O(n) 时间、O(1) 空间，输入保证有解。思路：把数组按中位数三分区，小于中位数的放一边、等于的中间、大于的另外一边。wiggle 的布局是偶数位（谷）放小数、奇数位（峰）放大数，让 mid 的落奇数位。三分区得到的是物理数组「mid」，不是交错布局，靠完美洗牌来交错：

n 为偶数：shuffle(0, n−1) 把数组交错成「右半、左半…」，再整体 reverse 一次，正好变成「左半、右半…」的 wiggle 布局。n 为奇数：shuffle(1, n−1)，位置 0 保持不动（它本来就是谷位），从 1 开始交错。

为什么要 reverse：洗牌后偶数位放的是原右半（大数）、奇数位放原左半（小数），wiggle 要偶数位小、奇数位大，整个反一下正好对调。奇数时位置 0 单独留在外面，是因为它没有相邻的左侧谷位，参与洗牌会把布局弄乱。

四、摇摆排序 II 实现

randomizedSelect 找第 n/2 名的中位数并顺手三分区，再按奇偶调洗牌。测试链接：LeetCode 324 摆动排序 II，题目保证有解。

#include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 1000005;

    int arr[MAXN];
    int start[MAXN];
    int split[MAXN];
    int size;
    int first, last;

    void swap(int i, int j) {
    	int t = arr[i];
    	arr[i] = arr[j];
    	arr[j] = t;
    }

    void partition(int l, int r, int x) {
    	first = l;
    	last = r;
    	int i = l;
    	while (i <= last) {
    		if (arr[i] == x) {
    			i++;
    		} else if (arr[i] < x) {
    			swap(first++, i++);
    		} else {
    			swap(i, last--);
    		}
    	}
    }

    int randomizedSelect(int n, int i) {
    	int ans = 0;
    	for (int l = 0, r = n - 1; l <= r;) {
    		partition(l, r, arr[l + rand() % (r - l + 1)]);
    		if (i < first) {
    			r = first - 1;
    		} else if (i > last) {
    			l = last + 1;
    		} else {
    			ans = arr[i];
    			break;
    		}
    	}
    	return ans;
    }

    void reverse(int l, int r) {
    	while (l < r) {
    		swap(l++, r--);
    	}
    }

    void rotate(int l, int m, int r) {
    	reverse(l, m);
    	reverse(m + 1, r);
    	reverse(l, r);
    }

    int to(int i, int l, int r) {
    	if (i <= (l + r) >> 1) {
    		return i + (i - l + 1);
    	} else {
    		return i - (r - i + 1);
    	}
    }

    void circle(int l, int r, int i) {
    	for (int j = 1, init, cur, next, curv, nextv; j <= i; j++) {
    		init = cur = l + start[j] - 1;
    		next = to(cur, l, r);
    		curv = arr[cur];
    		while (next != init) {
    			nextv = arr[next];
    			arr[next] = curv;
    			curv = nextv;
    			cur = next;
    			next = to(cur, l, r);
    		}
    		arr[init] = curv;
    	}
    }

    void build(int n) {
    	size = 0;
    	for (int s = 1, p = 2; p <= n; s *= 3, p = s * 3 - 1) {
    		start[++size] = s;
    		split[size] = p;
    	}
    }

    void shuffle(int l, int r) {
    	int n = r - l + 1;
    	build(n);
    	for (int i = size, m; n > 0;) {
    		if (split[i] <= n) {
    			m = (l + r) / 2;
    			rotate(l + split[i] / 2, m, m + split[i] / 2);
    			circle(l, l + split[i] - 1, i);
    			l += split[i];
    			n -= split[i];
    		} else {
    			i--;
    		}
    	}
    }

    void wiggleSort(int n) {
    	randomizedSelect(n, n / 2);
    	if ((n & 1) == 0) {
    		shuffle(0, n - 1);
    		reverse(0, n - 1);
    	} else {
    		shuffle(1, n - 1);
    	}
    }

    int main() {
    	ios::sync_with_stdio(false);
    	cin.tie(nullptr);
    	srand(time(nullptr));
    	int n;
    	cin >> n;
    	for (int i = 0; i < n; i++) {
    		cin >> arr[i];
    	}
    	wiggleSort(n);
    	for (int i = 0; i < n; i++) {
    		cout << arr[i] << " ";
    	}
    	cout << endl;
    	return 0;
    }

总结

完美洗牌把「原地交叉一段」归结为下标映射 to(i) 的循环分解。特殊长度 (3^i+1)/2 时循环个数和起点有闭合公式（第 j 个起点 3^(j−1)），任意偶数长度从大到小拆成特殊长度块，每块三段旋转加循环怼，整段 O(n)、O(1)。摇摆排序 II 的进阶解是两个模块的组合：随机选择把中位数三分区，再按奇偶调完美洗牌交错，偶数洗完整段反转、奇数从位置 1 开始洗。交错后逐对验证就能确认布局。这两个技巧都不开额外数组，适合内存受限的场景。
