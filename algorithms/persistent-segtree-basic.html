可持久化线段树：改一次，多一个版本

普通线段树改一个点，整棵树的旧状态就没了。可持久化线段树把每次修改留痕：新版本和老版本共存，查任意历史版本都是 O(log n)。

一、版本从哪来

线段树的修改操作是原地进行的：update 递归下去，沿途把节点的值改掉。改完这棵树就变成了新样子，旧样子没有备份。

可持久化的做法是「复制路径」：update 每走到一个节点，先把它复制一份，在副本上改；没有走到的那部分子树，副本直接指向老节点。这样一次单点修改新建 O(log n) 个节点，老版本纹丝不动。

每个版本存一个根节点编号，放进 root 数组：root[0] 是初始版本，root[k] 是第 k 次操作后的版本。查询「第 k 个版本」就是拿 root[k] 当入口下钻，复杂度 O(log n)。

总空间：初始树 4n 个节点，之后每次修改新建 O(log n) 个节点，n 次修改一共 O(n log n)。

二、单点改、单点查

单点修改是最朴素的可持久化，不需要懒更新，代码也好写。以模版题为例：原始数组是 0 号版本，每次操作基于某个版本生成新版本，版本号就是操作计数。

核心就两个函数。update 先复制当前节点，抄下左右孩子和值，再决定往左还是往右递归；query 从指定版本根节点一路下钻到目标位置。注意一个细节：查询操作也生成新版本，虽然内容跟来源版本一致，但版本号照常 +1。

单点修改 + 单点查询（P3919）

#include <iostream>

    using namespace std;

    const int MAXN = 1000001;
    const int MAXT = MAXN * 23;
    int n, m;
    int arr[MAXN];
    int root[MAXN];
    int ls[MAXT];
    int rs[MAXT];
    int value[MAXT];
    int cnt = 0;

    int build(int l, int r) {
        int rt = ++cnt;
        if (l == r) {
            value[rt] = arr[l];
        } else {
            int mid = (l + r) >> 1;
            ls[rt] = build(l, mid);
            rs[rt] = build(mid + 1, r);
        }
        return rt;
    }

    int update(int jobi, int jobv, int l, int r, int i) {
        int rt = ++cnt;
        ls[rt] = ls[i];
        rs[rt] = rs[i];
        value[rt] = value[i];
        if (l == r) {
            value[rt] = jobv;
        } else {
            int mid = (l + r) >> 1;
            if (jobi <= mid) {
                ls[rt] = update(jobi, jobv, l, mid, ls[rt]);
            } else {
                rs[rt] = update(jobi, jobv, mid + 1, r, rs[rt]);
            }
        }
        return rt;
    }

    int query(int jobi, int l, int r, int i) {
        if (l == r) {
            return value[i];
        }
        int mid = (l + r) >> 1;
        if (jobi <= mid) {
            return query(jobi, l, mid, ls[i]);
        } else {
            return query(jobi, mid + 1, r, rs[i]);
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
        }
        root[0] = build(1, n);
        for (int i = 1, version, op, x, y; i <= m; i++) {
            cin >> version >> op >> x;
            if (op == 1) {
                cin >> y;
                root[i] = update(x, y, 1, n, root[version]);
            } else {
                root[i] = root[version];
                cout << query(x, 1, n, root[i]) << '\n';
            }
        }
        return 0;
    }

三、范围查询：版本做差

单点查询只能问「某个位置的值」。要问区间信息，用「版本差」：把线段树改成值域线段树，叶子是值域上的每个值，节点的 siz 记这个值域里有几个数。对数组每个前缀位置建一个版本：第 i 个版本就是 arr[1..i] 插完后的值域树。

这样区间 [l, r] 的信息 = 版本 r 减去版本 l-1：两个版本从根一起往下走，每个节点的 siz 差就是这个值域里在 [l, r] 内的数的个数。这是可持久化线段树最常考的组合：单点修改 + 范围查询。

第 k 小的查法：从两个版本的根同时出发，看左子树的 siz 差。差 ≥ k 就走左；否则 k 减去差、走右。走到叶子就是答案。值域先离散化，把 n 个不同值压缩成 1..s，树高就是 O(log n)。

区间第 k 小（P3834）

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 200001;
    const int MAXT = MAXN * 22;
    int n, m, s;
    int arr[MAXN];
    int sorted[MAXN];
    int root[MAXN];
    int ls[MAXT];
    int rs[MAXT];
    int siz[MAXT];
    int cnt;

    int kth(int num) {
        int left = 1, right = s, mid, ans = 0;
        while (left <= right) {
            mid = (left + right) / 2;
            if (sorted[mid] <= num) {
                ans = mid;
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        return ans;
    }

    int build(int l, int r) {
        int rt = ++cnt;
        siz[rt] = 0;
        if (l < r) {
            int mid = (l + r) / 2;
            ls[rt] = build(l, mid);
            rs[rt] = build(mid + 1, r);
        }
        return rt;
    }

    int insert(int jobi, int l, int r, int i) {
        int rt = ++cnt;
        ls[rt] = ls[i];
        rs[rt] = rs[i];
        siz[rt] = siz[i] + 1;
        if (l < r) {
            int mid = (l + r) / 2;
            if (jobi <= mid) {
                ls[rt] = insert(jobi, l, mid, ls[rt]);
            } else {
                rs[rt] = insert(jobi, mid + 1, r, rs[rt]);
            }
        }
        return rt;
    }

    int query(int jobk, int l, int r, int u, int v) {
        if (l == r) {
            return l;
        }
        int lsiz = siz[ls[v]] - siz[ls[u]];
        int mid = (l + r) / 2;
        if (lsiz >= jobk) {
            return query(jobk, l, mid, ls[u], ls[v]);
        } else {
            return query(jobk - lsiz, mid + 1, r, rs[u], rs[v]);
        }
    }

    void prepare() {
        cnt = 0;
        for (int i = 1; i <= n; i++) {
            sorted[i] = arr[i];
        }
        sort(sorted + 1, sorted + n + 1);
        s = 1;
        for (int i = 2; i <= n; i++) {
            if (sorted[s] != sorted[i]) {
                sorted[++s] = sorted[i];
            }
        }
        root[0] = build(1, s);
        for (int i = 1, x; i <= n; i++) {
            x = kth(arr[i]);
            root[i] = insert(x, 1, s, root[i - 1]);
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
        }
        prepare();
        for (int i = 1, l, r, k, rank; i <= m; i++) {
            cin >> l >> r >> k;
            rank = query(k, 1, s, root[l - 1], root[r]);
            cout << sorted[rank] << '\n';
        }
        return 0;
    }

五、总结

可持久化线段树的核心就一句话：修改只复制路径，版本号做入口。单点修改加范围查询是它的主战场，区间第 k 小就是「版本差」的典型用法。下一种玩法是范围修改的可持久化，那要处理懒更新的可持久化问题。
