差分约束：判负环找一组解

给 n 个变量和一组 Xi − Xj ≤ Ci 的不等式，要判断有没有解，有解就输出一组。把每个不等式当成一条带权边，问题就变成「图里有没有负环」：没有负环，spfa 跑完的 dist 就是一组解；有负环，不等式组无解。

零、两种形式，一个转法

形式1 是不等式 Xi − Xj ≤ Ci，判负环用最短路；形式2 是 Xi − Xj ≥ Ci，判「无限增加的环」用最长路。形式上两种，其实只需要形式1 的代码：两边乘 −1，Xi − Xj ≥ Ci 变成 Xj − Xi ≤ −Ci，正好是形式1。

为什么有解等于没有负环：把不等式当约束，spfa 的松弛就是在给变量找一组不冲突的值。负环意味着绕一圈还能继续收紧，这条环上的约束加起来小于 0，凑不出任何一组值。那在形式上就是矛盾。

一、建图：不等式变成边

Xi − Xj ≤ Ci 挪一下是 Xi ≤ Xj + Ci。spfa 松弛时 dist[v] ≤ dist[u] + w 对应一条 u→v 权 w 的边，所以这条不等式建成边 j→i、权 Ci。

方向别弄反：不等号左边是 Xi，右边是 Xj + Ci，按「右边推到左边」建 j→i。建反了整张图都是错的，判负环跟着一起错。

跑完 spfa，dist 满足每条边约束 dist[i] ≤ dist[j] + Ci，逐条对应回不等式 Xi − Xj ≤ Ci，所以 dist 就是一组解。

二、超级源点：孤立点也要有人管

只有从起点能到达的点，它身上的负环才会被 spfa 判到。变量可能没出现在任何不等式里，或者图本身不连通，负环在另一个分量里就漏了。建一个 0 号源点，向 1..n 每个点连一条权 0 的边，所有点都从 0 出发，负环在哪都能判到。这一步 n 条边，换整张图的覆盖，便宜得很。

三、spfa 判负环

一个点被入队次数超过 n 次就说明它在环上转：最短路最多经过 n 个点，入队超过 n 次意味着路径在无限绕。判到就输出 NO；没判到，dist[1..n] 就是一组解。

输入给的是 Xu − Xv ≤ c，正好对应建 v→u 权 c 的边，不用再转。n、m ≤ 5000，边最多 10000，链式前向星数组按 2m 开有余量。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;
    const int MAXN = 5005, MAXM = 10005, MAXQ = 5000005;
    int head[MAXN], nxt[MAXM], to[MAXM], w[MAXM], cnt;
    int dist[MAXN], upd[MAXN], que[MAXQ], h, t;
    bool enter[MAXN];
    int n, m;

    void add(int u, int v, int ww) {
      nxt[cnt] = head[u]; to[cnt] = v; w[cnt] = ww; head[u] = cnt++;
    }

    bool spfa(int s) {
      h = t = 0;
      fill(dist, dist + n + 1, INT_MAX);
      fill(upd, upd + n + 1, 0);
      fill(enter, enter + n + 1, false);
      dist[s] = 0; upd[s] = 1;
      que[t++] = s; enter[s] = true;
      while (h < t) {
        int u = que[h++]; enter[u] = false;
        for (int ei = head[u]; ei > 0; ei = nxt[ei]) {
          int v = to[ei], ww = w[ei];
          if (dist[v] > dist[u] + ww) {
            dist[v] = dist[u] + ww;
            if (!enter[v]) {
              if (++upd[v] > n) return true;
              que[t++] = v; enter[v] = true;
            }
          }
        }
      }
      return false;
    }

    int main() {
      scanf("%d%d", &n, &m);
      cnt = 1;
      fill(head, head + n + 1, 0);
      for (int i = 1; i <= n; i++) add(0, i, 0);
      for (int i = 1, u, v, ww; i <= m; i++) {
        scanf("%d%d%d", &u, &v, &ww);
        add(v, u, ww);
      }
      if (spfa(0)) {
        printf("NO\n");
      } else {
        for (int i = 1; i <= n; i++) printf("%d ", dist[i]);
        printf("\n");
      }
      return 0;
    }

实测（对拍）：5000 组随机小不等式，变量 2..4、系数 −10..10、30% 故意注入矛盾环，课程输出与「枚举 x∈[−12,12] 判可行 + 有解时逐条验证 dist」完全一致。

四、题2 小K农场：三种关系翻译

小K农场把关系换成三种说法，翻译成边就三行：

关系1 a b c：a 比 b 至少多种 c → xa − xb ≥ c → xb − xa ≤ −c → 边 a→b 权 −c。

关系2 a b c：至多多 c → xa − xb ≤ c → 边 b→a 权 c。

关系3 a b：一样多 → xa = xb → 双向 0 边。

然后同一个 spfa：有负环输出 No，没有输出 Yes。注意这里输出是首字母大写的 No/Yes，不是模板题的 NO。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;
    const int MAXN = 5005, MAXM = 20005, MAXQ = 20000005;
    int head[MAXN], nxt[MAXM], to[MAXM], w[MAXM], cnt;
    int dist[MAXN], upd[MAXN], que[MAXQ], h, t;
    bool enter[MAXN];
    int n, m;

    void add(int u, int v, int ww) {
      nxt[cnt] = head[u]; to[cnt] = v; w[cnt] = ww; head[u] = cnt++;
    }

    bool spfa(int s) {
      h = t = 0;
      fill(dist, dist + n + 1, INT_MAX);
      fill(upd, upd + n + 1, 0);
      fill(enter, enter + n + 1, false);
      dist[s] = 0; upd[s] = 1;
      que[t++] = s; enter[s] = true;
      while (h < t) {
        int u = que[h++]; enter[u] = false;
        for (int ei = head[u]; ei > 0; ei = nxt[ei]) {
          int v = to[ei], ww = w[ei];
          if (dist[v] > dist[u] + ww) {
            dist[v] = dist[u] + ww;
            if (!enter[v]) {
              if (++upd[v] > n) return true;
              que[t++] = v; enter[v] = true;
            }
          }
        }
      }
      return false;
    }

    int main() {
      scanf("%d%d", &n, &m);
      cnt = 1;
      fill(head, head + n + 1, 0);
      for (int i = 1; i <= n; i++) add(0, i, 0);
      for (int i = 1, type, u, v, ww; i <= m; i++) {
        scanf("%d%d%d", &type, &u, &v);
        if (type == 1) {
          scanf("%d", &ww);
          add(u, v, -ww);
        } else if (type == 2) {
          scanf("%d", &ww);
          add(v, u, ww);
        } else {
          add(u, v, 0);
          add(v, u, 0);
        }
      }
      printf("%s\n", spfa(0) ? "No" : "Yes");
      return 0;
    }

五、解整体平移

一组解 (x1, x2, ..., xn) 每个同时加同一个常数 d，代入每个不等式，两边各多一个 d，抵消。所以解从来不是一条，是一族。题目只要任意一组，输出 dist 就好；无解才是真正要判的事。

总结

形式1 的 Xi − Xj ≤ Ci 建边 j→i 权 Ci，形式2 两边乘 −1 变回形式1，判负环的代码只有一份。

0 号源点向所有点连 0 权边，保证图里任何位置的负环都逃不过 spfa；入队次数超过 n 就是负环，没判到就把 dist 当解输出。

农场的三种关系翻译成边：≥ 转 −c、≤ 转 +c、等号拆两条 0 边，同一个 spfa 判矛盾。解整体平移仍是解，所以「给一组解」宽松，判无解才是重点。
