左偏树应用：猴王、派遣与数字序列

可并堆最常见的用法是「每个集合挂一个堆，操作要么取集合极值、要么把两个集合合并」。猴王每次打斗要把两边的最大值减半再合并，派遣要从子树里弹出高工资直到预算够，数字序列要用堆维护一段数的中位数。三题都靠 merge 的 O(log n)。

零、背景

左偏树的代码骨架只有 merge、pop、find 三个函数，应用题的差别全在「弹出堆顶之后怎么处理」：猴王把堆顶值减半再放回去，派遣把堆顶直接丢掉，数字序列弹出堆里一半的数只留中位数。看懂这三个变体，可并堆就算会用。

一、P1456 猴王

n 只猴子各带战斗力，初始各自成阵营。每次冲突给 x、y：同一阵营打印 -1；不同阵营，两边最强猴子打斗，各自战斗力减半（向下取整），其余猴子不变，然后两阵营合并，打印合并后阵营的最大战斗力。可能有多个测试用例，读输入直到结束。

课程代码的 fight 分四步：find 出两个堆顶；pop 掉两个堆顶（左右子树合并成空位）；把两个堆顶的值各自除 2；把两个减半的猴子连回子树，整体 merge 成新阵营。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 100001;
    int n, m;
    int num[MAXN];
    int ls[MAXN];
    int rs[MAXN];
    int dist[MAXN];
    int fa[MAXN];

    void prepare() {
        dist[0] = -1;
        for(int i = 1; i <= n; i++) {
            ls[i] = rs[i] = dist[i] = 0;
            fa[i] = i;
        }
    }

    int find(int i) {
        fa[i] = fa[i] == i ? i : find(fa[i]);
        return fa[i];
    }

    int merge(int i, int j) {
        if (i == 0 || j == 0) {
            return i + j;
        }
        if (num[i] < num[j]) {
            swap(i, j);
        }
        rs[i] = merge(rs[i], j);
        if (dist[ls[i]] < dist[rs[i]]) {
            swap(ls[i], rs[i]);
        }
        dist[i] = dist[rs[i]] + 1;
        fa[ls[i]] = fa[rs[i]] = i;
        return i;
    }

    int pop(int i) {
        fa[ls[i]] = ls[i];
        fa[rs[i]] = rs[i];
        fa[i] = merge(ls[i], rs[i]);
        ls[i] = rs[i] = dist[i] = 0;
        return fa[i];
    }

    int fight(int x, int y) {
        int a = find(x);
        int b = find(y);
        if (a == b) {
            return -1;
        }
        int l = pop(a);
        int r = pop(b);
        num[a] /= 2;
        num[b] /= 2;
        fa[a] = fa[b] = fa[l] = fa[r] = merge(merge(l, a), merge(r, b));
        return num[fa[a]];
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        while (cin >> n) {
            prepare();
            for (int i = 1; i <= n; i++) {
                cin >> num[i];
            }
            cin >> m;
            for (int i = 1; i <= m; i++) {
                int x, y;
                cin >> x >> y;
                cout << fight(x, y) << endl;
            }
        }
        return 0;
    }

这里有个细节：pop 之后 l、r 分别是两个堆顶左右子树的合并结果，merge(merge(l, a), merge(r, b)) 把减半的 a、b 和它们原来的子树拼回一棵树，新阵营的堆顶就是新最大值。

二、P1552 派遣

n 个忍者有上级编号、工资、能力三个属性，上级编号一定小于本人编号，1 号是整棵树头。预算 m：选一棵子树，在这棵子树里任选忍者（不必选头），工资总和不超过 m。收益 = 雇佣人数 × 子树头的能力，求最大收益。n ≤ 105，m ≤ 109。

贪心：对每棵子树，在预算内尽量多雇人。工资从低往高雇，人数才最多。等价于「从子树里不断弹出工资最高的忍者，直到工资总和不超过 m」。每棵子树的头算一次收益，取最大。

课程代码从 n 到 1 倒序遍历：每个节点先自成一个集合（堆、siz、sum），算完以它为头的收益后，把整个集合合并到它的上级。因为上级编号一定更小，倒序保证处理到上级时，子树的忍者已经全在它的堆里。并查集上额外维护两个字段：siz（人数）和 sum（工资总和）。

#include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 100001;
    int n, m;
    int leader[MAXN];
    long long cost[MAXN];
    long long ability[MAXN];
    int ls[MAXN];
    int rs[MAXN];
    int dist[MAXN];
    int fa[MAXN];
    int siz[MAXN];
    long long sum[MAXN];

    int find(int i) {
        return fa[i] = (fa[i] == i ? i : find(fa[i]));
    }

    int merge(int i, int j) {
        if (i == 0 || j == 0) {
            return i + j;
        }
        if (cost[i] < cost[j]) {
            swap(i, j);
        }
        rs[i] = merge(rs[i], j);
        if (dist[ls[i]] < dist[rs[i]]) {
            swap(ls[i], rs[i]);
        }
        dist[i] = dist[rs[i]] + 1;
        fa[ls[i]] = fa[rs[i]] = i;
        return i;
    }

    int pop(int i) {
        fa[ls[i]] = ls[i];
        fa[rs[i]] = rs[i];
        fa[i] = merge(ls[i], rs[i]);
        ls[i] = rs[i] = dist[i] = 0;
        return fa[i];
    }

    void prepare() {
        dist[0] = -1;
        for (int i = 1; i <= n; i++) {
            ls[i] = rs[i] = dist[i] = 0;
            siz[i] = 1;
            sum[i] = cost[i];
            fa[i] = i;
        }
    }

    long long compute() {
        long long ans = 0;
        int p, psize, h, hsize;
        long long hsum, psum;
        for (int i = n; i >= 1; i--) {
            h = find(i);
            hsize = siz[h];
            hsum = sum[h];
            while (hsum > m) {
                pop(h);
                hsize--;
                hsum -= cost[h];
                h = find(i);
            }
            ans = max(ans, (long long)hsize * ability[i]);
            if (i > 1) {
                p = find(leader[i]);
                psize = siz[p];
                psum = sum[p];
                fa[p] = fa[h] = merge(p, h);
                siz[fa[p]] = psize + hsize;
                sum[fa[p]] = psum + hsum;
            }
        }
        return ans;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            cin >> leader[i] >> cost[i] >> ability[i];
        }
        prepare();
        cout << compute() << endl;
        return 0;
    }

合并时新堆的 siz、sum 直接相加：siz[fa[p]] = psize + hsize。这样每个集合维护「当前预算内能雇的最大人数」，pop 高工资时同步减 siz 和 sum。

三、P4331 数字序列

给定长度 n 的数组 A，构造递增数组 B 使 |A[i] − B[i]| 的总和最小，输出最小值和一个 B。n ≤ 106，A[i] ≤ 232 − 1。

先做一步转化：递增转不降。令 arr[i] = A[i] − i，对 arr 求不降数组 C，则 B[i] = C[i] + i 就是递增的，且 |A[i] − B[i]| = |arr[i] − C[i]|。

不降数组怎么构造最省？一段连续的值取它们的中位数最优（奇数个取中间，偶数个取上中位数）。如果相邻两段的中位数破坏了不降序，就合并两段，重新取中位数。单调栈维护这些段，左偏树维护每段的所有值，栈顶段的中位数比新来的数大就弹栈合并。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 1000001;
    int n;
    long long arr[MAXN];
    int ls[MAXN];
    int rs[MAXN];
    int dist[MAXN];
    int fa[MAXN];
    int from[MAXN];
    int to[MAXN];
    int siz[MAXN];
    int stk[MAXN];
    long long ans[MAXN];

    void prepare() {
        dist[0] = -1;
        for (int i = 1; i <= n; i++) {
            ls[i] = rs[i] = dist[i] = 0;
            fa[i] = from[i] = to[i] = i;
            siz[i] = 1;
        }
    }

    int find(int i) {
        fa[i] = fa[i] == i ? i : find(fa[i]);
        return fa[i];
    }

    int merge(int i, int j) {
        if (i == 0 || j == 0) {
            return i + j;
        }
        if (arr[i] < arr[j]) {
            swap(i, j);
        }
        rs[i] = merge(rs[i], j);
        if (dist[ls[i]] < dist[rs[i]]) {
            swap(ls[i], rs[i]);
        }
        dist[i] = dist[rs[i]] + 1;
        fa[ls[i]] = fa[rs[i]] = i;
        return i;
    }

    int pop(int i) {
        fa[ls[i]] = ls[i];
        fa[rs[i]] = rs[i];
        fa[i] = merge(ls[i], rs[i]);
        ls[i] = rs[i] = dist[i] = 0;
        return fa[i];
    }

    long long compute() {
        int stackSize = 0;
        for (int i = 1, pre, cur, s; i <= n; i++) {
            while (stackSize > 0) {
                pre = find(stk[stackSize]);
                cur = find(i);
                if (arr[pre] <= arr[cur]) {
                    break;
                }
                s = siz[pre] + siz[cur];
                cur = merge(pre, cur);
                while (s > ((i - from[pre] + 1 + 1) / 2)) {
                    cur = pop(cur);
                    s--;
                }
                from[cur] = from[pre];
                to[cur] = i;
                siz[cur] = s;
                stackSize--;
            }
            stk[++stackSize] = i;
        }
        long long sum = 0;
        for (int i = 1, cur; i <= stackSize; i++) {
            cur = find(stk[i]);
            for (int j = from[cur]; j <= to[cur]; j++) {
                ans[j] = arr[cur];
                sum += llabs(ans[j] - arr[j]);
            }
        }
        return sum;
    }

    int main() {
        ios_base::sync_with_stdio(false);
        cin.tie(NULL);
        cin >> n;
        prepare();
        long long x;
        for (int i = 1; i <= n; i++) {
            cin >> x;
            arr[i] = x - i;
        }
        long long res = compute();
        cout << res << "\n";
        for (int i = 1; i <= n; i++) {
            cout << ans[i] + i << (i == n ? '\n' : ' ');
        }
        return 0;
    }

课程里每段用大根堆存所有值，段长 len 的中位数位置是 (len + 1) / 2：while (s > (段长 + 1) / 2) pop(cur) 不断弹出较大的一半，堆里剩下的最大值就是上中位数。from、to 记录段的区间，弹栈合并时 from[cur] = from[pre]、to[cur] = i。

最后每个段把堆顶值赋给段内所有位置：ans[j] = 堆顶，累加 |ans[j] − arr[j]| 就是最小代价，输出 ans[j] + i 还原递增 B。

总结

猴王、派遣、数字序列是左偏树三类典型应用：弹出堆顶后「改值放回」（猴王减半）、「直接丢弃直到满足约束」（派遣弹高工资）、「弹到只剩中位数」（数字序列分段）。共同点是一个集合只维护一棵可并堆，合并两个集合就是一次 merge。数字序列的「递增转不降 + 分段取中位数」是独立的贪心技巧，左偏树只是实现手段。
