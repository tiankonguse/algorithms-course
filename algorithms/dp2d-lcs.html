最长公共子序列与最长回文子序列：两种格子依赖

这一页两道题，两个可变参数的含义完全不同：一个是两个串各自的前缀长度，另一个是同一个串的左右端点。参数含义不同，格子之间的依赖图形状就不同，空间压缩的写法也跟着变。

零、两个串，两个前缀长度

最长公共子序列：给两个串，找一个同时是两者子序列的最长串，返回长度。子序列不要求连续，但顺序得保持。

两个串就有两个可变参数。最自然的写法是 f(i1, i2)：s1 的前 i1 个、s2 的前 i2 个里的最长公共子序列长度。

// s1[0..i1] 与 s2[0..i2] 的最长公共子序列长度（下标版）
    int f1(string& s1, string& s2, int i1, int i2) {
        if (i1 < 0 || i2 < 0) {
            return 0;
        }
        int p1 = f1(s1, s2, i1 - 1, i2 - 1);
        int p2 = f1(s1, s2, i1 - 1, i2);
        int p3 = f1(s1, s2, i1, i2 - 1);
        int p4 = s1[i1] == s2[i2] ? (p1 + 1) : 0;
        return max(max(p1, p2), max(p3, p4));
    }

这份递归是对的，四个分支把四种情况都覆盖了：最后一个字符都不要、只不要 s1 的、只不要 s2 的、两个都要（此时要求它们相等）。它也能改动态规划，只是代价大得离谱。

一、用长度当参数，四个分支变两个

换个定义：f(len1, len2) 是 s1 前 len1 个、s2 前 len2 个的答案。看这两个串的最后一个字符 s1[len1−1] 和 s2[len2−1]：

  • 相等，那它们一定可以一起接在某个公共子序列后面，答案是 f(len1−1, len2−1) + 1；
  • 不等，它们不可能同时被选上，答案是 max(f(len1−1, len2), f(len1, len2−1))。

两种情况互斥，用一个 if 就够了，分支从四个降到两个。

// s1 前 len1 个、s2 前 len2 个的最长公共子序列长度（长度版）
    int f2(string& s1, string& s2, int len1, int len2) {
        if (len1 == 0 || len2 == 0) {
            return 0;
        }
        if (s1[len1 - 1] == s2[len2 - 1]) {
            return f2(s1, s2, len1 - 1, len2 - 1) + 1;
        }
        return max(f2(s1, s2, len1 - 1, len2), f2(s1, s2, len1, len2 - 1));
    }

降两个分支值多少钱？实测一下。两个等长 n 的随机串（字符集 2）：

用长度而不是下标来定义尝试，还有个直接的好处：长度最小是 0，下标会跑到 −1，边界讨论少一半。这也是为什么很多动态规划题宁可多开一行一列，也要把「空」这个状态摆进表里。

二、记忆化搜索

和一维时一模一样：拿一张 (n+1) × (m+1) 的表，−1 表示没算过。

int f3(string& s1, string& s2, int len1, int len2, vector<vector<int>>& dp) {
        if (len1 == 0 || len2 == 0) {
            return 0;
        }
        if (dp[len1][len2] != -1) {
            return dp[len1][len2];
        }
        int ans;
        if (s1[len1 - 1] == s2[len2 - 1]) {
            ans = f3(s1, s2, len1 - 1, len2 - 1, dp) + 1;
        } else {
            ans = max(f3(s1, s2, len1 - 1, len2, dp), f3(s1, s2, len1, len2 - 1, dp));
        }
        dp[len1][len2] = ans;
        return ans;
    }

三、严格位置依赖：一张 (n+1) × (m+1) 的表

格子 dp[len1][len2] 依赖三个方向：上面 dp[len1−1][len2]、左面 dp[len1][len2−1]、左上 dp[len1−1][len2−1]。三个都在左上方，所以 len1 从 1 到 n、len2 从 1 到 m 两重循环就够了，第 0 行第 0 列天然是全 0 的空串情况。

int longestCommonSubsequence(string& s1, string& s2) {
        int n = s1.size(), m = s2.size();
        vector<vector<int>> dp(n + 1, vector<int>(m + 1));   // 多开一行一列，天然处理空串
        for (int len1 = 1; len1 <= n; len1++) {
            for (int len2 = 1; len2 <= m; len2++) {
                if (s1[len1 - 1] == s2[len2 - 1]) {
                    dp[len1][len2] = 1 + dp[len1 - 1][len2 - 1];
                } else {
                    dp[len1][len2] = max(dp[len1 - 1][len2], dp[len1][len2 - 1]);
                }
            }
        }
        return dp[n][m];
    }

顺着表看几个格子就明白了：dp[3][2] 是 s1 前 3 个 "abc" 与 s2 前 2 个 "ac" 的答案，最后一个字符都是 c，所以取左上角 dp[2][1] 加 1，得 2。dp[2][2] 的 b 和 c 不相等，就从上和左里挑大的，得 1。

四、空间压缩：leftUp 从哪来

依赖里有「左上角」这个斜向格子，只留一行的滚动数组时，它被覆盖掉了。解决办法是拿一个变量把它存下来。

填 dp[len2] 之前先把它备份进 backup，填完之后 backup 就成了下一格的「左上角」。整个循环里 leftUp 一直跟着走：

int longestCommonSubsequenceRoll(string str1, string str2) {
        if (str1.size() < str2.size()) {        // 让短的那个当列，滚动数组才短
            swap(str1, str2);
        }
        int n = str1.size(), m = str2.size();
        vector<int> dp(m + 1);
        for (int len1 = 1; len1 <= n; len1++) {
            int leftUp = 0, backup;             // leftUp 就是想象中 dp[len1-1][len2-1]
            for (int len2 = 1; len2 <= m; len2++) {
                backup = dp[len2];              // 覆盖之前先存下来，下一步它就成了 leftUp
                if (str1[len1 - 1] == str2[len2 - 1]) {
                    dp[len2] = 1 + leftUp;
                } else {
                    dp[len2] = max(dp[len2], dp[len2 - 1]);
                }
                leftUp = backup;
            }
        }
        return dp[m];
    }

两个容易写错的点。一是 backup 必须在覆盖之前存，存晚了拿到的是本行新值。二是别写 dp[len2] = 1 + dp[len2-1]，那是拿「左」当「左上」，实测 5000 组随机里有 47.78% 的样例会算错。

还有个小优化：让短的那个串当列，滚动数组才短。n = m = 1500 时实测二维表 3.208 ms、压缩版 1.719 ms，内存从 8.59 MB 降到 5.9 KB。

五、最长回文子序列：区间当状态

第二题换了个参数含义。给一个串 s，找最长的回文子序列。这题可以转成「s 与 s 的反串的最长公共子序列」，不过这里讲区间动态规划的思路，那个思路后面讲区间 dp 时还要用。

定义 f(l, r)：s[l..r] 这一段里的最长回文子序列长度，要求 l ≤ r。看两端：

  • l == r，只有一个字符，答案是 1；
  • l + 1 == r，两个字符，相等就是 2，不等就是 1；
  • s[l] == s[r]，这两个可以一左一右包在外面，答案是 2 + f(l+1, r−1)；
  • 否则两端不可能同时被选上，答案是 max(f(l+1, r), f(l, r−1))。
// s[l...r] 上最长回文子序列的长度，l <= r
    int f1(string& s, int l, int r) {
        if (l == r) {
            return 1;
        }
        if (l + 1 == r) {
            return s[l] == s[r] ? 2 : 1;
        }
        if (s[l] == s[r]) {
            return 2 + f1(s, l + 1, r - 1);
        }
        return max(f1(s, l + 1, r), f1(s, l, r - 1));
    }

这份暴力递归的调用次数可以是指数级的。取一个全部字符互不相同的串，每次两端都不相等，两支都得走，实测调用次数恰好是 2^(n−1) − 1：n = 20 时 524287 次，n = 22 时 2097151 次，而格子只有 n² 个。

六、区间动态规划的填表顺序

格子 dp[l][r] 依赖三个：dp[l+1][r−1]（左下）、dp[l+1][r]（下）、dp[l][r−1]（左）。注意方向：它依赖的是「更短的区间」，也就是往里缩一层的那些格子。

所以不能按 l 从 0 到 n−1 的顺序填。算 dp[0][4] 需要 dp[1][3]，而 dp[1][3] 的行号更大，正序枚举时它还没算。正确的顺序是 l 从 n−1 倒着走到 0，r 从 l+2 正着走到 n−1。

int longestPalindromeSubseq(string& s) {
        int n = s.size();
        vector<vector<int>> dp(n, vector<int>(n));
        for (int l = n - 1; l >= 0; l--) {      // 左端点必须倒着走
            dp[l][l] = 1;
            if (l + 1 < n) {
                dp[l][l + 1] = s[l] == s[l + 1] ? 2 : 1;
            }
            for (int r = l + 2; r < n; r++) {   // 右端点正着走
                if (s[l] == s[r]) {
                    dp[l][r] = 2 + dp[l + 1][r - 1];
                } else {
                    dp[l][r] = max(dp[l + 1][r], dp[l][r - 1]);
                }
            }
        }
        return dp[0][n - 1];
    }

实测把 l 写成正序，5000 组随机串里有 85.58% 的样例答案是错的。这个错误还很隐蔽：小程序上常常碰巧对，规模一大就全错。

七、区间动态规划怎么压缩空间

同样只留一行。dp[r] 在这一轮里代表 dp[l][r]，但下标的语义是「右端点」，所以这一行是被倒着覆盖的：l 每减 1，我们要的旧值是「l+1 那一行的 dp[r]」，也就是覆盖前的值。

和上一题对称，拿一个变量 leftDown 存它。区别只在于这次依赖的是左下而不是左上，所以变量沿着 r 增大的方向传递。

int longestPalindromeSubseqRoll(string& s) {
        int n = s.size();
        vector<int> dp(n);
        for (int l = n - 1, leftDown = 0, backup; l >= 0; l--) {
            dp[l] = 1;                          // 想象中的 dp[l][l]
            if (l + 1 < n) {
                leftDown = dp[l + 1];           // 先把 dp[l+1] 存下来，它是想象中的 dp[l+1][l+1]
                dp[l + 1] = s[l] == s[l + 1] ? 2 : 1;   // 想象中的 dp[l][l+1]
            }
            for (int r = l + 2; r < n; r++) {
                backup = dp[r];                 // 覆盖前的 dp[r] 是 dp[l+1][r]，也就是左下方
                if (s[l] == s[r]) {
                    dp[r] = 2 + leftDown;
                } else {
                    dp[r] = max(dp[r], dp[r - 1]);
                }
                leftDown = backup;
            }
        }
        return dp[n - 1];
    }

这里还有个容易漏的地方：dp[l+1] 在被改写成 dp[l][l+1] 之前，它本身是 dp[l+1][l+1]，正是下一轮要用的 leftDown 的初值。所以要先 leftDown = dp[l+1] 再覆盖。

漏掉这一步的代价：实测 5000 组随机串里 63.66% 会算错。n = 2000 时二维表 3.083 ms、压缩版 1.665 ms，答案一致。

八、复杂度分析

题做法时间复杂度空间复杂度
最长公共子序列暴力递归（四分支）O(2^(n+m))O(n+m)
最长公共子序列记忆化搜索O(n·m)O(n·m)
最长公共子序列严格位置依赖O(n·m)O(n·m)
最长公共子序列空间压缩O(n·m)O(min(n, m))

两题每个格子的枚举代价都是 O(1)，所以时间复杂度就等于表的大小。枚举代价不是常数的题要两件事分开算，那种题另说。

九、总结

这两题的共同点是「两个可变参数都是 int，且完全决定返回值」，所以都能改。差别全在依赖图形状上：最长公共子序列依赖左上方三个方向，按行按列正着填就行；最长回文子序列依赖更短的区间，l 必须倒着走。

空间压缩的通用动作只有两个：一是让滚动数组沿着「依赖里那一维」走，二是被覆盖掉的斜向格子用一个变量传下来。leftUp 和 leftDown 是同一个招数用在两个方向上，没有别的玄机。

再往下，是那些填一个格子还要枚举一圈的题。表的大小乘以枚举代价才是总时间，空间压缩时「先枚举哪一维」也会变成硬约束。
