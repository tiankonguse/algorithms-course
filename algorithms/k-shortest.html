K 短路：可持久化左偏树的最优解

A* 算法也能做 k 短路，复杂度要高一个量级。可持久化左偏树把「相对最短路的偏离」组织成堆，展开一条候选 O(log m)，总复杂度 O((k + m) log m)。洛谷 P2483 就是这样过的。

零、题意与三步框架

n 个点、m 条有向正权边。从 1 号点走到 n 号点算一次旅行，旅行中边不能重复选、点可以重复经过，一到达 n 就立刻停止。任何两次旅行只要选的边稍有不同就算不同方案。钱数是 money，旅行一次扣一次路费，问最多能旅行几次。

注意「边不能重复选」说的是同一次旅行内部的边不能重复；课程的可持久化左偏树解法里，最短路树上的边段可以反复出现在不同候选里，因为每次旅行是不同的方案。费用全是正数，按费用从小到大依次扣钱即可。

解法分三步：

（1）反图跑 dijkstra，得到每个点到 n 的最短距离 dis，以及最短路树上的父边 path。

（2）mergeRoad 为每个点建一棵可持久化左偏树 rt[u]，装的是「从 u 出发的所有偏离边」的代价增量。

（3）expand 从最短路径开始，反复弹出最小的偏离增量候选，钱够就计数，并展开这个候选的三个后继。

一、反图最短路与最短路树

反图就是所有边反向。从 n 号点跑 dijkstra，dis[u] 就是 u 到 n 的最短距离。松弛成功时记下 path[v] = e，e 是反图上让 dis[v] 收敛的入边编号。因为正图边与反图边成对添加（编号一一对应），正图上 u → tog[path[u]] 就是 u 沿最短路树走向 n 的那条边，它的边权恰好等于 dis[u] - dis[tog[path[u]]]。

二、偏离边堆 rt[u]

mergeRoad 逆着 dijkstra 的弹出顺序建堆。对每个点 u，遍历正图出边：

（1）跳过最短路树边 e == path[u]，它是沿树前进的边，不算偏离。

（2）其余每条出边 (u→v, w) 都是一个偏离选项，代价增量 = w + dis[v] - dis[u]：本来沿树边从 u 走到 n 花 dis[u]，改走这条边到 v 再沿树边到 n 花 w + dis[v]，多花的钱就是它。全为正数，所以可持久化左偏树按代价增量排小根堆。

（3）处理完 u 自己的偏离边，把 rt[tog[path[u]]] 也 merge 进来：沿树边走到树子后，还能继续偏离。这就是「前缀共享」：从 u 出发的候选包含从树子出发的全部候选。

可持久化 merge 在这里是必须的：rt[1] 和 rt[2] 都要包含 rt[3] 的子树，如果不拷贝，修改一棵就会破坏另一棵。

三、展开：A* 式弹出候选

expand 先扣最短路费用 money -= dis[1]，能付就计一次旅行。然后候选堆里放第一组候选：rt[1] 的堆顶，预算是 dis[1] + cost[rt[1]]，即最短路加上第一个偏离增量。

每次弹出一个候选 h、预算 w，钱够就计数，然后展开三个后继：

（1）左儿子 ls[h]：同偏离点下一条更贵的偏离边，预算换成 w - cost[h] + cost[ls[h]]。

（2）右儿子 rs[h]：同理。

（3）偏离终点 rt[to[h]]：走到偏离边终点后继续偏离，预算 w + cost[rt[to[h]]]。

候选的预算就是路径的真实费用，因为第一个候选已经加上了 dis[1]。money 扣到负数就停，输出计数。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 5001;
    const int MAXM = 200001;
    const int MAXT = 1000001;
    const int MAXH = 4200001;
    const double INF = 1e18;

    int n, m;
    double money;

    int headg[MAXN];
    int tog[MAXM];
    int nextg[MAXM];
    double weightg[MAXM];
    int cntg = 0;

    int headr[MAXN];
    int tor[MAXM];
    int nextr[MAXM];
    double weightr[MAXM];
    int cntr = 0;

    int to[MAXT];
    double cost[MAXT];
    int ls[MAXT];
    int rs[MAXT];
    int dist[MAXT];
    int cntt = 0;

    int rt[MAXN];

    int key[MAXH];
    double val[MAXH];
    int heap[MAXH];
    int cntd, cnth;

    bool vis[MAXN];
    int path[MAXN];
    double dis[MAXN];

    void addEdgeG(int u, int v, double w){
        nextg[++cntg] = headg[u];
        tog[cntg] = v;
        weightg[cntg] = w;
        headg[u] = cntg;
    }

    void addEdgeR(int u, int v, double w){
        nextr[++cntr] = headr[u];
        tor[cntr] = v;
        weightr[cntr] = w;
        headr[u] = cntr;
    }

    int init(int t, double v){
        to[++cntt] = t;
        cost[cntt] = v;
        ls[cntt] = rs[cntt] = dist[cntt] = 0;
        return cntt;
    }

    int clone(int i){
        to[++cntt] = to[i];
        cost[cntt] = cost[i];
        ls[cntt] = ls[i];
        rs[cntt] = rs[i];
        dist[cntt] = dist[i];
        return cntt;
    }

    int merge(int i, int j){
        if(i == 0 || j == 0){
            return i + j;
        }
        if(cost[i] > cost[j]){
            swap(i, j);
        }
        int h = clone(i);
        rs[h] = merge(rs[h], j);
        if(dist[ls[h]] < dist[rs[h]]){
            swap(ls[h], rs[h]);
        }
        dist[h] = dist[rs[h]] + 1;
        return h;
    }

    void heapAdd(int k, double v){
        key[++cntd] = k;
        val[cntd] = v;
        heap[++cnth] = cntd;
        int cur = cnth, father = cur / 2;
        while(cur > 1 && val[heap[father]] > val[heap[cur]]){
            swap(heap[father], heap[cur]);
            cur = father;
            father = cur / 2;
        }
    }

    int heapPop(){
        int ans = heap[1];
        heap[1] = heap[cnth--];
        int cur = 1, l = cur * 2, r = l + 1, best;
        while(l <= cnth){
            best = r <= cnth && val[heap[r]] < val[heap[l]] ? r : l;
            best = val[heap[best]] < val[heap[cur]] ? best : cur;
            if(best == cur) {
                break;
            }
            swap(heap[best], heap[cur]);
            cur = best;
            l = cur * 2;
            r = l + 1;
        }
        return ans;
    }

    bool heapEmpty(){
        return cnth == 0;
    }

    void dijkstra(){
        fill(dis, dis + MAXN, INF);
        dis[n] = 0;
        cntd = cnth = 0;
        heapAdd(n, 0.0);
        while(!heapEmpty()){
            int top = heapPop();
            int u = key[top];
            double w = val[top];
            if(!vis[u]){
                vis[u] = true;
                for(int e = headr[u], v; e != 0; e = nextr[e]){
                    v = tor[e];
                    if(dis[v] > w + weightr[e]){
                        dis[v] = w + weightr[e];
                        path[v] = e;
                        heapAdd(v, dis[v]);
                    }
                }
            }
        }
    }

    void mergeRoad(){
        cntd = cnth = 0;
        for(int i = 1; i <= n; i++){
            heapAdd(i, dis[i]);
        }
        dist[0] = -1;
        while(!heapEmpty()){
            int top = heapPop();
            int u = key[top];
            for(int e = headg[u], v; e != 0; e = nextg[e]){
                v = tog[e];
                if(e != path[u]){
                    rt[u] = merge(rt[u], init(v, weightg[e] + dis[v] - dis[u]));
                }
            }
            if(path[u] != 0){
                rt[u] = merge(rt[u], rt[tog[path[u]]]);
            }
        }
    }

    int expand(){
        int ans = 0;
        money -= dis[1];
        if(money >= 0){
            ans++;
            cntd = cnth = 0;
            if(rt[1] != 0){
                heapAdd(rt[1], dis[1] + cost[rt[1]]);
            }
            while(!heapEmpty()){
                int top = heapPop();
                int h = key[top];
                double w = val[top];
                money -= w;
                if(money < 0){
                    break;
                }
                ans++;
                if(ls[h] != 0){
                    heapAdd(ls[h], w - cost[h] + cost[ls[h]]);
                }
                if(rs[h] != 0){
                    heapAdd(rs[h], w - cost[h] + cost[rs[h]]);
                }
                if(to[h] != 0 && rt[to[h]] != 0){
                    heapAdd(rt[to[h]], w + cost[rt[to[h]]]);
                }
            }
        }
        return ans;
    }

    int main(){
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m >> money;
        int u, v;
        double w;
        for(int i = 1; i <= m; i++){
            cin >> u >> v >> w;
            if(u != n){
                addEdgeG(u, v, w);
                addEdgeR(v, u, w);
            }
        }
        dijkstra();
        mergeRoad();
        int ans = expand();
        cout << ans << endl;
        return 0;
    }

mergeRoad 里 e != path[u] 的比较依赖正图边与反图边编号一一对应：main 里每条输入边都同时 addEdgeG 和 addEdgeR，两套编号同步递增。这是这道题里最容易被忽略的前提。

三条候选路径依次弹出：最短路 1→2→3 费用 2；偏离直连 1→3 费用 4；再偏离 2→1 后沿树回 n，路径 1→2→1→2→3 费用 6。如果 money 是 13，这三条都付得起，还能继续展开后面的候选，直到钱不够为止。

总结

k 短路的最优解把问题拆成三段：反图 dijkstra 提供 dis 与最短路树；每个点的偏离边堆 rt[u] 用可持久化左偏树维护，前缀靠拷贝共享；expand 从最短路开始按预算从小到大展开候选，每弹一个就是下一条路径。偏离增量的定义 w + dis[v] - dis[u] 是全部正确性的核心：它保证候选预算严格递增，所以弹出的顺序就是路径费用的顺序。可持久化左偏树在这里的价值是拷贝：同一个偏离边节点被多个点的堆共享，展开任何一个都不会破坏其他候选。
