同余最短路：以最小步长为模

给一堆正数、每种个数无限，问能拼出哪些数、拼不出哪个最大。这类题直接做完全背包会撞上超大的上限（楼层数 h 能到 2^63）。换个思路：拿一个数当模，把「拼数」变成「余数在图上跑最短路」，图的点数只有模长那么多，跑完答案直接数出来。

零、同余分组：拼数变成图上最短路

设这些正数里最小的一个是 x。任何一个拼出的数 v，都可以写成 v = d + k·x：先把 x 看成「基准」，每多拼一个 x 只改变大小、不改变 v 对 x 的余数。于是同一个余数 r 下的可拼数，是从某个最小值 dist[r] 起、每加一个 x 得到一串：dist[r], dist[r]+x, dist[r]+2x, …

那「能拼出的数有多少个 ≤ 上限」就变成：对每个余数 r，数一下这串里 ≤ 上限的有几个，即 (上限 − dist[r]) / x + 1（当 dist[r] ≤ 上限时）。

剩下的问题是 dist[r] 怎么求。建一张图：点是余数 0..x−1；每出现一个别的数 y，连边 r → (r+y) mod x、边权 y。从 0 出发跑最短路，dist[r] 就是「余数为 r 的最小可拼值」。图只有 x 个点，x 多大图就多大。

这个分组是整套方法的骨架：模数越小图越小。x 取最小的那个数最划算，但先排序再取第一个也成立，只要 x 本身在可拼集合里，dist[r]+k·x 就都是可拼的。

一、题1 跳楼机：楼层上限 2^63 也能数

电梯在 1 层，每次能向上 x、y、z 层，或者回 1 层。问 1..h 里能到几层。h 读进来先减 1：回 1 层意味着任意时刻都能清零重来，所以能到的楼层 = 1 + 从偏移 0 出发能拼出的偏移个数，偏移上限就是 H = h−1。

模数直接取 x。连两条边：i → (i+y) mod x 权 y、i → (i+z) mod x 权 z。dijkstra 跑出 dist[r]，再对每个余数套统计公式。x、y、z 最大 10^5，x 个点、2x 条边，dijkstra 的 O(x log x) 完全够。

#include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 100001;
    const int MAXM = 200001;

    long long h;
    int x, y, z;

    int head[MAXN], nxt[MAXM], to[MAXM], cnt;
    long long weight[MAXM];

    long long distanceArr[MAXN];
    bool visitedArr[MAXN];

    void prepare() {
    	cnt = 1;
    	fill(head, head + x, 0);
    	fill(distanceArr, distanceArr + x, LLONG_MAX);
    	fill(visitedArr, visitedArr + x, false);
    }

    void addEdge(int u, int v, long long w) {
    	nxt[cnt] = head[u];
    	to[cnt] = v;
    	weight[cnt] = w;
    	head[u] = cnt++;
    }

    void dijkstra() {
    	priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> heap;
    	heap.push({0, 0});
    	distanceArr[0] = 0;
    	while (!heap.empty()) {
    		auto cur = heap.top();
    		heap.pop();
    		int u = cur.second;
    		long long w = cur.first;
    		if (visitedArr[u]) {
    			continue;
    		}
    		visitedArr[u] = true;
    		for (int ei = head[u], v; ei > 0; ei = nxt[ei]) {
    			v = to[ei];
    			if (!visitedArr[v] && distanceArr[v] > w + weight[ei]) {
    				distanceArr[v] = w + weight[ei];
    				heap.push({distanceArr[v], v});
    			}
    		}
    	}
    }

    long long compute() {
    	dijkstra();
    	long long ans = 0;
    	for (int i = 0; i < x; i++) {
    		if (distanceArr[i] <= h) {
    			ans += (h - distanceArr[i]) / x + 1;
    		}
    	}
    	return ans;
    }

    int main() {
    	cin >> h >> x >> y >> z;
    	h = h - 1;
    	prepare();
    	for (int i = 0; i < x; i++) {
    		addEdge(i, (i + y) % x, y);
    		addEdge(i, (i + z) % x, z);
    	}
    	cout << compute() << endl;
    	return 0;
    }

实测（对拍）：2000 组随机小用例（x,y,z ∈ [1,8]，h ≤ 60），课程输出与暴力 BFS 枚举可达偏移完全一致。有个细节：模数取 x 而不是取 x、y、z 的最小值也正确，只是图更大一点，课程代码直接用第一个数 x。

二、题2 牛场围栏：拼不出的最大长度

每种木棍长度 arr[i]，每根可以消掉最多 m 的长度，所以一根 arr[i] 的木棍实际能用 [max(1, arr[i]−m), arr[i]] 里任意长度。把全部可用长度收进一个集合（去重），最短的那个做模 x。

x 等于 1 时所有长度都能拼（模 1 只有一个余数 0，dist[0]=0），返回 −1。每个可用长度 j 连边 k → (k+j) mod x 权 j。跑完 dijkstra：

某余数完全不可达（dist[r] = ∞）→ 这个余数的长度一个都拼不出，不可拼的长度无穷多，返回 −1；否则答案 = max(dist[r] − x)。为什么减 x：dist[r] 是该余数下最小可拼值，dist[r]−x 一定拼不出（拼得出就轮不到 dist[r] 当最小），它就是该余数下最大的不可拼值，所有余数取最大。

两个 −1 别混：x==1 是「全都能拼」，dist[r]==∞ 是「拼不出的有无穷多」，题目对这两种都要求输出 −1，但含义相反。

#include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 101;
    const int MAXV = 3001;
    const int MAXM = 30001;

    int inf = INT_MAX;
    int n, m, x;
    int arr[MAXN];
    bool setArr[MAXV];

    int head[MAXV], nxt[MAXM], to[MAXM], weight[MAXM], cnt;
    int distanceArr[MAXV];
    bool visitedArr[MAXV];

    void prepare() {
    	cnt = 1;
    	fill(setArr, setArr + MAXV, false);
    	fill(head, head + x, 0);
    	fill(distanceArr, distanceArr + x, inf);
    	fill(visitedArr, visitedArr + x, false);
    }

    void addEdge(int u, int v, int w) {
    	nxt[cnt] = head[u];
    	to[cnt] = v;
    	weight[cnt] = w;
    	head[u] = cnt++;
    }

    void dijkstra() {
    	priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> heap;
    	heap.push({0, 0});
    	distanceArr[0] = 0;
    	while (!heap.empty()) {
    		auto cur = heap.top();
    		heap.pop();
    		int u = cur.second, w = cur.first;
    		if (visitedArr[u]) {
    			continue;
    		}
    		visitedArr[u] = true;
    		for (int ei = head[u], v; ei > 0; ei = nxt[ei]) {
    			v = to[ei];
    			if (!visitedArr[v] && distanceArr[v] > w + weight[ei]) {
    				distanceArr[v] = w + weight[ei];
    				heap.push({distanceArr[v], v});
    			}
    		}
    	}
    }

    int compute() {
    	int ans = 0;
    	if (x == 1) {
    		ans = -1;
    	} else {
    		for (int i = 1; i <= n; i++) {
    			for (int j = max(1, arr[i] - m); j <= arr[i]; j++) {
    				if (!setArr[j]) {
    					setArr[j] = true;
    					for (int k = 0; k < x; k++) {
    						addEdge(k, (k + j) % x, j);
    					}
    				}
    			}
    		}
    		dijkstra();
    		for (int i = 1; i < x; i++) {
    			if (distanceArr[i] == inf) {
    				ans = -1;
    				break;
    			}
    			ans = max(ans, distanceArr[i] - x);
    		}
    	}
    	return ans;
    }

    int main() {
    	cin >> n >> m;
    	x = inf;
    	for (int i = 1; i <= n; i++) {
    		cin >> arr[i];
    		x = min(x, max(1, arr[i] - m));
    	}
    	prepare();
    	cout << compute() << endl;
    	return 0;
    }

实测（对拍）：3000 组随机小用例（n ≤ 6，arr ∈ [2,10]，m ∈ [1,5]），课程输出与「完全背包打表找最大不可拼」完全一致；「无穷多」情形（余数不可达）两边都判 −1。

总结

同余最短路把「拼数」压成「模 x 的余数图」：x 个点，每个可用数 y 连一条余数转移边，dist[r] = 余 r 的最小可拼值，答案按 (上限−dist[r])/x+1 逐余数数出来。

跳楼机把楼层偏移做成余数图直接数个数；围栏先收集可用长度、最短的做模，dist[r]−x 就是该余数下最大不可拼值，某余数不可达就是无穷多。两道题都是同一个 dijkstra，差别只在建边和统计。

下一步的变体：边权只有 0 和 1 时用 01bfs 更快，以及不建图、按子环转两圈的 O(n·x) 做法，后两页分别讲。
