最假女选手：最大值和最小值两路一起压

大于 x 的数砍到 x，小于 x 的数抬到 x，两个方向都要整段做完。靠的是两边各自三个量：「最大值、最大值个数、严格次大」和「最小值、最小值个数、严格次小」。六种操作摆在一起，结点上就是六个信息加一个加法标记，下发的时候四件事要按固定顺序做。

零、背景

先只看压最大值这一半。抬最小值的时候，动的是当前等于最小值的那批格子，动的幅度是 x 减去最小值，需要的也是「最小值、最小值个数、严格次小」。

麻烦在两种操作会互相干扰。一段里既有被抬上来的、又有被压下去的，最大值和最小值可能在同一次操作里先后改变。所以最大值那一头和最小值那一头不能合用，得各记一套三个量。

再叠上区间加，一共三种修改。加法会整体平移所有数，压和抬的基准线也跟着平移，这是下发顺序必须讲究的原因。

一、题目

给定一个长度为 n 的数组 arr，然后 m 条操作：

  • 操作 1 l r x：arr[l..r] 范围内每个数都加上 x
  • 操作 2 l r x：arr[l..r] 范围内小于 x 的数都变成 x
  • 操作 3 l r x：arr[l..r] 范围内大于 x 的数都变成 x
  • 操作 4 l r：查询 arr[l..r] 的累加和
  • 操作 5 l r：查询 arr[l..r] 的最大值
  • 操作 6 l r：查询 arr[l..r] 的最小值

其中 n 和 m 都不超过 5×10⁵，累加和要用 long long。

二、一个结点装六个信息

三个管最大值那一头：mx 是最大值，cntMax 是有几个格子等于它，sem 是严格次大值。另外三个管最小值那一头，写法完全对称：mn、cntMin、sim（严格次小值）。再加一个 sum 和一个 addTag。

没有次大值的时候 sem 记成负无穷，没有次小值的时候 sim 记成正无穷。叶子结点两者都不存在，这正是它作为递归终点时不需要额外判断的原因。

long long sum[MAXN << 2];
    int addTag[MAXN << 2];

    int mx[MAXN << 2], cntMax[MAXN << 2], sem[MAXN << 2];   // 最大值、个数、严格次大
    int mn[MAXN << 2], cntMin[MAXN << 2], sim[MAXN << 2];   // 最小值、个数、严格次小

    void up(int i) {
        int l = i << 1, r = i << 1 | 1;
        sum[i] = sum[l] + sum[r];
        if (mx[l] > mx[r]) {
            mx[i] = mx[l];  cntMax[i] = cntMax[l];  sem[i] = max(sem[l], mx[r]);
        } else if (mx[l] < mx[r]) {
            mx[i] = mx[r];  cntMax[i] = cntMax[r];  sem[i] = max(mx[l], sem[r]);
        } else {
            mx[i] = mx[l];  cntMax[i] = cntMax[l] + cntMax[r];  sem[i] = max(sem[l], sem[r]);
        }
        if (mn[l] < mn[r]) {
            mn[i] = mn[l];  cntMin[i] = cntMin[l];  sim[i] = min(sim[l], mn[r]);
        } else if (mn[l] > mn[r]) {
            mn[i] = mn[r];  cntMin[i] = cntMin[r];  sim[i] = min(mn[l], sim[r]);
        } else {
            mn[i] = mn[l];  cntMin[i] = cntMin[l] + cntMin[r];  sim[i] = min(sim[l], sim[r]);
        }
    }

up 里三组分支要注意：两个孩子最大值相等时 cntMax 是相加而不是取 max，sem 也从两边的次大值里取大的那一个。最小值那一头照抄一遍，取 min 的地方换成取 max。

void build(int l, int r, int i) {
        if (l == r) {
            sum[i] = mx[i] = mn[i] = arr[l];
            sem[i] = -INF;   // 没有次大值
            sim[i] = INF;    // 没有次小值
            cntMax[i] = cntMin[i] = 1;
        } else {
            int mid = (l + r) >> 1;
            build(l, mid, i << 1);
            build(mid + 1, r, i << 1 | 1);
            up(i);
        }
        addTag[i] = 0;
    }

三、两种「压平」操作

先说抬最小值。整段一次做完的条件是 jobv < sim[i]：次小值还在上面顶着，说明比 jobv 小的只有当前这批最小值，把它们抬到 jobv 之后不会和别人合并。合并一旦发生，cntMin 就得重新数，而这里的代码故意不数它。

压最大值同理，条件是 sem[i] < jobv。两个不等号都是严格的，这保证了整段命中时那批被改动的格子数量就是 cntMin 或 cntMax，不用重新统计。

void lazyAdd(int i, int n, int v) {
        sum[i] += 1LL * v * n;
        mx[i] += v;  sem[i] += sem[i] == -INF ? 0 : v;
        mn[i] += v;  sim[i] += sim[i] == INF ? 0 : v;
        addTag[i] += v;
    }

    // 把最大值压到 v：只有「当前就等于最大值」的那些格子会动
    void lazySetMin(int i, int v) {
        sum[i] += 1LL * (v - mx[i]) * cntMax[i];
        if (mn[i] == mx[i]) mn[i] = v;          // 整段只有一个值，最小值跟着走
        else if (sim[i] == mx[i]) sim[i] = v;   // 次小值正好是最大值，也要压下来
        mx[i] = v;
    }

    // 把最小值抬到 v，对称的写法
    void lazySetMax(int i, int v) {
        sum[i] += 1LL * (v - mn[i]) * cntMin[i];
        if (mx[i] == mn[i]) mx[i] = v;
        else if (sem[i] == mn[i]) sem[i] = v;
        mn[i] = v;
    }

    void down(int i, int ln, int rn) {
        int l = i << 1, r = i << 1 | 1;
        if (addTag[i] != 0) {                   // 先发加法，后面两个赋值才知道基准在哪
            lazyAdd(l, ln, addTag[i]);
            lazyAdd(r, rn, addTag[i]);
            addTag[i] = 0;
        }
        if (mx[l] > mx[i]) lazySetMin(l, mx[i]);
        if (mn[l] < mn[i]) lazySetMax(l, mn[i]);
        if (mx[r] > mx[i]) lazySetMin(r, mx[i]);
        if (mn[r] < mn[i]) lazySetMax(r, mn[i]);
    }

lazySetMin 里那两句 if 值得单独看。整段只有一个值时 mn[i] 和 mx[i] 相等，压掉最大值的同时最小值也得跟着下来；否则要检查次小值是不是正好等于最大值，等于的话它也得一起压。抬最小值那一边是对称的两句。

// 小于 x 的数都变成 x，也就是整体往上抬到 x
    void setMax(int jobl, int jobr, int jobv, int l, int r, int i) {
        if (jobv <= mn[i]) return;                       // 连最小值都够不着
        if (jobl <= l && r <= jobr && jobv < sim[i]) {   // 次小值还在上面顶着
            lazySetMax(i, jobv);
        } else {
            int mid = (l + r) >> 1;
            down(i, mid - l + 1, r - mid);
            if (jobl <= mid) setMax(jobl, jobr, jobv, l, mid, i << 1);
            if (jobr > mid) setMax(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
            up(i);
        }
    }

    // 大于 x 的数都变成 x
    void setMin(int jobl, int jobr, int jobv, int l, int r, int i) {
        if (jobv >= mx[i]) return;
        if (jobl <= l && r <= jobr && sem[i] < jobv) {
            lazySetMin(i, jobv);
        } else {
            int mid = (l + r) >> 1;
            down(i, mid - l + 1, r - mid);
            if (jobl <= mid) setMin(jobl, jobr, jobv, l, mid, i << 1);
            if (jobr > mid) setMin(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
            up(i);
        }
    }
void add(int jobl, int jobr, int jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            lazyAdd(i, r - l + 1, jobv);
        } else {
            int mid = (l + r) >> 1;
            down(i, mid - l + 1, r - mid);
            if (jobl <= mid) add(jobl, jobr, jobv, l, mid, i << 1);
            if (jobr > mid) add(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
            up(i);
        }
    }

四、下发的时候先发加法

down 一共做五件事，顺序不能乱。第一件是把 addTag 交给两个孩子，因为后面四个判断全部建立在「孩子的 mx 和 mn 已经和父亲对齐」这个前提上。

四个判断的形式都是拿孩子和父亲比：孩子的最大值比父亲的大，说明父亲被压过而孩子还不知道，要补一次 lazySetMin；孩子的最小值比父亲的小，同理补一次 lazySetMax。父亲自己是不会比孩子更极端的，所以只有「孩子越界」这一种情况需要处理。

把 addTag 挪到最后发会怎样？孩子的 mx 和 mn 还是加上加法之前的旧值，四个判断照着旧值做，等于把「先加后压」变成了「先压后加」。图上那组数据里左半的累加和会从 18 变成 21。

五、复杂度分析

项代价
建树O(n)
区间加O(log n)
两种压平O(log n) 均摊
三种查询O(log n)
空间O(n)：八个四倍数组

均摊那一栏是两套势能叠加的结果。抬最小值和压最大值各自有一套标签回收分析，两者写在同一个结点上并不互相破坏：一次下钻要么走最大那一头、要么走最小那一头，势能的账各自记各自的。

按 n 等于 5×10⁵ 算：一个 long long 数组加七个 int 数组，每个长度都是四倍，一共 36×4×500001 字节，约 68.7 MiB。三个管最大值、三个管最小值，加上 sum 和 addTag，一个都不能省。

实测（g++ -O2，同一份操作跑 5 遍取最好）：n 和 m 都是 5×10⁵ 时 316.969 ms；n 是 5×10⁵、m 是 2×10⁵ 时 138.692 ms。对照的朴素做法是每次操作扫一遍区间，m 是 5×10⁵ 时按前 2000 次外推要 10316.9 ms，差了三十倍多。

六、几种写错的样子

同样每组 3000 个小样本、每个样本最多 60 步，值域分三档：

  • down 里把 addTag 挪到四个判断之后才发：96.33% / 97.23% / 97.10%。这是整套结构里最容易犯也最容易被抓到的错。
  • up 里最小值的个数取 max 而不是相加：75.87% / 72.43% / 71.77%。两个孩子最小值相等时这一批要合并。
  • lazySetMin 里删掉「次小值正好等于最大值时一起压」那一句：10.50% / 18.13% / 17.73%。漏改的 sim 只在后续操作正好摸到那个结点时才现形，所以抓取率很低但确实是错的。
  • 抬最小值的整段命中条件写成 jobv <= sim[i]：三档都是 0.00%，答案上完全等价。多出来的情形是 jobv 正好等于 sim[i]，这时抬上来的那批和原来就在 sim 上的那批合并，cntMin 偏小；但同一个动作也让 sim[i] 落到和 mn[i] 相等，整段命中条件「jobv 严格小于 sim」从此被挡住，偏小的 cntMin 不会被拿去算累加和。往上的祖先也是如此：最小值等于这个数的，sim 跟着塌下来，一样被挡住；最小值比它小的，压根不数这一批。为了确认不是「样本没跑到」，数了两种情形的进入次数：全包住且 jobv 等于 sim 的，三档分别是 6530、202、0 次，同期严格小于 sim 的是 73790、86271、85117 次。相等的情形确实跑到过，值域只有 5 的时候占了百分之八，但一次都没造成答案不同。另拿值域 2、3、5、200 四档各 3000 组和朴素数组对拍，答案错 0 组，这条分支分别进了 7600、6037、4321、144 次。

七、总结

六种操作放在一起，结点上的信息分成对称的两组，每组三个量加一个加法标记。up 是两组各自合并，down 是先发加法、再按「孩子比父亲更极端」补四次压平。

整段命中的两个条件都用严格不等号，这一点决定了 lazySetMin 和 lazySetMax 里不用重新数个数。等价写法那条提醒我们：不等号放宽一格未必出错，要看放宽后多出来的那种情形会不会让错误的状态被读到。

最后

一是 down 的第一句必须是 addTag，四个判断之后不能再补发一次。

二是 lazySetMin 里两句 if 的顺序：先判「整段只有一个值」，再判「次小值等于最大值」，两个条件可能同时成立。

三是 sem 和 sim 的无穷值分别用负无穷和正无穷，加标记时要先判断再平移，否则无穷会被加成一个有限的数。
