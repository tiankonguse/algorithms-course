糖果药片与树上游戏：配对 dp 当钦定

前两页的「钦定」靠排列数和隔板法直接数，这一页两道题的钦定要 dp 先数出来：糖果药片用一维递推数「恰好 i 个糖果大配对」，树上游戏用树形背包数「恰好 i 个非平局回合」，再乘上剩下配对的任意排列当 g，最后形式四反演。两道题都是 n 到 2000~5000 的 O(n²)。

零、糖果药片（P4859）

n 颗糖果、n 片药片，每个糖果的甜度 a[i]、每片药片的药效 b[i] 各不相同。糖果和药片一一配对，糖果甜度大于药片药效叫「糖果大」配对，反之叫「药片大」配对。要求糖果大配对的数量 = 药片大配对的数量 + k，模 1e9+9。n 到 2000。

糖果大 + 药片大 = n，糖果大 − 药片大 = k，解出糖果大 = (n+k)/2。n+k 是奇数时无解输出 0。目标变成「恰好 (n+k)/2 个糖果大配对」。

先把 a、b 都排序。对每个糖果 a[i]，small[i] = 比它小的药片数量（图上灰色药片）。dp[i][j] = 前 i 个糖果里恰好 j 个糖果大配对：第 i 个糖果要么不配大（继承 dp[i−1][j]），要么配大：前面 j−1 个大的已用掉 j−1 片药，还剩 small[i]−(j−1) 片可配，乘 (small[i]−j+1)。

钦定：g(i) = fac[n−i]·dp[n][i]。先钦定 i 个糖果大配对（dp[n][i] 种选法），剩下 n−i 对任意配（(n−i)! 种）。形式四反演恰好 k 个：

f(k) = Σi=k..n (−1)^(i−k) · C(i,k) · g(i)

#include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 2001;
    const int MOD = 1000000009;

    int n, k;
    int a[MAXN], b[MAXN];
    long long fac[MAXN], c[MAXN][MAXN], g[MAXN];
    long long dp[MAXN][MAXN];
    int small[MAXN];

    void build() {
        fac[0] = 1;
        for (int i = 1; i <= n; i++) fac[i] = fac[i - 1] * i % MOD;
        for (int i = 0; i <= n; i++) {
            c[i][0] = 1;
            for (int j = 1; j <= i; j++) c[i][j] = (c[i - 1][j] + c[i - 1][j - 1]) % MOD;
        }
    }

    long long compute() {
        build();
        sort(a + 1, a + n + 1);
        sort(b + 1, b + n + 1);
        // small[i] = 比 a[i] 小的药片数量
        for (int i = 1, cnt = 0; i <= n; i++) {
            while (cnt + 1 <= n && b[cnt + 1] < a[i]) cnt++;
            small[i] = cnt;
        }
        // dp[i][j] = 前 i 个糖果里恰好 j 个「糖果大」配对
        dp[0][0] = 1;
        for (int i = 1; i <= n; i++) {
            dp[i][0] = dp[i - 1][0];
            for (int j = 1; j <= i; j++) {
                dp[i][j] = (dp[i - 1][j] + dp[i - 1][j - 1] * (small[i] - j + 1) % MOD) % MOD;
            }
        }
        // g[i] = 钦定至少 i 个糖果大配对：先挑 i 个，剩下 n-i 个任意配
        for (int i = 0; i <= n; i++) g[i] = fac[n - i] * dp[n][i] % MOD;
        // 形式四反演：恰好 k 个
        long long ans = 0;
        for (int i = k; i <= n; i++) {
            if (((i - k) & 1) == 0) ans = (ans + c[i][k] * g[i]) % MOD;
            else ans = (ans + c[i][k] * g[i] % MOD * (MOD - 1)) % MOD;
        }
        return ans;
    }

    int main() {
        cin >> n >> k;
        for (int i = 1; i <= n; i++) cin >> a[i];
        for (int i = 1; i <= n; i++) cin >> b[i];
        // 糖果大 + 药片大 = n，糖果大 = 药片大 + k
        // => 糖果大 = (n + k) / 2，n + k 为奇数时无解
        if (((n + k) & 1) == 0) {
            k = (n + k) / 2;
            cout << compute() << endl;
        } else {
            cout << 0 << endl;
        }
        return 0;
    }

dp 表 O(n²) 填，fac、组合数表 O(n²) 建，总复杂度 O(n²)。

一、游戏（P6478）

一棵 n 个点的树（n 到 5000，偶数），n/2 个点属于小 A、n/2 个点属于小 B（字符串 arr，0 是 A、1 是 B）。玩 m = n/2 个回合，每回合两人各从自己剩下的点里选一个当前点。若小 A 当前点在对方当前点的子树里，小 A 胜；若小 B 当前点在对方当前点的子树里，小 B 胜；否则平局。问恰好 k 回合非平局的游戏方法数，k 从 0 到 m 全输出，模 998244353。

先定游戏怎么计数：两场游戏不同，当且仅当存在小 A 的点 x，小 B 在小 A 选择 x 的那个回合所选择的点不同。也就是说，每个 A 点配哪个 B 点定了，游戏就定了，A 点自己的选择顺序不参与计数。所以游戏 = A 点集到 B 点集的双射，一共 m! 种。对拍时这一条最容易踩：写 A 排列 × B 排列会多出 m! 倍。

图上 n=6 的树，A 点画白、B 点画黑。B 点在 A 点的子树里 → A 胜；A 点在 B 点的子树里 → B 胜；互不包含 → 平局。树型 dp 统计非平局回合：子树合并是卷积，u 自己参与的回合要求对手点在 u 的子树里，这保证非平局（谁胜由方向定）。

dp[u][j] = u 的子树内恰有 j 个非平局回合的配对方式。合并子树：枚举左子树 l 个、右子树 r 个，卷积相加，上界用子树节点数压到 sz/2。u 自己配对手：对手数量 num = belong[u][arr[u]^1]（u 子树里对方阵营的点数），前面已配 j−1 个用掉 j−1 个对手，还剩 num−j+1 个可配：

dp[u][j] += dp[u][j−1] · (num−j+1)

钦定：g(i) = dp[1][i]·fac[m−i]，先钦定 i 个非平局回合（dp[1][i]），剩下 m−i 个回合的 B 点任意排列。形式四反演：

f(k) = Σi=k..m (−1)^(i−k) · C(i,k) · g(i)

#include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 5001;
    const int MOD = 998244353;

    int n, m;
    int arr[MAXN];
    long long fac[MAXN], c[MAXN][MAXN];
    int head[MAXN], nxt[MAXN << 1], to[MAXN << 1], cnt;
    int sz[MAXN];
    int belong[MAXN][2];
    long long dp[MAXN][MAXN];
    long long backup[MAXN];
    long long g[MAXN], f[MAXN];

    void build() {
        cnt = 1;
        fac[0] = 1;
        for (int i = 1; i <= n; i++) {
            head[i] = 0;
            fac[i] = fac[i - 1] * i % MOD;
        }
        for (int i = 0; i <= n; i++) {
            c[i][0] = 1;
            for (int j = 1; j <= i; j++) c[i][j] = (c[i - 1][j] + c[i - 1][j - 1]) % MOD;
        }
    }

    void addEdge(int u, int v) {
        nxt[cnt] = head[u];
        to[cnt] = v;
        head[u] = cnt++;
    }

    // 树型 dp（递归版）：dp[u][j] = 以 u 为根的子树里恰有 j 个「A 胜」回合
    void dfs(int u, int fa) {
        sz[u] = 1;
        belong[u][arr[u]] = 1;
        dp[u][0] = 1;
        for (int e = head[u], v; e > 0; e = nxt[e]) {
            v = to[e];
            if (v != fa) {
                dfs(v, u);
                // 子树合并：卷积，枚举上界用子树节点数优化
                for (int i = 0; i <= min(sz[u] / 2, m); i++) {
                    backup[i] = dp[u][i];
                    dp[u][i] = 0;
                }
                for (int l = 0; l <= min(sz[u] / 2, m); l++) {
                    for (int r = 0; r <= min(sz[v] / 2, m - l); r++) {
                        dp[u][l + r] = (dp[u][l + r] + backup[l] * dp[v][r]) % MOD;
                    }
                }
                sz[u] += sz[v];
                belong[u][0] += belong[v][0];
                belong[u][1] += belong[v][1];
            }
        }
        // 把 u 自己也配进一个回合：对手数量 num，可配 (num - i + 1) 种
        int num = belong[u][arr[u] ^ 1];
        for (int i = 1; i <= min(num, m); i++) backup[i] = dp[u][i];
        for (int i = 1; i <= min(num, m); i++) {
            dp[u][i] = (dp[u][i] + backup[i - 1] * (num - i + 1)) % MOD;
        }
    }

    void compute() {
        dfs(1, 0);
        // g[i] = 钦定至少 i 个 A 胜回合：先挑 i 个，剩下 m-i 个回合任意配
        for (int i = 0; i <= m; i++) g[i] = dp[1][i] * fac[m - i] % MOD;
        // 形式四反演：恰好 k 个非平局
        for (int k = 0; k <= m; k++) {
            for (int i = k; i <= m; i++) {
                if (((i - k) & 1) == 0) f[k] = (f[k] + c[i][k] * g[i]) % MOD;
                else f[k] = (f[k] + c[i][k] * g[i] % MOD * (MOD - 1)) % MOD;
            }
        }
    }

    int main() {
        cin >> n;
        m = n >> 1;
        build();
        string str;
        cin >> str;
        for (int i = 1; i <= n; i++) arr[i] = str[i - 1] - '0';
        for (int i = 1, u, v; i < n; i++) {
            cin >> u >> v;
            addEdge(u, v);
            addEdge(v, u);
        }
        compute();
        for (int k = 0; k <= m; k++) cout << f[k] << '\n';
        return 0;
    }

链式前向星存边，树形背包按 sz 限界，每对子树合并一次，总复杂度 O(n²)。

总结

两道题的骨架一样：先用 dp 数出「恰好 i 个目标配对」的选法，乘剩下配对的任意排列得到 g(i)（钦定至少 i 个），再形式四反演成恰好 k 个。糖果药片的一维 dp 靠排序后的 small 计数，转移是 dp[i][j] = dp[i−1][j] + dp[i−1][j−1]·(small[i]−j+1)；树上游戏是树形背包，u 配对手乘 (num−j+1)，子树合并按 sz 限界卷积。差别只在「配对」的语义：一个是数值大小，一个是树的包含关系。游戏计数记得是 A 到 B 的双射 m! 种，A 点顺序不参与。
