FHQ Treap 区间操作：文本编辑器与文艺平衡树

上一页的 split 按 key 值劈树，能回答「比我小的有几个」这种问题。这一页要处理的是连续区间：光标后删 n 个字符、把 [l, r] 整段翻转。连续区间在树里是一段连续的中序位置，所以 split 换成按排名劈：前 rank 个节点一棵，剩下的另一棵。区间移动、区间翻转都是「取中段、打标记、拼回去」三板斧。

零、背景

按排名分裂的 split 和按值分裂的代码几乎一样，只是递归条件从 key[i] ≤ num 换成 siz[ls[i]] + 1 ≤ rank。区间操作的标准姿势：

1）split(head, y)：前 y 个节点成 lm，剩下的成 r。
2）split(lm, x - 1)：前 x - 1 个成 l，中间的 [x, y] 成 m。
3）对 m 做操作（删、取中序、打翻转标记），再 merge(merge(l, m), r) 拼回。

三步里只有第三步不同，区间操作之间的差异全在「对 m 干什么」。

一、文本编辑器 P4008：区间移动

文本编辑器要维护一段文本和一个光标，支持 6 种操作：Move 移动光标、Insert 在光标处插入字符串、Delete 删光标后 n 个字符、Get 输出光标后 n 个字符、Prev / Next 移动一位。光标位置 pos 直接当排名用：插入在 pos 后，就是 split(head, pos) 把前 pos 个字符分出来，新字符逐个 merge 到左侧树的尾部，再和右侧拼回。

Insert 操作有个过滤规则：题目保证字符串里 ASCII 在 [32, 126] 范围的字符一定有 n 个，其他字符要过滤掉。读取时逐个判断字符落在范围里才插进树。

Delete 和 Get 都要求光标后至少有 n 个字符，所以先 split(head, pos + n) 取前 pos + n 个，再 split 出中间的 n 个：Delete 直接丢掉，Get 中序遍历输出。操作之后树要拼回原状，光标不动。

二、题3 完整实现（P4008 文本编辑器）

字符用 char 数组存，中序遍历结果暂存在另一个数组里再批量输出。Get 保证有足够字符，不需要处理越界。初始文本为空，光标在开头。

#include <iostream>
    #include <vector>
    #include <cstdlib>
    #include <cstring>
    #include <algorithm>
    using namespace std;

    const int MAXN = 2000001;

    int head = 0;
    int cnt = 0;
    char key[MAXN];
    int ls[MAXN];
    int rs[MAXN];
    int siz[MAXN];
    double priority[MAXN];
    char ans[MAXN];
    int ansi;

    void up(int i) {
        siz[i] = siz[ls[i]] + siz[rs[i]] + 1;
    }

    void split(int l, int r, int i, int rank) {
        if (i == 0) {
            rs[l] = ls[r] = 0;
        } else {
            if (siz[ls[i]] + 1 <= rank) {
                rs[l] = i;
                split(i, r, rs[i], rank - siz[ls[i]] - 1);
            } else {
                ls[r] = i;
                split(l, i, ls[i], rank);
            }
            up(i);
        }
    }

    int merge(int l, int r) {
        if (l == 0 || r == 0) {
            return l + r;
        }
        if (priority[l] >= priority[r]) {
            rs[l] = merge(rs[l], r);
            up(l);
            return l;
        } else {
            ls[r] = merge(l, ls[r]);
            up(r);
            return r;
        }
    }

    void inorder(int i) {
        if (i != 0) {
            inorder(ls[i]);
            ans[++ansi] = key[i];
            inorder(rs[i]);
        }
    }

    int main() {
        srand(time(0));
        int pos = 0, len, l, m, lm, r;
        int n;
        cin >> n;
        for (int i = 1; i <= n; i++) {
            char op[10];
            cin >> op;
            if (op[0] == 'P') {
                pos--;
            } else if (op[0] == 'N') {
                pos++;
            } else if (op[0] == 'M') {
                cin >> pos;
            } else if (op[0] == 'I') {
                cin >> len;
                split(0, 0, head, pos);
                l = rs[0];
                r = ls[0];
                for (int j = 1; j <= len; j++) {
                    char ch = getchar();
                    while (ch < 32 || ch > 126) {
                        ch = getchar();
                    }
                    key[++cnt] = ch;
                    siz[cnt] = 1;
                    priority[cnt] = (double)rand() / RAND_MAX;
                    l = merge(l, cnt);
                }
                head = merge(l, r);
            } else if (op[0] == 'D') {
                cin >> len;
                split(0, 0, head, pos + len);
                lm = rs[0];
                r = ls[0];
                split(0, 0, lm, pos);
                l = rs[0];
                head = merge(l, r);
            } else {
                cin >> len;
                split(0, 0, head, pos + len);
                lm = rs[0];
                r = ls[0];
                split(0, 0, lm, pos);
                l = rs[0];
                m = ls[0];
                ansi = 0;
                inorder(m);
                head = merge(merge(l, m), r);
                for (int j = 1; j <= ansi; j++) {
                    cout << ans[j];
                }
                cout << '\n';
            }
        }
        return 0;
    }

三、文艺平衡树 P3391：懒翻转

翻转 [l, r] 就是把这一段中序倒过来。如果每次翻转都把 m 段的左右子树全换一遍，一次操作 O(len) 就废了。用线段树的懒更新思路：只给 m 段的根打一个 rev 标记，等后续 split / merge 走到这个节点时，再把标记往下传。

标记怎么传：down(i) 里如果 rev[i] 为真，交换 ls[i] 和 rs[i]，把 rev 异或传给两个子树，清掉自己的标记。整段翻转两次标记又变回 0，等于没翻。中序遍历时走到打了标记的节点，左右已经换过，输出就是翻转后的顺序。

关键点在于 rev 只打在 m 段的根上，m 段内部的节点一个都没动，代价 O(1)。后续 split 或 merge 走到 m 的子树时，down 把标记逐层传下去，保证每一层访问到的子树都是「已经翻转过」的状态。

演示：序列 1..6 连续翻转三次（树节点标 rev 的为懒标记，下方中序实时变化）

四、题4 完整实现（P3391 文艺平衡树）

n、k 都到 1e5。初始序列 1..n 按顺序建树，priority 随机生成。每次翻转 split 两刀取中段，rev[m] ^= 1，拼回。最后中序遍历输出。split 和 merge 里都要先 down 再递归，否则标记会漏传。

#include <iostream>
    #include <cstdio>
    #include <cstdlib>
    #include <cmath>
    #include <ctime>
    #include <cstring>
    #include <algorithm>
    using namespace std;

    const int MAXN = 100001;

    int head = 0;
    int cnt = 0;
    int key[MAXN];
    int ls[MAXN];
    int rs[MAXN];
    int siz[MAXN];
    double priority[MAXN];
    bool rev[MAXN];
    int ans[MAXN];
    int ansi;

    void up(int i) {
        siz[i] = siz[ls[i]] + siz[rs[i]] + 1;
    }

    void down(int i) {
        if (rev[i]) {
            swap(ls[i], rs[i]);
            rev[ls[i]] ^= 1;
            rev[rs[i]] ^= 1;
            rev[i] = false;
        }
    }

    void split(int l, int r, int i, int rank) {
        if (i == 0) {
            rs[l] = ls[r] = 0;
        } else {
            down(i);
            if (siz[ls[i]] + 1 <= rank) {
                rs[l] = i;
                split(i, r, rs[i], rank - siz[ls[i]] - 1);
            } else {
                ls[r] = i;
                split(l, i, ls[i], rank);
            }
            up(i);
        }
    }

    int merge(int l, int r) {
        if (l == 0 || r == 0) {
            return l + r;
        }
        if (priority[l] >= priority[r]) {
            down(l);
            rs[l] = merge(rs[l], r);
            up(l);
            return l;
        } else {
            down(r);
            ls[r] = merge(l, ls[r]);
            up(r);
            return r;
        }
    }

    void inorder(int i) {
        if (i != 0) {
            down(i);
            inorder(ls[i]);
            ans[++ansi] = key[i];
            inorder(rs[i]);
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        srand(time(0));
        int n, k;
        cin >> n >> k;
        for (int i = 1; i <= n; i++) {
            key[++cnt] = i;
            siz[cnt] = 1;
            priority[cnt] = (double)rand() / RAND_MAX;
            head = merge(head, cnt);
        }
        for (int i = 1, x, y, l, m, lm, r; i <= k; i++) {
            cin >> x >> y;
            split(0, 0, head, y);
            lm = rs[0];
            r = ls[0];
            split(0, 0, lm, x - 1);
            l = rs[0];
            m = ls[0];
            rev[m] ^= 1;
            head = merge(merge(l, m), r);
        }
        ansi = 0;
        inorder(head);
        for (int i = 1; i <= ansi; i++) {
            cout << ans[i] << " ";
        }
        cout << endl;
        return 0;
    }

五、总结

区间操作把「按值劈」换成「按排名劈」，代价只是递归条件从 key 比较换成 siz 比较。删除、取中序、打懒标记这三种「对 m 干什么」覆盖了文本编辑器和文艺平衡树。懒标记的核心纪律：split 和 merge 里每一次触碰节点前先 down，保证被拆开、被合并的子树都处于已翻转状态；rev 异或两次自动还原，省去显式清区间。这一页的按排名 split 也是下一页可持久化版本的基础。
