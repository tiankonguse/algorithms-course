带权并查集：一维数轴相对距离模型

普通并查集只回答「在不在一个集合」，带权并查集在路径压缩时顺手维护每个节点到集合头节点的距离，于是还能回答「l 和 r 相差多少」。区间累加和这类题，把区间端点当成数轴上的点，直接套这个模型。

零、一维数轴相对距离模型

把并查集里每个集合想成数轴上的一簇点，集合头节点是坐标原点。dist[i] 存节点 i 到头的距离，路径压缩之前这个值不一定对，find 的过程会把它修正。

三个操作，整节课的所有变化都从它们身上来：

1）find(i)：找 i 所在集合的头，顺带修正 dist[i]。路径压缩之前 i 的父是 t，压缩之后 dist[i] += dist[t]。父节点一路到头，差值一路累加，最后 dist[i] 就是从 i 到头节点的真实距离。

2）union(l, r, v)：l 和 r 属于两个集合，并且 l 到 r 的距离为 v，把两个集合合并。先 find 出 lf、rf，find 过程已经把 dist[l]、dist[r] 修正成各自到头 lf、rf 的距离，然后 father[lf] = rf，dist[lf] = v + dist[r] - dist[l]。

为什么是 v + dist[r] - dist[l]？合并后要求 dist[l] - dist[r] = v。dist[l] 变成 dist[lf] 加上旧的 dist[l]（lf 挂到 rf 下面，l 到头的新距离多出一截），dist[r] 不动。所以 dist[lf] + dist[l] - dist[r] = v，移项就是代码里那行。

3）query(l, r)：查询 l 和 r 的距离。find(l) == find(r) 才有距离关系，答案是 dist[l] - dist[r]；不同集合返回「没有答案」。方向别弄反：是 dist[l] 减 dist[r]，不是反过来。

一、区间累加和套进模型

有 n 个数字，给定若干区间累加和，再查一些区间累加和。关键一步是把它翻译成前缀和。前缀和 S[i] = a[1] + ... + a[i-1]，注意是前 i-1 项，不是前 i 项：S[1] = 0，S[n+1] 才是全部数字的和。于是区间 [l, r] 的累加和 = S[r+1] - S[l]。

「给定 l~r 累加和为 v」就是「S[r+1] 到 S[l] 的距离为 v」，写成 union(l, r+1, v)。「查询 l~r 累加和」就是 query(l, r+1)。下标 +1 是本页最容易被手滑写错的地方：区间右端 r 对应的是 S[r+1]，不是 S[r]。

为什么 S 要往后错一位？因为区间 [l, r] 包含 a[l] 到 a[r]，它的和用 S[r+1] - S[l] 刚好把 a[l] 到 a[r] 全框进来，S[r] - S[l-1] 也等价，但并查集节点从 1 开始编号时，S 下标用「右端 +1、左端原样」最省事，正好一个操作一行代码。

二、推导部分和（P8779）

模版题 1，直接把上面的模型搬上去。m 个给定先 union 建关系，q 个查询逐个 answer：同根输出 dist[l] - dist[r]，查不到输出 UNKNOWN。n 读进来先 +1，因为节点是 S[1..n+1]。

这里有个细节：给定操作全部当真，不存在矛盾检测，给什么就 union 什么。判断查询能不能答，只看 l 和 r+1 在不在一个集合。

推导部分和

// 推导部分和，带权并查集模版题1
    // n 个数字，下标 1~n；m 个区间累加和，q 个区间累加和查询
    // 查询能推出答案就打印，否则 UNKNOWN
    // 1 <= n, m, q <= 1e5，累加和不超过 long 范围
    // 测试链接 : https://www.luogu.com.cn/problem/P8779
    #include <bits/stdc++.h>
    using namespace std;

    const long long INF = LLONG_MAX;
    const int MAXN = 100002;

    int n, m, q;
    int father[MAXN];
    long long dist[MAXN];

    void prepare() {
        for (int i = 0; i <= n; i++) {
            father[i] = i;
            dist[i] = 0;
        }
    }

    int find(int i) {
        if (i != father[i]) {
            int tmp = father[i];
            father[i] = find(tmp);
            dist[i] += dist[tmp];
        }
        return father[i];
    }

    void unionSet(int l, int r, long long v) {
        int lf = find(l), rf = find(r);
        if (lf != rf) {
            father[lf] = rf;
            dist[lf] = v + dist[r] - dist[l];
        }
    }

    long long query(int l, int r) {
        if (find(l) != find(r)) {
            return INF;
        }
        return dist[l] - dist[r];
    }

    int main() {
        scanf("%d%d%d", &n, &m, &q);
        n++;
        prepare();
        for (int i = 1, l, r; i <= m; i++) {
            long long v;
            scanf("%d%d%lld", &l, &r, &v);
            r++;
            unionSet(l, r, v);
        }
        for (int i = 1, l, r; i <= q; i++) {
            scanf("%d%d", &l, &r);
            r++;
            long long v = query(l, r);
            if (v == INF) {
                printf("UNKNOWN\n");
            } else {
                printf("%lld\n", v);
            }
        }
        return 0;
    }

三、狡猾的商人（P2294）

模版题 2，多了一个矛盾检测：每个操作先 check，find(l) == find(r) 时如果 dist[l] - dist[r] != v，说明新操作和旧信息对不上，数据自相矛盾。矛盾的操作用不上 union，但 ans 已经置成 false，后续操作照常读入照常 check。

题目保证总收入不超过 int 范围，所以 dist 用 int 就够，这是三份模版代码里唯一不用 long long 的地方。

狡猾的商人

// 狡猾的商人，带权并查集模版题2
    // n 个月的收入，m 个操作 l r v 表示第 l 到 r 月总收入为 v
    // 判断给定数据是否自洽，矛盾打印 false
    // 1 <= n <= 100，1 <= m <= 1000，总收入不超过 int 范围
    // 测试链接 : https://www.luogu.com.cn/problem/P2294
    #include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 102;

    int t, n, m;
    bool ans;
    int father[MAXN];
    int dist[MAXN];

    void prepare() {
        ans = true;
        for (int i = 1; i <= n; i++) {
            father[i] = i;
            dist[i] = 0;
        }
    }

    int find(int i) {
        if (i != father[i]) {
            int tmp = father[i];
            father[i] = find(tmp);
            dist[i] += dist[tmp];
        }
        return father[i];
    }

    void unionSet(int l, int r, int v) {
        int lf = find(l), rf = find(r);
        if (lf != rf) {
            father[lf] = rf;
            dist[lf] = v + dist[r] - dist[l];
        }
    }

    bool check(int l, int r, int v) {
        if (find(l) == find(r)) {
            if ((dist[l] - dist[r]) != v) {
                return false;
            }
        }
        return true;
    }

    int main() {
        scanf("%d", &t);
        for (int c = 1; c <= t; c++) {
            scanf("%d%d", &n, &m);
            n++;
            prepare();
            for (int i = 1, l, r, v; i <= m; i++) {
                scanf("%d%d%d", &l, &r, &v);
                r++;
                if (!check(l, r, v)) {
                    ans = false;
                } else {
                    unionSet(l, r, v);
                }
            }
            printf("%s\n", ans ? "true" : "false");
        }
        return 0;
    }

四、错误答案数量（P3038）

模版题 3，和题 2 结构几乎一样，区别是输出矛盾操作的个数，而且多测，读到 EOF 结束。矛盾的操作「不进行」：count 加一，但不 union，所以后面可能还有别的操作和它撞上。

写多测循环时用 while (scanf(...) == 2) 而不是读一行字符串，能省掉整行解析的麻烦。

错误答案数量

// 错误答案数量，带权并查集模版题3
    // n 个数字，m 个操作 l r v 表示 l~r 累加和为 v
    // 与之前信息矛盾的操作为错误操作，不执行，打印错误操作数量
    // 1 <= n <= 2e5，1 <= m <= 4e4，多测到 EOF
    // 测试链接 : https://acm.hdu.edu.cn/showproblem.php?pid=3038
    #include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 200002;

    int n, m, ans;
    int father[MAXN];
    int dist[MAXN];

    void prepare() {
        ans = 0;
        for (int i = 0; i <= n; i++) {
            father[i] = i;
            dist[i] = 0;
        }
    }

    int find(int i) {
        if (i != father[i]) {
            int tmp = father[i];
            father[i] = find(tmp);
            dist[i] += dist[tmp];
        }
        return father[i];
    }

    void unionSet(int l, int r, int v) {
        int lf = find(l), rf = find(r);
        if (lf != rf) {
            father[lf] = rf;
            dist[lf] = v + dist[r] - dist[l];
        }
    }

    bool check(int l, int r, int v) {
        if (find(l) == find(r)) {
            if ((dist[l] - dist[r]) != v) {
                return false;
            }
        }
        return true;
    }

    int main() {
        while (scanf("%d%d", &n, &m) == 2) {
            n++;
            prepare();
            for (int i = 1, l, r, v; i <= m; i++) {
                scanf("%d%d%d", &l, &r, &v);
                r++;
                if (!check(l, r, v)) {
                    ans++;
                } else {
                    unionSet(l, r, v);
                }
            }
            printf("%d\n", ans);
        }
        return 0;
    }

五、总结

带权并查集 = 普通并查集 + 每个节点到头节点的距离。三句话记牢：find 路径压缩时 dist[i] += dist[t]；union 合并时 dist[lf] = v + dist[r] - dist[l]；query 只有同根才有答案，距离 = dist[l] - dist[r]。区间累加和题记得下标 +1（右端对应 S[r+1]）。

「距离」只是名字，换成乘法就是倍数关系，换成 %3 就是种类关系，换成异或就是异或关系。下两页分别展开这三种变形。
