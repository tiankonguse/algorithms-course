标记永久化：标记不下发

懒更新标记可以不下发。标记只属于它挂着的那个范围，上下级标记互不影响；查询时从上往下把标记叠加起来，一样算得对。

一、标记属于范围，不属于路径

普通懒更新的标记要下发：父节点有标记，查询/修改到孩子之前得先把标记推下去。标记永久化反着来：标记挂在父范围就不动了，孩子节点的标记是自己的，跟父标记无关。

查询时从头往下走，把沿途每个节点的标记累加进一个 addHistory，走到完全覆盖的节点，答案 = 节点 sum + addHistory × 区间长度。标记叠加是「求和式」的，先后顺序无所谓，所以能这么算。

二、不是所有操作都能永久化

标记永久化的前提是「修改和查询的性质可叠加」：范围加 + 区间求和可以（加多少就是多少，叠加无顺序问题）。范围重置 + 查询最值就不行：父范围的重置和子范围的修改互相覆盖，不下发就说不清谁先谁后。

普通线段树用标记永久化实现，代码比经典懒更新还短：add 函数里，sum 先加上「本节点与操作区间的交集长度 × v」，只有操作完全覆盖本节点时才在 addTag 上累加 v；query 把 addHistory 传下去。

标记永久化的普通线段树（P3372）

#include <iostream>
    #include <cstdio>
    #include <cstring>
    #include <algorithm>
    using namespace std;

    const int MAXN = 100001;
    long long arr[MAXN];
    long long sum[MAXN << 2];
    long long addTag[MAXN << 2];

    void build(int l, int r, int i) {
        if (l == r) {
            sum[i] = arr[l];
        } else {
            int mid = (l + r) / 2;
            build(l, mid, i << 1);
            build(mid + 1, r, i << 1 | 1);
            sum[i] = sum[i << 1] + sum[i << 1 | 1];
        }
        addTag[i] = 0;
    }

    void add(int jobl, int jobr, long long jobv, int l, int r, int i) {
        int a = max(jobl, l), b = min(jobr, r);
        sum[i] += jobv * (b - a + 1);
        if (jobl <= l && r <= jobr) {
            addTag[i] += jobv;
        } else {
            int mid = (l + r) / 2;
            if (jobl <= mid) {
                add(jobl, jobr, jobv, l, mid, i << 1);
            }
            if (jobr > mid) {
                add(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
            }
        }
    }

    long long query(int jobl, int jobr, long long addHistory, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return sum[i] + addHistory * (r - l + 1);
        }
        int mid = (l + r) >> 1;
        long long ans = 0;
        if (jobl <= mid) {
            ans += query(jobl, jobr, addHistory + addTag[i], l, mid, i << 1);
        }
        if (jobr > mid) {
            ans += query(jobl, jobr, addHistory + addTag[i], mid + 1, r, i << 1 | 1);
        }
        return ans;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        int n, m;
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
        }
        build(1, n, 1);
        int op, jobl, jobr;
        long long jobv;
        for (int i = 1; i <= m; i++) {
            cin >> op;
            if (op == 1) {
                cin >> jobl >> jobr >> jobv;
                add(jobl, jobr, jobv, 1, n, 1);
            } else {
                cin >> jobl >> jobr;
                cout << query(jobl, jobr, 0, 1, n, 1) << "\n";
            }
        }
        return 0;
    }

三、可持久化 + 标记永久化

范围修改的可持久化如果用经典懒更新，查询也新建节点，空间 O(n log n + m log n)。换成标记永久化：add 不触发下发，查询只读不建，总空间回到 O(n·4 + n·log n)。

代码上 add 每层新建节点（复制 sum 和 addTag），完全覆盖才累加标记；query 同样带 addHistory，全程不新建。操作语义与经典版完全一致，C/Q/H/B 照旧，版本号同样是覆盖式写入。

可持久化 + 标记永久化（HDU4348）

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 100001;
    const int MAXT = MAXN * 25;
    int n, m, t = 0;
    long long arr[MAXN];
    int root[MAXN];
    int ls[MAXT];
    int rs[MAXT];
    long long sum[MAXT];
    long long addTag[MAXT];
    int cnt = 0;

    int build(int l, int r) {
        int rt = ++cnt;
        addTag[rt] = 0;
        if (l == r) {
            sum[rt] = arr[l];
        } else {
            int mid = (l + r) / 2;
            ls[rt] = build(l, mid);
            rs[rt] = build(mid + 1, r);
            sum[rt] = sum[ls[rt]] + sum[rs[rt]];
        }
        return rt;
    }

    int add(int jobl, int jobr, long long jobv, int l, int r, int i) {
        int rt = ++cnt, a = max(jobl, l), b = min(jobr, r);
        ls[rt] = ls[i];
        rs[rt] = rs[i];
        sum[rt] = sum[i] + jobv * (b - a + 1);
        addTag[rt] = addTag[i];
        if (jobl <= l && r <= jobr) {
            addTag[rt] += jobv;
        } else {
            int mid = (l + r) / 2;
            if (jobl <= mid) {
                ls[rt] = add(jobl, jobr, jobv, l, mid, ls[rt]);
            }
            if (jobr > mid) {
                rs[rt] = add(jobl, jobr, jobv, mid + 1, r, rs[rt]);
            }
        }
        return rt;
    }

    long long query(int jobl, int jobr, long long addHistory, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return sum[i] + addHistory * (r - l + 1);
        }
        int mid = (l + r) / 2;
        long long ans = 0;
        if (jobl <= mid) {
            ans += query(jobl, jobr, addHistory + addTag[i], l, mid, ls[i]);
        }
        if (jobr > mid) {
            ans += query(jobl, jobr, addHistory + addTag[i], mid + 1, r, rs[i]);
        }
        return ans;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
        }
        root[0] = build(1, n);
        char op;
        int x, y;
        long long z;
        for (int i = 1; i <= m; i++) {
            cin >> op;
            if (op == 'C') {
                cin >> x >> y >> z;
                root[t + 1] = add(x, y, z, 1, n, root[t]);
                t++;
            } else if (op == 'Q') {
                cin >> x >> y;
                cout << query(x, y, 0, 1, n, root[t]) << "\n";
            } else if (op == 'H') {
                cin >> x >> y >> z;
                cout << query(x, y, 0, 1, n, root[z]) << "\n";
            } else {
                cin >> x;
                t = x;
            }
        }
        return 0;
    }

五、总结

标记永久化用「标记叠加」换掉「标记下发」，查询不建节点，空间省一个 O(m log n)。代价是只适用于可叠加的修改查询对，范围重置配最值这类题老老实实走经典方式。
