块状数组经典题：区间众数

上一节的有序副本能对付「≥ v 计数」，因为有序性对这类查询直接有用。这一节的查询换成区间众数：它连两块的答案都合并不出来，有序副本彻底失效。分块的对策是换一种预处理，把「块与块之间」的答案直接存成表，查询时让散块里的数字来竞争。

零、背景

先看众数为什么难。区间和可以两半相加，区间最值可以两半取大，众数不行：两半各自的众数，拼起来可能谁都当不上。所以线段树这类靠信息合并的结构在这里无从下手，节点上存什么都答不出一个临时区间的众数。

分块的出路是把整块部分的答案提前算好。任意查询 [l,r] 被拆成中间的连续整块加两端散块：整块部分的众数查表 O(1) 拿到，散块最多 2√n 个数，让它们逐个竞争。竞争的胜负要用到「某个值在某段整块里出现了几次」，这也要提前存。

这一节用三道题把这个套路走完整：蒲公英问众数的值；一道空间卡得死的题只问众数出现了几次；作诗把问题换成出现正偶数次的数有几个。三道题全部强制在线，预处理表正好不吃在线的影响。

一、题目：蒲公英

给定长度为 n 的数组，m 次查询，每次打印 arr[l..r] 上的众数，多个众数时打印值最小的。n ≤ 4×10⁴，m ≤ 5×10⁴，值在 1 到 10⁹，强制在线。测试链接：洛谷 P4168。

预处理两张表。第一张 freq[i][j]：前 i 块中值 j 出现的次数。按块一层层累加就能得到，它是「值 j 在块 l 到块 r 之间出现了几次」的差分来源：freq[r][j] − freq[l−1][j]。值先离散化，值域收敛到不同值的个数 s。

第二张 mode[i][j]：第 i 块到第 j 块的众数。它按 j 递推：候选一开始是 mode[i][j−1]，再把第 j 块里的每个数拿来竞争，各自的次数用 freq 一减就知道。平次数时取值小的。

查询 [l,r] 时，中间整块的众数直接读 mode[bi[l]+1][bi[r]−1]，次数用 freq 差补上；然后扫两端散块的每个数，它的总次数等于「整块里的次数加散块里已扫到的次数」，逐个与当前众数竞争。同块查询就纯暴力。每个数至多被扫两遍，单次查询 O(√n)。

强制在线的解码照题目规则来：读入 a、b 后换成 (a + lastAns − 1) % n + 1，再取小的一个当 l。散块竞争时用 numCnt 记散块内每个值已扫到的次数，用完清零，数组就总能复用。

下面把一次跨块查询走一遍。数组分成 4 块、块长 3，查询 [2,11]：中间的块 2、块 3 查表拿众数，两端散块的数字逐个竞争，注意平次数时值小的获胜。

数组分成 4 块，块长 3。点「下一步」开始查询 [2,11] 的众数。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 40001;
    const int MAXB = 201;
    int n, m, s;
    int arr[MAXN];
    int sortv[MAXN];

    int blen, bnum;
    int bi[MAXN];
    int bl[MAXB];
    int br[MAXB];

    int freq[MAXB][MAXN];
    int mode[MAXB][MAXB];
    int numCnt[MAXN];

    int lower(int num) {
        int l = 1, r = s, m, ans = 0;
        while (l <= r) {
            m = (l + r) >> 1;
            if (sortv[m] >= num) {
                ans = m;
                r = m - 1;
            } else {
                l = m + 1;
            }
        }
        return ans;
    }

    int getCnt(int l, int r, int v) {
        return freq[r][v] - freq[l - 1][v];
    }

    void prepare() {
        blen = (int)sqrt(n);
        bnum = (n + blen - 1) / blen;
        for (int i = 1; i <= n; i++) {
            bi[i] = (i - 1) / blen + 1;
        }
        for (int i = 1; i <= bnum; i++) {
            bl[i] = (i - 1) * blen + 1;
            br[i] = min(i * blen, n);
        }
        for (int i = 1; i <= n; i++) {
            sortv[i] = arr[i];
        }
        sort(sortv + 1, sortv + n + 1);
        s = 1;
        for (int i = 2; i <= n; i++) {
            if (sortv[s] != sortv[i]) {
                sortv[++s] = sortv[i];
            }
        }
        for (int i = 1; i <= n; i++) {
            arr[i] = lower(arr[i]);
        }
        for (int i = 1; i <= bnum; i++) {
            for (int j = bl[i]; j <= br[i]; j++) {
                freq[i][arr[j]]++;
            }
            for (int j = 1; j <= s; j++) {
                freq[i][j] += freq[i - 1][j];
            }
        }
        for (int i = 1; i <= bnum; i++) {
            for (int j = i; j <= bnum; j++) {
                int most = mode[i][j - 1];
                int mostCnt = getCnt(i, j, most);
                for (int k = bl[j]; k <= br[j]; k++) {
                    int cur = arr[k];
                    int curCnt = getCnt(i, j, cur);
                    if (curCnt > mostCnt || (curCnt == mostCnt && cur < most)) {
                        most = cur;
                        mostCnt = curCnt;
                    }
                }
                mode[i][j] = most;
            }
        }
    }

    int query(int l, int r) {
        int most = 0;
        if (bi[l] == bi[r]) {
            for (int i = l; i <= r; i++) {
                numCnt[arr[i]]++;
            }
            for (int i = l; i <= r; i++) {
                if (numCnt[arr[i]] > numCnt[most] || (numCnt[arr[i]] == numCnt[most] && arr[i] < most)) {
                    most = arr[i];
                }
            }
            for (int i = l; i <= r; i++) {
                numCnt[arr[i]] = 0;
            }
        } else {
            for (int i = l; i <= br[bi[l]]; i++) {
                numCnt[arr[i]]++;
            }
            for (int i = bl[bi[r]]; i <= r; i++) {
                numCnt[arr[i]]++;
            }
            most = mode[bi[l] + 1][bi[r] - 1];
            int mostCnt = getCnt(bi[l] + 1, bi[r] - 1, most) + numCnt[most];
            for (int i = l; i <= br[bi[l]]; i++) {
                int cur = arr[i];
                int curCnt = getCnt(bi[l] + 1, bi[r] - 1, cur) + numCnt[cur];
                if (curCnt > mostCnt || (curCnt == mostCnt && cur < most)) {
                    most = cur;
                    mostCnt = curCnt;
                }
            }
            for (int i = bl[bi[r]]; i <= r; i++) {
                int cur = arr[i];
                int curCnt = getCnt(bi[l] + 1, bi[r] - 1, cur) + numCnt[cur];
                if (curCnt > mostCnt || (curCnt == mostCnt && cur < most)) {
                    most = cur;
                    mostCnt = curCnt;
                }
            }
            for (int i = l; i <= br[bi[l]]; i++) {
                numCnt[arr[i]] = 0;
            }
            for (int i = bl[bi[r]]; i <= r; i++) {
                numCnt[arr[i]] = 0;
            }
        }
        return sortv[most];
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
        }
        prepare();
        int lastAns = 0, a, b, l, r;
        for (int i = 1; i <= m; i++) {
            cin >> a >> b;
            a = (a + lastAns - 1) % n + 1;
            b = (b + lastAns - 1) % n + 1;
            l = min(a, b);
            r = max(a, b);
            lastAns = query(l, r);
            cout << lastAns << '\n';
        }
        return 0;
    }

二、题目：空间少求众数的次数

给定长度为 n 的数组，m 次查询，每次打印 arr[l..r] 上众数到底出现了几次。所有数据在 1 到 5×10⁵，内存空间只有 64MB，强制在线。测试链接：洛谷 P5048。

只问次数不问值，看起来是蒲公英的降级，但 64MB 卡死了 freq 表：√n × n 的 int 到了 5×10⁵ 的规模要上 GB。所以第二张表保留，第一张表必须换掉。

这里有个细节：强制在线的意思是输入里的 l、r 要异或上一条答案才是真区间，代码里 l ^= lastAns 那两行不能省。另一处是预处理用的 numCnt 按值开桶，清空时只清 1 到 n，所以数组里的数值要落在 1 到 n 之间，值域更大就先离散化一遍。

替身是 sortList：把所有（值，下标）对按值排序、同值按下标排序。同一种值的下标在 sortList 里连成一段且严格递增。listIdx[i] 记录 arr[i] 在 sortList 里的位置。想知道某个值在 [l,r] 里出现了几次，不需要频次表，沿着 sortList 里这一段跳格子就行。

查询的骨架不变：ans 先取整块的 modeCnt（结构与 mode 表一样，只是存次数）。然后扫散块，以左散块的元素 arr[i] 为例，设它在 sortList 里的位置是 idx：如果 sortList[idx + ans] 与它同值、且下标不超过 r，说明这个值在 [l,r] 里至少出现 ans+1 次，ans 加一，继续判。右散块对称，往 idx − ans 方向跳，判下标不小于 l。

为什么每次只加一还要用 while 循环：一个位置上的值可能让答案不止增大一次，跳成功就继续跳，直到跳到别的值或越出 [l,r]。答案的总量不超过 2√n，每跳一步是 O(1)，单次查询还是 O(√n)。ans 只增不减，这是整条链路成立的前提。

这里有个细节：预处理里词频数组 numCnt 的清零范围是下标 1 到 n，这份代码按「值就是下标、值域不超过 n」在使用它。本题所有数据同阶，实际使用时先离散化一遍会更稳。

#include <bits/stdc++.h>

    using namespace std;

    struct Node {
        int v, i;
    };

    bool NodeCmp(Node a, Node b) {
        if (a.v != b.v) {
            return a.v < b.v;
        }
        return a.i < b.i;
    }

    const int MAXN = 500001;
    const int MAXB = 801;
    int n, m;
    int arr[MAXN];

    int blen, bnum;
    int bi[MAXN];
    int bl[MAXB];
    int br[MAXB];

    Node sortList[MAXN];
    int listIdx[MAXN];

    int modeCnt[MAXB][MAXB];
    int numCnt[MAXN];

    void prepare() {
        blen = (int)sqrt(n);
        bnum = (n + blen - 1) / blen;
        for (int i = 1; i <= n; i++) {
            bi[i] = (i - 1) / blen + 1;
        }
        for (int i = 1; i <= bnum; i++) {
            bl[i] = (i - 1) * blen + 1;
            br[i] = min(i * blen, n);
        }
        for (int i = 1; i <= n; i++) {
            sortList[i].v = arr[i];
            sortList[i].i = i;
        }
        sort(sortList + 1, sortList + n + 1, NodeCmp);
        for (int i = 1; i <= n; i++) {
            listIdx[sortList[i].i] = i;
        }
        for (int i = 1; i <= bnum; i++) {
            for (int j = i; j <= bnum; j++) {
                int cnt = modeCnt[i][j - 1];
                for (int k = bl[j]; k <= br[j]; k++) {
                    cnt = max(cnt, ++numCnt[arr[k]]);
                }
                modeCnt[i][j] = cnt;
            }
            for (int j = 1; j <= n; j++) {
                numCnt[j] = 0;
            }
        }
    }

    int query(int l, int r) {
        int ans = 0;
        if (bi[l] == bi[r]) {
            for (int i = l; i <= r; i++) {
                ans = max(ans, ++numCnt[arr[i]]);
            }
            for (int i = l; i <= r; i++) {
                numCnt[arr[i]] = 0;
            }
        } else {
            ans = modeCnt[bi[l] + 1][bi[r] - 1];
            for (int i = l, idx; i <= br[bi[l]]; i++) {
                idx = listIdx[i];
                while (idx + ans <= n && sortList[idx + ans].v == arr[i] && sortList[idx + ans].i <= r) {
                    ans++;
                }
            }
            for (int i = bl[bi[r]], idx; i <= r; i++) {
                idx = listIdx[i];
                while (idx - ans >= 1 && sortList[idx - ans].v == arr[i] && sortList[idx - ans].i >= l) {
                    ans++;
                }
            }
        }
        return ans;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
        }
        prepare();
        for (int i = 1, l, r, lastAns = 0; i <= m; i++) {
            cin >> l >> r;
            l ^= lastAns;
            r ^= lastAns;
            lastAns = query(l, r);
            cout << lastAns << '\n';
        }
        return 0;
    }

三、题目：作诗

给定长度为 n 的数组，值域为 c，m 次查询，每次打印 arr[l..r] 上有多少个数出现正偶数次。所有数据在 1 到 10⁵，强制在线。测试链接：洛谷 P4135。

统计的对象从「最多次数」换成了「偶数次的个数」，但结构原样照搬：freq 表回归（这题内存 512MB，用得起），第二张表换成 even[i][j]：第 i 块到第 j 块里出现正偶数次的数的个数。

递推的关键是一个小函数 delta(pre)：某个值已经出现了 pre 次，再出现一次时，「正偶数次」的计数怎么变。pre 为 0 时不变（1 次还不是偶数次）；pre 为奇数时加一（1 到 2、3 到 4，挤进偶数俱乐部）；pre 为非零偶数时减一（2 到 3，掉了出去）。

查询照抄竞争的骨架：ans 先取 even[bi[l]+1][bi[r]−1]，再扫两端散块，每个数的总次数是「freq 差加散块里已扫到的次数」，套上 delta 累加。同块查询纯暴力。值域是 c 而不是 n，freq 表清零、遍历都要按 c 来，这也是它和蒲公英唯一的差别。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 100001;
    const int MAXB = 401;
    int n, c, m;
    int arr[MAXN];

    int blen, bnum;
    int bi[MAXN];
    int bl[MAXB];
    int br[MAXB];

    int freq[MAXB][MAXN];
    int even[MAXB][MAXB];
    int numCnt[MAXN];

    int getCnt(int l, int r, int v) {
        return freq[r][v] - freq[l - 1][v];
    }

    int delta(int pre) {
        if (pre == 0) {
            return 0;
        }
        if ((pre & 1) == 0) {
            return -1;
        }
        return 1;
    }

    void prepare() {
        blen = (int)sqrt(n);
        bnum = (n + blen - 1) / blen;
        for (int i = 1; i <= n; i++) {
            bi[i] = (i - 1) / blen + 1;
        }
        for (int i = 1; i <= bnum; i++) {
            bl[i] = (i - 1) * blen + 1;
            br[i] = min(i * blen, n);
        }
        for (int i = 1; i <= bnum; i++) {
            for (int j = bl[i]; j <= br[i]; j++) {
                freq[i][arr[j]]++;
            }
            for (int j = 1; j <= c; j++) {
                freq[i][j] += freq[i - 1][j];
            }
        }
        for (int i = 1; i <= bnum; i++) {
            for (int j = i; j <= bnum; j++) {
                even[i][j] = even[i][j - 1];
                for (int k = bl[j]; k <= br[j]; k++) {
                    even[i][j] += delta(numCnt[arr[k]]);
                    numCnt[arr[k]]++;
                }
            }
            for (int j = 1; j <= c; j++) {
                numCnt[j] = 0;
            }
        }
    }

    int query(int l, int r) {
        int ans = 0;
        if (bi[l] == bi[r]) {
            for (int i = l; i <= r; i++) {
                ans += delta(numCnt[arr[i]]);
                numCnt[arr[i]]++;
            }
            for (int i = l; i <= r; i++) {
                numCnt[arr[i]] = 0;
            }
        } else {
            ans = even[bi[l] + 1][bi[r] - 1];
            for (int i = l; i <= br[bi[l]]; i++) {
                ans += delta(getCnt(bi[l] + 1, bi[r] - 1, arr[i]) + numCnt[arr[i]]);
                numCnt[arr[i]]++;
            }
            for (int i = bl[bi[r]]; i <= r; i++) {
                ans += delta(getCnt(bi[l] + 1, bi[r] - 1, arr[i]) + numCnt[arr[i]]);
                numCnt[arr[i]]++;
            }
            for (int i = l; i <= br[bi[l]]; i++) {
                numCnt[arr[i]] = 0;
            }
            for (int i = bl[bi[r]]; i <= r; i++) {
                numCnt[arr[i]] = 0;
            }
        }
        return ans;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> c >> m;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
        }
        prepare();
        for (int i = 1, a, b, l, r, lastAns = 0; i <= m; i++) {
            cin >> a >> b;
            a = (a + lastAns) % n + 1;
            b = (b + lastAns) % n + 1;
            l = min(a, b);
            r = max(a, b);
            lastAns = query(l, r);
            cout << lastAns << '\n';
        }
        return 0;
    }

四、复杂度分析

三道题的骨架一致：一张关于值的统计表加一张块与块之间的答案表，预处理 O(n√n)，单次查询 O(√n)。差别全在第一张表怎么来、多大。

题目预处理时间空间单次查询
蒲公英：freq 前缀 + mode 表O(n·√n)O(n·√n)，n = 4×10⁴ 时 freq 约 32MBO(√n)
空间少求众数次数：sortList + modeCntO(n·√n)O(n)，二维频次表被 sortList 顶替O(√n)
作诗：freq 前缀 + even 表O(n·√n)O(c·√n)，c 为值域，本题内存给到 512MBO(√n)

mode 表和 modeCnt 表的递推都是三层循环：块数 × 块数 × 块长 = √n × √n × √n = n√n。查询侧每次扫 2√n 个散块元素，蒲公英和作诗每个 O(1)，众数次数那题每跳一步 O(1) 但总增量封顶 2√n，量级相同。

五、最后

这三道题合起来看是一个完整的方法论：整块的信息合并不出来时，就把块与块之间的答案提前算成表；散块是答案唯一的变数，让它逐个竞争。freq 表、sortList、delta 分别解决「次数从哪来」的三个变体。

到这里块状数组告一段落。下一节把块从「静态的数组」变成「链起来的块」，插入删除一段内容时，后面的元素再也不用整体搬家。

《完》
