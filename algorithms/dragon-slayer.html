屠龙勇士：选剑、列方程、修正答案

每只龙要打固定 ans 下才死，ans 对所有龙是同一个数。题目最后变成「n 个形如 ans·atk ≡ hp (mod recovery) 的同余条件求最小解」，但真正的坑藏在最后：同余方程的解域是整个整数，而恢复次数不能是负数，答案还得再往上凑。

零、背景

n 只龙按顺序讨伐，每只有初始血量 hp、恢复力 recovery（被打到 ≤ 0 后每回合回 recovery 血）。手里有一批剑，每把有攻击力。对每只龙只能用固定的一把剑，规则是：攻击力不高于当前龙血量的剑里选最大的一把，没有就选攻击力最小的一把。龙死后用的剑消失，奖励一把新剑。

策略是全程只打 ans 下：每只龙挨 ans 剑后血量 ≤ 0，然后在恢复过程中恰好某刻归零就被杀死。求最小的 ans，做不到输出 -1。这个题把「有序表选剑」「同余方程组」「答案下界修正」三件事串在一起。

一、选剑与打击下界

剑的集合用有序表维护：攻击力不高于 hp 的最大一把，就是 upper_bound(hp) 的前一个元素；没有这样的剑，就取整个集合的最小值。选完删掉这把剑、插入奖励的剑。同时记下每只龙「砍到 ≤ 0」需要的次数 ceil(hp / atk)，全程取最大为 max。任何可行的 ans 都必须 ≥ max，这是后面修正的依据。

选完剑后把血量取模：hp = hp % recovery。这是为下一步铺路。血量为正时「恢复中恰好到 0」等价于「攻击 ans 下后的剩余血量能由若干次恢复恰好补回」，而这个条件只跟 hp mod recovery 有关。

二、转成同余方程组

对第 i 只龙，攻击 ans 下后血量是 hp − ans·atk，它要能通过非负次恢复回到 0：hp − ans·atk ≡ 0 (mod recovery)，也就是 ans·atk ≡ hp (mod recovery)。前面把 hp 取过模，这里直接用余数。

n 个这样的同余式，模数 recovery 之间没有任何互质保证，就用扩展中国剩余定理那套迭代合并。唯一区别是左边系数不再是 1：合并第 i 个方程时解的是 atk·lcm·x + recovery·y = hp − atk·tail，a = (atk·lcm) mod recovery，c = (hp − atk·tail) mod recovery，流程与 EXCRT 完全一致。

三、答案修正：恢复次数必须非负

合并完得到 ans = k·lcm + tail。这里的 k 每取一个整数都满足同余条件，但同余条件默认「恢复次数可以是任意整数」，实际它必须 ≥ 0。恢复次数 ≥ 0 等价于「ans 下就把血打到 ≤ 0」，也就是 ans ≥ max（max 是第一段算出的最大打击需求）。

所以：tail ≥ max 时答案就是 tail；tail < max 时要在 ans = tail + k·lcm 里找第一个 ≥ max 的，k 向上取整：k = ceil((max − tail) / lcm)，ans = k·lcm + tail。这步不做，答案会偏小。

四、实现

选剑用 multiset 模拟有序表，upper_bound 的前驱就是「不高于 hp 的最大剑」，取 begin 就是最小剑。合并部分和扩展中国剩余定理同一套代码，只改 a 和 c 的算法。乘法全程龟速乘。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;
    multiset<ll> swords;
    ll attack[100005];
    ll allocate(int n, int m, ll hp[], ll recovery[], ll reward[], ll init[]) {
      swords.clear();
      for (int i = 1; i <= m; i++) swords.insert(init[i]);
      ll maxv = 0;
      for (int i = 1; i <= n; i++) {
        auto it = swords.upper_bound(hp[i]);
        ll sword;
        if (it == swords.begin()) sword = *swords.begin();
        else {
          --it;
          sword = *it;
        }
        attack[i] = sword;
        swords.erase(swords.find(sword));
        swords.insert(reward[i]);
        maxv = max(maxv, (hp[i] + attack[i] - 1) / attack[i]);
        hp[i] %= recovery[i];
      }
      return maxv;
    }
#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;
    ll d, x, y, px, py;
    void exgcd(ll a, ll b) {
      if (b == 0) {
        d = a;
        x = 1;
        y = 0;
      } else {
        exgcd(b, a % b);
        px = x;
        py = y;
        x = py;
        y = px - py * (a / b);
      }
    }
    ll multiply(ll a, ll b, ll mod) {
      a = (a % mod + mod) % mod;
      b = (b % mod + mod) % mod;
      ll ans = 0;
      while (b != 0) {
        if (b & 1) ans = (ans + a) % mod;
        a = (a + a) % mod;
        b >>= 1;
      }
      return ans;
    }
    ll compute(int n, int m, ll hp[], ll recovery[], ll reward[], ll init[]) {
      ll maxv = allocate(n, m, hp, recovery, reward, init);
      ll tail = 0, lcm = 1, tmp, a, b, c, x0;
      for (int i = 1; i <= n; i++) {
        a = multiply(attack[i], lcm, recovery[i]);
        b = recovery[i];
        c = ((hp[i] - attack[i] * tail) % b + b) % b;
        exgcd(a, b);
        if (c % d != 0) return -1;
        x0 = multiply(x, c / d, b / d);
        tmp = lcm * (b / d);
        tail = (tail + multiply(x0, lcm, tmp)) % tmp;
        lcm = tmp;
      }
      ll ans;
      if (tail >= maxv) ans = tail;
      else ans = (maxv - tail + lcm - 1) / lcm * lcm + tail;
      return ans;
    }
    int main() {
      int cases;
      cin >> cases;
      while (cases--) {
        int n, m;
        cin >> n >> m;

        for (int i = 1; i <= n; i++) cin >> recovery[i];
        for (int i = 1; i <= n; i++) cin >> reward[i];
        for (int i = 1; i <= m; i++) cin >> init[i];
        cout << compute(n, m, hp, recovery, reward, init) << endl;
      }

总结

屠龙勇士三步走：有序表选剑并算打击下界 max；把每只龙的条件整理成 ans·atk ≡ hp (mod recovery) 并迭代合并；最后按「恢复次数非负」修正答案，tail 不足 max 时向上凑到第一个 ≥ max 的同余解。无解发生在合并过程中 c 不被 gcd 整除，返回 -1。这个题的教训是：同余方程只管同余，实际问题里变量还有取值范围约束，公式解出来后还要回头核一遍。
