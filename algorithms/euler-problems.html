欧拉路径的难题分析

这一讲最后三道题：垃圾车把欧拉回路拆成简单环，邮递员把边压缩再解压，丁香之路用配平和最小生成树凑出欧拉回路。模版没变，变的是把实际问题翻译成图的手段，一道比一道绕。

零、背景

前两页的题都是一次判定加一次生成，这一页三道题要在两者之间加一层翻译。垃圾车把「车怎么走」翻译成简单环，邮递员把「序列要连续」翻译成边与边的连接关系，丁香之路把「终点任意」翻译成配平与连通性。

三道题都只用第一页的模版，没有新算法。难点全在建模：想清楚合法解长什么样，剩下的 Hierholzer 就那么几行。

一、题目9：垃圾车

n 个点、m 条无向边，不保证连通。每条边有属性 u v s t：s 是初始状态，t 是最终状态，状态只有 0 和 1，车每通过一次，状态翻转。一辆车指定起点出发回到起点，沿途其他点不能重复经过。所有边都要达到最终状态，需要若干辆车。有方案就打印车数，再对每辆车打印边数和依次到达的点；无解打印 NIE。n 到 10⁵，m 到 10⁶。测试链接：Luogu P3520。

先想一辆车是什么。从某点出发回到该点、中间点不重复，就是一个简单环。车经过一条边一次，状态翻一次；要达到 t，等价于走过奇数次。走过三次和走过一次效果一样但更浪费，所以最优就是恰好一次：s 等于 t 的边直接扔掉，剩下的边每个端点的度数都必须是偶数，否则无解。

度数全偶之后，每个连通区存在欧拉回路。但欧拉回路不一定简单：8 字形的回路会两次经过中间点。题目要的是简单环，所以要把回路拆开。

拆法藏在递归栈里。欧拉路径的 path 数组本来就是一条回路；走的过程中如果遇到一个已经在栈上的点 x，说明栈里从 x 到栈顶的这一段自己就是一个简单环，把它整段弹出来记成一辆车。实现是 popCircle(u)：u 已在栈上时，把栈顶到 u 之间的点依次弹出，首尾补上 u。

完整代码。注意输出按弹栈的先后，环的顺序不唯一，题目接受任何方案。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 100001;
    const int MAXM = 2000001;
    int n, m;

    int head[MAXN];
    int nxt[MAXM];
    int to[MAXM];
    int eid[MAXM];
    int cntg;

    bool visNode[MAXN];
    bool visEdge[MAXM];

    int deg[MAXN];
    int cur[MAXN];

    bool inpath[MAXN];
    int path[MAXM];
    int cntp;

    int ansArr[MAXM];
    int ansl[MAXM];
    int ansr[MAXM];
    int idx;
    int cnta;

    void addEdge(int u, int v, int id) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        eid[cntg] = id;
        head[u] = cntg;
    }

    void popCircle(int u) {
        cnta++;
        ansArr[++idx] = u;
        ansl[cnta] = idx;
        for (int pop = path[cntp--]; pop != u; pop = path[cntp--]) {
            inpath[pop] = false;
            ansArr[++idx] = pop;
        }
        inpath[u] = false;
        ansArr[++idx] = u;
        ansr[cnta] = idx;
    }

    void euler(int u) {
        visNode[u] = true;
        for (int e = cur[u]; e > 0; e = cur[u]) {
            cur[u] = nxt[e];
            if (!visEdge[eid[e]]) {
                visEdge[eid[e]] = true;
                euler(to[e]);
            }
        }
        if (inpath[u]) {
            popCircle(u);
        }
        inpath[u] = true;
        path[++cntp] = u;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1, u, v, s, t; i <= m; i++) {
            cin >> u >> v >> s >> t;
            if (s != t) {
                deg[u]++;
                deg[v]++;
                addEdge(u, v, i);
                addEdge(v, u, i);
            }
        }
        for (int i = 1; i <= n; i++) {
            cur[i] = head[i];
        }
        bool check = true;
        for (int i = 1; i <= n; i++) {
            if ((deg[i] & 1) == 1) {
                check = false;
                break;
            }
        }
        if (!check) {
            cout << "NIE\n";
        } else {
            for (int i = 1; i <= n; i++) {
                if (!visNode[i]) {
                    euler(i);
                }
            }
            cout << cnta << "\n";
            for (int i = 1; i <= cnta; i++) {
                cout << (ansr[i] - ansl[i]) << " ";
                for (int j = ansl[i]; j <= ansr[i]; j++) {
                    cout << ansArr[j] << " ";
                }
                cout << "\n";
            }
        }
        return 0;
    }

二、题目10：邮递员

n 个点、m 条有向边，无重边无自环。邮递员从 1 号点出发回到 1 号点，每条边恰好走一次，走过的点形成路径。再给 t 个序列，每个序列先给长度 k，再给 k 个点的编号，要求每个序列都是路径的连续子段。不存在打印 NIE，存在打印 TAK 和依次走过的节点。n 到 5×10⁴，m 到 2×10⁵，序列节点总量到 2×10⁶。测试链接：Luogu P3443。

序列是连续子段，翻译过来：序列里相邻两个点 (a, b) 必须是图上一条边，而且这条边在路径里必须紧跟在前一条序列边的后面。于是序列的职责变成了把边串成链：边 i 的后面固定接边 j，记 etoe[i] = j，边 j 从此不能再当别的边的后继，也标记为不再是链头。

这里有个细节：无重边只是题目保证没有完全相同的两条边，但序列经过同一个点两次时，同一对 (a, b) 可能要接两条不同的边。用 map 把 (a, b) 映射到边的编号，只能记一条，序列指定哪条就是哪条；如果两条不同的链想接在同一个位置，直接 NIE。

还有一个隐藏的不合法：链不能成环。序列自己不是环，所以 etoe 沿着走不能绕回来，用 vis 标记在找链尾时检测。链成环等价于邮递员在这几条边里原地打转，出不去也回不来 1。

链建好后就压缩：链头到链尾的整段当成一条边，起点是链头的头，终点是链尾的尾，同时记下链头编号方便解压。压缩图上跑欧拉回路，度数不平衡就 NIE。最后沿 etoe 把每条压缩边展开，得到原始路径。

完整代码。linkEdge 建链，compress 压缩并判定度数，euler 走回路，decompress 展开。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 50005;
    const int MAXM = 200005;
    const int MAXC = 2000005;
    int n, m, t;
    int a[MAXM];
    int b[MAXM];
    int seq[MAXC];

    map<pair<int, int>, int> pairEdge;

    int headg[MAXN];
    int nextg[MAXM];
    int tog[MAXM];
    int chainHead[MAXM];
    int cntg;
    int edgeCnt;

    int cur[MAXN];
    int inDeg[MAXN];
    int outDeg[MAXN];

    bool isHead[MAXM];
    int etoe[MAXM];
    bool vis[MAXM];

    int path[MAXM];
    int cntp;

    int ans[MAXM];
    int cnta;

    void addEdge(int x, int y, int h) {
        nextg[++cntg] = headg[x];
        tog[cntg] = y;
        chainHead[cntg] = h;
        headg[x] = cntg;
    }

    bool linkEdge() {
        for (int i = 1; i <= m; i++) {
            pairEdge[{a[i], b[i]}] = i;
            isHead[i] = true;
        }
        int siz = seq[1], l = 2, r = l + siz - 1;
        int a, b, ledge, redge;
        while (siz > 0) {
            ledge = 0;
            for (int i = l; i < r; i++) {
                a = seq[i];
                b = seq[i + 1];
                auto it = pairEdge.find({a, b});
                if (it == pairEdge.end()) {
                    return false;
                }
                redge = it->second;
                if (ledge != 0) {
                    if (etoe[ledge] != 0 && etoe[ledge] != redge) {
                        return false;
                    }
                    etoe[ledge] = redge;
                    isHead[redge] = false;
                }
                ledge = redge;
            }
            siz = seq[r + 1];
            l = r + 2;
            r = l + siz - 1;
        }
        return true;
    }

    int getLinkEnd(int edge) {
        while (etoe[edge] != 0) {
            if (vis[edge] == true) {
                return -1;
            }
            vis[edge] = true;
            edge = etoe[edge];
        }
        return edge;
    }

    bool compress() {
        for (int i = 1; i <= m; i++) {
            if (isHead[i]) {
                int x = a[i];
                int y = b[i];
                if (etoe[i] != 0) {
                    int end = getLinkEnd(i);
                    if (end == -1) {
                        return false;
                    }
                    y = b[end];
                }
                outDeg[x]++;
                inDeg[y]++;
                edgeCnt++;
                addEdge(x, y, i);
            }
        }
        for (int i = 1; i <= n; i++) {
            if (inDeg[i] != outDeg[i]) {
                return false;
            }
        }
        for (int i = 1; i <= n; i++) {
            cur[i] = headg[i];
        }
        return true;
    }

    void euler(int u, int h) {
        for (int e = cur[u]; e > 0; e = cur[u]) {
            cur[u] = nextg[e];
            euler(tog[e], chainHead[e]);
        }
        path[++cntp] = h;
    }

    void decompress() {
        ans[++cnta] = 1;
        for (int i = cntp - 1; i >= 1; i--) {
            int e = path[i];
            while (e > 0) {
                ans[++cnta] = b[e];
                e = etoe[e];
            }
        }
    }

    bool compute() {
        if (!linkEdge()) {
            return false;
        }
        if (!compress()) {
            return false;
        }
        euler(1, -1);
        if (cntp != edgeCnt + 1) {
            return false;
        }
        decompress();
        return cnta == m + 1;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1; i <= m; i++) {
            cin >> a[i] >> b[i];
        }
        cin >> t;
        for (int i = 1, siz, idx = 0; i <= t; i++) {
            cin >> siz;
            seq[++idx] = siz;
            for (int j = 1; j <= siz; j++) {
                cin >> seq[++idx];
            }
        }
        bool check = compute();
        if (!check) {
            cout << "NIE\n";
        } else {
            cout << "TAK\n";
            for (int i = 1; i <= cnta; i++) {
                cout << ans[i] << "\n";
            }
        }
        return 0;
    }

三、题目11：丁香之路

n 个点构成完全图，点 i 和点 j 的距离是两者编号差的绝对值。给定 m 个点对表示被标记的无向边，以及起点 s。不管终点是几，每条被标记的边都要至少走一次。对每个终点 i，打印从 s 出发、走过所有标记边、最后停在 i 的最小路程。n 到 2500。测试链接：Luogu P6628。

先把「开环」变成「闭环」。想象终点 i 向起点 s 连一条代价为 0 的无向边：这条边也必须走，于是每条要走的边都变成「至少一次」，整个问题变成找一个边集加粗后的欧拉回路，代价最小。

标记边的端点度数各加一，s 和 i 的度数也各加一（0 代价边）。度数为奇的点必须配平：两个奇度点之间补一条虚边。距离是数轴上的绝对值，所以配平最优策略是把奇度点排序后相邻两两配对，代价就是相邻差之和。

配平之后还要连通，不连通就没有欧拉回路。这里有一个精巧的构造：一条从 u 到 v 的虚边 |u-v|，可以拆成 u 到 u+1、u+1 到 u+2、一直到 v-1 到 v 的一串碎边，总代价不变，但连通性瞬间变大，途中每个点都被串起来了。这些碎边先粗略连一气，再用最小生成树补齐剩下的缺口，树上的边都要走两次。

最终答案 = 标记边代价之和 + 相邻配平的代价 + 最小生成树代价的两倍。对每个终点 i 都独立算一遍，排序和 kruskal 都是 O(n log n)，整体 O(n² log n)，n 等于 2500 时正好能过。

完整代码。compute(s, i) 对每个终点独立计算，prepare 把标记边和并查集预处理一次。

#include <bits/stdc++.h>

    using namespace std;

    using ll = long long;

    struct Edge {
        int u, v, w;
    };

    bool EdgeCmp(Edge e1, Edge e2) {
        if (e1.w != e2.w) {
            return e1.w < e2.w;
        }
        if (e1.u != e2.u) {
            return e1.u < e2.u;
        }
        return e1.v < e2.v;
    }

    const int MAXN = 3001;
    const int MAXM = 3000001;
    int n, m, s;
    int a[MAXM];
    int b[MAXM];

    bool fix[MAXN];
    set<int> nodeSet;
    int nodeArr[MAXN];
    int cnt;

    int fa1[MAXN];
    int deg1[MAXN];
    int fa2[MAXN];
    int deg2[MAXN];

    Edge edgeArr[MAXN];

    ll sum;

    int dist(int x, int y) {
        return abs(x - y);
    }

    int find(int fa[], int x) {
        if (x != fa[x]) {
            fa[x] = find(fa, fa[x]);
        }
        return fa[x];
    }

    void Union(int fa[], int x, int y) {
        int fx = find(fa, x);
        int fy = find(fa, y);
        if (fx != fy) {
            fa[fx] = fy;
        }
    }

    ll kruskal() {
        ll cost = 0;
        int edgeCnt = cnt - 1;
        sort(edgeArr + 1, edgeArr + edgeCnt + 1, EdgeCmp);
        for (int i = 1; i <= edgeCnt; i++) {
            int u = edgeArr[i].u;
            int v = edgeArr[i].v;
            int w = edgeArr[i].w;
            if (find(fa2, u) != find(fa2, v)) {
                cost += 1LL * w * 2;
                Union(fa2, u, v);
            }
        }
        return cost;
    }

    ll compute(int start, int end) {
        cnt = 0;
        for (int x : nodeSet) {
            nodeArr[++cnt] = x;
        }
        for (int i = 1; i <= n; i++) {
            fa2[i] = find(fa1, i);
            deg2[i] = deg1[i];
        }
        ll ans = sum;
        deg2[start]++;
        deg2[end]++;
        Union(fa2, start, end);
        for (int i = 1, p = 0; i <= cnt; i++) {
            if ((deg2[nodeArr[i]] & 1) == 1) {
                if (p == 0) {
                    p = i;
                } else {
                    ans += dist(nodeArr[p], nodeArr[i]);
                    while (p < i) {
                        Union(fa2, nodeArr[p++], nodeArr[i]);
                    }
                    p = 0;
                }
            }
        }
        for (int i = 1; i <= cnt - 1; i++) {
            edgeArr[i].u = nodeArr[i];
            edgeArr[i].v = nodeArr[i + 1];
            edgeArr[i].w = nodeArr[i + 1] - nodeArr[i];
        }
        ans += kruskal();
        return ans;
    }

    void prepare() {
        fix[s] = true;
        nodeSet.insert(s);
        sum = 0;
        for (int i = 1; i <= n; i++) {
            fa1[i] = i;
        }
        for (int i = 1, u, v; i <= m; i++) {
            u = a[i];
            v = b[i];
            deg1[u]++;
            deg1[v]++;
            fix[u] = true;
            fix[v] = true;
            nodeSet.insert(u);
            nodeSet.insert(v);
            Union(fa1, u, v);
            sum += dist(u, v);
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m >> s;
        for (int i = 1; i <= m; i++) {
            cin >> a[i] >> b[i];
        }
        prepare();
        for (int i = 1; i <= n; i++) {
            if (!fix[i]) {
                nodeSet.insert(i);
            }
            cout << compute(s, i) << " ";
            if (!fix[i]) {
                nodeSet.erase(i);
            }
        }
        cout << "\n";
        return 0;
    }

四、复杂度分析

三道题三个量级：垃圾车线性，邮递员多一个 map 的对数，丁香之路是平方对数。

操作时间复杂度空间复杂度
题目9：建图 + 判奇度 + 欧拉回路拆环O(n + m)O(n + m)
题目10：建链（map 查边）+ 压缩 + 欧拉回路 + 解压O((m + k) log m)O(m + k)（k 为序列节点总量）
题目11：n 个终点各一次配平 + kruskalO(n² log n)O(n + m)

题目9 的 m 到 10⁶，链式前向星和 vis 数组是唯一选择；题目10 的 map 可以换成排序加哈希压掉一个 log；题目11 的瓶颈在 n 个终点各跑一遍 kruskal，点列相邻差边让树边的候选只有 n-1 条，省了不少。

五、最后

这一讲十一道题全部收束在一个模版上：判定看度数、构造靠递归栈、连通性交给路径长度。三道难题给的教训是，翻译成图之后别急着写，先把「合法结构长什么样」想清楚：简单环、链、配平后的回路，想清楚了代码都是短的。

欧拉路径到此收尾，下一讲换强连通分量：从「把每条边走一遍」回到有向图，研究点与点之间能不能互相走到。
