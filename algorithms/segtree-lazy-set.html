线段树：范围重置的懒标记

把一段范围整段重置成同一个值，懒信息就从一个可以累加的量变成一个具体的值。值本身可能正好是 0，所以账本旁边还得配一位开关。

零、背景

两次落在同一段上的区间加可以合成一次。add 记 3、再记 2，等价于记 5，下发一次就够了。

把操作换成「整段重置成 v」，这两次就叠不起来了。[1, 8] 先重置成 5、再重置成 2，最后能算数的只有 2，前面那次被彻底取消。同一本账装不下这两种语义，账本得换一种写法。

一、问题

长度为 n 的数组 a，两种操作交替出现。

  • 修改：区间 [l, r] 里每个数都重置成 v。
  • 查询：区间 [l, r] 的累加和，或者最大值。

这两问都没有公开测试链接，课件用对数器验证：随机生成操作序列，线段树和暴力各跑一遍，逐条比对答案。

二、懒信息是一个值加一位开关

整段重置成 v 之后，节点上的信息同样能 O(1) 算出来：和是 v 乘元素个数，最大值就是 v。所以懒标记这条路照样走得通。

区别在账本的内容。加法的账本记的是「还欠多少」，欠 0 就表示没欠；重置的账本记的是「要变成的那个值」，而 0 是一个完全合法的重置值。用 change 是否为 0 来判断「这个节点上有没有待下发的重置」，会把「重置成 0」和「从未重置」看成同一件事。

所以重置的懒信息要占两个格子：change 装那个值，update 装一位开关。判断有没有账要发，看的是 update，不是 change。

三、lazy 里是赋值不是累加

加法的 lazy 是 sum[i] += v 乘 len，重置的 lazy 是 sum[i] = v 乘 len，一个加号一个等号，这一处写错就是把重置当成了加。

这一处也决定了重叠操作的语义。同一节点上先落到 5、再落到 2，第二次直接把 change[i] 覆盖成 2，于是下发出去的只有 2；如果写成累加，两次会合成 7，那是加法才有的行为。

// 查询信息: 累加和
    // 懒信息: change 记录「整段要变成的值」, update 是一位开关
    void up(int i) {
      sum[i] = sum[i << 1] + sum[i << 1 | 1];
    }

    // 当前范围 l~r 整段重置成 v
    void lazy(int i, ll v, int len) {
      sum[i] = v * len;
      change[i] = v;
      update[i] = true;
    }

    void down(int i, int ln, int rn) {
      if (update[i]) {
        lazy(i << 1, change[i], ln);
        lazy(i << 1 | 1, change[i], rn);
        update[i] = false;
      }
    }

    void build(int l, int r, int i) {
      if (l == r) {
        sum[i] = arr[l];
      } else {
        int mid = (l + r) >> 1;
        build(l, mid, i << 1);
        build(mid + 1, r, i << 1 | 1);
        up(i);
      }
      change[i] = 0;
      update[i] = false;
    }

    // [jobl, jobr] 整段重置成 jobv
    void rangeUpdate(int jobl, int jobr, ll jobv, int l, int r, int i) {
      if (jobl <= l && r <= jobr) {
        lazy(i, jobv, r - l + 1);
      } else {
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        if (jobl <= mid) {
          rangeUpdate(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
          rangeUpdate(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
      }
    }

    ll query(int jobl, int jobr, int l, int r, int i) {
      if (jobl <= l && r <= jobr) {
        return sum[i];
      }
      int mid = (l + r) >> 1;
      down(i, mid - l + 1, r - mid);
      ll ans = 0;
      if (jobl <= mid) {
        ans += query(jobl, jobr, l, mid, i << 1);
      }
      if (jobr > mid) {
        ans += query(jobl, jobr, mid + 1, r, i << 1 | 1);
      }
      return ans;
    }

五、把查询信息换成最大值

重置换成最大值，改动比区间加那一版还少。up 从求和改成取最大，lazy 里去掉长度参数，down 跟着少传两个参数。

重置这套懒信息不需要长度，是因为「整段变成 v」这个动作本身与范围有多长无关：最大值直接是 v，和才需要乘以长度。

// 查询信息: 最大值。懒信息: change 加一位 update 开关
    void up(int i) {
      mx[i] = max(mx[i << 1], mx[i << 1 | 1]);
    }

    // 整段重置成 v，最大值就是 v
    // 这里也不需要 len
    void lazy(int i, ll v) {
      mx[i] = v;
      change[i] = v;
      update[i] = true;
    }

    void down(int i) {
      if (update[i]) {
        lazy(i << 1, change[i]);
        lazy(i << 1 | 1, change[i]);
        update[i] = false;
      }
    }

    void build(int l, int r, int i) {
      if (l == r) {
        mx[i] = arr[l];
      } else {
        int mid = (l + r) >> 1;
        build(l, mid, i << 1);
        build(mid + 1, r, i << 1 | 1);
        up(i);
      }
      change[i] = 0;
      update[i] = false;
    }

    // [jobl, jobr] 整段重置成 jobv
    void rangeUpdate(int jobl, int jobr, ll jobv, int l, int r, int i) {
      if (jobl <= l && r <= jobr) {
        lazy(i, jobv);
      } else {
        int mid = (l + r) >> 1;
        down(i);
        if (jobl <= mid) {
          rangeUpdate(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
          rangeUpdate(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
      }
    }

    ll query(int jobl, int jobr, int l, int r, int i) {
      if (jobl <= l && r <= jobr) {
        return mx[i];
      }
      int mid = (l + r) >> 1;
      down(i);
      ll ans = LLONG_MIN;
      if (jobl <= mid) {
        ans = max(ans, query(jobl, jobr, l, mid, i << 1));
      }
      if (jobr > mid) {
        ans = max(ans, query(jobl, jobr, mid + 1, r, i << 1 | 1));
      }
      return ans;
    }

六、错一行会怎样

拿「重置 + 查最大值」这一版，每组随机用例里插一个区间、问一次区间，3000 组统计答案出过错的比例。改动都在懒信息这一小块上。

随机用例规模lazy 忘置开关down 忘清开关lazy 忘写 change正解
n ≤ 8，值 ≤ 527.37%11.53%26.60%0.00%
n ≤ 20，值 ≤ 1045.43%23.33%43.47%0.00%
n ≤ 60，值 ≤ 100059.03%33.13%54.47%0.00%
n ≤ 200，值 ≤ 10⁵68.03%41.27%63.13%0.00%

三处错的严重程度不一样。忘置开关最狠，整个节点的重置根本不往下走，父节点的旧账会一直压在上面；忘清开关轻一些，多发的那个值本身是对的，只是等儿子后来被改过之后再发一次，就把儿子的新值盖回了旧值；忘写 change 是下发时用了上一次的旧值，重叠重置的时候才现形。

再补一组对照：把 n 固定成 20，值域从 ±2 一路拉到 ±10⁵，三档分别是 55.30%、57.27%、58.57%、58.57%（忘置开关）。值域涨了五万倍，抓取率只动了三个百分点。这三处错依赖的是「操作序列有没有撞上那个形状」，跟数值大不大、相不相等没关系，所以随机数据要把规模铺开才抓得准。

七、复杂度分析

操作时间空间
建树 buildO(n)两条信息数组 + 一条 change + 一条开关
范围重置O(log n)递归栈 O(log n)
范围查询O(log n)递归栈 O(log n)
整体O((n + m) log n)O(n)

比区间加多出来的是一位布尔开关，按字节算也不值一提。n = 10⁶ 时四组信息数组各开 4n 个 long long，加起来约 32 MB，和区间加那一版同量级。

八、总结

重置和加共用「完全覆盖就懒住、要用再下发一层、回程重算父亲」这套骨架，差别全在懒信息上：加记一个可以累加的量，重置记一个具体的值再配一位开关。开关是必须的，因为 0 既是合法的重置值，又是「什么都没记」的默认值，光看值分不出这两种状态。

下一步把两种修改放在同一棵树上。到那时懒信息的形态就不止一种，节点上同时挂着待重置的值的待补发的加，谁先谁后就成了必须先想清楚的事。
