有一次错误称重：枚举删除与精确整数消元

n+1 条称重里恰好藏着一条错的，把它找出来的办法很笨但有效：一条一条删，删完剩下的 n 条能解出合法方案的，就是那条错的。合法方案要求重量全是正整数、最重物品唯一，这两个条件在 double 下差点翻车。

零、把题面翻译成方程组

n 个物品，重量未知。每次称重格式是「3 2 5 6 10」：本次涉及 3 个物品，编号 2、5、6，总重 10。这是一条方程：物品 2、5、6 的系数是 1，其余系数 0，右边是 10。

一共 n+1 条称重。删掉恰好 1 条错的，剩下 n 条方程、n 个未知数，正好用高斯消元解。合法方案三个条件：

1）每个重量都是正整数；

2）最重物品有且仅有 1 个；

3）只有删掉错误那条称重才能求出合法方案：枚举 n+1 次删除，恰好 1 次能解出合法方案（times == 1），才输出最重编号，否则打印 illegal。

一、枚举删除 + double 判整

主流程是枚举：把第 k 条称重换到末尾，只取前 n 条，用上一节的严格三分模板解，检查合法性，再换回来。check 里三条硬规则：没有主元直接 0；解不是正整数直接 0；最重并列直接 0。

#include <bits/stdc++.h>
    using namespace std;

    // 枚举删除第 k 条称重，剩下 n 条解 n 个未知数
    // data[i][j] = 1 表示第 i 条称重包含物品 j，data[i][n + 1] 是该条总重
    const int MAXN = 102;
    int data[MAXN][MAXN];
    double mat[MAXN][MAXN];
    int n;

    void swapData(int i, int j) {
      for (int k = 1; k <= n + 1; k++) {
        swap(data[i][k], data[j][k]);
      }
    }

    int main() {
      scanf("%d", &n);
      for (int i = 1; i <= n + 1; i++) {
        int m;
        scanf("%d", &m);
        for (int j = 1, cur; j <= m; j++) {
          scanf("%d", &cur);
          data[i][cur] = 1;
        }
        scanf("%d", &data[i][n + 1]);
      }
      int ans = 0, times = 0;
      for (int k = 1; k <= n + 1; k++) {
        swapData(k, n + 1);  // 第 k 条换到末尾，只取前 n 条
        for (int i = 1; i <= n; i++) {
          for (int j = 1; j <= n + 1; j++) {
            mat[i][j] = data[i][j];
          }
        }
        swapData(k, n + 1);  // 换回来
        int cur = check();
        if (cur != 0) {
          times++;
          ans = cur;
        }
      }
      // 恰好一条称重是错的，排除它之后合法方案必须唯一
      if (times != 1) {
        printf("illegal\n");
      } else {
        printf("%d\n", ans);
      }
      return 0;
    }
#include <bits/stdc++.h>
    using namespace std;

    // double 版：解必须正好是正整数，最重物品有且仅有 1 个
    // gauss 用的是严格三分模板
    const double sml = 1e-7;
    double mat[102][102];
    int n;

    int check() {
      gauss(n);
      double maxv = 0;
      int maxt = 0, ans = 0;
      for (int i = 1; i <= n; i++) {
        if (fabs(mat[i][i]) < sml) {
          return 0;  // 没有主元，解不出来
        }
        if (mat[i][n + 1] <= 0 || mat[i][n + 1] != (int)mat[i][n + 1]) {
          return 0;  // 不是正整数
        }
        if (maxv < mat[i][n + 1]) {
          maxv = mat[i][n + 1];
          maxt = 1;
          ans = i;
        } else if (maxv == mat[i][n + 1]) {
          maxt++;  // 最重并列，不合法
        }
      }
      return maxt == 1 ? ans : 0;
    }

double 版的判整写的是 mat[i][n + 1] != (int)mat[i][n + 1]。重量是整数、方程系数是 0/1，理论上解出来就是整数，double 表示小整数精确。问题出在数值过程：消元中间的除法会产生小数，最后一步虽然回到了整数，但 double 的舍入可能让它变成 2.9999999999996 这种数，(int) 截断成 2，和原值不相等，判整误杀。原题起初能过，后来加了测试数据，double 版就稳不住了。

二、整数版：辗转消元与回代

既然解是整数，就别碰 double。矩阵换成 long，整个消元在整数域里做，有两个改动：

1）主元行不归一化。long 除法会截断分数，除以主元会把解毁掉，所以消元只消下方：rate = mat[i][i] / mat[j][i]，主元行减去第 j 行的 rate 倍，再交换两行，直到第 j 行第 i 列变 0。这和辗转相除求 gcd 是同一个动作，系数只会越来越小，不溢出。

2）消完是上三角，倒序回代。第 i 行可能还带着编号更大的变量，从 n 到 1 逐个减掉它们的贡献，再用取模判断能不能整除：mat[i][n + 1] % mat[i][i] != 0 就是非整数解。

#include <bits/stdc++.h>
    using namespace std;

    // 整数版：mat 换成 long，主元行不再归一化，只消下方（辗转相除风格）
    // 之后倒序回代，用取模精确判断是否整数
    long mat[102][102];
    int n;

    void swapRows(int a, int b) {
      for (int j = 1; j <= n + 1; j++) {
        swap(mat[a][j], mat[b][j]);
      }
    }

    void gauss(int n) {
      for (int i = 1; i <= n; i++) {
        int max = i;
        for (int j = 1; j <= n; j++) {
          if (j < i && mat[j][j] != 0) {
            continue;
          }
          if (llabs(mat[j][i]) > llabs(mat[max][i])) {
            max = j;
          }
        }
        swapRows(i, max);
        if (mat[i][i] == 0) {
          return;  // 找不到主元，交给 check 判不合法
        }
        for (int j = i + 1; j <= n; j++) {
          // 反复整行相减再交换，直到 mat[j][i] 变成 0
          while (mat[j][i] != 0) {
            long rate = mat[i][i] / mat[j][i];
            if (rate != 0) {
              for (int k = i; k <= n + 1; k++) {
                mat[i][k] -= mat[j][k] * rate;
              }
            }
            swapRows(i, j);
          }
        }
      }
    }

    int check() {
      gauss(n);
      // 倒序回代：第 i 行可能还带着编号更大的变量
      for (int i = n; i >= 1; i--) {
        if (mat[i][i] == 0) {
          return 0;
        }
        for (int j = i + 1; j <= n; j++) {
          mat[i][n + 1] -= mat[i][j] * mat[j][n + 1];
        }
        if (mat[i][n + 1] % mat[i][i] != 0) {
          return 0;  // 除不尽，不是整数解
        }
        mat[i][n + 1] /= mat[i][i];
      }
      long maxv = 0;
      int maxt = 0, ans = 0;
      for (int i = 1; i <= n; i++) {
        if (mat[i][i] == 0 || mat[i][n + 1] <= 0) {
          return 0;
        }
        if (maxv < mat[i][n + 1]) {
          maxv = mat[i][n + 1];
          maxt = 1;
          ans = i;
        } else if (maxv == mat[i][n + 1]) {
          maxt++;
        }
      }
      return maxt == 1 ? ans : 0;
    }

回代是必须的：只消下方的话，第 i 行的解在消元结束时不在最后一列，要先把第 i+1..n 行的解代回来。取模判整除替代了 double 的浮点判整，整数域里没有精度问题，错杀不存在了。

三、复杂度

枚举 n+1 条，每次高斯消元 O(n3)，总共 O(n3·n) = O(n4)。n ≤ 100，实测能跑，再大就要想别的办法跳过重复消元。

演示：n=3 的 4 条称重，逐条删除并判定合法性

四、总结

错误称重把「找错」翻译成「枚举删除 + 解方程组」，剩下的活全是高斯消元。double 判整在数据变硬后翻车，换成整数矩阵 + 辗转消元 + 回代取模，语义不变、精度归零。以后遇到「恰好一条数据是错的」这类题，枚举删除是首选框架；遇到整数解的方程组，优先想整数消元而不是 double。
