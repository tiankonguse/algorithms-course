线段树分治：换一根轴

上一页的轴是时间。这一页把轴换掉：一根是答案轴，考察每个值能不能当 mex；一根是颜色轴，考察每个颜色的边在路径统计里贡献了多少。骨架不变，挂任务的规则跟着轴走。

零、轴是什么

线段树分治的骨架是：一根轴、一堆「在某段轴值上生效」的任务、一个能撤销的数据结构。轴不一定是时间。

换轴的关键是想清楚两件事：轴上的每个位置代表什么、每条任务在哪些轴值上生效。想清楚了，剩下的和模版题一模一样。

一、题目3：最小 mex 生成树

n 个点、m 条边的无向连通图，边有边权。求一棵生成树，让边权集合的 mex 尽可能小。这里 mex(S) 指没出现在 S 里的最小自然数，0 算自然数。1 ≤ n ≤ 10⁶，1 ≤ m ≤ 2 × 10⁶，0 ≤ 边权 ≤ 10⁵。测试链接：P5631。

先翻译一遍：生成树 T 的 mex 等于 w，意思是 T 上一条边权为 w 的边都没有。所以 mex(T) = w 当且仅当「不用权 w 的边也能生成树」，去掉所有权 w 的边后图仍然连通。反过来，只要 G 去掉权 w 的边后连通，就存在一棵不含权 w 边的生成树，它的 mex 不超过 w。

于是答案就是最小的 w，使得去掉权 w 的边后图连通。从 0 开始逐个考察 w，这正是把「答案」当轴。

轴是 0 到 V（V 为值域上界）。权 w 的边在轴值 w 上必须缺席，在其余轴值上都生效，所以挂两段：[0, w-1] 和 [w+1, V]。

part 的维护和可撤销并查集绑在一起：每成功并一次 part 减一，undo 一次 part 加一。叶子上 part == 1 就是全连通。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 1000001;
    const int MAXV = 100001;
    const int MAXT = 30000001;

    int n, m, v;

    int father[MAXN];
    int siz[MAXN];
    int rollback[MAXN][2];
    int opsize = 0;

    int head[MAXV << 2];
    int nxt[MAXT];
    int tox[MAXT];
    int toy[MAXT];
    int cnt = 0;

    int part;

    void addEdge(int i, int x, int y) {
        nxt[++cnt] = head[i];
        tox[cnt] = x;
        toy[cnt] = y;
        head[i] = cnt;
    }

    int find(int i) {
        while (i != father[i]) {
            i = father[i];
        }
        return i;
    }

    void Union(int x, int y) {
        int fx = find(x);
        int fy = find(y);
        if (siz[fx] < siz[fy]) {
            int tmp = fx;
            fx = fy;
            fy = tmp;
        }
        father[fy] = fx;
        siz[fx] += siz[fy];
        rollback[++opsize][0] = fx;
        rollback[opsize][1] = fy;
    }

    void undo() {
        int fx = rollback[opsize][0];
        int fy = rollback[opsize--][1];
        father[fy] = fy;
        siz[fx] -= siz[fy];
    }

    void add(int jobl, int jobr, int jobx, int joby, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            addEdge(i, jobx, joby);
        } else {
            int mid = (l + r) >> 1;
            if (jobl <= mid) {
                add(jobl, jobr, jobx, joby, l, mid, i << 1);
            }
            if (jobr > mid) {
                add(jobl, jobr, jobx, joby, mid + 1, r, i << 1 | 1);
            }
        }
    }

    int dfs(int l, int r, int i) {
        int unionCnt = 0;
        for (int ei = head[i]; ei > 0; ei = nxt[ei]) {
            int fx = find(tox[ei]);
            int fy = find(toy[ei]);
            if (fx != fy) {
                Union(fx, fy);
                part--;
                unionCnt++;
            }
        }
        int ans = -1;
        if (l == r) {
            if (part == 1) {
                ans = l;
            }
        } else {
            int mid = (l + r) >> 1;
            ans = dfs(l, mid, i << 1);
            if (ans == -1) {
                ans = dfs(mid + 1, r, i << 1 | 1);
            }
        }
        for (int k = 1; k <= unionCnt; k++) {
            undo();
            part++;
        }
        return ans;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        v = MAXV;
        for (int i = 1; i <= n; i++) {
            father[i] = i;
            siz[i] = 1;
        }
        for (int i = 1; i <= m; i++) {
            int x, y, w;
            cin >> x >> y >> w;
            if (w > 0) {
                add(0, w - 1, x, y, 0, v, 1);
            }
            add(w + 1, v, x, y, 0, v, 1);
        }
        part = n;
        cout << dfs(0, v, 1) << '\n';
        return 0;
    }

二、题目4：独特事件

n 个点的树，每条边有颜色。f(u, v) 定义为 u 到 v 的简单路径上恰好出现一次的颜色个数，求所有点对 f 的总和。1 ≤ 颜色值 ≤ n ≤ 2 × 10⁵。测试链接：CF1681F。

直接对每对点数路径颜色是 O(n²) 的，得换贡献视角：对每个颜色分别算，有多少对点的路径上这个颜色恰好出现一次。

固定颜色 c。把所有颜色不是 c 的边并起来，树被切成若干连通块；颜色 c 的边在这些块之间连出去，而且每条 c 色边的两端在不同的块里（不然树上就有环了）。

一条 c 色边 (x, y) 的贡献是：x 所在块的大小乘以 y 所在块的大小。原因：路径一端落在 x 的块、另一端落在 y 的块时，路径必须经过这条边；而树上的路径唯一，它映射到「块缩点图」上就是 x 块到 y 块的这一条边，除它之外不经过任何别的 c 色边。恰好一次，就是这个意思。

和上一题的差别在叶子上：这里叶子不做查询，做「统计」。c 色边的列表提前按颜色分好类挂在颜色下标上，到叶子直接累加。

还有个细节：叶子上的并查集装的是全部非 c 色边，它们是森林，合并互不成环，所以 Union 不用判 find 是否相等，直接并。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 500001;
    const int MAXT = 10000001;
    int n, v;

    int father[MAXN];
    int siz[MAXN];
    int rollback[MAXN][2];
    int opsize = 0;

    int headc[MAXN];
    int nxtc[MAXN];
    int xc[MAXN];
    int yc[MAXN];
    int cntc = 0;

    int headt[MAXN << 2];
    int nxtt[MAXT];
    int xt[MAXT];
    int yt[MAXT];
    int cntt = 0;

    long long ans = 0;

    void addEdgeC(int i, int x, int y) {
        nxtc[++cntc] = headc[i];
        xc[cntc] = x;
        yc[cntc] = y;
        headc[i] = cntc;
    }

    void addEdgeS(int i, int x, int y) {
        nxtt[++cntt] = headt[i];
        xt[cntt] = x;
        yt[cntt] = y;
        headt[i] = cntt;
    }

    int find(int i) {
        while (i != father[i]) {
            i = father[i];
        }
        return i;
    }

    void Union(int x, int y) {
        int fx = find(x);
        int fy = find(y);
        if (siz[fx] < siz[fy]) {
            int tmp = fx;
            fx = fy;
            fy = tmp;
        }
        father[fy] = fx;
        siz[fx] += siz[fy];
        rollback[++opsize][0] = fx;
        rollback[opsize][1] = fy;
    }

    void undo() {
        int fx = rollback[opsize][0];
        int fy = rollback[opsize--][1];
        father[fy] = fy;
        siz[fx] -= siz[fy];
    }

    void add(int jobl, int jobr, int jobx, int joby, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            addEdgeS(i, jobx, joby);
        } else {
            int mid = (l + r) >> 1;
            if (jobl <= mid) {
                add(jobl, jobr, jobx, joby, l, mid, i << 1);
            }
            if (jobr > mid) {
                add(jobl, jobr, jobx, joby, mid + 1, r, i << 1 | 1);
            }
        }
    }

    void dfs(int l, int r, int i) {
        int unionCnt = 0;
        for (int ei = headt[i]; ei > 0; ei = nxtt[ei]) {
            Union(xt[ei], yt[ei]);
            unionCnt++;
        }
        if (l == r) {
            for (int ei = headc[l], fx, fy; ei > 0; ei = nxtc[ei]) {
                fx = find(xc[ei]);
                fy = find(yc[ei]);
                ans += 1LL * siz[fx] * siz[fy];
            }
        } else {
            int mid = (l + r) >> 1;
            dfs(l, mid, i << 1);
            dfs(mid + 1, r, i << 1 | 1);
        }
        for (int k = 1; k <= unionCnt; k++) {
            undo();
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n;
        v = n;
        for (int i = 1, x, y, c; i < n; i++) {
            cin >> x >> y >> c;
            addEdgeC(c, x, y);
            if (c > 1) {
                add(1, c - 1, x, y, 1, v, 1);
            }
            if (c < v) {
                add(c + 1, v, x, y, 1, v, 1);
            }
        }
        for (int i = 1; i <= n; i++) {
            father[i] = i;
            siz[i] = 1;
        }
        dfs(1, v, 1);
        cout << ans << '\n';
        return 0;
    }

三、复杂度分析

题目 3：m 条边各拆两段，挂任务 O(m log V)；遍历时每段任务一次并查集操作，O(log n)。

题目 4：n-1 条树边各拆两段，O(n log n) 份任务；每色叶子上把该色边全扫一遍，总和还是 O(n)。

操作时间复杂度空间复杂度
题目 3：挂任务 + 遍历O(m log V log n)O(m log V)：任务链表，V 为边权值域
题目 4：挂任务 + 遍历 + 叶子统计O(n log² n)O(n log n)：任务链表；颜色邻接表 O(n)
一次 Union / undoO(log n)回滚栈 O(m log V)

四、最后

两道题的轴都不是时间，但「任务在一段轴值上生效、叶子上兑现」的形状没变。下一页回到时间轴，处理另一种麻烦：题目根本不直接告诉你边什么时候存在，得从操作序列里自己整理出时间段。
