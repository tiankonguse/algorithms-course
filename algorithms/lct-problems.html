LCT 模版题目

模版和三种最直接的用法在上一页。这一页是剩下的四道，共同点是都要在模版上多维护一点东西：一道要记住哪些边正在开战，一道要汇总路径上的点权和，一道要把「跳几步」数成路径上的点数，最后一道要按颜色把整棵树复制成好几份。

一、题目4：部落冲突

n 个点，给定 n-1 条边，所有点组成一棵树。m 条操作：

  • C x y：x 和 y 在树中相邻，之前没开战，现在开战，这条直接边变成不可通行。每条 C 操作是一次战争事件，从 1 开始编号，每次自增。
  • U x：第 x 号战争事件结束，对应的直接边恢复可通行，保证战争不会重复结束。
  • Q x y：查 x 和 y 是否连通，打印 Yes 或 No。

n、m 到 3×10⁵。

三种操作正好对上模版的三个动作。开战就是 cut(x, y)，停战就是 link(x, y)，查询就是比两次 findroot。没有任何信息要汇总，up 都省了。

要额外做的只有一件事：U 操作给的是战争事件的编号，而数组里得能按编号找回那次开战的两个端点。做法是开两个数组 cutx、cuty，遇到 C 操作就先切边、再把端点按顺序存进去，下标就是这次战争的编号。U x 取 cutx[x] 和 cuty[x] 把边连回来。

完整代码：

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 300001;
    int n, m;

    int fa[MAXN];
    int ls[MAXN];
    int rs[MAXN];
    int sta[MAXN];
    bool rev[MAXN];

    int cutx[MAXN];
    int cuty[MAXN];
    int cutc;

    bool isroot(int x) {
        return ls[fa[x]] != x && rs[fa[x]] != x;
    }

    int lr(int x) {
        return ls[fa[x]] == x ? 0 : 1;
    }

    void reverse(int x) {
        if (x != 0) {
            swap(ls[x], rs[x]);
            rev[x] = !rev[x];
        }
    }

    void down(int x) {
        if (rev[x]) {
            reverse(ls[x]);
            reverse(rs[x]);
            rev[x] = false;
        }
    }

    void rotate(int x) {
        int f = fa[x], g = fa[f];
        if (lr(x) == 0) {
            ls[f] = rs[x];
            if (ls[f] != 0) {
                fa[ls[f]] = f;
            }
            rs[x] = f;
        } else {
            rs[f] = ls[x];
            if (rs[f] != 0) {
                fa[rs[f]] = f;
            }
            ls[x] = f;
        }
        if (!isroot(f)) {
            if (lr(f) == 0) {
                ls[g] = x;
            } else {
                rs[g] = x;
            }
        }
        fa[f] = x;
        fa[x] = g;
    }

    void splay(int x) {
        int siz = 0;
        sta[++siz] = x;
        for (int y = x; !isroot(y); y = fa[y]) {
            sta[++siz] = fa[y];
        }
        while (siz != 0) {
            down(sta[siz--]);
        }
        while (!isroot(x)) {
            int f = fa[x];
            if (!isroot(f)) {
                if (lr(x) == lr(f)) {
                    rotate(f);
                } else {
                    rotate(x);
                }
            }
            rotate(x);
        }
    }

    void access(int x) {
        for (int y = 0; x != 0; y = x, x = fa[x]) {
            splay(x);
            rs[x] = y;
        }
    }

    void makeroot(int x) {
        access(x);
        splay(x);
        reverse(x);
    }

    int findroot(int x) {
        access(x);
        splay(x);
        down(x);
        while (ls[x] != 0) {
            x = ls[x];
            down(x);
        }
        splay(x);
        return x;
    }

    void link(int x, int y) {
        makeroot(x);
        if (findroot(y) != x) {
            fa[x] = y;
        }
    }

    void cut(int x, int y) {
        makeroot(x);
        if (findroot(y) == x && fa[y] == x && ls[y] == 0 && rs[x] == y) {
            fa[y] = rs[x] = 0;
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1, x, y; i < n; i++) {
            cin >> x >> y;
            link(x, y);
        }
        for (int i = 1, x, y; i <= m; i++) {
            string op;
            cin >> op;
            if (op == "C") {
                cin >> x >> y;
                cut(x, y);
                cutx[++cutc] = x;
                cuty[cutc] = y;
            } else if (op == "U") {
                cin >> x;
                link(cutx[x], cuty[x]);
            } else {
                cin >> x >> y;
                if (findroot(x) == findroot(y)) {
                    cout << "Yes" << "\n";
                } else {
                    cout << "No" << "\n";
                }
            }
        }
        return 0;
    }

二、题目5：群岛

n 个点，每个点有点权，初始时没有边。m 条操作：

  • bridge x y：如果 x 和 y 已经连通，打印 no；否则打印 yes，然后加一条无向边。
  • penguins x y：把 x 的点权改成 y。
  • excursion x y：如果 x 和 y 不连通，打印 impossible；否则打印 x 到 y 路径上点权的累加和。

n 到 3×10⁴，m 到 3×10⁵。

这是一道标准的汇总题：路径上的点权和，交给 up 维护一个 sum 就行。

void up(int x) {
    sum[x] = sum[ls[x]] + sum[rs[x]] + arr[x];
}

三个操作各对应一段现成的逻辑。bridge 先用 findroot 判连通，顺便决定打印哪个词；不连通就 link。penguins 是单点改权，splay(x) 之后改 arr[x] 再 up(x)。excursion 判连通后 split(x, y)，答案就是 sum[y]。

link 内部其实也会判一次连通，bridge 外面这层判断是为了拿到该打印的信息，不是为了防重边。

完整代码：

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 30001;
    int n, m;
    int arr[MAXN];

    int fa[MAXN];
    int ls[MAXN];
    int rs[MAXN];
    int sta[MAXN];
    bool rev[MAXN];

    int sum[MAXN];

    void up(int x) {
        sum[x] = sum[ls[x]] + sum[rs[x]] + arr[x];
    }

    bool isroot(int x) {
        return ls[fa[x]] != x && rs[fa[x]] != x;
    }

    int lr(int x) {
        return ls[fa[x]] == x ? 0 : 1;
    }

    void reverse(int x) {
        if (x != 0) {
            swap(ls[x], rs[x]);
            rev[x] = !rev[x];
        }
    }

    void down(int x) {
        if (rev[x]) {
            reverse(ls[x]);
            reverse(rs[x]);
            rev[x] = false;
        }
    }

    void rotate(int x) {
        int f = fa[x], g = fa[f];
        if (lr(x) == 0) {
            ls[f] = rs[x];
            if (ls[f] != 0) {
                fa[ls[f]] = f;
            }
            rs[x] = f;
        } else {
            rs[f] = ls[x];
            if (rs[f] != 0) {
                fa[rs[f]] = f;
            }
            ls[x] = f;
        }
        if (!isroot(f)) {
            if (lr(f) == 0) {
                ls[g] = x;
            } else {
                rs[g] = x;
            }
        }
        fa[f] = x;
        fa[x] = g;
        up(f);
        up(x);
    }

    void splay(int x) {
        int siz = 0;
        sta[++siz] = x;
        for (int y = x; !isroot(y); y = fa[y]) {
            sta[++siz] = fa[y];
        }
        while (siz != 0) {
            down(sta[siz--]);
        }
        while (!isroot(x)) {
            int f = fa[x];
            if (!isroot(f)) {
                if (lr(x) == lr(f)) {
                    rotate(f);
                } else {
                    rotate(x);
                }
            }
            rotate(x);
        }
    }

    void access(int x) {
        for (int y = 0; x != 0; y = x, x = fa[x]) {
            splay(x);
            rs[x] = y;
            up(x);
        }
    }

    void makeroot(int x) {
        access(x);
        splay(x);
        reverse(x);
    }

    int findroot(int x) {
        access(x);
        splay(x);
        down(x);
        while (ls[x] != 0) {
            x = ls[x];
            down(x);
        }
        splay(x);
        return x;
    }

    void split(int x, int y) {
        makeroot(x);
        access(y);
        splay(y);
    }

    void link(int x, int y) {
        makeroot(x);
        if (findroot(y) != x) {
            fa[x] = y;
        }
    }

    void cut(int x, int y) {
        makeroot(x);
        if (findroot(y) == x && fa[y] == x && ls[y] == 0 && rs[x] == y) {
            fa[y] = rs[x] = 0;
            up(x);
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
            sum[i] = arr[i];
        }
        cin >> m;
        string op;
        int x, y;
        for (int i = 1; i <= m; i++) {
            cin >> op >> x >> y;
            if (op == "bridge") {
                if (findroot(x) == findroot(y)) {
                    cout << "no" << "\n";
                } else {
                    cout << "yes" << "\n";
                    link(x, y);
                }
            } else if (op == "penguins") {
                splay(x);
                arr[x] = y;
                up(x);
            } else {
                if (findroot(x) != findroot(y)) {
                    cout << "impossible" << "\n";
                } else {
                    split(x, y);
                    cout << sum[y] << "\n";
                }
            }
        }
        return 0;
    }

三、题目6：弹飞绵羊

n 个装置编号 0 到 n-1，从左到右摆成一条直线，每个装置有一个弹力 f。比如 3 号装置弹力为 5，从 3 号可以向右跳到 8 号。如果跳到的位置大于等于 n，也就是不存在这个装置，就认为被弹飞了。

m 条操作：1 x 查从 x 出发跳几次会弹飞，2 x y 把 x 号装置的弹力改成 y。

n 到 2×10⁵，m 到 10⁵。

这题的图和前面几题不一样：它是从左往右一排装置，不是一棵给定的树。但每个装置只往一个方向跳，于是可以凭空造一棵树出来，加一个虚拟终点代表「弹飞」，每个装置向它跳到的那个位置连一条边。

点编号从 0 开始，代码换成从 1 开始，虚拟终点就是 n+1。装置 x 连向 min(x + force[x], n + 1)，跳到界外的一律连到 n+1。

树建好之后，查询就变成数路径上的边数。模版里 siz[x] = siz[ls[x]] + siz[rs[x]] + 1 数的是子树点数，所以从 x 到 n+1 的路径点数减一就是跳跃次数。

split(x, n + 1);
cout << siz[n + 1] - 1 << "\n";

下面把从 1 号出发的跳跃过程走一遍。

从 1 号装置出发，跳几次弹飞

橙色是当前所在的装置，蓝色箭头是走过的跳跃边。点「下一步」一路走到虚拟终点。

走过的跳跃边 当前装置 还没走到

改弹力就是把旧的边切断、换成新的边。注意要先算旧的下一跳，再改 force[x]，否则第二次 min 用的就是你刚写进去的新值。

cut(x, min(x + force[x], n + 1));
force[x] = y;
link(x, min(x + force[x], n + 1));

还有一个容易忘的细节：题目编号从 0 开始，读入 x 之后要 x++ 对齐到 LCT 的下标。虚拟终点 n+1 也要在一开始把 siz 初始化成 1。

        }
    }

    void down(int x) {

            reverse(ls[x]);
            reverse(rs[x]);
            rev[x] = false;
        }
    }

    void rotate(int x) {
        int f = fa[x], g = fa[f];
        if (lr(x) == 0) {
            ls[f] = rs[x];
            if (ls[f] != 0) {
                fa[ls[f]] = f;
            }
            rs[x] = f;
        } else {
            rs[f] = ls[x];
            if (rs[f] != 0) {
                fa[rs[f]] = f;
            }
            ls[x] = f;
        }
        if (!isroot(f)) {
            if (lr(f) == 0) {
                ls[g] = x;
            } else {
                rs[g] = x;
            }
        }
        fa[f] = x;
        fa[x] = g;
        up(f);
        up(x);
    }

    void splay(int x) {
        int size = 0;
        sta[++size] = x;
        for (int y = x; !isroot(y); y = fa[y]) {
            sta[++size] = fa[y];

        }

        reverse(x);
    }

                link(x, min(x + force[x], n + 1));
            }
        }

四、题目7：网络

无向图有 n 个点，点有点权，有 m 条边，边有颜色，颜色总数为 C，取值 0 到 C-1。图保证没有重边和自环，并且始终满足两条性质：

  • 性质1：对任意节点，它连出去的边里，相同颜色的边不超过两条。
  • 性质2：不存在同色的环，也就是相同颜色的边构成的图里没有环。

q 条操作：0 x y 把点 x 的权改成 y；2 c x y 在颜色 c 的边构成的图里打印 x 到 y 路径上的点权最大值，不连通打印 -1；1 x y c 把边 (x,y) 的颜色改成 c，按规则打印结果。

n 到 10⁴，C 不超过 10，m、q 到 10⁵。

先看两条性质有什么用。

性质1 限制了每个点在每种颜色下的度数不超过 2。度数不超过 2 的图，每个连通块只能是一条链。性质2 说同色不成环。两条合起来：只看某个颜色 c 的边，得到的是一片由链组成的森林。

森林正好是 LCT 能维护的东西。于是给每个颜色单独开一份 LCT，一共 C 份。颜色 c 里的原图点 x，在这份结构里的编号是 c * n + x。

点权的处理要小心。原图的点 x 在所有颜色层里都要有同一份权值，改一次就得改 C 层。

void updateNode(int x, int v) {
    for (int c = 0; c < C; c++) {
        int cur = node(c, x);
        splay(cur);
        arr[cur] = v;
        up(cur);
    }
}

改边颜色是这题最绕的部分，四种结果对应四道检查，顺序不能反。

先查这条直接边是否存在，不存在直接 No such edge.。存在的话，看新颜色 c 那一层里，x 和 y 的度数是不是已经到 2，到了再加边就会违反性质1，打印 Error 1.。接着在 c 层里看 x 和 y 是否已经连通，连通说明新边会在 c 层里成环，违反性质2，打印 Error 2.。Error 1 优先于 Error 2，所以必须按这个顺序判。都通过，就断开旧颜色的边、在 c 层里连上新边，打印 Success.。

int updateEdge(int x, int y, int c) {
    auto it = edgeColor.find(edge(x, y));
    if (it == edgeColor.end()) {
        return 3;
    }
    int p = it->second;
    if (p == c) {
        return 4;
    }
    if (nodeDegree[node(c, x)] >= 2 || nodeDegree[node(c, y)] >= 2) {
        return 1;
    }
    if (findroot(node(c, x)) == findroot(node(c, y))) {
        return 2;
    }
    disconnect(p, x, y);
    connect(c, x, y);
    return 4;
}

度数单独用数组 nodeDegree 记，比去 LCT 里数快。改颜色已经等于原色时也直接算成功，不用重复切连。

查询就简单了：在 c 层里判连通，连通就 split 出来读最大值。

完整代码：

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 100001;
    int n, m, C, q;

    int fa[MAXN];
    int ls[MAXN];
    int rs[MAXN];
    int sta[MAXN];
    bool rev[MAXN];

    int arr[MAXN];
    int maxv[MAXN];

    int nodeDegree[MAXN];
    map<pair<int, int>, int> edgeColor;

    void up(int x) {
        maxv[x] = max(max(maxv[ls[x]], maxv[rs[x]]), arr[x]);
    }

    bool isroot(int x) {
        return ls[fa[x]] != x && rs[fa[x]] != x;
    }

    int lr(int x) {
        return ls[fa[x]] == x ? 0 : 1;
    }

    void reverse(int x) {
        if (x != 0) {
            swap(ls[x], rs[x]);
            rev[x] = !rev[x];
        }
    }

    void down(int x) {
        if (rev[x]) {
            reverse(ls[x]);
            reverse(rs[x]);
            rev[x] = false;
        }
    }

    void rotate(int x) {
        int f = fa[x], g = fa[f];
        if (lr(x) == 0) {
            ls[f] = rs[x];
            if (ls[f] != 0) {
                fa[ls[f]] = f;
            }
            rs[x] = f;
        } else {
            rs[f] = ls[x];
            if (rs[f] != 0) {
                fa[rs[f]] = f;
            }
            ls[x] = f;
        }
        if (!isroot(f)) {
            if (lr(f) == 0) {
                ls[g] = x;
            } else {
                rs[g] = x;
            }
        }
        fa[f] = x;
        fa[x] = g;
        up(f);
        up(x);
    }

    void splay(int x) {
        int siz = 0;
        sta[++siz] = x;
        for (int y = x; !isroot(y); y = fa[y]) {
            sta[++siz] = fa[y];
        }
        while (siz != 0) {
            down(sta[siz--]);
        }
        while (!isroot(x)) {
            int f = fa[x];
            if (!isroot(f)) {
                if (lr(x) == lr(f)) {
                    rotate(f);
                } else {
                    rotate(x);
                }
            }
            rotate(x);
        }
    }

    void access(int x) {
        for (int y = 0; x != 0; y = x, x = fa[x]) {
            splay(x);
            rs[x] = y;
            up(x);
        }
    }

    void makeroot(int x) {
        access(x);
        splay(x);
        reverse(x);
    }

    int findroot(int x) {
        access(x);
        splay(x);
        down(x);
        while (ls[x] != 0) {
            x = ls[x];
            down(x);
        }
        splay(x);
        return x;
    }

    void split(int x, int y) {
        makeroot(x);
        access(y);
        splay(y);
    }

    void link(int x, int y) {
        makeroot(x);
        if (findroot(y) != x) {
            fa[x] = y;
        }
    }

    void cut(int x, int y) {
        makeroot(x);
        if (findroot(y) == x && fa[y] == x && ls[y] == 0 && rs[x] == y) {
            fa[y] = rs[x] = 0;
            up(x);
        }
    }

    int node(int c, int x) {
        return c * n + x;
    }

    pair<int, int> edge(int x, int y) {
        int a = min(x, y);
        int b = max(x, y);
        return {a, b};
    }

    void connect(int c, int x, int y) {
        int u = node(c, x);
        int v = node(c, y);
        nodeDegree[u]++;
        nodeDegree[v]++;
        link(u, v);
        edgeColor[edge(x, y)] = c;
    }

    void disconnect(int c, int x, int y) {
        int u = node(c, x);
        int v = node(c, y);
        nodeDegree[u]--;
        nodeDegree[v]--;
        cut(u, v);
        edgeColor.erase(edge(x, y));
    }

    void updateNode(int x, int v) {
        for (int c = 0; c < C; c++) {
            int cur = node(c, x);
            splay(cur);
            arr[cur] = v;
            up(cur);
        }
    }

    int updateEdge(int x, int y, int c) {
        auto it = edgeColor.find(edge(x, y));
        if (it == edgeColor.end()) {
            return 3;
        }
        int p = it->second;
        if (p == c) {
            return 4;
        }
        if (nodeDegree[node(c, x)] >= 2 || nodeDegree[node(c, y)] >= 2) {
            return 1;
        }
        if (findroot(node(c, x)) == findroot(node(c, y))) {
            return 2;
        }
        disconnect(p, x, y);
        connect(c, x, y);
        return 4;
    }

    int query(int c, int x, int y) {
        int u = node(c, x);
        int v = node(c, y);
        if (findroot(u) != findroot(v)) {
            return -1;
        }
        split(u, v);
        return maxv[v];
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m >> C >> q;
        for (int i = 1, v; i <= n; i++) {
            cin >> v;
            for (int c = 0; c < C; c++) {
                int cur = node(c, i);
                arr[cur] = v;
                maxv[cur] = v;
            }
        }
        for (int i = 1, x, y, c; i <= m; i++) {
            cin >> x >> y >> c;
            connect(c, x, y);
        }
        for (int i = 1, op, x, y, c; i <= q; i++) {
            cin >> op;
            if (op == 0) {
                cin >> x >> y;
                updateNode(x, y);
            } else if (op == 1) {
                cin >> x >> y >> c;
                int ans = updateEdge(x, y, c);
                if (ans == 1) {
                    cout << "Error 1." << "\n";
                }
                if (ans == 2) {
                    cout << "Error 2." << "\n";
                }
                if (ans == 3) {
                    cout << "No such edge." << "\n";
                }
                if (ans == 4) {
                    cout << "Success." << "\n";
                }
            } else {
                cin >> c >> x >> y;
                cout << query(c, x, y) << "\n";
            }
        }
        return 0;
    }

五、复杂度分析

题目4、5、6 都是普通 LCT，点数 n，每个操作均摊 O(log n)。

题目7 不同：它是 C 份 LCT 拼起来，总点数为 C·n。单层里的操作是 O(log(C·n))，但点权修改要扫过 C 层，代价 O(C·log(C·n))。C 不超过 10，log 那项也到不了 20，整体够用。

题目 / 操作时间复杂度
题目4：单次 C / U / QO(log n) 均摊
题目5：单次 bridge / penguins / excursionO(log n) 均摊
题目6：单次查询或改弹力O(log n) 均摊
题目6：建图（n 次 link）O(n log n)
题目7：查询 / 改边颜色O(log(C·n)) 均摊
题目7：改点权（扫 C 层）O(C · log(C·n)) 均摊
题目4、5：总时间O(m log n)
题目7：总时间O(q · C · log(C·n))

空间上，题目4、5、6 各是 O(n) 的数组。题目7 要开 C 份 LCT，数组大小是 C·n；度数数组和颜色映射表也是这个量级，整体 O(C·n)。

六、最后

四道题落在四个不同的维护对象上：边的开关、路径点权和、路径点数、分层最大值。

但它们改的都是同一件东西，up 里汇总什么、点节点该取什么值。模版本身一行没动。

模版题到此为止。下一节开始维护更复杂的路径信息，先乘后加的标记、颜色段的拼接、信号的逐层翻转，还是这副骨架。
