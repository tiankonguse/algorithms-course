CDQ 分治：时序上的平面问题

操作一条条发生，查询只认它之前发生过的修改。时序一旦占住分治的那一维，平面的 x 就只能靠每层排序维持，y 也就只能交给树状数组。本页两道题：摩基亚是带修改的矩形求和，天使玩偶是曼哈顿距离最近点。

零、背景

上一页的两道题所有点一开始就给定，x 可以在最开始排一次序，分治维和排序维共用同一维，y 于是能被双指针白拿。

这一页多了操作时序。事件在数组里的先后就是时间，分治只能按时间来切，否则「只统计之前的修改」这个前提就没了。x 这一维被挤出去之后，只能退到每层 merge 里：扫完按 x 排序，让上一层能双指针推进。y 剩下来就只能上树状数组。

一、题目5：摩基亚

给定数字 w，表示一个 w × w 的正方形区域，所有位置都在其中。接下来 m 条操作，两种类型：操作 1 x y v 表示坐标 (x, y) 位置增加了 v 个人；操作 2 a b c d 打印左上角 (a, b)、右下角 (c, d) 区域里的人数。1 ≤ w ≤ 2×10⁶，1 ≤ m ≤ 2×10⁵，0 ≤ v ≤ 10⁴。测试链接：洛谷 P4390。

和上一页一样，先把矩形查询拆成四个前缀事件，系数是 +1、−1、−1、+1。加人操作直接当成一个 op = 1 的事件。数组里的先后顺序就是时序，这就是大顺序。

merge 里 p2 从左往右扫右半，p1 把左半里 x 不超过 arr[p2].x 的元素推过来，遇到加人事件就 add(arr[p1].y, arr[p1].v)，遇到查询事件就 ans[q] += v × query(arr[p2].y)。扫完先把 add 过的位置逐个减回去，再把整段按 x 排序。

摩基亚：一次 merge 里的 x 双指针与 y 树状数组

左半（早，按 x）
右半（晚，按 x）

演示里 ans 拿到的是 +8 和 −5 两块，本层合计 +3。左半内部、右半内部还会各自补上自己的部分，最后按查询编号打印。

#include <bits/stdc++.h>

    using namespace std;

    struct Node {
        int op, x, y, v, q;
    };

    bool NodeCmp(Node a, Node b) {
        return a.x < b.x;
    }

    const int MAXM = 200001;
    const int MAXV = 2000002;
    const int INF = 1000000001;
    int w;

    Node arr[MAXM];
    int cnte = 0;
    int cntq = 0;

    int tree[MAXV];

    int ans[MAXM];

    int lowbit(int i) {
        return i & -i;
    }

    void add(int i, int v) {
        while (i <= w) {
            tree[i] += v;
            i += lowbit(i);
        }
    }

    int query(int i) {
        int ret = 0;
        while (i > 0) {
            ret += tree[i];
            i -= lowbit(i);
        }
        return ret;
    }

    void addPeople(int x, int y, int v) {
        arr[++cnte].op = 1;
        arr[cnte].x = x;
        arr[cnte].y = y;
        arr[cnte].v = v;
    }

    void addQuery(int x, int y, int v, int q) {
        arr[++cnte].op = 2;
        arr[cnte].x = x;
        arr[cnte].y = y;
        arr[cnte].v = v;
        arr[cnte].q = q;
    }

    void merge(int l, int m, int r) {
        int p1, p2;
        for (p1 = l - 1, p2 = m + 1; p2 <= r; p2++) {
            while (p1 + 1 <= m && arr[p1 + 1].x <= arr[p2].x) {
                p1++;
                if (arr[p1].op == 1) {
                    add(arr[p1].y, arr[p1].v);
                }
            }
            if (arr[p2].op == 2) {
                ans[arr[p2].q] += arr[p2].v * query(arr[p2].y);
            }
        }
        for (int i = l; i <= p1; i++) {
            if (arr[i].op == 1) {
                add(arr[i].y, -arr[i].v);
            }
        }
        sort(arr + l, arr + r + 1, NodeCmp);
    }

    void cdq(int l, int r) {
        if (l == r) {
            return;
        }
        int mid = (l + r) / 2;
        cdq(l, mid);
        cdq(mid + 1, r);
        merge(l, mid, r);
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        int tmp;
        cin >> tmp >> w;
        w++;
        int op, x, y, v, a, b, c, d;
        cin >> op;
        while (op != 3) {
            if (op == 1) {
                cin >> x >> y >> v;
                x++; y++;
                addPeople(x, y, v);
            } else {
                cin >> a >> b >> c >> d;
                a++; b++; c++; d++;
                addQuery(c, d, 1, ++cntq);
                addQuery(a - 1, b - 1, 1, cntq);
                addQuery(a - 1, d, -1, cntq);
                addQuery(c, b - 1, -1, cntq);
            }
            cin >> op;
        }
        cdq(1, cnte);
        for (int i = 1; i <= cntq; i++) {
            cout << ans[i] << '\n';
        }
        return 0;
    }

二、题目6：天使玩偶

规定 (x₁, y₁) 和 (x₂, y₂) 之间的距离是 |x₁ − x₂| + |y₁ − y₂|。一开始给定 n 个点的位置，接下来 m 条操作：操作 1 x y 添加一个点，操作 2 x y 打印已经添加的所有点中到 (x, y) 最近的那个点的距离。1 ≤ n、m ≤ 3×10⁵，0 ≤ x、y ≤ 10⁶。测试链接：洛谷 P4169。

先只看左下方的点，也就是 xⱼ ≤ xᵢ 且 yⱼ ≤ yᵢ 的那些。这时候绝对值可以直接拆掉，距离写成 (xᵢ + yᵢ) − (xⱼ + yⱼ)。于是问题变成：在 y 不超过 yᵢ 的点里，找 xⱼ + yⱼ 最大的那个。这正是树状数组能维护的前缀最大值。

分治的形状和上一题完全一样：时序当大顺序，merge 里按 x 双指针，y 交给树状数组。区别只有树状数组里存的东西：上一题是前缀人数和，这一题是前缀最大的 x + y，所以 add 改成取 max，query 也改成取 max，初始值不是 0 而是 −INF。

清空同样不能整体重置。代码里的做法是：add 时把经过的每个位置更新成最大值，撤销时沿同样的路径把这些位置写回 −INF。因为只走 add 过的那些位置，代价和本层 add 的次数同阶。

天使玩偶：四次变换与最近距离

演示里 v = 6（坐标最大值 5 加一）。四次变换各自命中一个方向上的点，距离依次是 4、无、2、5，取最小得 2。真实的四个点里最近的是 (5, 4)，曼哈顿距离 1 + 1 = 2，对得上。

#include <bits/stdc++.h>

    using namespace std;

    struct Node {
        int op, x, y, q;
    };

    bool NodeCmp(Node a, Node b) {
        return a.x < b.x;
    }

    const int MAXN = 300001;
    const int MAXV = 1000002;
    const int INF = 1000000001;
    int n, m, v;

    Node tim[MAXN << 1];
    Node arr[MAXN << 1];
    int cnte = 0;
    int cntq = 0;

    int tree[MAXV];

    int ans[MAXN];

    int lowbit(int i) {
        return i & -i;
    }

    void more(int i, int num) {
        while (i <= v) {
            tree[i] = max(tree[i], num);
            i += lowbit(i);
        }
    }

    int query(int i) {
        int ret = -INF;
        while (i > 0) {
            ret = max(ret, tree[i]);
            i -= lowbit(i);
        }
        return ret;
    }

    void clear(int i) {
        while (i <= v) {
            tree[i] = -INF;
            i += lowbit(i);
        }
    }

    void merge(int l, int m, int r) {
        int p1, p2;
        for (p1 = l - 1, p2 = m + 1; p2 <= r; p2++) {
            while (p1 + 1 <= m && arr[p1 + 1].x <= arr[p2].x) {
                p1++;
                if (arr[p1].op == 1) {
                    more(arr[p1].y, arr[p1].x + arr[p1].y);
                }
            }
            if (arr[p2].op == 2) {
                ans[arr[p2].q] = min(ans[arr[p2].q], arr[p2].x + arr[p2].y - query(arr[p2].y));
            }
        }
        for (int i = l; i <= p1; i++) {
            if (arr[i].op == 1) {
                clear(arr[i].y);
            }
        }
        sort(arr + l, arr + r + 1, NodeCmp);
    }

    void cdq(int l, int r) {
        if (l == r) {
            return;
        }
        int mid = (l + r) >> 1;
        cdq(l, mid);
        cdq(mid + 1, r);
        merge(l, mid, r);
    }

    void to1() {
        for (int i = 1; i <= cnte; i++) {
            arr[i] = tim[i];
        }
        cdq(1, cnte);
    }

    void to2() {
        for (int i = 1; i <= cnte; i++) {
            arr[i] = tim[i];
            arr[i].x = v - arr[i].x;
        }
        cdq(1, cnte);
    }

    void to3() {
        for (int i = 1; i <= cnte; i++) {
            arr[i] = tim[i];
            arr[i].x = v - arr[i].x;
            arr[i].y = v - arr[i].y;
        }
        cdq(1, cnte);
    }

    void to4() {
        for (int i = 1; i <= cnte; i++) {
            arr[i] = tim[i];
            arr[i].y = v - arr[i].y;
        }
        cdq(1, cnte);
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1, x, y; i <= n; i++) {
            cin >> x >> y;
            tim[++cnte].op = 1;
            tim[cnte].x = ++x;
            tim[cnte].y = ++y;
            v = max(v, max(x, y));
        }
        for (int i = 1, op, x, y; i <= m; i++) {
            cin >> op >> x >> y;
            tim[++cnte].op = op;
            tim[cnte].x = ++x;
            tim[cnte].y = ++y;
            if (op == 2) {
                tim[cnte].q = ++cntq;
            }
            v = max(v, max(x, y));
        }
        v++;
        for (int i = 1; i <= v; i++) {
            tree[i] = -INF;
        }
        for (int i = 1; i <= cntq; i++) {
            ans[i] = INF;
        }
        to1();
        to2();
        to3();
        to4();
        for (int i = 1; i <= cntq; i++) {
            cout << ans[i] << '\n';
        }
        return 0;
    }

三、复杂度分析

两题的骨架一致：分治 O(log E) 层，每层线性扫描加上树状数组的 O(log V) 调用。天使玩偶要多跑三遍，乘一个 4 的常数。

题目事件规模时间复杂度空间复杂度
摩基亚E = m + 4·查询数O(E·log E·log w)O(E + w)
天使玩偶E = n + mO(4·E·log E·log V)O(E + V)

摩基亚的 E 不超过 10⁶，log E 与 log w 都在 20 附近，总量约 4×10⁸ 次基本操作，卡着过。天使玩偶的 E 是 6×10⁵，四遍之后也是这个量级。

空间上树状数组按值域开：摩基亚开到 w + 1，天使玩偶开到 V = 最大坐标 + 2，都是 10⁶ 量级的静态数组。事件数组与临时数组各一份，递归栈 O(log E)。

四、最后

回顾这三页，CDQ 分治的分工始终没变：一维排序、一维分治、一维数据结构。变的只是哪一维腾给谁。

三维偏序里 a 能提前排好，二维数点里 x 能提前排好，到了带时序的题，时间占了分治的位置，x 就只能在每层里现排。
