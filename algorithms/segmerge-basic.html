线段树合并的五道题：子树统计、连通区第 k 小、树上差分与逆序对

上一页把合并的三条规则和 O(n log n) 的势能证明过了一遍。这一页看它能接住多少种不同的信息：子树里更大的能力值、会随加边变化的连通区第 k 小、树上差分之后的最大词频、跑步者路线拆成上行下行两套差分，以及合并时顺手数出来的交叉逆序对。五道题的 merge 函数几乎一模一样，变的只有 up 和叶子上的那一句。

零、背景

五道题都来自洛谷，链接分别是 P3605、P3224、P4556、P1600、P3521。数据规模都在 10⁵ 到 3·10⁵ 之间，值域离散化之后不超过 n，所以数组按 MAXN * 40 到 MAXN * 50 开就够。

题目2 要并查集，题目3、题目4 要树上倍增求 LCA 和树上差分。这几样都不是这一讲的重点，下面只讲它们和合并怎么配合。

一、题目1：晋升者计数（P3605）

n 个人组成公司的组织树，1 号是老板，从 2 号开始给出每个人的上司。每个人有一个能力值 arr[i]，要打印第 i 号人为头的子树里，能力值严格大于 arr[i] 的人数。n 到 10⁵，能力值到 10⁹。

能力值大到不能直接当值域，先离散化：排序去重之后二分出每个值的排名，值域就压到 1 到 cntv。然后每个点开一棵权值线段树，只插自己这个排名。后序遍历，儿子的树并进来，自己的答案就是这棵汇总树上区间 [arr[u] + 1, cntv] 的计数和。

这里有个细节：查询写的是 query(arr[u] + 1, cntv, ...)，用的是排名而不是原值。排名相等的能力值不算「更大」，离散化天然处理了这一点。另外 query 里第一句是 jobl > jobr || i == 0 就返回 0，因为最大的那个排名查出来区间是空的。

晋升者计数：后序合并与区间查询

七个点，圆里是能力值（正好是 1~7 各一个，排名就是自己）。右上是当前点那棵线段树里七个值的计数，橙框是要查的区间。点「下一步」按后序走一遍。

完整代码：

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 100001;
    const int MAXT = MAXN * 40;
    int n;

    int head[MAXN];
    int nxt[MAXN << 1];
    int to[MAXN << 1];
    int cntg;

    int arr[MAXN];
    int sorted[MAXN];
    int cntv;

    int root[MAXN];
    int ls[MAXT];
    int rs[MAXT];
    int siz[MAXT];
    int cntt;

    int ans[MAXN];

    void addEdge(int u, int v) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        head[u] = cntg;
    }

    int kth(int num) {
        int left = 1, right = cntv, ret = 0;
        while (left <= right) {
            int mid = (left + right) >> 1;
            if (sorted[mid] <= num) {
                ret = mid;
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        return ret;
    }

    void up(int i) {
        siz[i] = siz[ls[i]] + siz[rs[i]];
    }

    int add(int jobi, int l, int r, int i) {
        int rt = i;
        if (rt == 0) {
            rt = ++cntt;
        }
        if (l == r) {
            siz[rt]++;
        } else {
            int mid = (l + r) >> 1;
            if (jobi <= mid) {
                ls[rt] = add(jobi, l, mid, ls[rt]);
            } else {
                rs[rt] = add(jobi, mid + 1, r, rs[rt]);
            }
            up(rt);
        }
        return rt;
    }

    int merge(int l, int r, int t1, int t2) {
        if (t1 == 0 || t2 == 0) {
            return t1 + t2;
        }
        if (l == r) {
            siz[t1] += siz[t2];
        } else {
            int mid = (l + r) >> 1;
            ls[t1] = merge(l, mid, ls[t1], ls[t2]);
            rs[t1] = merge(mid + 1, r, rs[t1], rs[t2]);
            up(t1);
        }
        return t1;
    }

    int query(int jobl, int jobr, int l, int r, int i) {
        if (jobl > jobr || i == 0) {
            return 0;
        }
        if (jobl <= l && r <= jobr) {
            return siz[i];
        }
        int mid = (l + r) >> 1;
        int ret = 0;
        if (jobl <= mid) {
            ret += query(jobl, jobr, l, mid, ls[i]);
        }
        if (jobr > mid) {
            ret += query(jobl, jobr, mid + 1, r, rs[i]);
        }
        return ret;
    }

    void calc(int u, int fa) {
        for (int e = head[u]; e; e = nxt[e]) {
            int v = to[e];
            if (v != fa) {
                calc(v, u);
                root[u] = merge(1, cntv, root[u], root[v]);
            }
        }
        ans[u] = query(arr[u] + 1, cntv, 1, cntv, root[u]);
    }

    void compute() {
        for (int i = 1; i <= n; i++) {
            sorted[i] = arr[i];
        }
        sort(sorted + 1, sorted + n + 1);
        cntv = 1;
        for (int i = 2; i <= n; i++) {
            if (sorted[cntv] != sorted[i]) {
                sorted[++cntv] = sorted[i];
            }
        }
        for (int i = 1; i <= n; i++) {
            arr[i] = kth(arr[i]);
        }
        for (int i = 1; i <= n; i++) {
            root[i] = add(arr[i], 1, cntv, root[i]);
        }
        calc(1, 0);
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
        }
        for (int i = 2, fa; i <= n; i++) {
            cin >> fa;
            addEdge(fa, i);
            addEdge(i, fa);
        }
        compute();
        for (int i = 1; i <= n; i++) {
            cout << ans[i] << '\n';
        }
        return 0;
    }

二、题目2：永无乡（P3224）

n 个岛，每个岛分到一个 1 到 n 里互不相同的数字。初始有 m 座桥，之后 q 条操作：B x y 在 x、y 之间修一座桥；Q x k 打印 x 所在连通区里第 k 小的数字来自几号岛，不存在打印 −1。

连通区会随操作变大，所以要并查集。每个并查集的代表元额外挂一棵权值线段树，装着这个区里所有数字。合并两个区时，除了 father[xfa] = yfa，还要把两棵树 merge 起来。查询第 k 小就是在代表元那棵树上做线段树二分：左儿子的 siz 不小于 k 就往左，否则减去左儿子的 siz 往右。叶子上 pos[l] 直接给出岛的编号。

这里有个细节：Union 里写的是 root[yfa] = merge(1, n, root[yfa], root[xfa])，t1 是 yfa 的树。顺序不影响结果，但决定了哪棵树被保留。还有一点，n 个数字互不重复、正好是 1 到 n 的排列，所以值域就是 1 到 n，不用离散化。

永无乡：并查集合并连通区与第 k 小查询

五个岛，数字分别是 4、3、2、5、1。下面是岛 2 所在连通区的权值线段树，橙色是第 k 小的查找路径。点「下一步」按样例的操作顺序走。

完整代码：

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 100001;
    const int MAXT = MAXN * 40;
    int n, m, q;
    int pos[MAXN];
    int father[MAXN];

    int root[MAXN];
    int ls[MAXT];
    int rs[MAXT];
    int siz[MAXT];
    int cntt;

    void up(int i) {
        siz[i] = siz[ls[i]] + siz[rs[i]];
    }

    int add(int jobi, int l, int r, int i) {
        int rt = i;
        if (rt == 0) {
            rt = ++cntt;
        }
        if (l == r) {
            siz[rt]++;
        } else {
            int mid = (l + r) >> 1;
            if (jobi <= mid) {
                ls[rt] = add(jobi, l, mid, ls[rt]);
            } else {
                rs[rt] = add(jobi, mid + 1, r, rs[rt]);
            }
            up(rt);
        }
        return rt;
    }

    int merge(int l, int r, int t1, int t2) {
        if (t1 == 0 || t2 == 0) {
            return t1 + t2;
        }
        if (l == r) {
            siz[t1] += siz[t2];
        } else {
            int mid = (l + r) >> 1;
            ls[t1] = merge(l, mid, ls[t1], ls[t2]);
            rs[t1] = merge(mid + 1, r, rs[t1], rs[t2]);
            up(t1);
        }
        return t1;
    }

    int query(int jobk, int l, int r, int i) {
        if (i == 0 || jobk > siz[i]) {
            return -1;
        }
        if (l == r) {
            return pos[l];
        }
        int mid = (l + r) >> 1;
        if (siz[ls[i]] >= jobk) {
            return query(jobk, l, mid, ls[i]);
        } else {
            return query(jobk - siz[ls[i]], mid + 1, r, rs[i]);
        }
    }

    int find(int i) {
        if (i != father[i]) {
            father[i] = find(father[i]);
        }
        return father[i];
    }

    void Union(int x, int y) {
        int xfa = find(x);
        int yfa = find(y);
        if (xfa != yfa) {
            father[xfa] = yfa;
            root[yfa] = merge(1, n, root[yfa], root[xfa]);
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1, num; i <= n; i++) {
            cin >> num;
            pos[num] = i;
            father[i] = i;
            root[i] = add(num, 1, n, root[i]);
        }
        for (int i = 1, x, y; i <= m; i++) {
            cin >> x >> y;
            Union(x, y);
        }
        cin >> q;
        char op;
        int x, y, k;
        for (int i = 1; i <= q; i++) {
            cin >> op;
            if (op == 'B') {
                cin >> x >> y;
                Union(x, y);
            } else {
                cin >> x >> k;
                cout << query(k, 1, n, root[find(x)]) << '\n';
            }
        }
        return 0;
    }

三、题目3：雨天的尾巴（P4556）

n 个点的树，m 条路径 x y v，表示 x 到 y 路径上每个点都收到一个数字 v。对每个点打印它收到的次数最多的数字，并列时打印值最小的；一次都没收到就打印 0。

路径加一用树上差分：设 c = lca(x, y)，在 root[x] 和 root[y] 上给 v 加 1，在 root[c] 和 root[fa(c)] 上给 v 减 1。然后从叶向根合并，一个点合并完所有儿子之后，它那棵树里 v 的计数正好是「经过这个点的路径条数」。全部合并完查一次最大词频即可。

这棵树维护的信息从「计数和」换成了「最大词频」，所以 up 改成取 max，叶子上改成 maxCnt[rt] += jobv（jobv 可以是 −1）。查询并列时取小的那个，靠的是 maxCnt[ls] >= maxCnt[rs] 就往左走，相等时走左边，左边的值更小。

// 维护「最大词频」的合并与查询：up 取 max，相同时取值更小的
    void up(int i) {
        maxCnt[i] = max(maxCnt[ls[i]], maxCnt[rs[i]]);
    }

    int merge(int l, int r, int t1, int t2) {
        if (t1 == 0 || t2 == 0) {
            return t1 + t2;
        }
        if (l == r) {
            maxCnt[t1] += maxCnt[t2];
        } else {
            int mid = (l + r) >> 1;
            ls[t1] = merge(l, mid, ls[t1], ls[t2]);
            rs[t1] = merge(mid + 1, r, rs[t1], rs[t2]);
            up(t1);
        }
        return t1;
    }

    int query(int l, int r, int i) {
        if (l == r) {
            return l;
        }
        int mid = (l + r) >> 1;
        if (maxCnt[ls[i]] >= maxCnt[rs[i]]) {
            return query(l, mid, ls[i]);
        } else {
            return query(mid + 1, r, rs[i]);
        }
    }

还有个细节：查询前先判 maxCnt[root[u]] == 0，是 0 就输出 0，否则 query 会一路走到底返回一个没收到过的值。完整代码：

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 100001;
    const int MAXV = 100000;
    const int MAXT = MAXN * 50;
    const int MAXP = 20;
    int n, m;

    int head[MAXN];
    int nxt[MAXN << 1];
    int to[MAXN << 1];
    int cntg;

    int dep[MAXN];
    int stjump[MAXN][MAXP];

    int root[MAXN];
    int ls[MAXT];
    int rs[MAXT];
    int maxCnt[MAXT];
    int cntt;

    int ans[MAXN];

    void addEdge(int u, int v) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        head[u] = cntg;
    }

    void dfs(int u, int fa) {
        dep[u] = dep[fa] + 1;
        stjump[u][0] = fa;
        for (int p = 1; p < MAXP; p++) {
            stjump[u][p] = stjump[stjump[u][p - 1]][p - 1];
        }
        for (int e = head[u]; e > 0; e = nxt[e]) {
            int v = to[e];
            if (v != fa) {
                dfs(v, u);
            }
        }
    }

    int getLca(int a, int b) {
        if (dep[a] < dep[b]) {
            swap(a, b);
        }
        for (int p = MAXP - 1; p >= 0; p--) {
            if (dep[stjump[a][p]] >= dep[b]) {
                a = stjump[a][p];
            }
        }
        if (a == b) {
            return a;
        }
        for (int p = MAXP - 1; p >= 0; p--) {
            if (stjump[a][p] != stjump[b][p]) {
                a = stjump[a][p];
                b = stjump[b][p];
            }
        }
        return stjump[a][0];
    }

    void up(int i) {
        maxCnt[i] = max(maxCnt[ls[i]], maxCnt[rs[i]]);
    }

    int add(int jobi, int jobv, int l, int r, int i) {
        int rt = i;
        if (rt == 0) {
            rt = ++cntt;
        }
        if (l == r) {
            maxCnt[rt] += jobv;
        } else {
            int mid = (l + r) >> 1;
            if (jobi <= mid) {
                ls[rt] = add(jobi, jobv, l, mid, ls[rt]);
            } else {
                rs[rt] = add(jobi, jobv, mid + 1, r, rs[rt]);
            }
            up(rt);
        }
        return rt;
    }

    int merge(int l, int r, int t1, int t2) {
        if (t1 == 0 || t2 == 0) {
            return t1 + t2;
        }
        if (l == r) {
            maxCnt[t1] += maxCnt[t2];
        } else {
            int mid = (l + r) >> 1;
            ls[t1] = merge(l, mid, ls[t1], ls[t2]);
            rs[t1] = merge(mid + 1, r, rs[t1], rs[t2]);
            up(t1);
        }
        return t1;
    }

    int query(int l, int r, int i) {
        if (l == r) {
            return l;
        }
        int mid = (l + r) >> 1;
        if (maxCnt[ls[i]] >= maxCnt[rs[i]]) {
            return query(l, mid, ls[i]);
        } else {
            return query(mid + 1, r, rs[i]);
        }
    }

    void calc(int u, int fa) {
        for (int e = head[u]; e > 0; e = nxt[e]) {
            int v = to[e];
            if (v != fa) {
                calc(v, u);
                root[u] = merge(1, MAXV, root[u], root[v]);
            }
        }
        if (maxCnt[root[u]] == 0) {
            ans[u] = 0;
        } else {
            ans[u] = query(1, MAXV, root[u]);
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1, u, v; i < n; i++) {
            cin >> u >> v;
            addEdge(u, v);
            addEdge(v, u);
        }
        dfs(1, 0);
        for (int i = 1, x, y, v; i <= m; i++) {
            cin >> x >> y >> v;
            int lca = getLca(x, y);
            int lcafa = stjump[lca][0];
            root[x] = add(v, 1, 1, MAXV, root[x]);
            root[y] = add(v, 1, 1, MAXV, root[y]);
            root[lca] = add(v, -1, 1, MAXV, root[lca]);
            root[lcafa] = add(v, -1, 1, MAXV, root[lcafa]);
        }
        calc(1, 0);
        for (int i = 1; i <= n; i++) {
            cout << ans[i] << '\n';
        }
        return 0;
    }

拿一个五点的小例子验证：树是 1-2、1-3、2-4、2-5，三条路径分别是 (4,5,3)、(4,3,3)、(1,5,2)。跑出来是 2 3 3 3 2。1 号和 5 号都收到了 2 和 3 各一次，并列取小，所以是 2。

四、题目4：天天爱跑步（P1600）

n 个点的树，每个点 i 有一个观察员，观测时刻 w[i]。m 个跑步者从 x 跑到 y，出发时刻 0，每过一条边耗时 1 秒。跑步者到达 i 的时刻正好等于 w[i] 时，才会被 i 的观察员看到。打印每个观察员看到几个人。n、m 到 3·10⁵。

一个跑步者的路线 x → y 以 c = lca(x, y) 为界拆成两段。上行段 x → c 上的点 u 满足 dep[x] − dep[u] = w[u]，也就是 dep[u] + w[u] = dep[x]；下行段 c → y 上的点 u 满足 dep[x] + dep[u] − 2·dep[c] = w[u]，也就是 dep[u] − w[u] = 2·dep[c] − dep[x]。两段各做一次树上差分、各用一棵线段树，这就是题目里说的「两种线段树的合并」。

最后 ans[u] = query(dep[u] + w[u], 1, n, rtUp[u]) + query(dep[u] - w[u], -n, n, rtDown[u])。rtDown 的值域是 −n 到 n，因为 dep[u] − w[u] 可以取到负数。样例里五个点、三条路线（3→1、1→4、5→5），观测时刻 0 1 0 3 0，答案是 1 2 1 0 1。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 300001;
    const int MAXT = MAXN * 50;
    const int MAXP = 20;
    int n, m;
    int w[MAXN];

    int head[MAXN];
    int nxt[MAXN << 1];
    int to[MAXN << 1];
    int cntg;

    int dep[MAXN];
    int stjump[MAXN][MAXP];
    int rtUp[MAXN];
    int rtDown[MAXN];
    int ls[MAXT];
    int rs[MAXT];
    int sum[MAXT];
    int cntt;

    int ans[MAXN];

    void addEdge(int u, int v) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        head[u] = cntg;
    }

    void dfs(int u, int fa) {
        dep[u] = dep[fa] + 1;
        stjump[u][0] = fa;
        for (int p = 1; p < MAXP; p++) {
            stjump[u][p] = stjump[stjump[u][p - 1]][p - 1];
        }
        for (int e = head[u]; e; e = nxt[e]) {
            int v = to[e];
            if (v != fa) {
                dfs(v, u);
            }
        }
    }

    int getLca(int a, int b) {
        if (dep[a] < dep[b]) {
            swap(a, b);
        }
        for (int p = MAXP - 1; p >= 0; p--) {
            if (dep[stjump[a][p]] >= dep[b]) {
                a = stjump[a][p];
            }
        }
        if (a == b) {
            return a;
        }
        for (int p = MAXP - 1; p >= 0; p--) {
            if (stjump[a][p] != stjump[b][p]) {
                a = stjump[a][p];
                b = stjump[b][p];
            }
        }
        return stjump[a][0];
    }

    void up(int i) {
        sum[i] = sum[ls[i]] + sum[rs[i]];
    }

    int add(int jobi, int jobv, int l, int r, int i) {
        int rt = i;
        if (rt == 0) {
            rt = ++cntt;
        }
        if (l == r) {
            sum[rt] += jobv;
        } else {
            int mid = (l + r) >> 1;
            if (jobi <= mid) {
                ls[rt] = add(jobi, jobv, l, mid, ls[rt]);
            } else {
                rs[rt] = add(jobi, jobv, mid + 1, r, rs[rt]);
            }
            up(rt);
        }
        return rt;
    }

    int merge(int l, int r, int t1, int t2) {
        if (t1 == 0 || t2 == 0) {
            return t1 + t2;
        }
        if (l == r) {
            sum[t1] += sum[t2];
        } else {
            int mid = (l + r) >> 1;
            ls[t1] = merge(l, mid, ls[t1], ls[t2]);
            rs[t1] = merge(mid + 1, r, rs[t1], rs[t2]);
            up(t1);
        }
        return t1;
    }

    int query(int jobi, int l, int r, int i) {
        if (jobi < l || jobi > r || i == 0) {
            return 0;
        }
        if (l == r) {
            return sum[i];
        }
        int mid = (l + r) >> 1;
        if (jobi <= mid) {
            return query(jobi, l, mid, ls[i]);
        } else {
            return query(jobi, mid + 1, r, rs[i]);
        }
    }

    void calc(int u, int fa) {
        for (int e = head[u]; e; e = nxt[e]) {
            int v = to[e];
            if (v != fa) {
                calc(v, u);
                rtUp[u] = merge(1, n, rtUp[u], rtUp[v]);
                rtDown[u] = merge(-n, n, rtDown[u], rtDown[v]);
            }
        }
        ans[u] = query(dep[u] + w[u], 1, n, rtUp[u]) + query(dep[u] - w[u], -n, n, rtDown[u]);
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1, u, v; i < n; i++) {
            cin >> u >> v;
            addEdge(u, v);
            addEdge(v, u);
        }
        for (int i = 1; i <= n; i++) {
            cin >> w[i];
        }
        dfs(1, 0);
        for (int i = 1, x, y; i <= m; i++) {
            cin >> x >> y;
            int lca = getLca(x, y);
            int lcafa = stjump[lca][0];
            rtUp[x] = add(dep[x], 1, 1, n, rtUp[x]);
            rtUp[lcafa] = add(dep[x], -1, 1, n, rtUp[lcafa]);
            rtDown[y] = add(2 * dep[lca] - dep[x], 1, -n, n, rtDown[y]);
            rtDown[lca] = add(2 * dep[lca] - dep[x], -1, -n, n, rtDown[lca]);
        }
        calc(1, 0);
        for (int i = 1; i <= n; i++) {
            cout << ans[i] << ' ';
        }
        return 0;
    }

五、题目5：最小化逆序对（P3521）

输入流给出一棵二叉树：读到非 0 的数表示叶子、权值是这个数，读到 0 表示非叶节点，先递归建左子树再递归建右子树。可以任选一些节点交换它的左右子树。目标是让先序遍历得到的叶子权值序列的逆序对数量尽量小，打印这个最小值。n 到 2·10⁵。

在某个节点上，左子树的所有叶子一定排在右子树的所有叶子之前，交换只改变这两块的先后顺序，块内怎么排跟这里无关。所以每个节点独立决策：块内的最小逆序对在左右子树里已经算好了，这里只要算「跨两块」的新增量，两种摆法取小的那个。

关键是怎么在合并的时候把这两个数算出来。合并的左树 t1、右树 t2 都是权值线段树，走到区间 [l, r] 时，t1 右半部分（值在 mid 以上）里的任何值和 t2 左半部分（值在 mid 及以下）里的任何值，前者一定大于后者，而且前者排在前面 ， 这就是 siz[rs[t1]] × siz[ls[t2]] 个逆序对。反过来 siz[rs[t2]] × siz[ls[t1]] 是交换后的。逐层累加，一次合并就把两个方向的增量都拿到了。

两个计数器在每次 dfs 进入非叶节点时清零，merge 完之后立刻取 min。注意清零必须放在两次递归之后、merge 之前，否则子树的贡献会被算进来。

// 合并时顺便统计「不交换」与「交换」两种摆放各自新增的交叉逆序对
    long long no, yes;

    int merge(int l, int r, int t1, int t2) {
        if (t1 == 0 || t2 == 0) {
            return t1 + t2;
        }
        if (l == r) {
            siz[t1] += siz[t2];
        } else {
            no += 1LL * siz[rs[t1]] * siz[ls[t2]];
            yes += 1LL * siz[rs[t2]] * siz[ls[t1]];
            int mid = (l + r) >> 1;
            ls[t1] = merge(l, mid, ls[t1], ls[t2]);
            rs[t1] = merge(mid + 1, r, rs[t1], rs[t2]);
            up(t1);
        }
        return t1;
    }

    int dfs() {
        int rt;
        int val;
        cin >> val;
        if (val == 0) {
            int left = dfs();
            int right = dfs();
            no = yes = 0;
            rt = merge(1, n, left, right);
            ans += min(no, yes);
        } else {
            rt = build(val, 1, n);
        }
        return rt;
    }

用 n = 4、树形 ((3,1),(4,2)) 试一下：左下角那个节点 no = 1、yes = 0，取 0，块变成 {1,3}；右下角 no = 1、yes = 0，取 0，块变成 {2,4}；根上 no = 1、yes = 3，取 1。答案是 1，对应的叶序 1 3 2 4，逆序对正好是 (3, 2) 这一个。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXT = 5000001;
    int n;

    int ls[MAXT];
    int rs[MAXT];
    int siz[MAXT];
    int cntt;

    long long ans;

    void up(int i) {
        siz[i] = siz[ls[i]] + siz[rs[i]];
    }

    int build(int jobi, int l, int r) {
        int rt = ++cntt;
        if (l == r) {
            siz[rt]++;
        } else {
            int mid = (l + r) >> 1;
            if (jobi <= mid) {
                ls[rt] = build(jobi, l, mid);
            } else {
                rs[rt] = build(jobi, mid + 1, r);
            }
            up(rt);
        }
        return rt;
    }

    long long no, yes;

    int merge(int l, int r, int t1, int t2) {
        if (t1 == 0 || t2 == 0) {
            return t1 + t2;
        }
        if (l == r) {
            siz[t1] += siz[t2];
        } else {
            no += 1LL * siz[rs[t1]] * siz[ls[t2]];
            yes += 1LL * siz[rs[t2]] * siz[ls[t1]];
            int mid = (l + r) >> 1;
            ls[t1] = merge(l, mid, ls[t1], ls[t2]);
            rs[t1] = merge(mid + 1, r, rs[t1], rs[t2]);
            up(t1);
        }
        return t1;
    }

    int dfs() {
        int rt;
        int val;
        cin >> val;
        if (val == 0) {
            int left = dfs();
            int right = dfs();
            no = yes = 0;
            rt = merge(1, n, left, right);
            ans += min(no, yes);
        } else {
            rt = build(val, 1, n);
        }
        return rt;
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n;
        dfs();
        cout << ans << '\n';
        return 0;
    }

六、复杂度分析

五道题的 merge 都是同一个结构，差别只在每棵树上维护的信息和查询方式。V 是离散化后的值域大小，除题目4 的 rtDown 是 2n + 1 之外，其余都不超过 n。

操作时间复杂度空间复杂度
题目1：建树 + 全部合并 + 每个点一次区间查询O(n log V)O(n log V)，MAXN × 40
题目2：建 n 棵单点树O(n log n)O(n log n)，MAXN × 40
题目2：一次 B 操作（并查集 + 合并）O(log n) 摊还不新建节点
题目2：一次 Q 操作（第 k 小）O(log n)—
题目3：m 条路径各四个单点修改O(m log V)O((n + m) log V)，MAXN × 50
题目3：全部合并 + 每个点一次最大词频查询O((n + m) log V)—
题目4：m 条路线各四个单点修改 + 全部合并O((n + m) log n)O((n + m) log n)，MAXN × 50
题目5：建 n 棵叶子树 + 全部合并O(n log n)O(n log n)，MAXT 取 5·10⁶

题目5 的 MAXT 取 5000001 而不是 MAXN × 40，是因为它每片叶子都建了一棵完整的根到叶路径（build 不复用任何已有节点），n = 2·10⁵ 时约为 n × log₂n ≈ 3.6·10⁶ 个节点，留了余量。题目3、题目4 的 MAXN × 50 同理，是给「m 条路径 × 4 次单点修改 × 每条路径 log V 个节点」留的空间。

反过来，题目1、题目2、题目3、题目4 的查询都是 O(log V) 一次，n 个点合起来 O(n log V)，不会成为瓶颈。整个流程里唯一需要注意规模的就是节点总数，而它只减不增。

七、最后

五道题看下来，merge 那十几行一次都没改过：题目1 维护计数和，题目3 换成最大值，题目5 在合并的同一层顺手做两次乘法。真正需要想清楚的只有两件事 ， 每个点上该维护什么信息，以及这个信息在叶子上、在 up 里分别怎么合成。
