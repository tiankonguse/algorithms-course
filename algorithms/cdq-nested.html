CDQ 套 CDQ 与四维偏序

三维偏序的套路是「排序一维、双指针一维、树状数组一维」。多出来的那一维没有别的办法，只能在分治里面再分一次。这一节两道题都是四维偏序加 dp，后一道还要数方案。

零、背景

三维偏序的标准做法是：第一维排序定大顺序，分治保证第二维的偏序，双指针扫第三维，树状数组统计。三个维度各有各的活。

四维偏序多了一维，双指针已经用在第三维上了，没有第二个双指针可用。所以第二维也不交给双指针，改成再套一层分治。

也就是说，外层分治管第一维，内层分治管第二维，双指针管第三维，树状数组管第四维。每多套一层，复杂度就多乘一次 O(log n)。

一、问题：德丽莎世界第一可爱

一共有 n 个怪兽，每个怪兽有 a、b、c、d 四个能力值，以及打败之后的收益 v。

可以选择任意顺序打怪兽，每次打的怪兽的四种能力值都不能小于上次打的怪兽。打印能获得的最大收益，可能所有怪兽收益都是负数，那也需要至少打一只怪兽。

1 ≤ n ≤ 5 * 10⁴，-10⁵ ≤ a、b、c、d ≤ +10⁵，-10⁹ ≤ v ≤ +10⁹。

测试链接：洛谷 P5621。

二、朴素做法

先按 a、b、c、d 的字典序排序，排序之后能转移到 i 的只可能在它前面。

for (int i = 1; i <= n; i++) {
    dp[i] = v[i];
    for (int j = 1; j < i; j++) {
        if (b[j] <= b[i] && c[j] <= c[i] && d[j] <= d[i]) dp[i] = max(dp[i], dp[j] + v[i]);
    }
}

O(n²)，n = 5 * 10⁴ 时是 2.5 * 10⁹ 次判断，不行。三个维度的比较一次都省不掉。

三、外层与内层各管一维

对象先按 (a, b, c, d) 排序，第一维的偏序由排序解决。

外层 cdq1 按下标分治，和前面优化 dp 时的顺序一样：先递归左半，再用左半整批更新右半，最后递归右半。

更新的做法是这样。把区间里的对象复制一份，给左半的对象打上 left 标记，然后把整段按 b 排序，交给内层 cdq2。

内层 cdq2 在按 b 排好序的数组上再做一次分治。它的 merge 里，左半是 b 较小的一批、右半是 b 较大的一批，于是跨中点的转移天然满足 b 的偏序。左右两半各自按 c 排序，双指针把 c 小的那一侧加进树状数组，按 d 组织，右半的查询就是一次前缀最大值。

这里有个细节：加进树状数组的必须是打过 left 标记的对象，被更新的必须是没有标记的。内层分治只是用来枚举满足 b 偏序的那些对，左组的对象完全可能落在内层的右半里，所以要靠标记区分，不能靠位置。

还有一个细节在排序上。b 相同的两个对象，一个属于左组、一个属于右组，它们之间的转移是合法的。所以按 b 排序时 b 相同要按原下标排，左组的下标都小于右组，这样左组的那个一定排在前面，不会被漏掉。

树状数组的初值是负无穷，查不到前驱时 dp 保持不变。清空的办法还是把加过的格子逐个还原。

四个能力值完全相同的一组怪兽要做合并。它们两两之间都能互相转移，排序后是相邻的，扫一遍合并成一个对象就行。收益为正的都该打，全部累加；收益为负的不打，累加进去反而拉低答案，所以代码里只在 v > 0 时才累加。

四、代码：德丽莎世界第一可爱

    
#include <bits/stdc++.h>

    using namespace std;

    struct Node {
        int a, b, c, d;
        long long v;
        int i;
        bool left;
    };

    bool Cmp1(Node x, Node y) {
        if (x.a != y.a) return x.a < y.a;
        if (x.b != y.b) return x.b < y.b;
        if (x.c != y.c) return x.c < y.c;
        if (x.d != y.d) return x.d < y.d;
        return x.v > y.v;
    }

    bool Cmp2(Node x, Node y) {
        if (x.b != y.b) return x.b < y.b;
        return x.i < y.i;
    }

    bool Cmp3(Node x, Node y) {
        if (x.c != y.c) return x.c < y.c;
        return x.i < y.i;
    }

    const int MAXN = 50001;
    const long long INF = 1e18 + 1;
    int n, s;

    Node arr[MAXN];
    int sortd[MAXN];

    Node tmp1[MAXN];
    Node tmp2[MAXN];
    long long tree[MAXN];
    long long dp[MAXN];

    int lowbit(int i) {
        return i & -i;
    }

    void more(int i, long long num) {
        while (i <= s) {
            tree[i] = max(tree[i], num);
            i += lowbit(i);
        }
    }

    long long query(int i) {
        long long ret = -INF;
        while (i > 0) {
            ret = max(ret, tree[i]);
            i -= lowbit(i);
        }
        return ret;
    }

    void clear(int i) {
        while (i <= s) {
            tree[i] = -INF;
            i += lowbit(i);
        }
    }

    void merge(int l, int m, int r) {
        for (int i = l; i <= r; i++) {
            tmp2[i] = tmp1[i];
        }
        sort(tmp2 + l, tmp2 + m + 1, Cmp3);
        sort(tmp2 + m + 1, tmp2 + r + 1, Cmp3);
        int p1, p2;
        for (p1 = l - 1, p2 = m + 1; p2 <= r; p2++) {
            while (p1 + 1 <= m && tmp2[p1 + 1].c <= tmp2[p2].c) {
                p1++;
                if (tmp2[p1].left) {
                    more(tmp2[p1].d, dp[tmp2[p1].i]);
                }
            }
            if (!tmp2[p2].left) {
                dp[tmp2[p2].i] = max(dp[tmp2[p2].i], query(tmp2[p2].d) + tmp2[p2].v);
            }
        }
        for (int i = l; i <= p1; i++) {
            if (tmp2[i].left) {
                clear(tmp2[i].d);
            }
        }
    }

    void cdq2(int l, int r) {
        if (l == r) return;
        int mid = (l + r) / 2;
        cdq2(l, mid);
        merge(l, mid, r);
        cdq2(mid + 1, r);
    }

    void cdq1(int l, int r) {
        if (l == r) return;
        int mid = (l + r) / 2;
        cdq1(l, mid);
        for (int i = l; i <= r; i++) {
            tmp1[i] = arr[i];
            tmp1[i].left = i <= mid;
        } 
        sort(tmp1 + l, tmp1 + r + 1, Cmp2);
        cdq2(l, r);
        cdq1(mid + 1, r);
    }

    int lower(long long num) {
        int l = 1, r = s, ans = 1;
        while (l <= r) {
            int m = (l + r) / 2;
            if (sortd[m] >= num) {
                ans = m;
                r = m - 1;
            } else {
                l = m + 1;
            }
        }
        return ans;
    }

    void prepare() {
        for (int i = 1; i <= n; i++) {
            sortd[i] = arr[i].d;
        }
        sort(sortd + 1, sortd + n + 1);
        s = 1;
        for (int i = 2; i <= n; i++) {
            if (sortd[s] != sortd[i]) {
                sortd[++s] = sortd[i];
            }
        }
        for (int i = 1; i <= n; i++) {
            arr[i].d = lower(arr[i].d);
        }
        sort(arr + 1, arr + n + 1, Cmp1);
        int m = 1;
        for (int i = 2; i <= n; i++) {
            if (arr[m].a == arr[i].a && arr[m].b == arr[i].b &&
                arr[m].c == arr[i].c && arr[m].d == arr[i].d) {
                if (arr[i].v > 0) {
                    arr[m].v += arr[i].v;
                }
            } else {
                arr[++m] = arr[i];
            }
        }
        n = m;
        for (int i = 1; i <= n; i++) {
            arr[i].i = i;
            dp[i] = arr[i].v;
        }
        for (int i = 1; i <= s; i++) {
            tree[i] = -INF;
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i].a >> arr[i].b >> arr[i].c >> arr[i].d >> arr[i].v;
        }
        prepare();
        cdq1(1, n);
        long long ans = -INF;
        for (int i = 1; i <= n; i++) {
            ans = max(ans, dp[i]);
        }
        cout << ans << '\n';
        return 0;
    }

五、问题：寻找宝藏

一共有 n 个宝藏，每个宝藏有 a、b、c、d 四个属性值，以及拿取之后的收益 v。

可以选择任意顺序拿取宝藏，每次拿的宝藏的四种属性值都不能小于上次拿的宝藏。打印能获得的最大收益，打印有多少种最佳拿取方法，方法数对 998244353 取余。

1 ≤ n ≤ 8 * 10⁴，1 ≤ a、b、c、d、v ≤ 10⁹。

测试链接：洛谷 P4849。

六、把方案数一起塞进树状数组

骨架和上一题完全一样，只是转移时不再只比收益，还要比方案数。

树状数组每个位置存两个值：当前最大的收益，以及达到这个收益的方案数。更新时遇到更大的收益就替换并把方案数重置，遇到相等的收益就把方案数累加。

dp 数组同样配一个 cnt 数组，转移的三种情况各自处理：更大就替换、相等就累加、更小就不管。

四个属性值完全相同的宝藏也要合并，这一题的收益都是正的，直接全部累加。

最后扫一遍 dp，取最大值，把所有达到这个最大值的方案数加起来取模，就是答案。

七、代码：寻找宝藏

    
#include <bits/stdc++.h>

    using namespace std;

    struct Node {
        int a, b, c, d;
        long long v;
        int i;
        bool left;
    };

    bool Cmp1(Node x, Node y) {
        if (x.a != y.a) return x.a < y.a;
        if (x.b != y.b) return x.b < y.b;
        if (x.c != y.c) return x.c < y.c;
        return x.d < y.d;
    }

    bool Cmp2(Node x, Node y) {
        if (x.b != y.b) return x.b < y.b;
        return x.i < y.i;
    }

    bool Cmp3(Node x, Node y) {
        if (x.c != y.c) return x.c < y.c;
        return x.i < y.i;
    }

    const int MAXN = 80001;
    const long long INF = 1e18 + 1;
    const int MOD = 998244353;
    int n, s;

    Node arr[MAXN];
    Node tmp1[MAXN];
    Node tmp2[MAXN];
    int sortd[MAXN];

    long long treeVal[MAXN];
    int treeCnt[MAXN];

    long long dp[MAXN];
    int cnt[MAXN];

    int lowbit(int i) {
        return i & -i;
    }

    void more(int i, long long v, int c) {
        while (i <= s) {
            if (v > treeVal[i]) {
                treeVal[i] = v;
                treeCnt[i] = c % MOD;
            } else if (v == treeVal[i]) {
                treeCnt[i] = (treeCnt[i] + c) % MOD;
            }
            i += lowbit(i);
        }
    }

    long long queryVal;
    int queryCnt;

    void query(int i) {
        queryVal = -INF;
        queryCnt = 0;
        while (i > 0) {
            if (treeVal[i] > queryVal) {
                queryVal = treeVal[i];
                queryCnt = treeCnt[i];
            } else if (treeVal[i] == queryVal) {
                queryCnt = (queryCnt + treeCnt[i]) % MOD;
            }
            i -= lowbit(i);
        }
    }

    void clear(int i) {
        while (i <= s) {
            treeVal[i] = -INF;
            treeCnt[i] = 0;
            i += lowbit(i);
        }
    }

    void merge(int l, int m, int r) {
        for (int i = l; i <= r; i++) {
            tmp2[i] = tmp1[i];
        }
        sort(tmp2 + l, tmp2 + m + 1, Cmp3);
        sort(tmp2 + m + 1, tmp2 + r + 1, Cmp3);
        int p1, p2, id;
        for (p1 = l - 1, p2 = m + 1; p2 <= r; p2++) {
            while (p1 + 1 <= m && tmp2[p1 + 1].c <= tmp2[p2].c) {
                p1++;
                if (tmp2[p1].left) {
                    more(tmp2[p1].d, dp[tmp2[p1].i], cnt[tmp2[p1].i]);
                }
            }
            if (!tmp2[p2].left) {
                query(tmp2[p2].d);
                id = tmp2[p2].i;
                if (queryVal + tmp2[p2].v > dp[id]) {
                    dp[id] = queryVal + tmp2[p2].v;
                    cnt[id] = queryCnt;
                } else if (queryVal + tmp2[p2].v == dp[id]) {
                    cnt[id] = (cnt[id] + queryCnt) % MOD;
                }
            }
        }
        for (int i = l; i <= p1; i++) {
            if (tmp2[i].left) {
                clear(tmp2[i].d);
            }
        }
    }

    void cdq2(int l, int r) {
        if (l == r) return;
        int mid = (l + r) / 2;
        cdq2(l, mid);
        merge(l, mid, r);
        cdq2(mid + 1, r);
    }

    void cdq1(int l, int r) {
        if (l == r) return;
        int mid = (l + r) / 2;
        cdq1(l, mid);
        for (int i = l; i <= r; i++) {
            tmp1[i] = arr[i];
            tmp1[i].left = i <= mid;
        }
        sort(tmp1 + l, tmp1 + r + 1, Cmp2);
        cdq2(l, r);
        cdq1(mid + 1, r);
    }

    int lower(int num) {
        int l = 1, r = s, ans = 1;
        while (l <= r) {
            int mid = (l + r) / 2;
            if (sortd[mid] >= num) {
                ans = mid;
                r = mid - 1;
            } else {
                l = mid + 1;
            }
        }
        return ans;
    }

    void prepare() {
        for (int i = 1; i <= n; i++) {
            sortd[i] = arr[i].d;
        }
        sort(sortd + 1, sortd + n + 1);
        s = 1;
        for (int i = 2; i <= n; i++) {
            if (sortd[s] != sortd[i]) {
                sortd[++s] = sortd[i];
            }
        }
        for (int i = 1; i <= n; i++) {
            arr[i].d = lower(arr[i].d);
        }
        sort(arr + 1, arr + n + 1, Cmp1);
        int m = 1;
        for (int i = 2; i <= n; i++) {
            if (arr[m].a == arr[i].a && arr[m].b == arr[i].b &&
                arr[m].c == arr[i].c && arr[m].d == arr[i].d) {
                arr[m].v += arr[i].v;
            } else {
                arr[++m] = arr[i];
            }
        }
        n = m;
        for (int i = 1; i <= n; i++) {
            arr[i].i = i;
            dp[i] = arr[i].v;
            cnt[i] = 1;
        }
        for (int i = 1; i <= s; i++) {
            treeVal[i] = -INF;
            treeCnt[i] = 0;
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        int m;
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i].a >> arr[i].b >> arr[i].c >> arr[i].d >> arr[i].v;
        }
        prepare();
        cdq1(1, n);
        long long best = 0;
        int ways = 0;
        for (int i = 1; i <= n; i++) {
            best = max(best, dp[i]);
        }
        for (int i = 1; i <= n; i++) {
            if (dp[i] == best) {
                ways = (ways + cnt[i]) % MOD;
            }
        }
        cout << best << '\n';
        cout << ways << '\n';
        return 0;
    }

样例：4 个宝藏 (a,b,c,d,v) = (1,1,1,1,5) (1,2,2,2,3) (2,1,2,2,3) (2,3,3,3,6)，答案是收益 14、方案 2。

八、复杂度分析

设 n 为对象个数。

操作时间复杂度
排序与合并相同对象O(n log n)
内层 cdq2 的一次 mergeO(len log len)（排序加双指针与树状数组）
内层 cdq2 整体O(n log² n)
外层 cdq1 整体（题 5、题 6）O(n log³ n)

外层有 log n 层，每层的代价是一次完整的 cdq2，也就是 O(n log²n)，乘起来是 O(n log³n)。三维偏序是 O(n log²n)，每多套一层分治就多一个 log n。

n 取 5 * 10⁴ 时这个量级已经不小，常数必须压住：代码里用的是数组而不是 vector，排序也只对必要的区间做。用 vector 和整段排序的写法，同样会超时。

空间上多了两份辅助数组 tmp1、tmp2 和树状数组，都是 O(n)。

九、最后

CDQ 分治这条线到这里：先统计点对，再优化 dp，最后套一层处理四维偏序。骨架始终没变，变的是每一维交给谁。

再往上加维度，套三层分治理论上也行，但 O(n log⁴n) 基本上就没法用了，那时候该换别的做法。
