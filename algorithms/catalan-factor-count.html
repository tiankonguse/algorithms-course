卡特兰数：因子计数法

前面几页遇到合数模数只能退 O(n²) 的公式4。n 到 10⁶ 时 O(n²) 直接爆炸，而 p 又可能是合数，逆元这条路也堵死。出路是把卡特兰数拆成质数幂再相乘，全程不碰除法。这就是本题的重点：公式2 + 质因子计数法。

零、取模的七种数据状况

求第 n 项卡特兰数对 p 取模，按 n 的规模和 p 是否质数分七种情况，先定数据再挑公式：

情况数据状况用哪个
1n ≤ 35，答案不溢出 long公式4，直接返回真实值
2n 是 10³ 规模，p 为质数公式1、2、3、4 都行
3n 是 10³ 规模，p 可能是合数公式4
4n 是 10⁶ 规模，p 为质数公式1、2、3
5n 是 10⁶ 规模，p 可能是合数公式2 + 因子计数法
6n 是 10³ 规模，要求打印完整真实答案大数乘法（树屋阶梯那页）
7n 是 10⁶ 规模，要求打印完整真实答案不可能，不取模一定算不完

情况 5 是这页的主角：n 到 10⁶，p 任意（可能不是质数）。公式4 O(n²) 算不动，公式2 里除法要逆元而 p 不给逆元，只能把 C(2n,n)/(n+1) 的质因子逐个数出来，再快速幂乘回去。

一、题目5：有趣的数列

洛谷 P3200。求第 n 项卡特兰数对 p 取模，n ≤ 10⁶，p ≤ 10⁹，p 可能不为质数。核心是把 (2n)!/(n!(n+1)!) 拆成质数幂：

做法分四步。第一步用欧拉筛拿到 2..2n 每个数的最小质因子，minpf[x]=0 说明 x 是质数。第二步给每个数设计数：2..n 各 −1（分母 n!），n+2..2n 各 +1（分子 (2n)!/(n+1)!，因为 (n+1)! 恰好消掉 n+1 这一项）。第三步从大到小遍历，合数 i 把自己的计数搬给两个质因子：counts[minpf[i]] += counts[i]，counts[i/minpf[i]] += counts[i]，然后清空自己。第四步对每个计数非零的数做快速幂，全部乘起来就是答案。

为什么从大到小：合数 i 的质因子都比 i 小，从 2n 往下扫保证处理 i 时它已经是分解终态。counts[i] 可能为负（分母多于分子）或零，快速幂照常算，符号由幂次的正负决定。复杂度 = 欧拉筛 O(n) + 计数分解 O(n) + 质数个数个快速幂 O(n log p)，实测 n=10⁶ 单组 0.03 秒。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;

    // 有趣的数列：求第 n 项卡特兰数对 p 取模
    // 1 <= n <= 10^6，1 <= p <= 10^9，p 可能不为质数
    // 公式2 + 质因子计数法：把 (2n)! / (n!(n+1)!) 分解成质数幂再相乘
    const int MAXN = 2000001;

    // minpf[i] == 0 代表 i 是质数；否则 i 是合数且最小质因子是 minpf[i]
    int minpf[MAXN];
    int prime[MAXN];
    int cnt;

    // 每个数的质因子计数：2..n 各 -1（分母），n+2..2n 各 +1（分子），n+1 抵消为 0
    int counts[MAXN];

    // 欧拉筛，时间复杂度 O(n)
    void euler(int n) {
    	memset(minpf, 0, sizeof(minpf));
    	cnt = 0;
    	for (int i = 2; i <= n; i++) {
    		if (minpf[i] == 0) {
    			prime[cnt++] = i;
    		}
    		for (int j = 0; j < cnt; j++) {
    			if ((ll)i * prime[j] > n) {
    				break;
    			}
    			minpf[i * prime[j]] = prime[j];
    			if (i % prime[j] == 0) {
    				break;
    			}
    		}
    	}
    }

    ll power(ll x, ll p, int mod) {
    	ll ans = 1;
    	while (p > 0) {
    		if (p & 1) {
    			ans = ans * x % mod;
    		}
    		x = x * x % mod;
    		p >>= 1;
    	}
    	return ans;
    }

    // 公式2 + 质因子计数法
    int compute(int n, int mod) {
    	euler(2 * n);
    	for (int i = 2; i <= n; i++) {
    		counts[i] = -1;
    	}
    	for (int i = n + 2; i <= 2 * n; i++) {
    		counts[i] = 1;
    	}
    	// 从大到小统计计数：合数按最小质因子分解，质数计数确定后快速幂
    	for (int i = 2 * n; i >= 2; i--) {
    		if (minpf[i] != 0) {
    			counts[minpf[i]] += counts[i];
    			counts[i / minpf[i]] += counts[i];
    			counts[i] = 0;
    		}
    	}
    	ll ans = 1;
    	for (int i = 2; i <= 2 * n; i++) {
    		if (counts[i] != 0) {
    			ans = ans * power(i, counts[i], mod) % mod;
    		}
    	}
    	return (int)ans;
    }

    int main() {
    	int n, mod;
    	cin >> n >> mod;
    	cout << compute(n, mod) << endl;
    	return 0;
    }

二、题目7：不含递增三元组的排列方法数

洛谷 SP7897 / SPOJ SKYLINE。数字 1..n 的排列，要求任意从左往右的三个位置不能依次递增，问有多少种排列，答案对 10⁶ 取模。这个题设定不复杂，结论居然是卡特兰数：把 n 小的情况暴力枚举全排列，符合要求的数量打出来就是 1, 1, 2, 5, 14…。目前没有合理解释，记住卡特兰数前几项 1, 1, 2, 5, 14, 42 就能认出它。

这也提醒一个通用招：入参简单的计数题，先用暴力跑出前几项，再拿去和已知序列对表。10⁶ 含很多因子，无法求逆元，n ≤ 1000 直接公式4 预建整张表，多组输入一次算完。

#include <bits/stdc++.h>
    using namespace std;
    typedef long long ll;

    // 不含递增三元组的排列方法数
    // 数字 1..n 的排列，任意从左往右的三个位置不能依次递增
    // 暴力枚举小数据发现答案就是卡特兰数，答案对 1000000 取模
    // 1000000 含很多因子无法求逆元，直接公式4 预建，n <= 1000
    const int MOD = 1000000;
    const int MAXN = 1001;

    ll f[MAXN];

    void build() {
    	f[0] = f[1] = 1;
    	for (int i = 2; i < MAXN; i++) {
    		for (int l = 0, r = i - 1; l < i; l++, r--) {
    			f[i] = (f[i] + f[l] * f[r] % MOD) % MOD;
    		}
    	}
    }

    int main() {
    	build();
    	int n;
    	while (cin >> n) {
    		if (n == 0) {
    			break;
    		}
    		cout << f[n] << endl;
    	}
    	return 0;
    }

总结

因子计数法专治「n 大 + p 合数」：欧拉筛出最小质因子表，给 2..2n 每个数一个 ±1 的计数，从大到小把合数的计数分解给质因子，最后对每个质数做快速幂乘起来。没有除法，p 是合数也无所谓，复杂度 O(n)。题 7 是另一类套路：小数据暴力枚举出前几项，认出卡特兰数后直接公式4。两题对拍全过，n=10⁶ 性能 0.03 秒。
