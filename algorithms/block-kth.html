分块好题（一）：区间第 k 小与时间分块

分块的骨架是「整块一步取、散块逐个算」。这一节看两道变形：一道把块内存的东西从和换成了有序副本，另一道换了个维度，分的是时间。

零、背景

序列分块那一节里，块内存的是区间和，查询一次 O(√n)。

块内其实可以存任何「能快速重建、能快速回答整块」的信息。存和，就能回答区间和；存有序副本，就能回答整块里有多少个数不超过某个值。

这一节第一道题就是这么来的：区间加、查区间第 k 小。第二道题换个方向，把分块用在时间轴上。

一、问题：由乃打扑克

一个长度为 n 的数组 arr，m 条操作。

  • 操作 1 l r v：查询 arr[l..r] 上第 v 小的数。
  • 操作 2 l r v：arr[l..r] 上每个数加 v，v 可能是负数。

1 ≤ n、m ≤ 10⁵，数组中的值在 -2 * 10⁴ 到 +2 * 10⁴。

测试链接：洛谷 P5356。

二、朴素做法

查询的时候把 [l, r] 拷出来排序，取第 v 个。

int query(int l, int r, int k) {
    for (int i = l; i <= r; i++) tmp[i] = arr[i];
    sort(tmp + l, tmp + r + 1);
    return tmp[l + k - 1];
}

一次查询 O(len log len)，len 是区间长度。区间拉满就是 O(n log n)，做 10⁵ 次显然不行。

可以发现，慢在每次都要重新排序。区间加这种修改，并不改变块内元素的相对顺序，排序结果是可以复用的。

三、块内存一份有序副本

沿用序列分块的骨架，把数组按位置切成块，块长 B，块数 n/B。

每个块额外存一份排好序的副本 sortv，以及一个 lazy，表示整块统一加了多少。

整块加 v 时，块内所有数都加 v，相对顺序不变，所以只写 lazy += v，sortv 不动。

散块加 v 时，只有块内一部分元素变了，sortv 失效，就把这块的 arr 拷回 sortv 重新排一遍，O(B log B)。

有了 sortv 和 lazy，整块内「有多少个数 ≤ v」可以二分得到：在 sortv 里找 v - lazy 的位置。

int blockCnt(int i, int v) {        // 第 i 块内 ≤ v 的个数
    v -= lazy[i];                       // 去掉整块加成，回到副本里的值
    int l = bl[i], r = br[i];
    if (sortv[l] > v) return 0;
    if (sortv[r] <= v) return r - l + 1;
    int find = l;
    while (l <= r) {                     // 二分找最后一个 ≤ v 的位置
        int m = (l + r) >> 1;
        if (sortv[m] <= v) { find = m; l = m + 1; }
        else r = m - 1;
    }
    return find - bl[i] + 1;
}

散块仍然逐个扫，整块走二分。于是「[l, r] 里有多少个数 ≤ v」就是两端的散块暴力加中间的整块二分。

四、二分答案

第 k 小不好直接算，但「有多少个数 ≤ v」能算。

这个计数随 v 单调不减，所以可以二分答案：找一个最小的 v，使得 [l, r] 里 ≤ v 的个数 ≥ k，这个 v 就是第 k 小。

二分的上下界取 [l, r] 的最小值和最大值。最小值是每块 sortv 的首元素加 lazy 再取 min，最大值是每块 sortv 的尾元素加 lazy 再取 max，散块逐个比。

这里有个细节：二分的是值，不是下标，所以答案一定落在真实出现过的值上，循环结束时的 ans 就是答案。

五、块长取多少

一次操作的时间可以写成这样，x 是块长。

y = a * (n / x) + b * x

前一项是块间操作，块数 n/x 乘上每块的常数 a；后一项是块内操作，块长 x 乘上每个元素的常数 b。

这个函数开口向上，导数为 0 时取最小值。

x最优 = √(n * a / b)

块间操作的常数大，就适当加大块长；块内操作的常数大，就适当减小块长。

本题查询比修改贵得多（查询要做一轮值域二分，每轮还有若干次块内二分），修改次数又多于查询次数，所以块长取小一点更划算，代码里取 √(n / 2)。

这种调块长的技巧只在卡常的时候有意义，正式比赛现在很少卡，了解即可。

六、代码

完整代码如下，块长取 (int)sqrt(n / 2)。

#include <bits/stdc++.h>

    using namespace std;

    const int MAXN = 100001;
    const int MAXB = 1001;

    int n, m;
    int arr[MAXN];
    int sortv[MAXN];

    int blen, bnum;
    int bi[MAXN];
    int bl[MAXB];
    int br[MAXB];
    int lazy[MAXB];

    void innerAdd(int l, int r, int v) {
        for (int i = l; i <= r; i++) {
            arr[i] += v;
        }
        for (int i = bl[bi[l]]; i <= br[bi[l]]; i++) {
            sortv[i] = arr[i];
        }
        sort(sortv + bl[bi[l]], sortv + br[bi[l]] + 1);
    }

    void add(int l, int r, int v) {
        if (bi[l] == bi[r]) {
            innerAdd(l, r, v);
        } else {
            innerAdd(l, br[bi[l]], v);
            innerAdd(bl[bi[r]], r, v);
            for (int i = bi[l] + 1; i <= bi[r] - 1; i++) {
                lazy[i] += v;
            }
        }
    }

    int getMin(int l, int r) {
        int lb = bi[l], rb = bi[r], ans = 10000000;
        if (lb == rb) {
            for (int i = l; i <= r; i++) {
                ans = min(ans, arr[i] + lazy[lb]);
            }
        } else {
            for (int i = l; i <= br[lb]; i++) {
                ans = min(ans, arr[i] + lazy[lb]);
            }
            for (int i = bl[rb]; i <= r; i++) {
                ans = min(ans, arr[i] + lazy[rb]);
            }
            for (int i = lb + 1; i <= rb - 1; i++) {
                ans = min(ans, sortv[bl[i]] + lazy[i]);
            }
        }
        return ans;
    }

    int getMax(int l, int r) {
        int lb = bi[l], rb = bi[r], ans = -10000000;
        if (lb == rb) {
            for (int i = l; i <= r; i++) {
                ans = max(ans, arr[i] + lazy[lb]);
            }
        } else {
            for (int i = l; i <= br[lb]; i++) {
                ans = max(ans, arr[i] + lazy[lb]);
            }
            for (int i = bl[rb]; i <= r; i++) {
                ans = max(ans, arr[i] + lazy[rb]);
            }
            for (int i = lb + 1; i <= rb - 1; i++) {
                ans = max(ans, sortv[br[i]] + lazy[i]);
            }
        }
        return ans;
    }

    int blockCnt(int i, int v) {
        v -= lazy[i];
        int l = bl[i], r = br[i];
        if (sortv[l] > v) {
            return 0;
        }
        if (sortv[r] <= v) {
            return r - l + 1;
        }
        int find = l;
        while (l <= r) {
            int m = (l + r) >> 1;
            if (sortv[m] <= v) {
                find = m;
                l = m + 1;
            } else {
                r = m - 1;
            }
        }
        return find - bl[i] + 1;
    }

    int getCnt(int l, int r, int v) {
        int lb = bi[l], rb = bi[r], ans = 0;
        if (lb == rb) {
            for (int i = l; i <= r; i++) {
                if (arr[i] + lazy[lb] <= v) {
                    ans++;
                }
            }
        } else {
            for (int i = l; i <= br[lb]; i++) {
                if (arr[i] + lazy[lb] <= v) {
                    ans++;
                }
            }
            for (int i = bl[rb]; i <= r; i++) {
                if (arr[i] + lazy[rb] <= v) {
                    ans++;
                }
            }
            for (int i = lb + 1; i <= rb - 1; i++) {
                ans += blockCnt(i, v);
            }
        }
        return ans;
    }

    int query(int l, int r, int k) {
        if (k < 1 || k > r - l + 1) {
            return -1;
        }
        int minv = getMin(l, r);
        int maxv = getMax(l, r);
        int ans = -1;
        while (minv <= maxv) {
            int midv = minv + (maxv - minv) / 2;
            if (getCnt(l, r, midv) >= k) {
                ans = midv;
                maxv = midv - 1;
            } else {
                minv = midv + 1;
            }
        }
        return ans;
    }

    void prepare() {
        blen = (int)sqrt(n / 2);
        bnum = (n + blen - 1) / blen;
        for (int i = 1; i <= n; i++) {
            bi[i] = (i - 1) / blen + 1;
        }
        for (int i = 1; i <= bnum; i++) {
            bl[i] = (i - 1) * blen + 1;
            br[i] = min(i * blen, n);
        }
        for (int i = 1; i <= n; i++) {
            sortv[i] = arr[i];
        }
        for (int i = 1; i <= bnum; i++) {
            sort(sortv + bl[i], sortv + br[i] + 1);
        }
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
        }
        prepare();
        for (int i = 1, op, l, r, v; i <= m; i++) {
            cin >> op >> l >> r >> v;
            if (op == 1) {
                cout << query(l, r, v) << '\n';
            } else {
                add(l, r, v);
            }
        }
        return 0;
    }

七、换一维：时间分块

再看另一道题，表面上和上面没什么关系。

长度为 n 的数组 arr，初始是第 0 秒，第 i 条操作发生在第 i 秒，共 m 条。

  • 操作 1 l r v：arr[l..r] 上每个数加 v。
  • 操作 2 x v：不含当前这一秒，查询过去多少秒的时间里 arr[x] ≥ v。

2 ≤ n、m ≤ 10⁵，测试链接：洛谷 P3863。

第二条操作说的是个数：在第 t 秒查询，就数一数第 0 到第 t-1 秒里，有多少个时刻满足 arr[x] ≥ v。

如果只有一个数（n = 1），这个问题很清楚：维护每个时刻这个数的值，查询就是「[0, t-1] 这段时间内有多少个时刻 ≥ v」，正好是序列分块的形状，块内存有序副本，区间加 + 区间计数。

n 个数时，直接对每个位置都维护一条时间轴，是 n 条长度 m 的轴，存不下，也没法一次改一段区间。

八、按位置扫描

换一维来看。以位置为扫描顺序，以时间为下标维护一条数组 tim。

第 t 秒的修改 [l, r] 加 v，对位置 x 的影响是：x 落在 [l, r] 里才生效。所以把它拆成两个事件挂在位置上：位置 l 处加 v，位置 r + 1 处加 -v。这就是差分。

扫描到位置 x 时，把挂在 x 上的修改事件作用到时间轴上：修改发生在第 t 秒，影响 t 及以后所有时刻，于是时间轴 [t, m] 整体加 v。

于是 tim[i] 就是「位置 x 在时刻 i 相对初始值的加成」，查询时再和 arr[x] 拼起来。

具体来说，查询 (x, t, v) 要数 [1, t-1] 里有多少个时刻 i 满足 tim[i] + arr[x] ≥ v，也就是 tim[i] ≥ v - arr[x]。这正是时间轴上的区间计数，分块照做。

所有事件按 (位置, 时刻) 排序，扫一遍即可。答案按问题编号存回去，最后按输入顺序输出。

演示 ②　时间轴上的事件推进（样例：n=3，arr = [1, 3, 5]）

九、代码

时间轴长度是 m + 1（第 0 秒也要占一个位置），块长按 m 取。

#include <bits/stdc++.h>

    using namespace std;

    struct Event {
        int op, x, t, v, q;
    };

    bool EventCmp(Event &a, Event &b) {
        return a.x != b.x ? a.x < b.x : a.t < b.t;
    }

    const int MAXN = 100001;
    const int MAXB = 501;
    int n, m;
    int arr[MAXN];

    Event event[MAXN << 2];
    int cnte, cntq;

    long long tim[MAXN];
    long long sortv[MAXN];

    int blen, bnum;
    int bi[MAXN];
    int bl[MAXB];
    int br[MAXB];
    long long lazy[MAXB];

    int ans[MAXN];

    void innerAdd(int l, int r, long long v) {
        for (int i = l; i <= r; i++) {
            tim[i] += v;
        }
        for (int i = bl[bi[l]]; i <= br[bi[l]]; i++) {
            sortv[i] = tim[i];
        }
        sort(sortv + bl[bi[l]], sortv + br[bi[l]] + 1);
    }

    void add(int l, int r, long long v) {
        if (l > r) {
            return;
        }
        if (bi[l] == bi[r]) {
            innerAdd(l, r, v);
        } else {
            innerAdd(l, br[bi[l]], v);
            innerAdd(bl[bi[r]], r, v);
            for (int i = bi[l] + 1; i <= bi[r] - 1; i++) {
                lazy[i] += v;
            }
        }
    }

    int innerQuery(int l, int r, long long v) {
        v -= lazy[bi[l]];
        int ans = 0;
        for (int i = l; i <= r; i++) {
            if (tim[i] >= v) {
                ans++;
            }
        }
        return ans;
    }

    int getCnt(int i, long long v) {
        v -= lazy[i];
        int l = bl[i], r = br[i], m, pos = br[i] + 1;
        while (l <= r) {
            m = (l + r) >> 1;
            if (sortv[m] >= v) {
                pos = m;
                r = m - 1;
            } else {
                l = m + 1;
            }
        }
        return br[i] - pos + 1;
    }

    int query(int l, int r, long long v) {
        if (l > r) {
            return 0;
        }
        int ans = 0;
        if (bi[l] == bi[r]) {
        	ans = innerQuery(l, r, v);
        } else {
        	ans += innerQuery(l, br[bi[l]], v);
        	ans += innerQuery(bl[bi[r]], r, v);
            for (int i = bi[l] + 1; i <= bi[r] - 1; i++) {
                ans += getCnt(i, v);
            }
        }
        return ans;
    }

    void addChange(int x, int t, int v) {
        event[++cnte].op = 1;
        event[cnte].x = x;
        event[cnte].t = t;
        event[cnte].v = v;
    }

    void addQuery(int x, int t, int v) {
        event[++cnte].op = 2;
        event[cnte].x = x;
        event[cnte].t = t;
        event[cnte].v = v;
        event[cnte].q = ++cntq;
    }

    void prepare() {
        blen = (int)sqrt(m);
        bnum = (m + blen - 1) / blen;
        for (int i = 1; i <= m; i++) {
            bi[i] = (i - 1) / blen + 1;
        }
        for (int i = 1; i <= bnum; i++) {
            bl[i] = (i - 1) * blen + 1;
            br[i] = min(i * blen, m);
        }
        sort(event + 1, event + cnte + 1, EventCmp);
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
        }
        m++;
        for (int t = 2, op, l, r, v, x; t <= m; t++) {
            cin >> op;
            if (op == 1) {
                cin >> l >> r >> v;
                addChange(l, t, v);
                addChange(r + 1, t, -v);
            } else {
                cin >> x >> v;
                addQuery(x, t, v);
            }
        }
        prepare();
        for (int i = 1; i <= cnte; i++) {
            if (event[i].op == 1) {
                add(event[i].t, m, event[i].v);
            } else {
                ans[event[i].q] = query(1, event[i].t - 1, 1LL * event[i].v - arr[event[i].x]);
            }
        }
        for (int i = 1; i <= cntq; i++) {
            cout << ans[i] << '\n';
        }
        return 0;
    }

十、复杂度分析

设 n 为数组长度，m 为操作条数，B 为块长，块数 n/B。第一道题的值域跨度记为 V。

操作时间复杂度
预处理（建块 + 块内排序）O(n log B)
题 1 区间加O(B log B + n/B)
题 1 查询第 k 小O(log V · (B + (n/B) log B))
题 2 每个事件O(B log B + m/B)
题 2 总计O(m(B log B + m/B))，B = √m 时是 O(m√m log m)

题 1 查询里那一项 log V 是值域二分的轮数，V = 最大值减最小值，题目数据下在 30 上下。

空间上，题 1 只多了 sortv（长度 n）和 lazy（长度 n/B），是 O(n)；题 2 多了 tim、sortv（长度都是 m 量级）和事件数组（长度 2m），也是 O(n + m)。

题 2 的时间复杂度看着比题 1 高，是因为它一条操作要当成一个事件做一次区间加或区间查询，而题 1 的单次是 O(√n) 量级。
