带权并查集：主动设置距离、倍数与余数

上一页的 find/union 骨架一个字母都不用改，只换 dist 的运算，能维护的关系就从「差值」变成别的：银河英雄传说拿集合大小当距离，除法求值存倍数，食物链存 %3 余数当种类。

零、合并时主动设置距离（P1196 银河英雄传说）

30000 艘战舰，M l r 把 l 号所在的整队移动到 r 号所在队伍的最末尾后面，C l r 查两只战舰中间隔着几艘。距离不是给定的，是合并动作自己产生的：l 队整体接到 r 队尾，那么 l 队队首到 r 队队首的距离正好是 r 队的大小。

所以 union(l, r) 里没有 v，dist[lf] 直接取 size[rf]：father[lf] = rf、dist[lf] += size[rf]、size[rf] += size[lf]。查询 C l r 同队时答案是 |dist[l] - dist[r]| - 1，减 1 是因为中间隔着的战舰数要把两端的自己去掉。

这里有个坑：递归 find 会爆栈。t 最大 5e5，合并链可能拉得很长，递归深度不是常数。代码用显式栈做路径压缩：先沿着 father 一路走到底，再倒着回去逐层压缩，dist 在回程累加。

银河英雄传说

// 银河英雄传说
    // 30000 艘战舰，初始各自一队，M l r 把 l 队整体移到 r 队末尾，C l r 查询中间隔几艘
    // 1 <= t <= 5e5，迭代 find 防爆栈
    // 测试链接 : https://www.luogu.com.cn/problem/P1196
    #include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 30001;

    int n = 30000;
    int father[MAXN];
    int dist[MAXN];
    int sizeArr[MAXN];
    int stackArr[MAXN];

    void prepare() {
        for (int i = 1; i <= n; i++) {
            father[i] = i;
            dist[i] = 0;
            sizeArr[i] = 1;
        }
    }

    int find(int i) {
        int si = 0;
        while (i != father[i]) {
            stackArr[++si] = i;
            i = father[i];
        }
        stackArr[si + 1] = i;
        for (int j = si; j >= 1; j--) {
            father[stackArr[j]] = i;
            dist[stackArr[j]] += dist[stackArr[j + 1]];
        }
        return i;
    }

    void unionSet(int l, int r) {
        int lf = find(l), rf = find(r);
        if (lf != rf) {
            father[lf] = rf;
            dist[lf] += sizeArr[rf];
            sizeArr[rf] += sizeArr[lf];
        }
    }

    int query(int l, int r) {
        if (find(l) != find(r)) {
            return -1;
        }
        return abs(dist[l] - dist[r]) - 1;
    }

    int main() {
        prepare();
        int t;
        scanf("%d", &t);
        char op;
        for (int i = 1, l, r; i <= t; i++) {
            scanf(" %c%d%d", &op, &l, &r);
            if (op == 'M') {
                unionSet(l, r);
            } else {
                printf("%d\n", query(l, r));
            }
        }
        return 0;
    }

一、倍数关系（P0399 除法求值）

等式是除法：l / r = v。dist[i] 存 i 到根的倍数，根的值当 1。合并时 l 到 r 的倍数关系是 v，find 的修正从累加变成累乘，union 公式从加减变成乘除。

变量是字符串，father 和 dist 都用哈希表。find 里 dist[x] = dist[x] * dist[tmp]，union 里 dist[lf] = dist[r] / dist[l] * v，query 返回 dist[l] / dist[r]。变量从没出现过，或者两个变量不在同一个集合，答案都是 -1.0。

这道题的输入是 LeetCode 的函数式接口，代码只写 calcEquation 一个函数，不写 main。页面给出的 C++ 版本可以直接塞进力扣提交。

除法求值

// 除法求值
    // 变量用字符串表示，等式 l/r = v，查询 l/r 的值，推不出返回 -1.0
    // 带权并查集维护倍数关系（乘法），dist 存到根的比值
    // 测试链接 : https://leetcode.cn/problems/evaluate-division/
    #include <bits/stdc++.h>
    using namespace std;

    unordered_map<string, string> father;
    unordered_map<string, double> dist;

    string findSet(const string& x) {
        if (!father.count(x)) {
            return "";
        }
        if (x != father[x]) {
            string tmp = father[x];
            string fa = findSet(tmp);
            dist[x] = dist[x] * dist[tmp];
            father[x] = fa;
        }
        return father[x];
    }

    void unionSet(const string& l, const string& r, double v) {
        string lf = findSet(l), rf = findSet(r);
        if (lf != rf) {
            father[lf] = rf;
            dist[lf] = dist[r] / dist[l] * v;
        }
    }

    double query(const string& l, const string& r) {
        string lf = findSet(l), rf = findSet(r);
        if (lf.empty() || rf.empty() || lf != rf) {
            return -1.0;
        }
        return dist[l] / dist[r];
    }

    vector<double> calcEquation(vector<vector<string>>& equations, vector<double>& values,
                                vector<vector<string>>& queries) {
        father.clear();
        dist.clear();
        for (auto& eq : equations) {
            for (auto& key : eq) {
                father[key] = key;
                dist[key] = 1.0;
            }
        }
        for (int i = 0; i < (int)values.size(); i++) {
            unionSet(equations[i][0], equations[i][1], values[i]);
        }
        vector<double> ans;
        for (auto& q : queries) {
            ans.push_back(query(q[0], q[1]));
        }
        return ans;
    }

二、余数关系（P2024 甄别食物链）

三类动物 A 吃 B、B 吃 C、C 吃 A，是循环的。给 n 只动物的编号和 k 句话（1 X Y 同类、2 X Y X 吃 Y），数假话。把「同类 / 吃 / 被吃」压进 dist 对 3 取模：dist[i] = 0 是 i 和头同类，1 是 i 吃头，2 是 i 被头吃。

find 的修正变成 dist[i] = (dist[i] + dist[t]) % 3。union 时 v 由语句类型决定：同类 v = 0，吃 v = 1，公式 dist[lf] = (dist[r] - dist[l] + v + 3) % 3，那个 +3 是防止负数取模。假话三判据：越界（l 或 r 大于 n）、自吃（op == 2 且 l == r）、同根时 dist 对不上。

这里有个细节：同类任意两只动物，如果它们不在一个集合，check 会放行并当成真话合并。也就是说假话只可能被「已经连通的集合」当场抓住，这正是题目说的「与前面的某些真话冲突」。

甄别食物链

// 甄别食物链
    // n 只动物，A/B/C 三类，A吃B、B吃C、C吃A
    // 1 X Y 同类，2 X Y X吃Y，统计假话数量
    // 带权并查集维护 %3 余数：0 同类、1 吃头、2 被头吃
    // 1 <= n <= 5e4，1 <= k <= 1e5
    // 测试链接 : https://www.luogu.com.cn/problem/P2024
    #include <bits/stdc++.h>
    using namespace std;

    const int MAXN = 50001;

    int n, k, ans;
    int father[MAXN];
    int dist[MAXN];

    void prepare() {
        ans = 0;
        for (int i = 1; i <= n; i++) {
            father[i] = i;
            dist[i] = 0;
        }
    }

    int find(int i) {
        if (i != father[i]) {
            int tmp = father[i];
            father[i] = find(tmp);
            dist[i] = (dist[i] + dist[tmp]) % 3;
        }
        return father[i];
    }

    // op == 1：l 和 r 是同类；op == 2：l 吃 r
    void unionSet(int op, int l, int r) {
        int lf = find(l), rf = find(r), v = op == 1 ? 0 : 1;
        if (lf != rf) {
            father[lf] = rf;
            dist[lf] = (dist[r] - dist[l] + v + 3) % 3;
        }
    }

    bool check(int op, int l, int r) {
        if (l > n || r > n || (op == 2 && l == r)) {
            return false;
        }
        if (find(l) == find(r)) {
            if (op == 1) {
                if (dist[l] != dist[r]) {
                    return false;
                }
            } else {
                if ((dist[l] - dist[r] + 3) % 3 != 1) {
                    return false;
                }
            }
        }
        return true;
    }

    int main() {
        scanf("%d%d", &n, &k);
        prepare();
        for (int i = 1, op, l, r; i <= k; i++) {
            scanf("%d%d%d", &op, &l, &r);
            if (!check(op, l, r)) {
                ans++;
            } else {
                unionSet(op, l, r);
            }
        }
        printf("%d\n", ans);
        return 0;
    }

三、总结

三种运算一套骨架：find 的修正（加法、乘法、加法取模）、union 的公式（v + dist[r] - dist[l]、dist[r]/dist[l]*v、(dist[r]-dist[l]+v+3)%3）、query 的答案（差值、比值、%3 差），全部跟着 dist 的含义换。银河英雄传说还演示了「距离由合并自己产生」：dist[lf] = size[rf]。

种类关系除了 %3 余数，还有更省的一派做法：压根不存余数，用 enemy 数组做二分类。异或关系更特殊，得先造一个值为 0 的虚点。下页讲这两块。
