CDQ 分治与二维空间计数

平面上给 n 个点和 m 个矩形查询，问每个矩形里有多少点、点权和是多少。事件化之后这就是一个二维偏序，CDQ 分治直接能套。本页两道题都不带修改，所以连数据结构都省了，每层只剩一个双指针。

零、背景

上一页把三维偏序拆成了三份：一维交给排序，一维交给分治，一维交给树状数组。这一页把维度换成平面坐标，事情反而变简单：只有 x 和 y 两维，分治之外只剩一个双指针。

静态二维数点其实有更朴素的做法：把点和查询按 x 排序，从左到右扫，用树状数组按 y 统计，一样是 O((n+m)·log)。CDQ 的写法价值不在这一页，而在于它和下一页那两道带时序的题是同一份框架，改几行就能搬过去。

一、题目3：园丁的烦恼

有 n 棵树，每棵给定位置坐标 (x, y)。接下来 m 条查询，格式是 a b c d，打印左上角 (a, b)、右下角 (c, d) 这个区域里有几棵树。0 ≤ n ≤ 5×10⁵，1 ≤ m ≤ 5×10⁵，坐标值在 0 到 10⁷。测试链接：洛谷 P2163。

每条查询拆成四个前缀事件，加上 n 个树事件，一共 n + 4m 个事件。事件结构体里 op 区分类型，x、y 是坐标，v 是容斥系数，q 是查询编号。

矩形查询：四个前缀怎么叠回矩形

排序的规则是 x 升序，x 相同就把树排在查询前面。这个 tie-break 不能省：x 相同的树必须被同一 x 的查询统计到，而分治只处理「左半 → 右半」，同 x 的两个事件如果左右分开，排在后面的那一个就拿不到贡献。

merge 的形状：p2 从左往右扫右半，p1 不断吞掉左半里 y 不超过 arr[p2].y 的元素，遇到树就把计数 tree 加一；然后如果 arr[p2] 是查询，就 ans[q] += tree × v。扫完把 [l, r] 按 y 归并成一个有序段交上去。

园丁的烦恼：顶层 merge 的双指针（8 个事件）

左半（按 y 有序）
右半（按 y 有序）

演示里这一层贡献 +2，左半内部贡献 −2，右半内部贡献 +2，加起来是 2，正好等于矩形 [3,4] × [1,5] 里的两棵树。每一对「树在前、查询在后」的组合只会在某一层被算一次，不重不漏。

#include <bits/stdc++.h>

    using namespace std;

    struct Node {
        int op, x, y, v, q;
    };

    bool NodeCmp(Node a, Node b) {
        if (a.x != b.x) {
            return a.x < b.x;
        }
        return a.op < b.op;
    }

    const int MAXN = 500001 * 5;
    int n, m;
    Node arr[MAXN];
    int cnt = 0;
    Node tmp[MAXN];
    int ans[MAXN];

    void addTree(int x, int y) {
        arr[++cnt].op = 1;
        arr[cnt].x = x;
        arr[cnt].y = y;
    }

    void addQuery(int x, int y, int v, int q) {
        arr[++cnt].op = 2;
        arr[cnt].x = x;
        arr[cnt].y = y;
        arr[cnt].v = v;
        arr[cnt].q = q;
    }

    void merge(int l, int m, int r) {
        int p1, p2, tree = 0;
        for (p1 = l - 1, p2 = m + 1; p2 <= r; p2++) {
            while (p1 + 1 <= m && arr[p1 + 1].y <= arr[p2].y) {
                p1++;
                if (arr[p1].op == 1) {
                    tree++;
                }
            }
            if (arr[p2].op == 2) {
                ans[arr[p2].q] += tree * arr[p2].v;
            }
        }
        p1 = l;
        p2 = m + 1;
        int i = l;
        while (p1 <= m && p2 <= r) {
            tmp[i++] = arr[p1].y <= arr[p2].y ? arr[p1++] : arr[p2++];
        }
        while (p1 <= m) {
            tmp[i++] = arr[p1++];
        }
        while (p2 <= r) {
            tmp[i++] = arr[p2++];
        }
        for (i = l; i <= r; i++) {
            arr[i] = tmp[i];
        }
    }

    void cdq(int l, int r) {
        if (l == r) {
            return;
        }
        int mid = (l + r) / 2;
        cdq(l, mid);
        cdq(mid + 1, r);
        merge(l, mid, r);
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1, x, y; i <= n; i++) {
            cin >> x >> y;
            addTree(x, y);
        }
        for (int i = 1, a, b, c, d; i <= m; i++) {
            cin >> a >> b >> c >> d;
            addQuery(c, d, 1, i);
            addQuery(a - 1, b - 1, 1, i);
            addQuery(a - 1, d, -1, i);
            addQuery(c, b - 1, -1, i);
        }
        sort(arr + 1, arr + cnt + 1, NodeCmp);
        cdq(1, cnt);
        for (int i = 1; i <= m; i++) {
            cout << ans[i] << '\n';
        }
        return 0;
    }

二、题目4：老C的任务

有 n 个基站，每个给定 x、y、v，表示基站在 (x, y) 位置、功率为 v。接下来 m 条查询，格式同样是 a b c d，打印左上角 (a, b)、右下角 (c, d) 区域里基站的功率和。1 ≤ n、m ≤ 10⁵，其余数值都在 int 范围。测试链接：洛谷 P3755。

和上一题的差别只有两处。第一处：树事件现在带上功率，merge 里不再是 tree++，而是 sum += arr[p1].v。第二处：查询的系数乘的是功率和，答案可能超过 int，ans 数组改成 long long。

这里有个细节：Node 里的 v 字段在两种事件里含义不同。基站事件里 v 是功率，查询事件里 v 是容斥系数 ±1。merge 里写的 ans[q] += sum × arr[p2].v，读起来是「前缀功率和乘上符号」，两个 v 各管一头，不冲突。

顺带一提，这两题的树和基站一开始就全部给定，所以 x 的排序在最开始做一次就行。事件数组大小为 n + 4m，题目给的是 5×10⁵，数组要按 5 倍开。

#include <bits/stdc++.h>

    using namespace std;

    struct Node {
        int op, x, y, v, q;
    };

    bool NodeCmp(Node a, Node b) {
        if (a.x != b.x) {
            return a.x < b.x;
        }
        return a.op < b.op;
    }

    const int MAXN = 500001;
    int n, m;
    Node arr[MAXN];
    int cnt = 0;
    Node tmp[MAXN];
    long long ans[MAXN];

    void addStation(int x, int y, int v) {
        arr[++cnt].op = 1;
        arr[cnt].x = x;
        arr[cnt].y = y;
        arr[cnt].v = v;
    }

    void addQuery(int x, int y, int v, int q) {
        arr[++cnt].op = 2;
        arr[cnt].x = x;
        arr[cnt].y = y;
        arr[cnt].v = v;
        arr[cnt].q = q;
    }

    void merge(int l, int m, int r) {
        int p1, p2;
        long long sum = 0;
        for (p1 = l - 1, p2 = m + 1; p2 <= r; p2++) {
            while (p1 + 1 <= m && arr[p1 + 1].y <= arr[p2].y) {
                p1++;
                if (arr[p1].op == 1) {
                    sum += arr[p1].v;
                }
            }
            if (arr[p2].op == 2) {
                ans[arr[p2].q] += sum * arr[p2].v;
            }
        }
        p1 = l;
        p2 = m + 1;
        int i = l;
        while (p1 <= m && p2 <= r) {
            tmp[i++] = arr[p1].y <= arr[p2].y ? arr[p1++] : arr[p2++];
        }
        while (p1 <= m) {
            tmp[i++] = arr[p1++];
        }
        while (p2 <= r) {
            tmp[i++] = arr[p2++];
        }
        for (i = l; i <= r; i++) {
            arr[i] = tmp[i];
        }
    }

    void cdq(int l, int r) {
        if (l == r) {
            return;
        }
        int mid = (l + r) / 2;
        cdq(l, mid);
        cdq(mid + 1, r);
        merge(l, mid, r);
    }

    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        cin >> n >> m;
        for (int i = 1, x, y, v; i <= n; i++) {
            cin >> x >> y >> v;
            addStation(x, y, v);
        }
        for (int i = 1, a, b, c, d; i <= m; i++) {
            cin >> a >> b >> c >> d;
            addQuery(c, d, 1, i);
            addQuery(a - 1, b - 1, 1, i);
            addQuery(a - 1, d, -1, i);
            addQuery(c, b - 1, -1, i);
        }
        sort(arr + 1, arr + cnt + 1, NodeCmp);
        cdq(1, cnt);
        for (int i = 1; i <= m; i++) {
            cout << ans[i] << '\n';
        }
        return 0;
    }

三、复杂度分析

两层循环都跑到头：分治有 O(log E) 层，每层 merge 的双指针是线性的，归并也是线性的。因为没用树状数组，每层里没有 log，整体比三维偏序少一个 log。

题目事件规模时间复杂度空间复杂度
园丁的烦恼E = n + 4mO(E·log E)O(E)
老C的任务E = n + 4mO(E·log E)O(E)

E 取 2.5×10⁶ 时，log E 约 21，每层 2.5×10⁶ 次比较，总量在 5×10⁷ 量级。老C的任务的答案是功率和，n = 10⁵、功率在 int 范围，最大约为 2×10¹⁴，必须用 long long。

空间上只有两个静态数组：事件数组和归并用的临时数组，加上答案数组。递归深度是 O(log E)，栈开销可以忽略。

四、最后

这两题的共同点是没有修改：所有点的 x 从一开始就能排好，分治维和排序维是同一维，y 于是可以被双指针白拿。

下一页的两道题有了操作时序，分治维被时间占住，x 只能在每层 merge 里靠排序维持。这时候 y 就推不动了，得上树状数组。
