线段树：重置与加同时存在

一棵树上同时开着两种修改，节点上就有两套懒信息。重置会彻底取消之前欠的加，而加只在重置的基础上叠加，所以下发的顺序只能是先重置、后加。

零、背景

两种修改的优先级并不对称。同一段上先加 3 再重置成 5，前面那次加连痕迹都不留，结果就是 5；反过来先重置成 5 再加 3，得到的是 8，两笔账都得记住。

所以当节点上同时压着一条重置和一条加，重置一定是更早的那条，否则它早就把加清掉了。下发的时候按这个顺序来，节点上的信息才对得上。

一、问题

长度为 n 的数组 a，三种操作交替出现。

  • 修改一：区间 [l, r] 里每个数重置成 v。
  • 修改二：区间 [l, r] 里每个数加上 v。
  • 查询：区间 [l, r] 的累加和，或者最大值。

洛谷 P1253「扶苏的问题」就是最大值版本：区间赋值、区间加、区间求最大，n、m 到 2 × 10⁶，答案要用 64 位整数装。

二、节点上同时挂着两套懒信息

前两页把加和重置分开看，各自只要一组数组。现在两者并存，节点上就要同时放：一份查询信息（和或最大值），一份加法的账 add，一份重置的值 change，外加 change 的开关 update。

加法的账和重置的账互不干扰，谁也不用去改另一方的数据结构，唯一的交叉点是那句「重置要把 add 清掉」。

三、重置要顺手把加法的账撕掉

整段重置成 v 的那一刻，这段范围上每个数的最终值就定了，前面欠的加再补发下去只会把事情弄坏。所以重置落到节点上时，除了写自己的 change 和开关，还要把 add[i] 清零。

这一行是全课最贵的一行。漏掉它之后，节点上「整段是 v」和「整段还欠着 +3」会同时成立，下一步 down 就会把那笔早就作废的加补到儿子身上去。

// 两种懒信息同时挂在节点上: change/update 管重置, add 管整段加

    void up(int i) {
      sum[i] = sum[i << 1] + sum[i << 1 | 1];
    }

    // 重置任务盖到当前范围: 整段变成 v，之前欠的加全部作废
    void updateLazy(int i, ll v, int len) {
      sum[i] = v * len;
      add[i] = 0;
      change[i] = v;
      update[i] = true;
    }

    // 加任务盖到当前范围: 在已有的懒信息之上再记一笔
    void addLazy(int i, ll v, int len) {
      sum[i] += v * len;
      add[i] += v;
    }

    // 父范围的懒信息往下发，两趟: 先重置，后加
    void down(int i, int ln, int rn) {
      if (update[i]) {
        updateLazy(i << 1, change[i], ln);
        updateLazy(i << 1 | 1, change[i], rn);
        update[i] = false;
      }
      if (add[i] != 0) {
        addLazy(i << 1, add[i], ln);
        addLazy(i << 1 | 1, add[i], rn);
        add[i] = 0;
      }
    }

    void build(int l, int r, int i) {
      if (l == r) {
        sum[i] = arr[l];
      } else {
        int mid = (l + r) >> 1;
        build(l, mid, i << 1);
        build(mid + 1, r, i << 1 | 1);
        up(i);
      }
      add[i] = 0;
      change[i] = 0;
      update[i] = false;
    }

    void rangeUpdate(int jobl, int jobr, ll jobv, int l, int r, int i) {
      if (jobl <= l && r <= jobr) {
        updateLazy(i, jobv, r - l + 1);
      } else {
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        if (jobl <= mid) {
          rangeUpdate(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
          rangeUpdate(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
      }
    }

    void rangeAdd(int jobl, int jobr, ll jobv, int l, int r, int i) {
      if (jobl <= l && r <= jobr) {
        addLazy(i, jobv, r - l + 1);
      } else {
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        if (jobl <= mid) {
          rangeAdd(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
          rangeAdd(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
      }
    }

    ll query(int jobl, int jobr, int l, int r, int i) {
      if (jobl <= l && r <= jobr) {
        return sum[i];
      }
      int mid = (l + r) >> 1;
      down(i, mid - l + 1, r - mid);
      ll ans = 0;
      if (jobl <= mid) {
        ans += query(jobl, jobr, l, mid, i << 1);
      }
      if (jobr > mid) {
        ans += query(jobl, jobr, mid + 1, r, i << 1 | 1);
      }
      return ans;
    }

反过来，加落到节点上时不去动 change：change 记的是「整段被重置成了什么」，后面叠加的加是另一笔账，两个字段各管一层。这也正是优先级不对称在代码里的样子。

四、down 分两趟走

父节点的懒信息往下发一层，要先发重置、再发加。顺序颠倒过来，发出去的重置会盖掉刚补发的加，儿子手里的数字就少了这一部分。

两趟都发完，父节点的两套账一起清空。

void down(int i, int ln, int rn) {
  if (update[i]) {
    updateLazy(i << 1, change[i], ln);
    updateLazy(i << 1 | 1, change[i], rn);
    update[i] = false;
  }
  if (add[i] != 0) {
    addLazy(i << 1, add[i], ln);
    addLazy(i << 1 | 1, add[i], rn);
    add[i] = 0;
  }
}

完全覆盖（懒住或直接取值） 部分相交（要先 down 再往下走） 这一步没碰

五、两份代码只差一个符号

把查询信息换成最大值，改动和前面几页一样：up 用 max，两个 lazy 去掉长度参数，rangeAdd 里的 v 乘 len 变成 v。

P1253 就是这一版，数据范围大，读入要用快一点的写法。

// 最大值版本: 两套懒信息都不用长度参数

    void up(int i) {
      mx[i] = max(mx[i << 1], mx[i << 1 | 1]);
    }

    // 整段重置成 v，最大值就是 v
    // 关键一句是 add[i] = 0: 重置把之前欠下的加作废
    void updateLazy(int i, ll v) {
      mx[i] = v;
      add[i] = 0;
      change[i] = v;
      update[i] = true;
    }

    // 整段加 v，最大值加 v
    void addLazy(int i, ll v) {
      mx[i] += v;
      add[i] += v;
    }

    // 先发重置、后发加
    void down(int i) {
      if (update[i]) {
        updateLazy(i << 1, change[i]);
        updateLazy(i << 1 | 1, change[i]);
        update[i] = false;
      }
      if (add[i] != 0) {
        addLazy(i << 1, add[i]);
        addLazy(i << 1 | 1, add[i]);
        add[i] = 0;
      }
    }

    void build(int l, int r, int i) {
      if (l == r) {
        mx[i] = arr[l];
      } else {
        int mid = (l + r) >> 1;
        build(l, mid, i << 1);
        build(mid + 1, r, i << 1 | 1);
        up(i);
      }
      add[i] = 0;
      change[i] = 0;
      update[i] = false;
    }

    void rangeUpdate(int jobl, int jobr, ll jobv, int l, int r, int i) {
      if (jobl <= l && r <= jobr) {
        updateLazy(i, jobv);
      } else {
        int mid = (l + r) >> 1;
        down(i);
        if (jobl <= mid) {
          rangeUpdate(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
          rangeUpdate(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
      }
    }

    void rangeAdd(int jobl, int jobr, ll jobv, int l, int r, int i) {
      if (jobl <= l && r <= jobr) {
        addLazy(i, jobv);
      } else {
        int mid = (l + r) >> 1;
        down(i);
        if (jobl <= mid) {
          rangeAdd(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
          rangeAdd(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
      }
    }

    ll query(int jobl, int jobr, int l, int r, int i) {
      if (jobl <= l && r <= jobr) {
        return mx[i];
      }
      int mid = (l + r) >> 1;
      down(i);
      ll ans = LLONG_MIN;
      if (jobl <= mid) {
        ans = max(ans, query(jobl, jobr, l, mid, i << 1));
      }
      if (jobr > mid) {
        ans = max(ans, query(jobl, jobr, mid + 1, r, i << 1 | 1));
      }
      return ans;
    }

六、错一行会怎样

这四种改法都只动一句，跑「重置 + 加 + 查求和」，3000 组随机用例里统计答案出过错的比例。

随机用例规模重置时不清 adddown 里先发加change 写成累加正解
n ≤ 8，值 ≤ 54.23%4.03%6.13%0.00%
n ≤ 20，值 ≤ 1015.33%14.07%20.70%0.00%
n ≤ 60，值 ≤ 100032.67%29.13%36.00%0.00%
n ≤ 200，值 ≤ 10⁵49.33%44.33%50.43%0.00%

小数据上这三处错几乎抓不到，n ≤ 8 时都在 6% 以下。原因是它们需要「同一段先被重置、又被加」或者反过来的形状，序列太短撞不上。规模放到 200，才涨到四五成。

值域不起作用。把 n 固定成 20、值域从 ±2 拉到 ±10⁵，三处的抓取率是 32.77%、32.63%、34.20%、33.13%，四个数字挤在 1.5 个百分点的区间里。

七、复杂度分析

操作时间空间
建树 buildO(n)查询信息 + add + change + update，各 4n
范围重置O(log n)递归栈 O(log n)
范围加O(log n)递归栈 O(log n)
范围查询O(log n)递归栈 O(log n)
整体O((n + m) log n)O(n)

实测 n = m = 10⁶ 的混合操作：求和版 725.26 ms，最大值版 724.17 ms。同一规模下只跑一种修改的版本是 0.61 秒到 0.63 秒，多出的一成来自 down 里多出来的那一趟判断。

八、总结

两种修改并存时的全部要点，就是一条时间关系：节点上同时留着重置和加时，重置一定更早，所以 down 先发重置再发加，而重置落到节点上时要顺手把 add 清零。

到这里线段树的模板已经能覆盖「区间加、区间赋值、区间求和、区间求最大」这四种常见组合。再往上走的信息就不是一两个数了，比如区间里最长的一段连续 1，那种信息父范围不能只靠左右儿子的答案拼出来，还得知道边界的形态。
