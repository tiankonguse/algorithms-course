#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MOD = 1000000007;

// ===== 课程版：Code01_GaussEor 改写 =====
const int MAXV = 2000;
int prime[MAXV+1], cnt;
bool vis[MAXV+1];
int pow2v[305];
int mat[305][305];   // [cnt][n+1]
int n;

void prepare() {
  for (int i = 2; i * i <= MAXV; i++) {
    if (!vis[i]) {
      for (int j = i * i; j <= MAXV; j += i) vis[j] = true;
    }
  }
  cnt = 0;
  for (int i = 2; i <= MAXV; i++) if (!vis[i]) prime[++cnt] = i;
  pow2v[0] = 1;
  for (int i = 1; i < 305; i++) pow2v[i] = (pow2v[i-1] * 2) % MOD;
}

void gauss(int gn) {
  for (int i = 1; i <= gn; i++) {
    for (int j = 1; j <= gn; j++) {
      if (j < i && mat[j][j] == 1) continue;
      if (mat[j][i] == 1) {
        for (int k = 1; k <= gn+1; k++) swap(mat[i][k], mat[j][k]);
        break;
      }
    }
    if (mat[i][i] == 1) {
      for (int j = 1; j <= gn; j++) {
        if (i != j && mat[j][i] == 1) {
          for (int k = i; k <= gn+1; k++) mat[j][k] ^= mat[i][k];
        }
      }
    }
  }
}

int course_solve(ll arr[]) {
  for (int i = 1; i <= cnt; i++)
    for (int j = 1; j <= cnt+1; j++) mat[i][j] = 0;
  ll cur;
  for (int i = 1; i <= n; i++) {
    cur = arr[i];
    for (int j = 1; j <= cnt && cur != 0; j++) {
      while (cur % prime[j] == 0) { mat[j][i] ^= 1; cur /= prime[j]; }
    }
  }
  gauss(cnt);
  int mainc = 0;
  for (int i = 1; i <= cnt; i++) if (mat[i][i] == 1) mainc++;
  return (pow2v[n - mainc] - 1 + MOD) % MOD;
}

// ===== 暴力版：枚举非空子集，质因子指数向量全偶 =====
int brute_solve(ll arr[]) {
  // 每项分解成指数向量 exp[i][p]
  static int expv[305][303];
  for (int i = 1; i <= n; i++) {
    ll cur = arr[i];
    for (int j = 1; j <= cnt && cur != 0; j++) {
      expv[i][j] = 0;
      while (cur % prime[j] == 0) { expv[i][j]++; cur /= prime[j]; }
    }
  }
  int ans = 0;
  for (int mask = 1; mask < (1 << n); mask++) {
    int ok = 1;
    for (int j = 1; j <= cnt; j++) {
      int s = 0;
      for (int i = 1; i <= n; i++) if (mask >> (i-1) & 1) s += expv[i][j];
      if (s & 1) { ok = 0; break; }
    }
    if (ok) ans++;
  }
  return ans;
}

// ===== 随机数据生成：质因子全 ≤ 2000 =====
mt19937 rng(20261016);
ll gen_arr() {
  // 选 1~4 个质数乘起来（指数 1~2），保证 ≤ 1e18
  while (true) {
    int k = 1 + rng() % 4;
    ll v = 1;
    for (int t = 0; t < k; t++) {
      int p = prime[1 + rng() % cnt];
      int e = 1 + rng() % 2;
      for (int x = 0; x < e; x++) {
        if (v > (ll)4e18 / p) { v = -1; break; }
        v *= p;
      }
      if (v < 0) break;
    }
    if (v > 0 && v <= (ll)1e18) return v;
  }
}

int main() {
  prepare();
  ll arr[305];
  for (int T = 0; T < 5000; T++) {
    n = 1 + rng() % 10;
    for (int i = 1; i <= n; i++) arr[i] = gen_arr();
    int c = course_solve(arr);
    int b = brute_solve(arr);
    if (c != b) {
      printf("MISMATCH T=%d n=%d course=%d brute=%d\n", T, n, c, b);
      for (int i = 1; i <= n; i++) printf("%lld ", arr[i]);
      printf("\n");
      return 1;
    }
  }
  printf("t1 OK 5000 groups\n");
  // 固定例：n=3, [4,9,36] -> 子集 {4*9=36, 4*36=144, 9*36=324, 4*9*36=1296} 4 个? 4=2^2,9=3^2,36=2^2*3^2
  // 指数向量: 4=(2,0) 9=(0,2) 36=(2,2). 非空子集和全偶: {4,9}: (2,2)✓ {36}: (2,2)✓ {4,9,36}:(4,4)✓ = 3 个
  n = 3; arr[1]=4; arr[2]=9; arr[3]=36;
  printf("fix1 [4,9,36] course=%d brute=%d (期望 3)\n", course_solve(arr), brute_solve(arr));
  // n=1, arr=[4] -> 子集 {4} 完全平方 ✓ -> 1
  n = 1; arr[1]=4;
  printf("fix2 [4] course=%d brute=%d (期望 1)\n", course_solve(arr), brute_solve(arr));
  // n=1, arr=[2] -> 非空子集 {2} 不是平方 -> 0
  n = 1; arr[1]=2;
  printf("fix3 [2] course=%d brute=%d (期望 0)\n", course_solve(arr), brute_solve(arr));
  return 0;
}

#include <bits/stdc++.h>
using namespace std;

// ===== 课程版：Code02_MinimumOperations 改写 =====
int mat[37][37];
int op[37];
int n, ans;

void prepare() {
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) mat[i][j] = 0;
    mat[i][i] = 1;
    mat[i][n+1] = 1;
    op[i] = 0;
  }
}

void gauss(int gn) {
  for (int i = 1; i <= gn; i++) {
    for (int j = 1; j <= gn; j++) {
      if (j < i && mat[j][j] == 1) continue;
      if (mat[j][i] == 1) {
        for (int k = 1; k <= gn+1; k++) swap(mat[i][k], mat[j][k]);
        break;
      }
    }
    if (mat[i][i] == 1) {
      for (int j = 1; j <= gn; j++) {
        if (i != j && mat[j][i] == 1) {
          for (int k = i; k <= gn+1; k++) mat[j][k] ^= mat[i][k];
        }
      }
    }
  }
}

void dfs(int i, int num) {
  if (num >= ans) return;
  if (i == 0) { ans = num; }
  else {
    if (mat[i][i] == 0) {
      op[i] = 0; dfs(i-1, num);
      op[i] = 1; dfs(i-1, num+1);
    } else {
      int cur = mat[i][n+1];
      for (int j = i+1; j <= n; j++) if (mat[i][j] == 1) cur ^= op[j];
      dfs(i-1, num + cur);
    }
  }
}

int course_solve(int adj[][37]) {
  prepare();
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) mat[i][j] = adj[i][j];
  for (int i = 1; i <= n; i++) { mat[i][i] = 1; mat[i][n+1] = 1; }
  gauss(n);
  int sign = 1;
  for (int i = 1; i <= n; i++) if (mat[i][i] == 0) { sign = 0; break; }
  if (sign == 1) {
    ans = 0;
    for (int i = 1; i <= n; i++) if (mat[i][n+1] == 1) ans++;
  } else {
    ans = n;
    dfs(n, 0);
  }
  return ans;
}

// ===== 暴力版：枚举 2^n 操作子集 =====
int brute_solve(int adj[][37]) {
  int best = n + 1;
  for (int mask = 0; mask < (1 << n); mask++) {
    int ok = 1;
    for (int j = 1; j <= n; j++) {
      int cntb = 0;
      if (mask >> (j-1) & 1) cntb++;
      for (int k = 1; k <= n; k++)
        if (adj[j][k] && (mask >> (k-1) & 1)) cntb++;
      if ((cntb & 1) == 0) { ok = 0; break; }  // 目标全 1：每个点被翻转奇数次
    }
    if (ok) best = min(best, __builtin_popcount(mask));
  }
  return best;
}

      int u = 1 + rng() % n, v = 1 + rng() % n;

    }

  n = 3; memset(adj, 0, sizeof(adj));

#include <bits/stdc++.h>
using namespace std;

// ===== 课程版：Code03_AlienInsectLegs 改写（位图）=====
const int BIT = 64;
const int MAXN = 2002;
const int MAXM = MAXN / BIT + 1;
long long mat[MAXN][MAXM];
int n, m, s;
int need;

void setbit(int row, int col, int v) {
  if (v == 0) mat[row][col / BIT] &= ~(1LL << (col % BIT));
  else mat[row][col / BIT] |= 1LL << (col % BIT);
}
int getbit(int row, int col) {
  return ((mat[row][col / BIT] >> (col % BIT)) & 1) == 1 ? 1 : 0;
}
void eor(int row1, int row2, int bits) {
  for (int k = 0; k <= bits / BIT; k++) mat[row2][k] ^= mat[row1][k];
}
void swaprows(int a, int b) {
  for (int k = 0; k < MAXM; k++) swap(mat[a][k], mat[b][k]);
}

void gauss_course(int gn) {
  need = 0;
  for (int i = 1; i <= gn; i++) {
    for (int j = i; j <= gn; j++) {
      if (getbit(j, i) == 1) {
        swaprows(i, j);
        need = max(need, j);
        break;
      }
    }
    if (getbit(i, i) == 0) return;
    for (int j = 1; j <= gn; j++) {
      if (i != j && getbit(j, i) == 1) eor(i, j, n + 2);
    }
  }
}

// 一次性构造课程输入
void build_course(vector<string>& rows, vector<int>& b, int nn, int mm) {
  n = nn; m = mm; s = max(n, m);
  memset(mat, 0, sizeof(mat));
  for (int i = 1; i <= m; i++) {
    for (int j = 1; j <= n; j++) setbit(i, j, rows[i-1][j-1] - '0');
    setbit(i, s + 1, b[i-1]);
  }
  gauss_course(s);
}

// 输出：need（或 -1 表示 Cannot Determine）、解向量
pair<int, vector<int>> course_solve(vector<string>& rows, vector<int>& b, int nn, int mm) {
  build_course(rows, b, nn, mm);
  int sign = 1;
  for (int i = 1; i <= n; i++) if (getbit(i, i) == 0) { sign = 0; break; }
  if (sign == 0) return {-1, {}};
  vector<int> x(n+1);
  for (int i = 1; i <= n; i++) x[i] = getbit(i, s + 1);
  return {need, x};
}

// ===== 参考版：前 k 条记录标准消元，判满秩；取最早 k = 参考 need =====
// 与课程一致：找主元 j 从 i..k，只对前 n 行消元
int matref[2010][2010];
int ref_solve(vector<string>& rows, vector<int>& b, int nn, int mm, int k, vector<int>& xout) {
  n = nn;
  memset(matref, 0, sizeof(matref));
  for (int i = 1; i <= k; i++) {
    for (int j = 1; j <= n; j++) matref[i][j] = rows[i-1][j-1] - '0';
    matref[i][n+1] = b[i-1];
  }
  for (int i = 1; i <= n; i++) {
    int p = -1;
    for (int j = i; j <= k; j++) if (matref[j][i]) { p = j; break; }
    if (p == -1) return 0;  // 该列无主元 -> 不满秩
    for (int j = 1; j <= n+1; j++) swap(matref[i][j], matref[p][j]);
    for (int j = 1; j <= k; j++) {
      if (j != i && matref[j][i]) {
        for (int c = i; c <= n+1; c++) matref[j][c] ^= matref[i][c];
      }
    }
  }
  // 唯一解
  for (int i = 1; i <= n; i++) xout[i] = matref[i][n+1];
  return 1;
}

// ===== 随机生成：n 条满秩基础记录 + 冗余，随机打乱 =====
mt19937 rng(20261016);

int main() {
  int tested = 0;
  while (tested < 3000) {
    n = 2 + rng() % 7;            // 2..8
    m = n + rng() % 5;            // n..n+4
    // 随机满秩 A（n x n）逐行生成保证独立
    int A[20][20];
    vector<int> piv(n+1, 0);
    memset(A, 0, sizeof(A));
    // 简化：用「上三角 + 行变换」构造可逆——直接随机尝试，用高斯消元检查秩
    for (int r = 1; r <= n; r++) {
      // 随机行，保证与已有行线性无关（检查）
      while (true) {
        for (int c = 1; c <= n; c++) A[r][c] = rng() & 1;
        // 检查 rank（对前 r 行）
        int B[20][20]; memset(B, 0, sizeof(B));
        for (int i = 1; i <= r; i++) for (int c = 1; c <= n; c++) B[i][c] = A[i][c];
        int rk = 0;
        for (int c = 1; c <= n && rk < r; c++) {
          int p = -1;
          for (int i = rk+1; i <= r; i++) if (B[i][c]) { p = i; break; }
          if (p == -1) continue;
          rk++;
          for (int j = 1; j <= n; j++) swap(B[rk][j], B[p][j]);
          for (int i = rk+1; i <= r; i++) if (B[i][c]) for (int j = c; j <= n; j++) B[i][j] ^= B[rk][j];
        }
        if (rk == r) break;
      }
    }
    // 随机解 x
    vector<int> x(n+1);
    for (int i = 1; i <= n; i++) x[i] = rng() & 1;
    vector<int> bx(n+1);
    for (int i = 1; i <= n; i++) {
      int t = 0;
      for (int j = 1; j <= n; j++) if (A[i][j] && x[j]) t ^= 1;
      bx[i] = t;
    }
    // 全部 m 条记录：前 n 条是基础记录，后 m-n 条是冗余（取 A 行的非零线性组合，
    // 常数取对应 bx 组合，保证与 x 一致；组合行是 A 行空间的元素，不补秩）
    vector<string> rows(m);
    vector<int> b(m);
    for (int i = 1; i <= n; i++) {
      string r;
      for (int j = 1; j <= n; j++) r += char('0' + A[i][j]);
      rows[i-1] = r;
      b[i-1] = bx[i];
    }
    for (int i = n+1; i <= m; i++) {
      // v[j]=1 表示取第 j 条基础记录，冗余行 = 这些行的异或，常数 = bx 对应异或
      vector<int> v(n+1);
      int nonz = 0;
      for (int j = 1; j <= n; j++) { v[j] = rng() & 1; nonz += v[j]; }
      if (nonz == 0 || nonz == n) { i--; continue; }  // 零行或全取（=所有行异或可能非零，也合法，但跳过简单化）
      string r; int t = 0;
      for (int c = 1; c <= n; c++) {
        int bit = 0;
        for (int j = 1; j <= n; j++) if (v[j] && A[j][c]) bit ^= 1;
        r += char('0' + bit);
      }
      for (int j = 1; j <= n; j++) if (v[j]) t ^= bx[j];
      rows[i-1] = r;
      b[i-1] = t;
    }
    // 随机打乱记录顺序
    for (int i = m-1; i > 0; i--) {
      int j = rng() % (i+1);
      swap(rows[i], rows[j]); swap(b[i], b[j]);
    }
    // 课程解
    auto cs = course_solve(rows, b, n, m);
    if (cs.first == -1) {
      printf("ERR: 课程说 Cannot Determine，但构造保证满秩\n"); return 1;
    }
    // 参考 need：最早前缀满秩
    int refneed = -1;
    for (int k = 1; k <= m; k++) {
      vector<int> xx(n+1);
      if (ref_solve(rows, b, n, m, k, xx)) { refneed = k; break; }
    }
    if (refneed == -1) { printf("ERR: 参考找不到满秩前缀\n"); return 1; }
    if (cs.first != refneed) {
      printf("MISMATCH-need n=%d m=%d course=%d ref=%d\n", n, m, cs.first, refneed);
      for (int i = 0; i < m; i++) printf("%s %d\n", rows[i].c_str(), b[i]);
      return 1;
    }
    // 参考解（用前 refneed 条）
    vector<int> xx(n+1);
    if (!ref_solve(rows, b, n, m, refneed, xx)) { printf("ERR ref solve\n"); return 1; }
    for (int i = 1; i <= n; i++) {
      if (cs.second[i] != xx[i] || cs.second[i] != x[i]) {
        printf("MISMATCH-x n=%d m=%d i=%d course=%d ref=%d true=%d\n", n, m, i, cs.second[i], xx[i], x[i]);
        for (int r = 0; r < m; r++) printf("%s %d\n", rows[r].c_str(), b[r]);
        return 1;
      }
    }
    tested++;
  }
  printf("t3 OK 3000 groups\n");

  // 固定例 1：PPT 例子（n=3 唯一解）

    vector<string> rows = {"110", "101", "011"};  // 全 1 常数的 x1^x2=1, x1^x3=1, x2^x3=1 -> 矛盾？
    vector<int> b = {1, 1, 1};

  }

           cs.first, cs.second[1], cs.second[2], cs.second[3]);
  }
  // 固定例 3：冗余让 need 变小——第 3 条记录能定 x3
  {
    n = 3; m = 3;
    vector<string> rows = {"110", "101", "001"};
    vector<int> b = {1, 1, 1};  // x1^x2=1, x1^x3=1, x3=1 -> x1=0,x2=1,x3=1 唯一
    auto cs = course_solve(rows, b, n, m);
    printf("fix3 n=3 need=%d x=(%d,%d,%d) (期望 need=3 x=(0,1,1))\n",
           cs.first, cs.second[1], cs.second[2], cs.second[3]);
  }
  // 固定例 4：need 用不到最后一条（4 条记录，前 3 条满秩）
  {
    n = 3; m = 4;
    vector<string> rows = {"100", "010", "001", "111"};
    vector<int> b = {1, 0, 1, 0};  // x1=1,x2=0,x3=1；第 4 条冗余 1^0^1=0 ✓
    auto cs = course_solve(rows, b, n, m);

           cs.first, cs.second[1], cs.second[2], cs.second[3]);

#include <bits/stdc++.h>
using namespace std;
// ShowDetails 4 例：验证课程 gauss 的结论分类
// 结论判定：遍历 i=1..n，对角线全 1 且无矛盾 -> 唯一解；有 0 行常数为 1 -> 矛盾；否则多解
int mat[101][101];
int n;
void gauss(int gn){
  for(int i=1;i<=gn;i++){
    for(int j=1;j<=gn;j++){
      if(j<i && mat[j][j]==1) continue;
      if(mat[j][i]==1){ for(int k=1;k<=gn+1;k++) swap(mat[i][k],mat[j][k]); break; }
    }
    if(mat[i][i]==1){
      for(int j=1;j<=gn;j++){
        if(i!=j && mat[j][i]==1){ for(int k=i;k<=gn+1;k++) mat[j][k]^=mat[i][k]; }
      }
    }
  }
}
// 结论：1 唯一解 / -1 矛盾 / 0 多解
int classify(){
  // 严格：矛盾优先（0 行 1 常数）；否则主元行数 == n 且无 0 行 -> 唯一解；否则多解
  for(int i=1;i<=n;i++){
    int allzero=1;
    for(int j=1;j<=n;j++) if(mat[i][j]) { allzero=0; break; }
    if(allzero && mat[i][n+1]) return -1;
  }
  int piv=0;
  for(int i=1;i<=n;i++) if(mat[i][i]) piv++;
  if(piv==n) return 1;
  return 0;
}
void dump(){
  for(int i=1;i<=n;i++){ for(int j=1;j<=n+1;j++) printf("%d ",mat[i][j]); printf("\n"); }
}
int main(){
  // 例1 唯一解
  n=4; memset(mat,0,sizeof(mat));
  mat[1][1]=1;mat[1][2]=1;mat[1][3]=1;mat[1][5]=0;

  mat[4][3]=1;mat[4][4]=1;mat[4][5]=0;
  gauss(4);
  printf("例1 唯一解: 结论=%d (期望1)\n", classify());
