欧拉路径与相关题目
左程云
通俗理解就是一笔画，每条边必须经过一次，必须经过所有的边，所形成的路起点和终点是同一个点叫欧拉回路（欧拉图），起点和终点不是同一个点叫欧拉路径（半欧拉图）有向图存在欧拉回路的条件1，忽略入度和出度都为0的点后，剩下每个点满足，入度 = 出度2，剩余图是强连通的（任意两点可以互相到达）构造性的证明1，从任意点s出发，如果遇到出边就选一条继续走，最终走到无法再走，一定会回到s，认为生成了主线2，如果有剩余的边没有走到，那么该边必然可被主线上的某点x找到，因为图是强连通的3，从点x出发，继续仿照步骤1，继续走没有走过的边，最终一定回到x，认为生成了支线4，把x走出的支线，插入到主线里，可以拼成一个欧拉回路5，主线不停插入支线，最终可以得到整个欧拉回路，这就是Hierholzer算法

#include <bits/stdc++.h>

using namespace std;

struct Edge {
    int u, v;
};

bool EdgeCmp(Edge e1, Edge e2) {
    if (e1.u != e2.u) {
        return e1.u < e2.u;
    }
    return e1.v < e2.v;
}

const int MAXN = 100001;
const int MAXM = 200002;
int n, m;
Edge edgeArr[MAXM];

int head[MAXN];
int nxt[MAXM];
int to[MAXM];
int cntg;

int cur[MAXN];
int outDeg[MAXN];
int inDeg[MAXN];

int path[MAXM];
int cntp;

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;
}

void connect() {
    sort(edgeArr + 1, edgeArr + m + 1, EdgeCmp);
    for (int l = 1, r = 1; l <= m; l = ++r) {
        while (r + 1 <= m && edgeArr[l].u == edgeArr[r + 1].u) {
            r++;
        }
        for (int i = r, u, v; i >= l; i--) {
            u = edgeArr[i].u;
            v = edgeArr[i].v;
            outDeg[u]++;
            inDeg[v]++;
            addEdge(u, v);
        }
    }
    for (int i = 1; i <= n; i++) {
        cur[i] = head[i];
    }
}

int directedStart() {
    int start = -1, end = -1;
    for (int i = 1; i <= n; i++) {
        int v = outDeg[i] - inDeg[i];
        if (v < -1 || v > 1 || (v == 1 && start != -1) || (v == -1 && end != -1)) {
            return -1;
        }
        if (v == 1) {
            start = i;
        }
        if (v == -1) {
            end = i;
        }
    }
    if ((start == -1) ^ (end == -1)) {
        return -1;
    }
    if (start != -1) {
        return start;
    }
    for (int i = 1; i <= n; i++) {
        if (outDeg[i] > 0) {
            return i;
        }
    }
    return -1;
}

void euler(int u) {
    for (int e = cur[u]; e > 0; e = cur[u]) {
        cur[u] = nxt[e];
        euler(to[e]);
    }
    path[++cntp] = u;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m;
    for (int i = 1; i <= m; i++) {
        cin >> edgeArr[i].u >> edgeArr[i].v;
    }
    connect();
    int start = directedStart();
    if (start == -1) {
        cout << "No\n";
    } else {
        euler(start);
        if (cntp != m + 1) {
            cout << "No\n";
        } else {
            for (int i = cntp; i >= 1; i--) {
                cout << path[i] << " ";
            }
            cout << "\n";
        }
    }
    return 0;
}

#include <bits/stdc++.h>

using namespace std;

struct Edge {
    int u, v, id;
};

bool EdgeCmp(Edge e1, Edge e2) {
    if (e1.u != e2.u) {
        return e1.u < e2.u;
    }
    return e1.v < e2.v;
}

const int MAXN = 501;
const int MAXM = 2001;
int n = 500, m, k;
Edge edgeArr[MAXM << 1];

int head[MAXN];
int nxt[MAXM << 1];
int to[MAXM << 1];
int eid[MAXM << 1];
int cntg;

int cur[MAXN];
int deg[MAXN];
bool vis[MAXM];

int path[MAXM];
int cntp;

void addEdge(int u, int v, int id) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    eid[cntg] = id;
    head[u] = cntg;
}

void connect() {
    sort(edgeArr + 1, edgeArr + k + 1, EdgeCmp);
    for (int l = 1, r = 1; l <= k; l = ++r) {
        while (r + 1 <= k && edgeArr[l].u == edgeArr[r + 1].u) {
            r++;
        }
        for (int i = r; i >= l; i--) {
            deg[edgeArr[i].u]++;
            addEdge(edgeArr[i].u, edgeArr[i].v, edgeArr[i].id);
        }
    }
    for (int i = 1; i <= n; i++) {
        cur[i] = head[i];
    }
}

int undirectedStart() {
    int odd = 0;
    for (int i = 1; i <= n; i++) {
        if ((deg[i] & 1) == 1) {
            odd++;
        }
    }
    if (odd != 0 && odd != 2) {
        return -1;
    }
    for (int i = 1; i <= n; i++) {
        if (odd == 0 && deg[i] > 0) {
            return i;
        }
        if (odd == 2 && (deg[i] & 1) == 1) {
            return i;
        }
    }
    return -1;
}

void euler(int u) {
    for (int e = cur[u]; e > 0; e = cur[u]) {
        cur[u] = nxt[e];
        if (!vis[eid[e]]) {
            vis[eid[e]] = true;
            euler(to[e]);
        }
    }
    path[++cntp] = u;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> m;
    k = 0;
    for (int i = 1, u, v; i <= m; i++) {
        cin >> u >> v;
        edgeArr[++k] = { u, v, i };
        edgeArr[++k] = { v, u, i };
    }
    connect();
    int start = undirectedStart();
    if (start == -1) {
        cout << "No\n";
    } else {
        euler(start);
        if (cntp != m + 1) {
            cout << "No\n";
        } else {
            for (int i = cntp; i >= 1; i--) {
                cout << path[i] << "\n";
            }
        }
    }
    return 0;
}

    if (a[i] != a[j]) {
        return a[i] < a[j];
    }
    return str[i] < str[j];
}

void addEdge(int u, int v, string w) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    weight[cntg] = w;
    head[u] = cntg;
}

int startNode(const string &str) {
    return (int)(str[0] - 'a' + 1);
}

int endNode(const string &str) {
    return (int)(str[(int)str.size() - 1] - 'a' + 1);
}

void connect() {
    sort(eidArr + 1, eidArr + m + 1, EdgeCmp);

        while (r + 1 <= m && a[eidArr[l]] == a[eidArr[r + 1]]) {
            r++;
        }
        for (int i = r, u, v; i >= l; i--) {
            u = a[eidArr[i]];
            v = b[eidArr[i]];
            outDeg[u]++;
            inDeg[v]++;
            addEdge(u, v, str[eidArr[i]]);
        }
    }
    for (int i = 1; i <= n; i++) {
        cur[i] = head[i];
    }
}

int directedStart() {

bool EdgeCmp(Edge e1, Edge e2) {
    if (e1.w != e2.w) {
        return e1.w < e2.w;
    }
    if (e1.u != e2.u) {
        return e1.u < e2.u;
    }
    return e1.v < e2.v;
}

int dist(int x, int y) {
    return abs(x - y);
}

int find(int fa[], int x) {
    if (x != fa[x]) {
        fa[x] = find(fa, fa[x]);
    }
    return fa[x];
}

void Union(int fa[], int x, int y) {
    int fx = find(fa, x);
    int fy = find(fa, y);
    if (fx != fy) {
        fa[fx] = fy;
    }
}

ll kruskal() {
    ll cost = 0;
    int edgeCnt = cnt - 1;
    sort(edgeArr + 1, edgeArr + edgeCnt + 1, EdgeCmp);
    for (int i = 1; i <= edgeCnt; i++) {
        int u = edgeArr[i].u;
        int v = edgeArr[i].v;
        int w = edgeArr[i].w;
        if (find(fa2, u) != find(fa2, v)) {
            cost += 1LL * w * 2;
            Union(fa2, u, v);
        }
    }
    return cost;
}

    cnt = 0;
    for (int x : nodeSet) {
        nodeArr[++cnt] = x;
    }
    for (int i = 1; i <= n; i++) {
        fa2[i] = find(fa1, i);
        deg2[i] = deg1[i];
    }
    ll ans = sum;
    deg2[start]++;
    deg2[end]++;
    Union(fa2, start, end);
    for (int i = 1, p = 0; i <= cnt; i++) {
        if ((deg2[nodeArr[i]] & 1) == 1) {
            if (p == 0) {
                p = i;
            } else {
                ans += dist(nodeArr[p], nodeArr[i]);
                while (p < i) {
                    Union(fa2, nodeArr[p++], nodeArr[i]);
                }
                p = 0;
            }
        }
    }
    for (int i = 1; i <= cnt - 1; i++) {
        edgeArr[i].u = nodeArr[i];
        edgeArr[i].v = nodeArr[i + 1];
        edgeArr[i].w = nodeArr[i + 1] - nodeArr[i];
    }

    return ans;
}

        markId[marks[i].second][marks[i].first] = i;
    }

    vector<vector<int>> dp(K, vector<int>(n + 1, INF));
    dp[0][s] = 0;
    // Dijkstra on (mask, v)
    using St = tuple<int,int,int>; // d, mask, v
    priority_queue<St, vector<St>, greater<St>> pq;

    while (!pq.empty()) {

                dp[nm][u] = nd;
                pq.push({nd, nm, u});
            }
        }
    }
    int FULL = K - 1;

        int best = dp[FULL][i];
        if (i > 1) cout << " ";
        cout << best;
    }

   }

       int l = 1, r = n, ans = 0;
       while (l <= r) {
           int mid = (l + r) >> 1;
           if (sortv[mid] >= num) {
               ans = mid;
               r = mid - 1;
           } else {
               l = mid + 1;
           }
       }
       return ans;
   }

       int len = 0;
       for (int i = 0; i < m; i++) {
           sortv[++len] = (*pair)[i][0];
           sortv[++len] = (*pair)[i][1];
       }
       sort(sortv + 1, sortv + len + 1);

       for (int i = 2; i <= len; i++) {

               sortv[++n] = sortv[i];
           }
       }
       cntg = cntp = 0;
       for (int i = 1; i <= n; i++) {
           head[i] = outDeg[i] = inDeg[i] = 0;
       }
   }

           u = kth((*pair)[i][0]);

           cur[i] = head[i];
       }
   }

               return i;
           }
       }

   void push(int u, int e) {
       sta[stacksize][0] = u;
       sta[stacksize][1] = e;
       stacksize++;
   }

   void pop() {
       stacksize--;
       u = sta[stacksize][0];
       e = sta[stacksize][1];
   }

   void prepare(int len, int num) {
       n = len;
       k = num;
       m = 1;
       for (int i = 1; i <= n - 1; i++) {
           m *= k;

# 题10 生成器：保证有欧拉回路的图 + 可能无解的序列混合
import random, sys
seed = int(sys.argv[1])
r = random.Random(seed)
for _ in range(2000):
    n = r.randint(2, 5)
    m = r.randint(3, 6)
    seen = set(); edges = []
    for _ in range(m):
        u = r.randint(1, n); v = r.randint(1, n)
        if u == v or (u, v) in seen:
            continue
        seen.add((u, v)); edges.append((u, v))
    if len(edges) < 3: continue
    # 欧拉回路条件：出度=入度、从1可达（含孤立点除外——所有出现点都可达）
    outd = [0]* (n+1); ind = [0]*(n+1)
    for u, v in edges: outd[u] += 1; ind[v] += 1
    if any(outd[i] != ind[i] for i in range(1, n+1)): continue
    # 连通性（有边点都从 1 可达）
    g = [[] for _ in range(n+1)]
    for u, v in edges: g[u].append(v)
    seen2 = {1}
    stack = [1]
    while stack:
        x = stack.pop()
        for y in g[x]:
            if y not in seen2: seen2.add(y); stack.append(y)
    if any(outd[i] > 0 for i in range(1, n+1) if i not in seen2): continue
    # 建一条已知欧拉回路（Hierholzer）用于取子段
    gg = [list(es) for es in [[] for _ in range(n+1)]]
    for u, v in edges: gg[u].append(v)
    # 课程风格 Hierholzer（迭代）
    for i in range(1, n+1): gg[i].sort()
    path = []
    cur = [0]*(n+1)
    stack = [1]
    while stack:
        x = stack[-1]
        if cur[x] < len(gg[x]):
            y = gg[x][cur[x]]; cur[x] += 1
            stack.append(y)
        else:
            path.append(x); stack.pop()
    circuit = list(reversed(path))
    # 序列：一半取自回路子段（必有解），一半随机（可能无解）
    appear = sorted({x for u, v in edges for x in (u, v)})
    seqs = []
    for _ in range(2):
        k = r.randint(1, 3)
        if r.random() < 0.5:
            st = r.randint(0, len(circuit) - k)
            seqs.append(circuit[st:st+k])
        else:
            seqs.append([r.choice(appear) for _ in range(k)])
    break
else:
    raise SystemExit('未找到欧拉图')
print(n, len(edges))
for u, v in edges: print(u, v)
print(len(seqs))
for sq in seqs:
    print(len(sq), *sq)
