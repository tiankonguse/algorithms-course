#include <bits/stdc++.h>

using namespace std;

const int MAXN = 20001;
const int MAXM = 100001;
int n, m;

int head[MAXN];
int nxt[MAXM << 1];
int to[MAXM << 1];
int cntg;

int dfn[MAXN];
int low[MAXN];
int cntd;

bool cutVertex[MAXN];

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;
}

void tarjan(int u, bool root) {
    dfn[u] = low[u] = ++cntd;
    int son = 0;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (dfn[v] == 0) {
            son++;
            tarjan(v, false);
            low[u] = min(low[u], low[v]);
            if (low[v] >= dfn[u]) {
                if (!root || son >= 2) {
                    cutVertex[u] = true;
                }
            }
        } else {
            low[u] = min(low[u], dfn[v]);
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m;
    for (int i = 1, u, v; i <= m; i++) {
        cin >> u >> v;
        addEdge(u, v);
        addEdge(v, u);
    }
    for (int i = 1; i <= n; i++) {
        if (dfn[i] == 0) {
            tarjan(i, true);
        }
    }
    int cutCnt = 0;
    for (int i = 1; i <= n; i++) {
        if (cutVertex[i]) {
            cutCnt++;
        }
    }
    cout << cutCnt << "\n";
    for (int i = 1; i <= n; i++) {
        if (cutVertex[i]) {
            cout << i << " ";
        }
    }
    cout << "\n";
    return 0;
}

#include <iostream>
#include <algorithm>

using namespace std;

const int MAXN = 10001;
const int MAXM = 20001;
int n, m;

int head[MAXN];
int nxt[MAXM << 1];
int to[MAXM << 1];
int cntg;

int dfn[MAXN];
int low[MAXN];
int cntd;

int block, ans;

void prepare() {
    cntg = cntd = block = ans = 0;
    for (int i = 1; i <= n; i++) {
        head[i] = dfn[i] = low[i] = 0;
    }
}

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;
}

void tarjan(int u, bool root) {
    dfn[u] = low[u] = ++cntd;
    int son = 0, cut = 0;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (dfn[v] == 0) {
            son++;
            tarjan(v, false);
            low[u] = min(low[u], low[v]);
            if (low[v] >= dfn[u] && !root) {
                cut++;
            }
        } else {
            low[u] = min(low[u], dfn[v]);
        }
    }
    if (root) {
        ans = max(ans, son);
    } else {
        ans = max(ans, cut + 1);
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin >> n >> m;
    while (n != 0 || m != 0) {
        prepare();
        for (int i = 1, u, v; i <= m; i++) {
            cin >> u >> v;
            u++;
            v++;
            addEdge(u, v);
            addEdge(v, u);
        }
        for (int i = 1; i <= n; i++) {
            if (dfn[i] == 0) {
                block++;
                tarjan(i, true);
            }
        }
        cout << (block + (ans - 1)) << "\n";
        cin >> n >> m;
    }
    return 0;
}

int cntd;

    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;

void tarjan(int u) {
    dfn[u] = low[u] = ++cntd;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (dfn[v] == 0) {
            tarjan(v);
            low[u] = min(low[u], low[v]);
            if (low[v] >= dfn[u] && u != a && u != b && dfn[b] >= dfn[v]) {
                isKey[u] = true;
            }
        } else {
            low[u] = min(low[u], dfn[v]);
        }
    }
}

    int u, v;
    cin >> u >> v;
    while (u != 0 || v != 0) {
        addEdge(u, v);
        addEdge(v, u);
        cin >> u >> v;
    }
    cin >> a >> b;
    tarjan(a);
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (isKey[i]) {
            ans = i;
            break;
        }
    }
    if (ans == 0) {
        cout << "No solution" << "\n";
    } else {
        cout << ans << "\n";
    }
    return 0;

int to[MAXM << 1];

    nxt[++cntg] = head[u];
    to[cntg] = v;

                vbccl[vbccCnt] = idx;

                do {
                    pop = sta[top--];
                    vbccSiz[vbccCnt]++;
                    vbccArr[++idx] = pop;

            }

        }

            addEdge(u, v);

#include <bits/stdc++.h>

using namespace std;

const int MAXN = 50001;
const int MAXM = 300001;
int n, m;

int head[MAXN];
int nxt[MAXM << 1];
int to[MAXM << 1];
int cntg;

int dfn[MAXN];
int low[MAXN];
int cntd;

int sta[MAXN];
int top;

vector<vector<int>> vbccArr;

bool VbccCmp(const vector<int> &o1, const vector<int> &o2) {
    int size = min(o1.size(), o2.size());
    for (int i = 0; i < size; i++) {
        if (o1[i] != o2[i]) {
            return o1[i] < o2[i];
        }
    }
    return o1.size() < o2.size();
}

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;
}

void tarjan(int u) {
    dfn[u] = low[u] = ++cntd;
    sta[++top] = u;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (dfn[v] == 0) {
            tarjan(v);
            low[u] = min(low[u], low[v]);
            if (low[v] == dfn[u]) {
                vector<int> list;
                list.push_back(u);
                int pop;
                do {
                    pop = sta[top--];
                    list.push_back(pop);
                } while (pop != v);
                vbccArr.push_back(list);
            }
        } else {
            low[u] = min(low[u], dfn[v]);
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m;
    for (int i = 1, u, v; i <= m; i++) {
        cin >> u >> v;
        addEdge(u, v);
        addEdge(v, u);
    }
    for (int i = 1; i <= n; i++) {
        if (dfn[i] == 0 && head[i] > 0) {
            tarjan(i);
        }
    }
    cout << vbccArr.size() << "\n";
    for (int i = 0; i < vbccArr.size(); i++) {
        sort(vbccArr[i].begin(), vbccArr[i].end());
    }
    sort(vbccArr.begin(), vbccArr.end(), VbccCmp);
    for (int i = 0; i < vbccArr.size(); i++) {
        for (int node : vbccArr[i]) {
            cout << node << " ";
        }
        cout << "\n";
    }
    return 0;
}

code/Code06.cpp

class Solution {
public:
    static const int MAXN = 100001;
    static const int MAXM = 100001;
    int n, m;

    int head[MAXN];
    int nxt[MAXM << 1];
    int to[MAXM << 1];
    int cntg;

    int dfn[MAXN];
    int low[MAXN];
    int cntd;

    int sta[MAXN];
    int top;

    bool cutVertex[MAXN];
    vector<vector<int>> vbccArr;

    void prepare() {
        cntg = cntd = top = 0;
        for (int i = 1; i <= n; i++) {
            head[i] = dfn[i] = low[i] = 0;
            cutVertex[i] = false;
        }
        vbccArr.clear();
    }

    void addEdge(int u, int v) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        head[u] = cntg;
    }

    void tarjan(int u, bool root) {
        dfn[u] = low[u] = ++cntd;
        sta[++top] = u;
        int son = 0;
        for (int e = head[u]; e > 0; e = nxt[e]) {
            int v = to[e];
            if (dfn[v] == 0) {
                son++;
                tarjan(v, false);
                low[u] = min(low[u], low[v]);
                if (low[v] >= dfn[u]) {
                    if (!root || son >= 2) {
                        cutVertex[u] = true;
                    }
                    vector<int> list;
                    list.push_back(u);
                    int pop;
                    do {
                        pop = sta[top--];
                        list.push_back(pop);
                    } while (pop != v);
                    vbccArr.push_back(list);
                }
            } else {
                low[u] = min(low[u], dfn[v]);
            }
        }
    }

    long long minimumCost(vector<int>& cost, vector<vector<int>>& roads) {
        n = cost.size();
        m = roads.size();
        prepare();
        for (int i = 0; i < m; i++) {
            int u = roads[i][0] + 1;
            int v = roads[i][1] + 1;
            addEdge(u, v);
            addEdge(v, u);
        }
        tarjan(1, true);
        long long ans = 0;
        if (vbccArr.size() == 1) {
            ans = LLONG_MAX;
            for (int i = 0; i < n; i++) {
                ans = min(ans, (long long)cost[i]);
            }
        } else {
            int maxVal = INT_MIN;
            for (auto &vbcc : vbccArr) {
                int cut = 0;
                int val = INT_MAX;
                for (int cur : vbcc) {
                    if (cutVertex[cur]) {
                        cut++;
                    } else {
                        val = min(val, cost[cur - 1]);
                    }
                }
                if (cut == 1) {
                    ans += val;
                    maxVal = max(maxVal, val);
                }
            }
            ans -= maxVal;
        }
        return ans;
    }
};

    void prepare() {
        cntg = cntd = top = 0;
        for (int i = 1; i <= n; i++) {
            head[i] = dfn[i] = low[i] = 0;
            cutVertex[i] = false;
        }
        vbccArr.clear();
    }

    void addEdge(int u, int v) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        head[u] = cntg;
    }

    void tarjan(int u, bool root) {
        dfn[u] = low[u] = ++cntd;
        sta[++top] = u;
        int son = 0;
        for (int e = head[u]; e > 0; e = nxt[e]) {
            int v = to[e];
            if (dfn[v] == 0) {
                son++;
                tarjan(v, false);
                low[u] = min(low[u], low[v]);
                if (low[v] >= dfn[u]) {
                    if (!root || son >= 2) {
                        cutVertex[u] = true;
                    }
                    vector<int> list;
                    list.push_back(u);
                    int pop;
                    do {
                        pop = sta[top--];
                        list.push_back(pop);
                    } while (pop != v);
                    vbccArr.push_back(list);
                }
            } else {
                low[u] = min(low[u], dfn[v]);
            }
        }
    }

    long long minimumCost(vector<int>& cost, vector<vector<int>>& roads) {
        n = cost.size();
        m = roads.size();
        prepare();
        for (int i = 0; i < m; i++) {
            int u = roads[i][0] + 1;
            int v = roads[i][1] + 1;
            addEdge(u, v);
            addEdge(v, u);
        }
        tarjan(1, true);
        long long ans = 0;
        if (vbccArr.size() == 1) {
            ans = LLONG_MAX;
            for (int i = 0; i < n; i++) {
                ans = min(ans, (long long)cost[i]);
            }
        } else {
            int maxVal = INT_MIN;
            for (auto &vbcc : vbccArr) {
                int cut = 0;
                int val = INT_MAX;
                for (int cur : vbcc) {
                    if (cutVertex[cur]) {
                        cut++;
                    } else {
                        val = min(val, cost[cur - 1]);
                    }
                }
                if (cut == 1) {
                    ans += val;
                    maxVal = max(maxVal, val);
                }
            }
            ans -= maxVal;
        }
        return ans;
    }
};

// ============ 点双通用：极大无割点连通点集 + 孤立点 ============
vector<vector<int>> vbccBrute(const vector<pair<int,int>>& es, int n, bool with_isolated) {
    vector<vector<int>> g(n + 1);
    vector<vector<int>> adj(n + 1);
    for (auto& e : es) {
        if (e.first == e.second) continue; // 忽略自环
        adj[e.first].push_back(e.second);
        adj[e.second].push_back(e.first);
    }
    int full = 1 << n;
    vector<char> good(full, 0);
    for (int S = 1; S < full; S++) {
        vector<int> nodes;
        for (int i = 0; i < n; i++) if (S >> i & 1) nodes.push_back(i + 1);
        int sz = nodes.size();
        if (sz == 1) {
            // 孤立点（无边）才成单块
            if (adj[nodes[0]].empty()) good[S] = 1;
            continue;
        }
        // 导出子图连通
        auto inS = [&](int x) { return S >> (x - 1) & 1; };
        int start = nodes[0];
        queue<int> q; q.push(start);
        vector<char> vis(n + 1, 0); vis[start] = 1;
        int cnt = 1;
        while (!q.empty()) {
            int x = q.front(); q.pop();
            for (int y : adj[x]) if (inS(y) && !vis[y]) { vis[y] = 1; cnt++; q.push(y); }
        }
        if (cnt != sz) continue;
        // 无割点：删任意点仍连通
        bool ok = true;
        for (int v : nodes) {
            vector<char> vis2(n + 1, 0);
            int st = (v == start) ? nodes[1] : start;
            queue<int> q2; q2.push(st); vis2[st] = 1;
            int c2 = 1;
            while (!q2.empty()) {
                int x = q2.front(); q2.pop();
                for (int y : adj[x]) if (inS(y) && y != v && !vis2[y]) { vis2[y] = 1; c2++; q2.push(y); }
            }
            if (c2 != sz - 1) { ok = false; break; }
        }
        if (ok) good[S] = 1;
    }
    // 极大化：按大小降序取，交集 ≥2 跳过
    vector<int> order;
    for (int S = 1; S < full; S++) if (good[S]) order.push_back(S);
    sort(order.begin(), order.end(), [](int a, int b) { return __builtin_popcount(a) > __builtin_popcount(b); });
    vector<vector<int>> res;
    for (int S : order) {
        vector<int> nodes;
        for (int i = 0; i < n; i++) if (S >> i & 1) nodes.push_back(i + 1);
        bool merged = false;
        for (auto& blk : res) {
            int inter = 0;
            for (int x : blk) if (S >> (x - 1) & 1) inter++;
            if (inter >= 2) { merged = true; break; }
        }
        if (!merged) res.push_back(nodes);
    }
    if (!with_isolated) {
        vector<vector<int>> tmp;
        for (auto& blk : res) if (blk.size() > 1 || !adj[blk[0]].empty()) tmp.push_back(blk);
        res = tmp;
    }
    return res;
}

        }
        return cnt == remain;
    };

        vector<char> taken(n, 0);

            bool free = false;
            for (int y : adj[x]) if (taken[y]) { free = true; break; }
            if (!free) tot += cost[x];
            taken[x] = 1;
            remain--;
            if (!unTakenConnected(taken, remain)) { ok = false; break; }
        }
        if (ok) best = min(best, tot);
    } while (next_permutation(perm.begin(), perm.end()));

// ============ 题8 圆桌骑士：存在经过 u 的奇环 ============
void solve8() {
    while (cin >> n >> m) {
        if (n == 0 && m == 0) break;
        vector<vector<char>> hate(n + 1, vector<char>(n + 1, 0));
        for (int i = 0; i < m; i++) {
            int u, v; cin >> u >> v;
            hate[u][v] = hate[v][u] = 1;
        }
        vector<vector<int>> adj(n + 1);
        for (int u = 1; u <= n; u++)
            for (int v = u + 1; v <= n; v++)
                if (!hate[u][v]) { adj[u].push_back(v); adj[v].push_back(u); }
        auto inCycle = [&](int u) {
            int full = 1 << n;
            for (int S = 1; S < full; S++) {
                if (!(S >> (u - 1) & 1)) continue;
                int sz = __builtin_popcount(S);
                if (sz < 3 || sz % 2 == 0) continue;
                // 导出子图每个点度 2 且连通 → 环
                bool ok = true;
                int deg0 = 0;
                for (int i = 1; i <= n; i++) {
                    if (!(S >> (i - 1) & 1)) continue;
                    int d = 0;
                    for (int j : adj[i]) if (S >> (j - 1) & 1) d++;
                    if (d != 2) { ok = false; break; }
                    deg0 = i;
                }
                if (!ok) continue;
                // 连通性（度 2 且边数 = 点数，无重边 → 每个分量是环；需单环）
                queue<int> q; q.push(deg0);
                vector<char> vis(n + 1, 0); vis[deg0] = 1;
                int c = 1;
                while (!q.empty()) {
                    int x = q.front(); q.pop();
                    for (int j : adj[x]) if ((S >> (j - 1) & 1) && !vis[j]) { vis[j] = 1; c++; q.push(j); }
                }
                if (c == sz) return true;
            }
            return false;
        };
        int ans = 0;
        for (int i = 1; i <= n; i++) if (!inCycle(i)) ans++;
        cout << ans << "\n";
        cin >> n >> m;
    }
}

def norm_blocks(t):
    lines = t.strip().split('\n')
    if not lines: return ''
    cnt = int(lines[0])
    blocks = []
    idx = 1
    for _ in range(cnt):
        size = int(lines[idx]); idx += 1
        nodes = sorted(int(x) for x in lines[idx].split()); idx += 1
        blocks.append(tuple(nodes))
    blocks.sort()
    return repr(blocks)

def norm5(t):
    lines = t.strip().split('\n')
    if not lines: return ''
    cnt = int(lines[0])
    blocks = []
    for i in range(1, cnt + 1):
        blocks.append(tuple(sorted(int(x) for x in lines[i].split())))
    blocks.sort()
    return repr(blocks)

import random, sys

def tree_edges(n, rnd):
    es = set()
    for i in range(2, n + 1):
        es.add((rnd.randint(1, i - 1), i))
    return es

def connected_edges(n, rnd):
    es = tree_edges(n, rnd)
    extra = rnd.randint(0, min(8, n * (n - 1) // 2 - (n - 1)))
    for _ in range(extra):
        u = rnd.randint(1, n); v = rnd.randint(1, n)
        if u != v:
            es.add((min(u, v), max(u, v)))
    return sorted(es)
