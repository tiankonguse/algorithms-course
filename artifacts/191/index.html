#include <bits/stdc++.h>

using namespace std;

const int MAXN = 500001;
const int MAXM = 2000001;
int n, m;

int head[MAXN];
int nxt[MAXM << 1];
int to[MAXM << 1];
int cntg;

int dfn[MAXN];
int low[MAXN];
int cntd;

bool cutEdge[MAXM];

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;
}

void tarjan(int u, int preEdge) {
    dfn[u] = low[u] = ++cntd;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        if ((e ^ 1) == preEdge) {
            continue;
        }
        int v = to[e];
        if (dfn[v] == 0) {
            tarjan(v, e);
            low[u] = min(low[u], low[v]);
            if (low[v] > dfn[u]) {
                cutEdge[e >> 1] = true;
            }
        } else {
            low[u] = min(low[u], dfn[v]);
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cntg = 1;
    cin >> n >> m;
    for (int i = 1, u, v; i <= m; i++) {
        cin >> u >> v;
        addEdge(u, v);
        addEdge(v, u);
    }
    for (int i = 1; i <= n; i++) {
        if (dfn[i] == 0) {
            tarjan(i, 0);
        }
    }
    int ansCnt = 0;
    for (int i = 1; i <= m; i++) {
        if (cutEdge[i]) {
            ansCnt++;
        }
    }
    cout << ansCnt << "\n";
    for (int i = 1; i <= m; i++) {
        if (cutEdge[i]) {
            cout << i << " ";
        }
    }
    cout << "\n";
    return 0;
}

class Solution {
public:
    static const int MAXN = 100001;
    static const int MAXM = 100001;
    int n, m;

    int a[MAXM];
    int b[MAXM];

    int head[MAXN];
    int nxt[MAXM << 1];
    int to[MAXM << 1];
    int cntg;

    int dfn[MAXN];
    int low[MAXN];
    int cntd;

    bool cutEdge[MAXM];

    void prepare() {
        cntg = 1;
        cntd = 0;
        for (int i = 1; i <= n; i++) {
            head[i] = dfn[i] = low[i] = 0;
        }
        for (int i = 1; i <= m; i++) {
            cutEdge[i] = false;
        }
    }

    void addEdge(int u, int v) {
        nxt[++cntg] = head[u];
        to[cntg] = v;
        head[u] = cntg;
    }

    void tarjan(int u, int preEdge) {
        dfn[u] = low[u] = ++cntd;
        for (int e = head[u]; e > 0; e = nxt[e]) {
            if ((e ^ 1) == preEdge) {
                continue;
            }
            int v = to[e];
            if (dfn[v] == 0) {
                tarjan(v, e);
                low[u] = min(low[u], low[v]);
                if (low[v] > dfn[u]) {
                    cutEdge[e >> 1] = true;
                }
            } else {
                low[u] = min(low[u], dfn[v]);
            }
        }
    }

    vector<vector<int>> criticalConnections(int nodeCnt, vector<vector<int>>& connections) {
        n = nodeCnt;
        m = (int)connections.size();
        prepare();
        for (int i = 1; i <= m; i++) {
            a[i] = connections[i - 1][0] + 1;
            b[i] = connections[i - 1][1] + 1;
            addEdge(a[i], b[i]);
            addEdge(b[i], a[i]);
        }
        tarjan(1, 0);
        vector<vector<int>> ans;
        for (int i = 1; i <= m; i++) {
            if (cutEdge[i]) {
                ans.push_back({a[i] - 1, b[i] - 1});
            }
        }
        return ans;
    }
};

int to[MAXM << 1];

int ebccArr[MAXN];

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;

void tarjan(int u, int preEdge) {
    dfn[u] = low[u] = ++cntd;
    sta[++top] = u;
    for (int e = head[u]; e > 0; e = nxt[e]) {

            tarjan(v, e);
            low[u] = min(low[u], low[v]);

            low[u] = min(low[u], dfn[v]);

    if (dfn[u] == low[u]) {
        ebccCnt++;
        ebccSiz[ebccCnt] = 0;
        ebccl[ebccCnt] = idx + 1;

        do {

            ebccSiz[ebccCnt]++;
            ebccArr[++idx] = pop;
        } while (pop != u);
        ebccr[ebccCnt] = idx;
    }
}

#include <iostream>
#include <algorithm>

using namespace std;

const int MAXN = 100001;
const int MAXM = 200001;
int t, n, m, q;
int a[MAXM];
int b[MAXM];

int head[MAXN];
int nxt[MAXM << 1];
int to[MAXM << 1];
int cntg;

int dfn[MAXN];
int low[MAXN];
int cntd;

int sta[MAXN];
int top;

int belong[MAXN];
int ebccCnt;

int up[MAXN];
int dep[MAXN];
int fa[MAXN];

void prepare() {
    cntg = 1;
    cntd = top = ebccCnt = 0;
    for (int i = 1; i <= n; i++) {
        head[i] = dfn[i] = low[i] = 0;
    }
}

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;
}

void tarjan(int u, int preEdge) {
    dfn[u] = low[u] = ++cntd;
    sta[++top] = u;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        if ((e ^ 1) == preEdge) {
            continue;
        }
        int v = to[e];
        if (dfn[v] == 0) {
            tarjan(v, e);
            low[u] = min(low[u], low[v]);
        } else {
            low[u] = min(low[u], dfn[v]);
        }
    }
    if (dfn[u] == low[u]) {
        ebccCnt++;
        int pop;
        do {
            pop = sta[top--];
            belong[pop] = ebccCnt;
        } while (pop != u);
    }
}

void condense() {
    cntg = 0;
    for (int i = 1; i <= ebccCnt; i++) {
        head[i] = 0;
    }
    for (int i = 1; i <= m; i++) {
        int ebcc1 = belong[a[i]];
        int ebcc2 = belong[b[i]];
        if (ebcc1 != ebcc2) {
            addEdge(ebcc1, ebcc2);
            addEdge(ebcc2, ebcc1);
        }
    }
}

void dfs(int u, int f) {
    dep[u] = dep[f] + 1;
    up[u] = f;
    fa[u] = u;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (v != f) {
            dfs(v, u);
        }
    }
}

int find(int i) {
    if (i != fa[i]) {
        fa[i] = find(fa[i]);
    }
    return fa[i];
}

void Union(int x, int y) {
    x = find(x);
    y = find(y);
    if (x != y) {
        if (dep[x] < dep[y]) {
            fa[y] = x;
        } else {
            fa[x] = y;
        }
    }
}

void link(int x, int y) {
    x = find(belong[x]);
    y = find(belong[y]);
    while (x != y) {
        if (dep[x] >= dep[y]) {
            Union(x, up[x]);
            x = find(x);
        } else {
            Union(y, up[y]);
            y = find(y);
        }
        ebccCnt--;
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    t = 0;
    cin >> n >> m;
    while (n != 0 || m != 0) {
        prepare();
        for (int i = 1; i <= m; i++) {
            cin >> a[i] >> b[i];
            addEdge(a[i], b[i]);
            addEdge(b[i], a[i]);
        }
        tarjan(1, 0);
        condense();
        dfs(1, 0);
        cout << "Case " << (++t) << ":\n";
        cin >> q;
        for (int i = 1, x, y; i <= q; i++) {
            cin >> x >> y;
            link(x, y);
            cout << ebccCnt - 1 << "\n";
        }
        cout << "\n";
        cin >> n >> m;
    }
    return 0;
}

int b[MAXM];

int head[MAXN];
int nxt[MAXM << 1];
int to[MAXM << 1];
int cntg;

int dfn[MAXN];
int low[MAXN];
int cntd;

int sta[MAXN];
int top;

int belong[MAXN];
int ebccCnt;

int degree[MAXN];

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;
}

void tarjan(int u, int preEdge) {
    dfn[u] = low[u] = ++cntd;
    sta[++top] = u;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        if ((e ^ 1) == preEdge) {
            continue;
        }
        int v = to[e];
        if (dfn[v] == 0) {
            tarjan(v, e);
            low[u] = min(low[u], low[v]);
        } else {
            low[u] = min(low[u], dfn[v]);
        }
    }
    if (dfn[u] == low[u]) {
        ebccCnt++;
        int pop;
        do {
            pop = sta[top--];
            belong[pop] = ebccCnt;
        } while (pop != u);
    }
}

int getLeaf() {
    for (int i = 1; i <= m; i++) {
        int ebcc1 = belong[a[i]];
        int ebcc2 = belong[b[i]];
        if (ebcc1 != ebcc2) {
            degree[ebcc1]++;
            degree[ebcc2]++;
        }
    }
    int ans = 0;
    for (int i = 1; i <= ebccCnt; i++) {
        if (degree[i] == 1) {
            ans++;
        }
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cntg = 1;
    cin >> n >> m;
    for (int i = 1; i <= m; i++) {
        cin >> a[i] >> b[i];
        addEdge(a[i], b[i]);
        addEdge(b[i], a[i]);
    }
    tarjan(1, 0);
    int leafCnt = getLeaf();
    cout << (leafCnt + 1) / 2 << "\n";
    return 0;
}

#include <bits/stdc++.h>

int n, m;
vector<pair<int, int>> es;  // 输入序 1-based
vector<vector<pair<int, int>>> g;  // g[u] = {(v, eid)}，无向边 eid*2 / eid*2+1 两条有向边

void build() {
    g.assign(n + 1, {});
    for (int i = 0; i < m; i++) {
        int u = es[i].first, v = es[i].second;
        g[u].push_back({v, i * 2});
        g[v].push_back({u, i * 2 + 1});
    }
}

// 删掉无向边 eid 后 u、v 是否还连通
bool connected_without(int eid) {
    int s = es[eid].first, t = es[eid].second;
    if (s == t) return true;  // 自环：删掉不影响自身连通
    vector<int> vis(n + 1, 0);
    queue<int> q;
    q.push(s); vis[s] = 1;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (auto& pr : g[u]) {
            int v = pr.first, id = pr.second;
            if (id / 2 == eid) continue;  // 该无向边禁用
            if (!vis[v]) { vis[v] = 1; q.push(v); }
        }
    }
    return vis[t];
}

bool is_bridge(int eid) { return !connected_without(eid); }

// 从 s 出发不经过边 eid 的连通块点集
vector<int> comp_without(int eid, int s) {
    vector<int> vis(n + 1, 0), res;
    queue<int> q;
    q.push(s); vis[s] = 1;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        res.push_back(u);
        for (auto& pr : g[u]) {
            int v = pr.first, id = pr.second;
            if (id / 2 == eid) continue;
            if (!vis[v]) { vis[v] = 1; q.push(v); }
        }
    }
    return res;
}

// 求全部割边（输入序 id 列表）
vector<int> all_bridges() {
    vector<int> res;
    for (int i = 0; i < m; i++) if (is_bridge(i)) res.push_back(i);
    return res;
}

import subprocess, sys
from collections import Counter

def run(argv, inp):
    p = subprocess.run(argv, input=inp, capture_output=True, text=True, timeout=10)
    return p.stdout

def norm(s):
    return [l for l in s.strip().split('\n') if l.strip()]

def check4(inp, cout):
    """题4 验证器：无桥→输出 m 行合法强连通定向；有桥→输出 0"""
    lines = inp.strip().split('\n')
    n, m = map(int, lines[0].split())
    edges = []
    for l in lines[1:1+m]:
        u, v = map(int, l.split())
        edges.append((u, v))
    # 邻接表（含边 id）
    adj = [[] for _ in range(n+1)]
    for i, (u, v) in enumerate(edges):
        adj[u].append((v, i)); adj[v].append((u, i))
    def conn_without(eid):
        s, t = edges[eid]
        if s == t: return True
        vis = [0]*(n+1); vis[s] = 1; q = [s]
        for u in q:
            for v, i in adj[u]:
                if i == eid: continue
                if not vis[v]: vis[v] = 1; q.append(v)
        return vis[t] == 1
    bridges = [i for i in range(m) if not conn_without(i)]
    cout_lines = norm(cout)
    if bridges:
        return cout_lines == ['0']
    # 无桥：验证 m 行定向
    if len(cout_lines) != m: return False
    used = [False]*m
    directed = [[] for _ in range(n+1)]
    for line in cout_lines:
        u, v = map(int, line.split())
        matched = -1
        for i in range(m):
            if used[i]: continue
            x, y = edges[i]
            if (u == x and v == y) or (u == y and v == x):
                matched = i; break
        if matched < 0: return False
        used[matched] = True
        directed[u].append(v)
    # 强连通
    for s in range(1, n+1):
        vis = [0]*(n+1); vis[s] = 1; q = [s]
        for u in q:
            for v in directed[u]:
                if not vis[v]: vis[v] = 1; q.append(v)
        if sum(vis) != n: return False
    return True

def compare(which, inp, cout, bout):
    if which == 4:
        return check4(inp, cout)
    if which == 3:
        cl = norm(cout); bl = norm(bout)
        if not cl or not bl: return cl == bl
        cntc, cntb = int(cl[0]), int(bl[0])
        if cntc != cntb: return False
        setc = Counter(tuple(map(int, l.split())) for l in cl[1:])
        setb = Counter(tuple(map(int, l.split())) for l in bl[1:])
        return setc == setb
    if which == 5:
        cl = norm(cout); bl = norm(bout)
        # cl[0]=分量数；cl[1:] 每分量一行 siz + 成员
        def parse(lines):
            if not lines: return None
            total = int(lines[0])
            comps = []
            i = 1
            for _ in range(total):
                if i >= len(lines): return None
                siz = int(lines[i]); i += 1
                members = []
                while len(members) < siz:
                    if i >= len(lines): return None
                    members += list(map(int, lines[i].split()))
                    i += 1
                comps.append((siz, frozenset(members[:siz])))
            return (total, Counter(comps))
        a, b = parse(cl), parse(bl)
        return a is not None and a == b
    # 题1/2/6/7：字符串比较（规范化）
    return norm(cout) == norm(bout)

#!/usr/bin/env python3
# 191 课对拍生成器：python3 gen191.py <seed> <题号>
# 无向图允许重边与自环（191 题面明确保证重边/自环下答案正确）
import random, sys

seed, which = int(sys.argv[1]), int(sys.argv[2])
rng = random.Random(seed)

def gen_edges(n, connected, allow_self=True, allow_multi=True):
    """返回无向边列表（1-based）。默认允许自环/重边；题4 禁两者（题面无此保证）"""
    edges = []
    if connected and n >= 2:
        # 树保证连通
        perm = list(range(2, n + 1))
        rng.shuffle(perm)
        for v in perm:
            p = rng.randint(1, v - 1)
            edges.append((p, v))
    # 附加随机边
    if allow_self and allow_multi:
        maxextra = n * n
        extra = rng.randint(0, maxextra)
        for _ in range(extra):
            u = rng.randint(1, n)
            v = rng.randint(1, n)
            edges.append((u, v))
    else:
        # 无自环无重边：从候选对里抽
        cand = [(u, v) for u in range(1, n + 1) for v in range(u + 1, n + 1)]
        rng.shuffle(cand)
        extra = rng.randint(0, len(cand))
        edges += cand[:extra]
    rng.shuffle(edges)
    return edges

def out_edges(edges, zero_based=False):
    lines = []
    for u, v in edges:
        if zero_based:
            lines.append(f"{u-1} {v-1}")
        else:
            lines.append(f"{u} {v}")
    return lines

n = rng.randint(2, 8)

    edges = gen_edges(n, connected=False)
    lines = [f"{n} {len(edges)}"] + out_edges(edges)
    sys.stdout.write("\n".join(lines) + "\n")

elif which == 2:  # 割边模版题2：连通，0-based
    edges = gen_edges(n, connected=True)
    lines = [f"{n} {len(edges)}"] + out_edges(edges, zero_based=True)
    sys.stdout.write("\n".join(lines) + "\n")

elif which == 3:  # 关键网络线路：连通，n m k l + k A + l B + 边
    edges = gen_edges(n, connected=True)
    k = rng.randint(1, n)
    l = rng.randint(1, n)
    A = sorted(rng.sample(range(1, n + 1), k))
    B = sorted(rng.sample(range(1, n + 1), l))
    lines = [f"{n} {len(edges)} {k} {l}"]
    lines += [str(x) for x in A]
    lines += [str(x) for x in B]
    lines += out_edges(edges)
    sys.stdout.write("\n".join(lines) + "\n")

elif which == 4:  # 贝尔敦道路：连通
    edges = gen_edges(n, connected=True, allow_self=False, allow_multi=False)

    sys.stdout.write("\n".join(lines) + "\n")

elif which == 5:  # 边双模版题：多连通区
    edges = gen_edges(n, connected=False)

    sys.stdout.write("\n".join(lines) + "\n")
