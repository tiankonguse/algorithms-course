// 题1 LC327 达标子数组个数：归并分治 + 树状数组离散化 vs 暴力
// 页面代码复现（dp-fenwick-range-sum.html）
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

/* ===== 课程实现：归并分治（页面代码） ===== */
const int MAXN = 100001;
ll sumv[MAXN], helpv[MAXN];
int lowv, upv;
int gn;  // 测试辅助：数组长度

int mergeCnt(int l, int m, int r) {
    int ans = 0;
    int wl = l, wr = l;
    ll mx, mn;
    for (int i = m + 1; i <= r; i++) {
        mx = sumv[i] - lowv;
        mn = sumv[i] - upv;
        while (wr <= m && sumv[wr] <= mx) wr++;
        while (wl <= m && sumv[wl] < mn) wl++;
        ans += wr - wl;
    }
    int p1 = l, p2 = m + 1, i = l;
    while (p1 <= m && p2 <= r) helpv[i++] = sumv[p1] <= sumv[p2] ? sumv[p1++] : sumv[p2++];
    while (p1 <= m) helpv[i++] = sumv[p1++];
    while (p2 <= r) helpv[i++] = sumv[p2++];
    for (i = l; i <= r; i++) sumv[i] = helpv[i];
    return ans;
}

int f(int l, int r) {
    if (l == r) {
        return lowv <= sumv[l] && sumv[l] <= upv ? 1 : 0;
    }
    int m = (l + r) / 2;
    return f(l, m) + f(m + 1, r) + mergeCnt(l, m, r);
}

int countRangeSumMerge(const vector<ll>& nums, int lower, int upper) {
    gn = nums.size();
    sumv[0] = nums[0];
    for (int i = 1; i < gn; i++) sumv[i] = sumv[i - 1] + nums[i];
    lowv = lower; upv = upper;
    return f(0, gn - 1);
}

/* ===== 课程实现：树状数组 + 离散化（页面代码） ===== */
ll sorted[MAXN];
int m;
int tree[MAXN];

int rankv(ll v) {
    int left = 1, right = m, mid;
    int ans = 0;
    while (left <= right) {
        mid = (left + right) / 2;
        if (sorted[mid] <= v) { ans = mid; left = mid + 1; }
        else right = mid - 1;
    }
    return ans;
}

void add(int i, int c) {
    while (i <= m) { tree[i] += c; i += i & -i; }
}

int presum(int i) {
    int ans = 0;
    while (i > 0) { ans += tree[i]; i -= i & -i; }
    return ans;
}

void buildBIT(const vector<ll>& nums) {
    int n = nums.size();
    for (int i = 1, j = 0; i <= n; i++, j++) sorted[i] = sorted[i - 1] + nums[j];
    sort(sorted + 1, sorted + n + 1);
    m = 1;
    for (int i = 2; i <= n; i++) if (sorted[m] != sorted[i]) sorted[++m] = sorted[i];
    for (int i = 1; i <= m; i++) tree[i] = 0;
}

int countRangeSumBIT(const vector<ll>& nums, int lower, int upper) {
    buildBIT(nums);
    ll s = 0;
    int ans = 0;
    int n = nums.size();
    for (int i = 0; i < n; i++) {
        s += nums[i];
        ans += presum(rankv(s - lower)) - presum(rankv(s - upper - 1));
        if (lower <= s && s <= upper) ans++;
        add(rankv(s), 1);
    }
    return ans;
}

/* ===== 暴力 ===== */
int brute(const vector<ll>& nums, int lower, int upper) {
    int n = nums.size(), ans = 0;
    for (int l = 0; l < n; l++) {
        ll s = 0;
        for (int r = l; r < n; r++) {
            s += nums[r];
            if (lower <= s && s <= upper) ans++;
        }
    }
    return ans;
}

int main() {
    mt19937 rng(20261016);
    int cases = 3000;
    for (int tc = 1; tc <= cases; tc++) {
        int n = rng() % 14 + 1;
        int lower = (int)(rng() % 41) - 20;
        int upper = lower + (int)(rng() % 21);
        vector<ll> nums(n);
        for (int i = 0; i < n; i++) nums[i] = (ll)(rng() % 41) - 20;
        int a = countRangeSumMerge(nums, lower, upper);
        int b = countRangeSumBIT(nums, lower, upper);
        int c = brute(nums, lower, upper);
        if (a != c || b != c) {
            printf("MISMATCH tc=%d n=%d lower=%d upper=%d merge=%d bit=%d brute=%d\n", tc, n, lower, upper, a, b, c);
            for (ll x : nums) printf("%lld ", x);
            printf("\n");
            return 1;
        }
    }
    printf("OK 3000 cases (merge==bit==brute)\n");
    // 固定小用例
    {
        vector<ll> n1 = {-2, 5, -1, 3};
        printf("例1 [-2,5,-1,3] l=-2 u=2: merge=%d bit=%d brute=%d\n",
               countRangeSumMerge(n1, -2, 2), countRangeSumBIT(n1, -2, 2), brute(n1, -2, 2));
        vector<ll> n2 = {1, 1, 1};
        printf("例2 [1,1,1] l=1 u=2: merge=%d bit=%d brute=%d\n",
               countRangeSumMerge(n2, 1, 2), countRangeSumBIT(n2, 1, 2), brute(n2, 1, 2));
    }
    return 0;
}

// 题2 LC2312 平衡子序列最大和（树状数组值域 max）+ 题4 LC2370 最长理想子序列（线段树 26）
// 页面代码复现（dp-fenwick-value-max.html）
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

/* ===== 题2：树状数组维护前缀最大值（页面代码） ===== */
const int M2 = 100001;
ll sortv[M2], treeMax[M2];
int m2, n2;

int rankv2(ll v) {
    int left = 1, right = m2, mid;
    int ans = 0;
    while (left <= right) {
        mid = (left + right) / 2;
        if (sortv[mid] <= v) { ans = mid; left = mid + 1; }
        else right = mid - 1;
    }
    return ans;
}

void update2(int i, ll v) {
    while (i <= m2) {
        treeMax[i] = max(treeMax[i], v);
        i += i & -i;
    }
}

ll query2(int i) {
    ll ans = LLONG_MIN;
    while (i > 0) {
        ans = max(ans, treeMax[i]);
        i -= i & -i;
    }
    return ans;
}

void build2(const vector<ll>& nums) {
    n2 = nums.size();
    for (int i = 1, j = 0; i <= n2; i++, j++) sortv[i] = nums[j] - j;
    sort(sortv + 1, sortv + n2 + 1);
    m2 = 1;
    for (int i = 2; i <= n2; i++) if (sortv[m2] != sortv[i]) sortv[++m2] = sortv[i];
    for (int i = 1; i <= m2; i++) treeMax[i] = LLONG_MIN;
}

ll maxBalancedSubsequenceSum(const vector<ll>& nums) {
    build2(nums);
    ll pre;
    for (int i = 0, k; i < n2; i++) {
        k = rankv2(nums[i] - i);
        pre = query2(k);
        if (pre < 0) update2(k, nums[i]);
        else update2(k, pre + nums[i]);
    }
    return query2(m2);
}

/* ===== 题2 暴力 ===== */
ll brute2(const vector<ll>& nums) {
    int n = nums.size();
    ll ans = LLONG_MIN;
    // dp[i] = nums[i] + max{dp[j]: j<i, nums[j]-j <= nums[i]-i}
    vector<ll> dp(n);
    for (int i = 0; i < n; i++) {
        dp[i] = nums[i];
        for (int j = 0; j < i; j++) {
            if (nums[j] - j <= nums[i] - i) dp[i] = max(dp[i], dp[j] + nums[i]);
        }
        ans = max(ans, dp[i]);
    }
    return ans;
}

/* ===== 题4：线段树 26 槽（页面代码） ===== */
int segmax[(26 + 1) << 2];

void up4(int i) {
    segmax[i] = max(segmax[i << 1], segmax[i << 1 | 1]);
}

void update4(int jobi, int jobv, int l, int r, int i) {
    if (l == r && jobi == l) {
        segmax[i] = jobv;
    } else {
        int m = (l + r) >> 1;
        if (jobi <= m) update4(jobi, jobv, l, m, i << 1);
        else update4(jobi, jobv, m + 1, r, i << 1 | 1);
        up4(i);
    }
}

int query4(int jobl, int jobr, int l, int r, int i) {
    if (jobl <= l && r <= jobr) return segmax[i];
    int m = (l + r) >> 1;
    int ans = 0;
    if (jobl <= m) ans = max(ans, query4(jobl, jobr, l, m, i << 1));
    if (jobr > m) ans = max(ans, query4(jobl, jobr, m + 1, r, i << 1 | 1));
    return ans;
}

int longestIdealString(const string& s, int k) {
    memset(segmax, 0, sizeof(segmax));
    int v, p;
    int ans = 0;
    for (char cha : s) {
        v = cha - 'a' + 1;
        p = query4(max(v - k, 1), min(v + k, 26), 1, 26, 1);
        ans = max(ans, p + 1);
        update4(v, p + 1, 1, 26, 1);
    }
    return ans;
}

/* ===== 题4 暴力 ===== */
int brute4(const string& s, int k) {
    int n = s.size();
    vector<int> dp(n, 1);
    int ans = 1;
    for (int i = 0; i < n; i++) {
        dp[i] = 1;
        for (int j = 0; j < i; j++) {
            if (abs((int)s[i] - (int)s[j]) <= k) dp[i] = max(dp[i], dp[j] + 1);
        }
        ans = max(ans, dp[i]);
    }
    return ans;
}

int main() {
    mt19937 rng(20261016);
    // 题2 对拍
    int cases = 4000;
    for (int tc = 1; tc <= cases; tc++) {
        int n = rng() % 12 + 1;
        vector<ll> nums(n);
        for (int i = 0; i < n; i++) nums[i] = (ll)(rng() % 41) - 20;
        ll a = maxBalancedSubsequenceSum(nums);
        ll b = brute2(nums);
        if (a != b) {
            printf("MISMATCH t2 tc=%d n=%d opt=%lld brute=%lld\n", tc, n, a, b);
            for (ll x : nums) printf("%lld ", x);
            printf("\n");
            return 1;
        }
    }
    printf("OK t2 %d cases\n", cases);
    // 题2 固定小用例
    {
        vector<ll> n1 = {3, 2, 5, 1, 4};
        printf("例 [3,2,5,1,4]: opt=%lld brute=%lld\n", maxBalancedSubsequenceSum(n1), brute2(n1));
        vector<ll> n2 = {-1, -2, -3};
        printf("例 [-1,-2,-3]: opt=%lld brute=%lld\n", maxBalancedSubsequenceSum(n2), brute2(n2));
        vector<ll> n3 = {5, -5, 5, -5};
        printf("例 [5,-5,5,-5]: opt=%lld brute=%lld\n", maxBalancedSubsequenceSum(n3), brute2(n3));
        vector<ll> n4 = {0, 3, 1, 6, 2};
        printf("例 [0,3,1,6,2]: opt=%lld brute=%lld\n", maxBalancedSubsequenceSum(n4), brute2(n4));
    }
    // 题4 对拍
    cases = 4000;
    for (int tc = 1; tc <= cases; tc++) {
        int n = rng() % 12 + 1;
        int k = rng() % 26;
        string s;
        for (int i = 0; i < n; i++) s += (char)('a' + rng() % 26);
        int a = longestIdealString(s, k);
        int b = brute4(s, k);
        if (a != b) {

        }
    }
    printf("OK t4 %d cases\n", cases);
    // 题4 固定小用例
    {
        printf("例 acfgbd k=2: opt=%d brute=%d\n", longestIdealString("acfgbd", 2), brute4("acfgbd", 2));
        printf("例 abcd k=1: opt=%d brute=%d\n", longestIdealString("abcd", 1), brute4("abcd", 1));
        printf("例 zyxzyx k=0: opt=%d brute=%d\n", longestIdealString("zyxzyx", 0), brute4("zyxzyx", 0));
        printf("例 pvpt k=2: opt=%d brute=%d\n", longestIdealString("pvpt", 2), brute4("pvpt", 2));
    }
    return 0;
}

// 题3 洛谷 P3287 方伯伯的玉米田：二维树状数组（高度 × 修改次数）vs 暴力
#include <bits/stdc++.h>
using namespace std;

/* ===== 课程实现（从 Code03 Java 改写） ===== */
const int MAXH = 5500, MAXK = 501, MAXN = 10001;
int arr[MAXN];
int tree2d[MAXH + 1][MAXK + 1];
int n3, k3;

void update2d(int x, int y, int v) {
    for (int i = x; i <= MAXH; i += i & -i) {
        for (int j = y; j <= k3 + 1; j += j & -j) {
            tree2d[i][j] = max(tree2d[i][j], v);
        }
    }
}

int query2d(int x, int y) {
    int ans = 0;
    for (int i = x; i > 0; i -= i & -i) {
        for (int j = y; j > 0; j -= j & -j) {
            ans = max(ans, tree2d[i][j]);
        }
    }
    return ans;
}

int compute3() {
    int v, dp;
    for (int i = 1; i <= n3; i++) {
        for (int j = k3; j >= 0; j--) {
            v = arr[i] + j;
            dp = query2d(v, j + 1) + 1;
            update2d(v, j + 1, dp);
        }
    }
    return query2d(MAXH, k3 + 1);
}

void reset3() {
    memset(tree2d, 0, sizeof(tree2d));
}

/* ===== 暴力 ===== */
int brute3(const vector<int>& a, int K) {
    int n = a.size();
    // f[i][j] = 第 i 个玉米（1 起）恰好多加 j 次，以它结尾的最长不下降子序列
    // f[i][j] = 1 + max{ f[p][q] : p<i, q<=j, a[p]+q <= a[i]+j }
    vector<vector<int>> f(n, vector<int>(K + 1, 0));
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j <= K; j++) {
            f[i][j] = 1;
            for (int p = 0; p < i; p++) {
                for (int q = 0; q <= j; q++) {
                    if (a[p] + q <= a[i] + j) f[i][j] = max(f[i][j], f[p][q] + 1);
                }
            }
            ans = max(ans, f[i][j]);
        }
    }
    return ans;
}

int main() {
    mt19937 rng(20261016);
    int cases = 800;
    for (int tc = 1; tc <= cases; tc++) {
        int n = rng() % 10 + 1;
        int K = rng() % 6 + 1;
        vector<int> a(n);
        for (int i = 0; i < n; i++) a[i] = (int)(rng() % 20) + 1;
        n3 = n; k3 = K;
        for (int i = 1; i <= n; i++) arr[i] = a[i - 1];
        reset3();
        int opt = compute3();
        int b = brute3(a, K);
        if (opt != b) {
            printf("MISMATCH tc=%d n=%d K=%d opt=%d brute=%d arr=", tc, n, K, opt, b);
            for (int x : a) printf("%d ", x);
            printf("\n");
            return 1;
        }
    }
    printf("OK %d cases\n", cases);
    // 固定小用例
    {
        vector<int> a1 = {1, 2, 3, 4, 5};
        n3 = 5; k3 = 2; for (int i = 1; i <= 5; i++) arr[i] = a1[i - 1];
        reset3();
        printf("例 [1,2,3,4,5] k=2: opt=%d brute=%d\n", compute3(), brute3(a1, 2));
        vector<int> a2 = {5, 4, 3, 2, 1};
        n3 = 5; k3 = 2; for (int i = 1; i <= 5; i++) arr[i] = a2[i - 1];
        reset3();
        printf("例 [5,4,3,2,1] k=2: opt=%d brute=%d\n", compute3(), brute3(a2, 2));
        vector<int> a3 = {3, 1, 4, 1, 5, 9};
        n3 = 6; k3 = 3; for (int i = 1; i <= 6; i++) arr[i] = a3[i - 1];
        reset3();
        printf("例 [3,1,4,1,5,9] k=3: opt=%d brute=%d\n", compute3(), brute3(a3, 3));
    }
    return 0;
}

// 题5 CF833B 面包店：线段树区间加（0~n 范围）vs 暴力
#include <bits/stdc++.h>
using namespace std;

/* ===== 课程实现（从 Code05 Java 改写） ===== */
const int MAXN = 35001;
int n5, k5;
int arr5[MAXN];
int dp5[MAXN];
int pre5[MAXN];
int segmax5[MAXN << 2];
int lazyadd5[MAXN << 2];

void up5(int i) {
    segmax5[i] = max(segmax5[i << 1], segmax5[i << 1 | 1]);
}

void lazy5(int i, int v) {
    segmax5[i] += v;
    lazyadd5[i] += v;
}

void down5(int i) {
    if (lazyadd5[i] != 0) {
        lazy5(i << 1, lazyadd5[i]);
        lazy5(i << 1 | 1, lazyadd5[i]);
        lazyadd5[i] = 0;
    }
}

void build5(int l, int r, int i) {
    if (l == r) {
        segmax5[i] = dp5[l];
    } else {
        int mid = (l + r) >> 1;
        build5(l, mid, i << 1);
        build5(mid + 1, r, i << 1 | 1);
        up5(i);
    }
    lazyadd5[i] = 0;
}

void add5(int jobl, int jobr, int jobv, int l, int r, int i) {
    if (jobl <= l && r <= jobr) {
        lazy5(i, jobv);
    } else {
        down5(i);
        int mid = (l + r) >> 1;
        if (jobl <= mid) add5(jobl, jobr, jobv, l, mid, i << 1);
        if (jobr > mid) add5(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        up5(i);
    }
}

int query5(int jobl, int jobr, int l, int r, int i) {
    if (jobl <= l && r <= jobr) return segmax5[i];
    down5(i);
    int mid = (l + r) >> 1;
    int ans = INT_MIN;
    if (jobl <= mid) ans = max(ans, query5(jobl, jobr, l, mid, i << 1));
    if (jobr > mid) ans = max(ans, query5(jobl, jobr, mid + 1, r, i << 1 | 1));
    return ans;
}

int compute5() {
    memset(dp5, 0, sizeof(dp5));  // dp[0..n]
    for (int t = 1; t <= k5; t++) {
        build5(0, n5, 1);
        memset(pre5, 0, sizeof(pre5));
        for (int i = 1; i <= n5; i++) {
            add5(pre5[arr5[i]], i - 1, 1, 0, n5, 1);
            if (i >= t) {
                dp5[i] = query5(0, i - 1, 0, n5, 1);
            }
            pre5[arr5[i]] = i;
        }
    }
    return dp5[n5];
}

/* ===== 暴力 ===== */
// dp[t][i] = max over p in [t-1, i-1] (dp[t-1][p] + distinct(p+1, i))
int brute5(const vector<int>& a, int K) {
    int n = a.size() - 1;  // a[0] 是哨兵，实际数据 a[1..n]
    vector<vector<int>> dp(K + 1, vector<int>(n + 1, 0));
    for (int t = 1; t <= K; t++) {
        for (int i = t; i <= n; i++) {
            int best = INT_MIN;
            for (int p = t - 1; p <= i - 1; p++) {
                set<int> st;
                for (int x = p + 1; x <= i; x++) st.insert(a[x]);
                best = max(best, dp[t - 1][p] + (int)st.size());
            }
            dp[t][i] = best;
        }
    }
    return dp[K][n];
}

int main() {
    mt19937 rng(20261016);
    int cases = 600;
    for (int tc = 1; tc <= cases; tc++) {
        int n = rng() % 9 + 1;
        int K = rng() % min(4, n) + 1;
        vector<int> a(n + 1);
        for (int i = 1; i <= n; i++) a[i] = (int)(rng() % n) + 1;
        n5 = n; k5 = K;
        for (int i = 1; i <= n; i++) arr5[i] = a[i];
        int opt = compute5();
        int b = brute5(a, K);
        if (opt != b) {
            printf("MISMATCH tc=%d n=%d K=%d opt=%d brute=%d arr=", tc, n, K, opt, b);
            for (int i = 1; i <= n; i++) printf("%d ", a[i]);
            printf("\n");
            return 1;
        }
    }
    printf("OK %d cases\n", cases);
    // 固定小用例
    {
        vector<int> a1 = {0, 1, 2, 1, 3};  // 1 起：1,2,1,3
        n5 = 4; k5 = 2; for (int i = 1; i <= 4; i++) arr5[i] = a1[i];
        printf("例 [1,2,1,3] k=2: opt=%d brute=%d\n", compute5(), brute5(a1, 2));
        vector<int> a2 = {0, 1, 2, 3, 4};  // 1,2,3,4
        n5 = 4; k5 = 3; for (int i = 1; i <= 4; i++) arr5[i] = a2[i];
        printf("例 [1,2,3,4] k=3: opt=%d brute=%d\n", compute5(), brute5(a2, 3));
    }
    return 0;
}

// 题6 洛谷 P2605 基站选址：线段树最小值 + 链式前向星预警 vs 暴力
#include <bits/stdc++.h>
using namespace std;

/* ===== 课程实现（从 Code06 Java 改写） ===== */
const int MAXN6 = 20002;
int n6, k6;
int dist6[MAXN6], fix6[MAXN6], range6[MAXN6], warranty6[MAXN6];
int lft6[MAXN6], rgt6[MAXN6];
int head6[MAXN6], nxt6[MAXN6], to6[MAXN6];
int cnt6;
int segmin6[MAXN6 << 2], lazyadd6[MAXN6 << 2];
int dp6[MAXN6];

int search6(int d) {
    int l = 1, r = n6, m;
    int ans = 0;
    while (l <= r) {
        m = (l + r) / 2;
        if (dist6[m] >= d) { ans = m; r = m - 1; }
        else l = m + 1;
    }
    return ans;
}

void addEdge6(int u, int v) {
    nxt6[cnt6] = head6[u];
    to6[cnt6] = v;
    head6[u] = cnt6++;
}

void prepare6() {
    cnt6 = 1;
    for (int i = 1; i <= n6; i++) {
        lft6[i] = search6(dist6[i] - range6[i]);
        rgt6[i] = search6(dist6[i] + range6[i]);
        if (dist6[rgt6[i]] > dist6[i] + range6[i]) {
            rgt6[i]--;
        }
        addEdge6(rgt6[i], i);
    }
}

void up6(int i) { segmin6[i] = min(segmin6[i << 1], segmin6[i << 1 | 1]); }

void lazy6(int i, int v) { segmin6[i] += v; lazyadd6[i] += v; }

void down6(int i) {
    if (lazyadd6[i] != 0) {
        lazy6(i << 1, lazyadd6[i]);
        lazy6(i << 1 | 1, lazyadd6[i]);
        lazyadd6[i] = 0;
    }
}

void build6(int l, int r, int i) {
    if (l == r) {
        segmin6[i] = dp6[l];
    } else {
        int mid = (l + r) >> 1;
        build6(l, mid, i << 1);
        build6(mid + 1, r, i << 1 | 1);
        up6(i);
    }
    lazyadd6[i] = 0;
}

void add6(int jobl, int jobr, int jobv, int l, int r, int i) {
    if (jobl <= l && r <= jobr) {
        lazy6(i, jobv);
    } else {
        down6(i);
        int mid = (l + r) >> 1;
        if (jobl <= mid) add6(jobl, jobr, jobv, l, mid, i << 1);
        if (jobr > mid) add6(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        up6(i);
    }
}

int query6(int jobl, int jobr, int l, int r, int i) {
    if (jobl <= l && r <= jobr) return segmin6[i];
    down6(i);
    int mid = (l + r) >> 1;
    int ans = INT_MAX;
    if (jobl <= mid) ans = min(ans, query6(jobl, jobr, l, mid, i << 1));
    if (jobr > mid) ans = min(ans, query6(jobl, jobr, mid + 1, r, i << 1 | 1));
    return ans;
}

int compute6() {
    for (int i = 1, w = 0; i <= n6; i++) {
        dp6[i] = w + fix6[i];
        for (int ei = head6[i]; ei != 0; ei = nxt6[ei]) {
            w += warranty6[to6[ei]];
        }
    }
    for (int t = 2; t <= k6 + 1; t++) {
        build6(1, n6, 1);
        for (int i = 1; i <= n6; i++) {
            if (i >= t) {
                dp6[i] = min(dp6[i], query6(1, i - 1, 1, n6, 1) + fix6[i]);
            }
            for (int ei = head6[i], uncover; ei != 0; ei = nxt6[ei]) {
                uncover = to6[ei];
                if (lft6[uncover] > 1) {
                    add6(1, lft6[uncover] - 1, warranty6[uncover], 1, n6, 1);
                }
            }
        }
    }
    return dp6[n6];
}

void setup6(const vector<int>& dist, const vector<int>& fix, const vector<int>& range,
            const vector<int>& warranty, int n0, int K) {
    n6 = n0 + 1;
    k6 = K;
    memset(dist6, 0, sizeof(dist6));
    memset(fix6, 0, sizeof(fix6));
    memset(range6, 0, sizeof(range6));
    memset(warranty6, 0, sizeof(warranty6));
    for (int i = 1; i <= n0; i++) {
        dist6[i] = dist[i];
        fix6[i] = fix[i];
        range6[i] = range[i];
        warranty6[i] = warranty[i];
    }
    dist6[n6] = INT_MAX;
    fix6[n6] = range6[n6] = warranty6[n6] = 0;
    memset(head6, 0, sizeof(head6));
    memset(dp6, 0, sizeof(dp6));
    prepare6();
}

/* ===== 暴力 ===== */
int brute6(const vector<int>& dist, const vector<int>& fix, const vector<int>& range,
           const vector<int>& warranty, int n0, int K) {
    int N = n0 + 1;
    vector<int> d(N + 1), fx(N + 1), rg(N + 1), wr(N + 1), lf(N + 1), rt(N + 1);
    for (int i = 1; i <= n0; i++) { d[i] = dist[i]; fx[i] = fix[i]; rg[i] = range[i]; wr[i] = warranty[i]; }
    d[N] = INT_MAX; fx[N] = rg[N] = wr[N] = 0;
    auto search = [&](int x) {
        int l = 1, r = N, ans = 0;
        while (l <= r) { int m = (l + r) / 2; if (d[m] >= x) { ans = m; r = m - 1; } else l = m + 1; }
        return ans;
    };
    for (int i = 1; i <= N; i++) {
        lf[i] = search(d[i] - rg[i]);
        rt[i] = search(d[i] + rg[i]);
        if (rt[i] >= 1 && d[rt[i]] > d[i] + rg[i]) rt[i]--;
    }
    auto serviced = [&](int j, int p) { return lf[j] <= p && p <= rt[j]; };
    const int INF = 1e9;
    vector<vector<int>> dp(K + 2, vector<int>(N + 1, INF));
    for (int i = 1; i <= N; i++) {
        int c = 0;
        for (int j = 1; j < i; j++) if (!serviced(j, i)) c += wr[j];
        dp[1][i] = fx[i] + c;
    }
    for (int t = 2; t <= K + 1; t++) {
        for (int i = 1; i <= N; i++) {
            dp[t][i] = dp[t - 1][i];  // 最多 t-1 个基站的方案仍合法（Java dp 就地 min 的语义）
            if (i < t) continue;
            int best = INF;
            for (int p = 1; p <= i - 1; p++) {
                int c = 0;
                for (int j = p + 1; j <= i - 1; j++) {
                    if (!serviced(j, p) && !serviced(j, i)) c += wr[j];
                }
                best = min(best, dp[t - 1][p] + c);
            }
            dp[t][i] = min(dp[t][i], fx[i] + best);
        }
    }
    return dp[K + 1][N];
}

int main() {
    mt19937 rng(20261016);
    int cases = 400;
    for (int tc = 1; tc <= cases; tc++) {
        int n0 = rng() % 8 + 1;
        int K = rng() % min(4, n0) + 1;
        vector<int> dist(n0 + 1), fix(n0 + 1), range(n0 + 1), warranty(n0 + 1);

        for (int i = 1; i <= n0; i++) {

            range[i] = (int)(rng() % 60) + 1;
            warranty[i] = (int)(rng() % 30) + 1;
        }
        setup6(dist, fix, range, warranty, n0, K);
        int opt = compute6();
        int b = brute6(dist, fix, range, warranty, n0, K);
        if (opt != b) {
            printf("MISMATCH tc=%d n=%d K=%d opt=%d brute=%d\n", tc, n0, K, opt, b);
            printf("dist: "); for (int i = 1; i <= n0; i++) printf("%d ", dist[i]); printf("\n");
            printf("fix: "); for (int i = 1; i <= n0; i++) printf("%d ", fix[i]); printf("\n");
            printf("range: "); for (int i = 1; i <= n0; i++) printf("%d ", range[i]); printf("\n");
            printf("wr: "); for (int i = 1; i <= n0; i++) printf("%d ", warranty[i]); printf("\n");
            return 1;
        }
    }
    printf("OK %d cases\n", cases);
    // 固定小用例
    {
        // 3 个村庄，dist=1,3,6；fix 各 1；range 足够大；warranty 各 5
        vector<int> dist = {0, 1, 3, 6};

        vector<int> range = {0, 100, 100, 100};

        setup6(dist, fix, range, warranty, 3, 1);
        printf("例 n=3 大range k=1: opt=%d brute=%d\n", compute6(), brute6(dist, fix, range, warranty, 3, 1));
        // 全不覆盖：range=1
        vector<int> dist2 = {0, 2, 4, 6};

        vector<int> range2 = {0, 1, 1, 1};
        vector<int> warranty2 = {0, 10, 10, 10};
        setup6(dist2, fix2, range2, warranty2, 3, 2);
        printf("例 n=3 小range k=2: opt=%d brute=%d\n", compute6(), brute6(dist2, fix2, range2, warranty2, 3, 2));
    }
    return 0;
}
