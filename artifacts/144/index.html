#include <bits/stdc++.h>
using namespace std;
const int MOD = 10007;
// 暴力：直接展开 (a*x + b*y)^k，取 x^n * y^m 系数 % MOD
int main() {
  int a, b, k, n, m; cin >> a >> b >> k >> n >> m;
  static long long coef[40][40];
  memset(coef, 0, sizeof(coef));
  coef[0][0] = 1 % MOD;
  for (int step = 0; step < k; step++) {
    static long long nx[40][40];
    memset(nx, 0, sizeof(nx));
    for (int i = 0; i <= step; i++)
      for (int j = 0; j <= step; j++) {
        if (!coef[i][j]) continue;
        nx[i+1][j] = (nx[i+1][j] + coef[i][j] * (a % MOD)) % MOD; // 乘 a*x
        nx[i][j+1] = (nx[i][j+1] + coef[i][j] * (b % MOD)) % MOD; // 乘 b*y
      }
    memcpy(coef, nx, sizeof(coef));
  }
  cout << coef[n][m] % MOD << "\n";
}

#include <bits/stdc++.h>
using namespace std;
// 暴力：独立建 %k 组合表（复用同款递推但只到 40），每组查询逐 (i,j) 计数
int main() {
  int t, k; cin >> t >> k;
  static int cc[41][41];
  for (int i = 0; i <= 40; i++) {
    cc[i][0] = 1 % k;
    for (int j = 1; j <= i; j++) cc[i][j] = (cc[i-1][j] + cc[i-1][j-1]) % k;
  }
  while (t--) {
    int n, m; cin >> n >> m;
    int ans = 0;
    for (int i = 0; i <= n; i++)
      for (int j = 0; j <= min(i, m); j++)
        if (cc[i][j] == 0) ans++;
    cout << ans << "\n";
  }
}

#include <bits/stdc++.h>
using namespace std;
// 暴力（课上版）：照 Code04_SplitWays1.ways1，B[i],C[i] >= 1
int fv(const vector<int>& arr, int i, int preb, int prec) {
  if (i == (int)arr.size()) return 1;
  int ans = 0;
  for (int b = 1, c = arr[i]-1; b < arr[i]; b++, c--)
    if (preb <= b && prec >= c) ans += fv(arr, i+1, b, c);
  return ans;
}
int ways1(const vector<int>& arr) {
  int ans = 0;
  for (int b = 1, c = arr[0]-1; b < arr[0]; b++, c--) ans += fv(arr, 1, b, c);
  return ans;
}
// 暴力（LeetCode 版）：B[i] 从 0..arr[i]（C[i] = arr[i]-B[i] >= 0）
int fv0(const vector<int>& arr, int i, int preb, int prec) {
  if (i == (int)arr.size()) return 1;
  int ans = 0;
  for (int b = 0, c = arr[i]; b <= arr[i]; b++, c--)
    if (preb <= b && prec >= c) ans += fv0(arr, i+1, b, c);
  return ans;
}
int ways0(const vector<int>& arr) {
  int ans = 0;
  for (int b = 0, c = arr[0]; b <= arr[0]; b++, c--) ans += fv0(arr, 1, b, c);
  return ans;
}
int main() {
  int n; cin >> n;
  vector<int> arr(n);
  for (int i = 0; i < n; i++) cin >> arr[i];
  cout << ways1(arr) << " " << ways0(arr) << "\n";
}

code/course_02.cpp

#include <bits/stdc++.h>
using namespace std;
const int MOD = 10007;
const int MAXK = 1001;
long long fac[MAXK+1], inv[MAXK+1];
long long power(long long x, int p) { long long ans=1; while(p>0){ if(p&1) ans=ans*x%MOD; x=x*x%MOD; p>>=1;} return ans; }
void build() {
  fac[0] = 1;  // 0! = 1
  fac[1] = 1;
  for(int i=2;i<=MAXK;i++) fac[i] = (long long)i*fac[i-1]%MOD;
  inv[MAXK]=power(fac[MAXK], MOD-2);
  for(int i=MAXK-1;i>=0;i--) inv[i] = (long long)(i+1)*inv[i+1]%MOD;
}
long long c(int n, int k){ return fac[n]*inv[k]%MOD*inv[n-k]%MOD; }
int main(){
  build();
  int a,b,k,n,m; cin>>a>>b>>k>>n>>m;
  cout << power(a,n)*power(b,m)%MOD*c(k,k-n)%MOD << "\n";
}

#include <bits/stdc++.h>
using namespace std;
const int MAXV = 2000, MAXN = 2002;
int c[MAXN][MAXN], f[MAXN][MAXN], sum[MAXN][MAXN];
int main() {
  int t, k; cin >> t >> k;
  for (int i = 0; i <= MAXV; i++) {
    c[i][0] = 1;
    for (int j = 1; j <= i; j++) c[i][j] = (c[i-1][j] + c[i-1][j-1]) % k;
  }
  for (int i = 1; i <= MAXV; i++)
    for (int j = 1; j <= i; j++) f[i][j] = c[i][j] % k == 0 ? 1 : 0;
  for (int i = 2; i <= MAXV; i++) {
    for (int j = 1; j <= i; j++)
      sum[i][j] = sum[i][j-1] + sum[i-1][j] - sum[i-1][j-1] + f[i][j];
    sum[i][i+1] = sum[i][i];
  }
  while (t--) {
    int n, m; cin >> n >> m;
    cout << (m > n ? sum[n][n] : sum[n][m]) << "\n";
  }
}

#include <bits/stdc++.h>
using namespace std;
const int MOD = 1000000007;
// 课上版（Code04_SplitWays1）：分裂方式 v-1 种，k = arr[0] - 1
long long power(long long x, long long p) { long long ans=1; while(p>0){ if(p&1) ans=ans*x%MOD; x=x*x%MOD; p>>=1;} return ans; }
long long comb(int n, int k) {
  long long fac=1, inv1=1, inv2=1;
  for (int i=1;i<=n;i++) {
    fac = fac*i%MOD;
    if (i==k) inv1 = power(fac, MOD-2);
    if (i==n-k) inv2 = power(fac, MOD-2);
  }
  return fac*inv1%MOD*inv2%MOD;
}
int main() {
  int n; cin >> n;
  vector<int> arr(n);
  for (int i=0;i<n;i++) cin >> arr[i];
  int k = arr[0]-1;
  for (int i=1;i<n && k>0;i++) if (arr[i-1] > arr[i]) k -= arr[i-1]-arr[i];
  if (k <= 0) { cout << 0 << "\n"; return 0; }
  cout << comb(k+n-1, n) << "\n";
}

// ===== 题2：计算系数 =====
const int MOD2 = 10007;
long long power2(long long x, int p, int mod) { long long ans=1; while(p>0){ if(p&1) ans=ans*x%mod; x=x*x%mod; p>>=1;} return ans; }
long long c2(int n, int k) {
  static long long fac[1002], inv[1002];
  static bool built = false;
  if (!built) {
    fac[0] = 1; fac[1] = 1;
    for (int i = 2; i <= 1001; i++) fac[i] = (long long)i * fac[i-1] % MOD2;
    inv[1001] = power2(fac[1001], MOD2-2, MOD2);
    for (int i = 1000; i >= 0; i--) inv[i] = (long long)(i+1) * inv[i+1] % MOD2;
    built = true;
  }
  return fac[n] * inv[k] % MOD2 * inv[n-k] % MOD2;
}
int course02(int a, int b, int k, int n, int m) {
  return (int)(power2(a, n, MOD2) * power2(b, m, MOD2) % MOD2 * c2(k, k-n) % MOD2);
}

const int MAXV3 = 2000;
static int cc3[2002][2002], f3[2002][2002], sum3[2002][2002];
void build3(int k) {
  static int builtk = -1;

  return (int)comb4(k+n-1, n);
}
int fv1(const vector<int>& arr, int i, int preb, int prec) {
  if (i == (int)arr.size()) return 1;
  int ans = 0;
  for (int b = 1, c = arr[i]-1; b < arr[i]; b++, c--)
    if (preb <= b && prec >= c) ans += fv1(arr, i+1, b, c);
  return ans;
}
int ways1(const vector<int>& arr) {
  int ans = 0;
  for (int b = 1, c = arr[0]-1; b < arr[0]; b++, c--) ans += fv1(arr, 1, b, c);
  return ans;
}
int fv0(const vector<int>& arr, int i, int preb, int prec) {
  if (i == (int)arr.size()) return 1;
  int ans = 0;
  for (int b = 0, c = arr[i]; b <= arr[i]; b++, c--)
    if (preb <= b && prec >= c) ans += fv0(arr, i+1, b, c);
  return ans;
}
int ways0(const vector<int>& arr) {
  int ans = 0;
  for (int b = 0, c = arr[0]; b <= arr[0]; b++, c--) ans += fv0(arr, 1, b, c);
  return ans;
}
