// 除法求值
// 变量用字符串表示，等式 l/r = v，查询 l/r 的值，推不出返回 -1.0
// 带权并查集维护倍数关系（乘法），dist 存到根的比值
// 测试链接 : https://leetcode.cn/problems/evaluate-division/
#include <bits/stdc++.h>
using namespace std;

unordered_map<string, string> father;
unordered_map<string, double> dist;

string findSet(const string& x) {
    if (!father.count(x)) {
        return "";
    }
    if (x != father[x]) {
        string tmp = father[x];
        string fa = findSet(tmp);
        dist[x] = dist[x] * dist[tmp];
        father[x] = fa;
    }
    return father[x];
}

void unionSet(const string& l, const string& r, double v) {
    string lf = findSet(l), rf = findSet(r);
    if (lf != rf) {
        father[lf] = rf;
        dist[lf] = dist[r] / dist[l] * v;
    }
}

double query(const string& l, const string& r) {
    string lf = findSet(l), rf = findSet(r);
    if (lf.empty() || rf.empty() || lf != rf) {
        return -1.0;
    }
    return dist[l] / dist[r];
}

vector<double> calcEquation(vector<vector<string>>& equations, vector<double>& values,
                            vector<vector<string>>& queries) {
    father.clear();
    dist.clear();
    for (auto& eq : equations) {
        for (auto& key : eq) {
            father[key] = key;
            dist[key] = 1.0;
        }
    }
    for (int i = 0; i < (int)values.size(); i++) {
        unionSet(equations[i][0], equations[i][1], values[i]);
    }
    vector<double> ans;
    for (auto& q : queries) {
        ans.push_back(query(q[0], q[1]));
    }
    return ans;
}

// 银河英雄传说
// 30000 艘战舰，初始各自一队，M l r 把 l 队整体移到 r 队末尾，C l r 查询中间隔几艘
// 1 <= t <= 5e5，迭代 find 防爆栈
// 测试链接 : https://www.luogu.com.cn/problem/P1196
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 30001;

int n = 30000;
int father[MAXN];
int dist[MAXN];
int sizeArr[MAXN];
int stackArr[MAXN];

void prepare() {
    for (int i = 1; i <= n; i++) {
        father[i] = i;
        dist[i] = 0;
        sizeArr[i] = 1;
    }
}

int find(int i) {
    int si = 0;
    while (i != father[i]) {
        stackArr[++si] = i;
        i = father[i];
    }
    stackArr[si + 1] = i;
    for (int j = si; j >= 1; j--) {
        father[stackArr[j]] = i;
        dist[stackArr[j]] += dist[stackArr[j + 1]];
    }
    return i;
}

void unionSet(int l, int r) {
    int lf = find(l), rf = find(r);
    if (lf != rf) {
        father[lf] = rf;
        dist[lf] += sizeArr[rf];
        sizeArr[rf] += sizeArr[lf];
    }
}

int query(int l, int r) {
    if (find(l) != find(r)) {
        return -1;
    }
    return abs(dist[l] - dist[r]) - 1;
}

int main() {
    prepare();
    int t;
    scanf("%d", &t);
    char op;
    for (int i = 1, l, r; i <= t; i++) {
        scanf(" %c%d%d", &op, &l, &r);
        if (op == 'M') {
            unionSet(l, r);
        } else {
            printf("%d\n", query(l, r));
        }
    }
    return 0;
}

// 关押罪犯
// n 个犯人分配到两个监狱，m 条仇恨记录 l r v
// 每个监狱的暴力值 = 该监狱内最深的仇恨，冲突值 = 两监狱暴力值最大值
// 最小化冲突值，返回最小值
// 贪心：按仇恨值降序，两人必须分到不同监狱（enemy 数组维护），
// 出现 must 同监狱的两人时，当前仇恨值就是答案
// 1 <= n <= 2e4，1 <= m <= 1e5，1 <= 仇恨值 <= 1e9
// 测试链接 : https://www.luogu.com.cn/problem/P1525
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 20002;
const int MAXM = 100001;

int n, m;
int father[MAXN];
int enemyArr[MAXN];
struct Edge { int l, r, v; };
Edge arr[MAXM];

void prepare() {
    for (int i = 1; i <= n; i++) {
        father[i] = i;
        enemyArr[i] = 0;
    }
}

int find(int i) {
    father[i] = father[i] == i ? i : find(father[i]);
    return father[i];
}

void unionSet(int l, int r) {
    father[find(l)] = find(r);
}

bool same(int l, int r) {
    return find(l) == find(r);
}

int compute() {
    sort(arr + 1, arr + m + 1, [](const Edge& a, const Edge& b) {
        return a.v > b.v;
    });
    int ans = 0;
    for (int i = 1; i <= m; i++) {
        int l = arr[i].l, r = arr[i].r, v = arr[i].v;
        if (same(l, r)) {
            ans = v;
            break;
        } else {
            if (enemyArr[l] == 0) {
                enemyArr[l] = r;
            } else {
                unionSet(enemyArr[l], r);
            }
            if (enemyArr[r] == 0) {
                enemyArr[r] = l;
            } else {
                unionSet(l, enemyArr[r]);
            }
        }
    }
    return ans;
}

int main() {
    scanf("%d%d", &n, &m);
    prepare();
    for (int i = 1; i <= m; i++) {
        scanf("%d%d%d", &arr[i].l, &arr[i].r, &arr[i].v);
    }
    printf("%d\n", compute());
    return 0;
}

// 甄别食物链
// n 只动物，A/B/C 三类，A吃B、B吃C、C吃A
// 1 X Y 同类，2 X Y X吃Y，统计假话数量
// 带权并查集维护 %3 余数：0 同类、1 吃头、2 被头吃
// 1 <= n <= 5e4，1 <= k <= 1e5
// 测试链接 : https://www.luogu.com.cn/problem/P2024
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 50001;

int n, k, ans;
int father[MAXN];
int dist[MAXN];

void prepare() {
    ans = 0;
    for (int i = 1; i <= n; i++) {
        father[i] = i;
        dist[i] = 0;
    }
}

int find(int i) {
    if (i != father[i]) {
        int tmp = father[i];
        father[i] = find(tmp);
        dist[i] = (dist[i] + dist[tmp]) % 3;
    }
    return father[i];
}

// op == 1：l 和 r 是同类；op == 2：l 吃 r
void unionSet(int op, int l, int r) {
    int lf = find(l), rf = find(r), v = op == 1 ? 0 : 1;
    if (lf != rf) {
        father[lf] = rf;
        dist[lf] = (dist[r] - dist[l] + v + 3) % 3;
    }
}

bool check(int op, int l, int r) {
    if (l > n || r > n || (op == 2 && l == r)) {
        return false;
    }
    if (find(l) == find(r)) {
        if (op == 1) {
            if (dist[l] != dist[r]) {
                return false;
            }
        } else {
            if ((dist[l] - dist[r] + 3) % 3 != 1) {
                return false;
            }
        }
    }
    return true;
}

int main() {
    scanf("%d%d", &n, &k);
    prepare();
    for (int i = 1, op, l, r; i <= k; i++) {
        scanf("%d%d%d", &op, &l, &r);
        if (!check(op, l, r)) {
            ans++;
        } else {
            unionSet(op, l, r);
        }
    }
    printf("%d\n", ans);
    return 0;
}

// 异或关系（HDU3234 / UVA12232）
// n 个数编号 0~n-1，m 次操作：
//   I x v     声明 第x个数 = v
//   I x y v   声明 第x个数 ^ 第y个数 = v
//   Q k a1..ak 查询这些数字异或起来的值
// 带权并查集维护异或关系 + 虚点 n（值为 0 的基准）
// 矛盾后不再处理任何操作；只有声明（I）计数，Q 不计数
// 1 <= n <= 2e4，1 <= m <= 4e4，1 <= k <= 15，多测，n=m=0 结束
// 测试链接 : https://acm.hdu.edu.cn/showproblem.php?pid=3234
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 20002;
const int MAXK = 21;

int n, m;
bool conflict;
int cnti;
int father[MAXN];
int exclu[MAXN];
int nums[MAXK];
int fas[MAXK];

void prepare() {
    conflict = false;
    cnti = 0;
    for (int i = 0; i <= n; i++) {
        father[i] = i;
        exclu[i] = 0;
    }
}

int find(int i) {
    if (i != father[i]) {
        int tmp = father[i];
        father[i] = find(tmp);
        exclu[i] ^= exclu[tmp];
    }
    return father[i];
}

bool opi(int l, int r, int v) {
    cnti++;
    int lf = find(l), rf = find(r);
    if (lf == rf) {
        if ((exclu[l] ^ exclu[r]) != v) {
            conflict = true;
            return false;
        }
    } else {
        if (lf == n) {
            lf = rf;
            rf = n;
        }
        father[lf] = rf;
        exclu[lf] = exclu[r] ^ exclu[l] ^ v;
    }
    return true;
}

int opq(int k) {
    int ans = 0;
    for (int i = 1; i <= k; i++) {
        int fa = find(nums[i]);
        ans ^= exclu[nums[i]];
        fas[i] = fa;
    }
    sort(fas + 1, fas + k + 1);
    for (int l = 1, r = 1; l <= k; l = ++r) {
        while (r + 1 <= k && fas[r + 1] == fas[l]) {
            r++;
        }
        if ((r - l + 1) % 2 != 0 && fas[l] != n) {
            return -1;
        }
    }
    return ans;
}

                enemyArr[r] = l;
            } else {

            }
        }
    }

// 除法求值
// 变量用字符串表示，等式 l/r = v，查询 l/r 的值，推不出返回 -1.0
// 带权并查集维护倍数关系（乘法），dist 存到根的比值
// 测试链接 : https://leetcode.cn/problems/evaluate-division/
#include <bits/stdc++.h>
using namespace std;

unordered_map<string, string> father;
unordered_map<string, double> dist;

string findSet(const string& x) {
    if (!father.count(x)) {
        return "";
    }
    if (x != father[x]) {
        string tmp = father[x];
        string fa = findSet(tmp);
        dist[x] = dist[x] * dist[tmp];
        father[x] = fa;
    }
    return father[x];
}

void unionSet(const string& l, const string& r, double v) {
    string lf = findSet(l), rf = findSet(r);
    if (lf != rf) {
        father[lf] = rf;
        dist[lf] = dist[r] / dist[l] * v;
    }
}

double query(const string& l, const string& r) {
    string lf = findSet(l), rf = findSet(r);
    if (lf.empty() || rf.empty() || lf != rf) {
        return -1.0;
    }
    return dist[l] / dist[r];
}

vector<double> calcEquation(vector<vector<string>>& equations, vector<double>& values,
                            vector<vector<string>>& queries) {
    father.clear();
    dist.clear();
    for (auto& eq : equations) {
        for (auto& key : eq) {
            father[key] = key;
            dist[key] = 1.0;
        }
    }
    for (int i = 0; i < (int)values.size(); i++) {
        unionSet(equations[i][0], equations[i][1], values[i]);
    }
    vector<double> ans;
    for (auto& q : queries) {
        ans.push_back(query(q[0], q[1]));
    }
    return ans;
}

// n 个数编号 0~n-1，m 次操作：
//   I x v     声明 第x个数 = v
//   I x y v   声明 第x个数 ^ 第y个数 = v
//   Q k a1..ak 查询这些数字异或起来的值
// 带权并查集维护异或关系 + 虚点 n（值为 0 的基准）
// 矛盾后不再处理任何操作；只有声明（I）计数，Q 不计数
// 1 <= n <= 2e4，1 <= m <= 4e4，1 <= k <= 15，多测，n=m=0 结束

const int MAXK = 21;

int exclu[MAXN];
int nums[MAXK];

    conflict = false;
    cnti = 0;
    for (int i = 0; i <= n; i++) {

        exclu[i] = 0;
    }
}

int find(int i) {
    if (i != father[i]) {
        int tmp = father[i];
        father[i] = find(tmp);
        exclu[i] ^= exclu[tmp];
    }
    return father[i];
}

bool opi(int l, int r, int v) {
    cnti++;
    int lf = find(l), rf = find(r);
    if (lf == rf) {
        if ((exclu[l] ^ exclu[r]) != v) {
            conflict = true;
            return false;
        }
    } else {
        if (lf == n) {
            lf = rf;
            rf = n;
        }
        father[lf] = rf;
        exclu[lf] = exclu[r] ^ exclu[l] ^ v;
    }
    return true;
}

int opq(int k) {
    int ans = 0;
    for (int i = 1; i <= k; i++) {
        int fa = find(nums[i]);
        ans ^= exclu[nums[i]];
        fas[i] = fa;
    }
    sort(fas + 1, fas + k + 1);
    for (int l = 1, r = 1; l <= k; l = ++r) {
        while (r + 1 <= k && fas[r + 1] == fas[l]) {
            r++;
        }
        if ((r - l + 1) % 2 != 0 && fas[l] != n) {
            return -1;
        }
    }
    return ans;
}

                if w not in seen:

        else:
            out.append("I don't know.")

def brute_criminal(n, edges):
    # 二分答案 + BFS 二染色：存在分配使 >w 的边全部跨监狱 <-> >w 的边构成二部图
    def feasible(w):
        adj = {i: [] for i in range(1, n + 1)}
        for (l, r, v) in edges:
            if v > w:
                adj[l].append(r)
                adj[r].append(l)
        color = {}
        for s in range(1, n + 1):
            if s in color:
                continue
            color[s] = 0
            stack = [s]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if v not in color:
                        color[v] = color[u] ^ 1
                        stack.append(v)
                    elif color[v] == color[u]:
                        return False
        return True
    # 答案在 {0} U {边权} 中取最小可行
    cands = sorted(set([0] + [v for (_, _, v) in edges]))
    for w in cands:
        if feasible(w):
            return w
    return max(v for (_, _, v) in edges)

    for (op, a, b) in ops:
        if op == 'F':
            union(a, b)
        else:
            for e in enemy[a]:
                union(e, b)
            enemy[a].add(b)
            for e in enemy[b]:
                union(e, a)
            enemy[b].add(a)
    return sum(1 for i in range(1, n + 1) if find(i) == i)

def fmt_interval_in(n, ops, queries):
    lines = ['%d %d %d' % (n, len(ops), len(queries))]
    for (l, r, v) in ops:
        lines.append('%d %d %d' % (l, r, v))
    for (l, r) in queries:
        lines.append('%d %d' % (l, r))
    return '\n'.join(lines) + '\n'

def fmt_interval_check_in(n, ops, multi=False):
    if multi:
        lines = ['1', '%d %d' % (n, len(ops))]
        for (l, r, v) in ops:
            lines.append('%d %d %d' % (l, r, v))
    else:
        lines = ['%d %d' % (n, len(ops))]
        for (l, r, v) in ops:
            lines.append('%d %d %d' % (l, r, v))
    return '\n'.join(lines) + '\n'

def fmt_div_in(eqs, vals, qs):
    lines = [str(len(eqs))]
    for (a, b), v in zip(eqs, vals):
        lines.append('%s %s %.6f' % (a, b, v))
    lines.append(str(len(qs)))
    for (a, b) in qs:
        lines.append('%s %s' % (a, b))
    return '\n'.join(lines) + '\n'

def fmt_food_in(n, ops):
    lines = ['%d %d' % (n, len(ops))]
    for (op, l, r) in ops:
        lines.append('%d %d %d' % (op, l, r))
    return '\n'.join(lines) + '\n'

def fmt_xor_in(n, stmts, queries):
    # 需要按时间顺序混合输出（I 与 Q 交错），但我们生成时分开存了
    # 改：重新交错 —— 简单方案：随机打乱顺序并记录原次序语义
    # 但课程语义：I 计数 + 查询都基于当前图。这里重新生成交错流：
    lines = ['%d %d' % (n, len(stmts) + len(queries))]
    # 为保持声明/查询顺序语义，交替生成
    # 简化：按 stmts 全部先声明再查询
    for (l, r, v) in stmts:
        if r == n:
            lines.append('I %d %d' % (l, v))
        else:
            lines.append('I %d %d %d' % (l, r, v))
    for qs in queries:
        lines.append('Q %d %s' % (len(qs), ' '.join(map(str, qs))))
    lines.append('0 0')
    return '\n'.join(lines) + '\n'

def fmt_galaxy_in(n, ops):
    lines = [str(len(ops))]
    for (op, l, r) in ops:
        lines.append('%s %d %d' % (op, l, r))

def fmt_criminal_in(n, edges):

def fmt_gang_in(n, ops):

def check(name, got, want, detail=''):
    got_s = got.strip().splitlines()
    want_s = want.strip().splitlines()
    if got_s == want_s:
        return True
    print('[%s] MISMATCH' % name)
    print('  got :', got_s[:12])
    print('  want:', want_s[:12])
    if detail:
        print('  data:', detail)
    return False
