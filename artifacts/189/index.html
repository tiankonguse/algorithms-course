强连通分量和缩点-上
左程云
强连通分量，SCC，Strongly Connected Components1，有向图中，一些点可以彼此相互到达，这些点就构成一个强连通分量2，一个强连通分量缩成一个点，原图会变成DAG(Directed Acyclic Graph)，有向无环图3，因为没有环，求解问题变得非常方便，最常用的求解手段就是拓扑排序4，需要保证缩点之后，求解答案不受影响5，求解强连通分量，掌握Tarjan算法即可，其他方法不再讲述课上重点图解，普通图缩点变成DAG

强连通分量和缩点-上

Tarjan算法的必备概念1，边分为三种类型，树边、回边、弃边2，dfn[u]，节点u获得的dfn序号3，low[u]，u及其子树上的点，最多走一条回边，能到达的树的最上方点，是什么dfn序号4，belong[u]，节点u分配的强连通分量序号，如果belong[u]为0，表示还没分配5，遍历图的过程中，准备一个栈sta，用来收集遍历到的节点6，遍历到的节点先进栈，一旦该节点分配了强连通分量，就从栈中弹出课上重点图解，边的类型，low的含义
--- slide 4 ---
强连通分量和缩点-上
左程云
Tarjan算法的过程，直观理解为 扎口袋算法1，初次来到u时，dfn[u] = low[u] = 分配的dfn序号，然后节点u进栈2，遍历u的每个儿子v，讨论u和v之间的边是什么类型，如果是弃边，什么也不做3，如果是树边，先递归计算low[v]，然后low[u] = min(low[u], low[v])4，如果是回边，直接low[u] = min(low[u], dfn[v])5，遍历结束后，判断dfn[u]是否等于low[u]，如果不等，代表节点u不能扎起口袋6，如果相等，代表节点u能扎起口袋，栈弹出节点，直到u出现时停止，这批节点属于一个强连通分量7，考察图中每个节点，一旦发现dfn[i] == 0，就从i节点开始，执行一遍Tarjan算法8，最终得到图中全部强连通分量，时间复杂度O(n + m)，n为节点数，m为边数课上重点图解，Tarjan算法的过程

code/Code01.cpp

#include <bits/stdc++.h>

using namespace std;

const int MAXN = 10001;
const int MAXM = 100001;
int n, m;

int head[MAXN];
int nxt[MAXM];
int to[MAXM];
int cntg;

int dfn[MAXN];
int low[MAXN];
int cntd;

int sta[MAXN];
int top;

int belong[MAXN];
int sccArr[MAXN];
int sccl[MAXN];
int sccr[MAXN];
int idx;
int sccCnt;

bool sccPrint[MAXN];

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;
}

void tarjan(int u) {
    dfn[u] = low[u] = ++cntd;
    sta[++top] = u;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (dfn[v] == 0) {
            tarjan(v);
            low[u] = min(low[u], low[v]);
        } else {
            if (belong[v] == 0) {
                low[u] = min(low[u], dfn[v]);
            }
        }
    }
    if (dfn[u] == low[u]) {
        sccCnt++;
        sccl[sccCnt] = idx + 1;
        int pop;
        do {
            pop = sta[top--];
            belong[pop] = sccCnt;
            sccArr[++idx] = pop;
        } while (pop != u);
        sccr[sccCnt] = idx;
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m;
    for (int i = 1, u, v; i <= m; i++) {
        cin >> u >> v;
        addEdge(u, v);
    }
    for (int i = 1; i <= n; i++) {
        if (dfn[i] == 0) {
            tarjan(i);
        }
    }
    cout << sccCnt << "\n";
    for (int i = 1; i <= sccCnt; i++) {
        sort(sccArr + sccl[i], sccArr + sccr[i] + 1);
    }
    for (int i = 1; i <= n; i++) {
        int scc = belong[i];
        if (!sccPrint[scc]) {
            sccPrint[scc] = true;
            for (int j = sccl[scc]; j <= sccr[scc]; j++) {
                cout << sccArr[j] << " ";
            }
            cout << "\n";
        }
    }
    return 0;
}

            addEdge(a, b);
        } else {
            addEdge(a, b);
            addEdge(b, a);

            tarjan(i);

// 用法: brute189.run <题号1..10>，读对应格式的输入，按题面输出
// 核心：Floyd 可达性 + SCC 集合 + 缩点度数/点权

static int n;
static vector<vector<bool>> R; // 可达矩阵

static vector<vector<int>> sccs;    // 每个分量内的点
static vector<int> belong;          // 点 -> 分量下标(0-based)
static vector<vector<bool>> C;      // 缩点图可达（分量间）

static void buildSCC(){
    // R 已算好（含自环）
    vector<bool> used(n+1,false);
    sccs.clear();
    belong.assign(n+1,-1);
    for(int i=1;i<=n;i++){
        if(used[i]) continue;
        // 分量 = 与 i 双向可达的点
        vector<int> g;
        for(int j=1;j<=n;j++){
            if(!used[j] && R[i][j] && R[j][i]) g.push_back(j);
        }
        for(int x:g){ used[x]=true; belong[x]=sccs.size(); }
        sort(g.begin(),g.end());
        sccs.push_back(g);
    }
    // 缩点图可达：分量 a 到 b（a!=b）存在原边
    int k=sccs.size();
    C.assign(k, vector<bool>(k,false));
    for(int a=0;a<k;a++){
        for(int b=0;b<k;b++){
            if(a==b) continue;
            for(int x:sccs[a]){
                for(int y:sccs[b]){
                    if(R[x][y]){ C[a][b]=true; break; }
                }
                if(C[a][b]) break;
            }
        }
    }
}

static void calcReach(const vector<pair<int,int>>& edges, int nn){
    n=nn;
    R.assign(n+1, vector<bool>(n+1,false));
    for(int i=1;i<=n;i++) R[i][i]=true;
    for(auto&e:edges) R[e.first][e.second]=true;
    for(int k2=1;k2<=n;k2++)
        for(int i=1;i<=n;i++) if(R[i][k2])
            for(int j=1;j<=n;j++) if(R[k2][j]) R[i][j]=true;
    buildSCC();
}

static int in0cnt(){ // 入度为 0 的分量数
    int k=sccs.size(), cnt=0;
    for(int a=0;a<k;a++){
        bool has=false;
        for(int b=0;b<k;b++) if(C[b][a]){ has=true; break; }
        if(!has) cnt++;
    }
    return cnt;
}
static int out0cnt(){
    int k=sccs.size(), cnt=0;
    for(int a=0;a<k;a++){
        bool has=false;
        for(int b=0;b<k;b++) if(C[a][b]){ has=true; break; }
        if(!has) cnt++;
    }
    return cnt;
}

def gen_directed(n, maxm):
    cap = n * (n - 1) // 2
    m = r.randint(0, min(maxm, cap))
    es = set()
    while len(es) < m:
        a = r.randint(1, n); b = r.randint(1, n)
        if a == b: continue
        es.add((a, b))
    return list(es)
