// descartes 对拍：随机数组 → 课程建树 → 校验 BST(中序=1..n) + 小根堆(父值<=子值)
#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1005;
int arr[MAXN], ls[MAXN], rs[MAXN], sta[MAXN], n;

void build() {
    int top = 0;
    for (int i = 1; i <= n; i++) {
        int pos = top;
        while (pos > 0 && arr[sta[pos]] > arr[i]) pos--;
        if (pos > 0) rs[sta[pos]] = i;
        if (pos < top) ls[i] = sta[pos + 1];
        sta[++pos] = i;
        top = pos;
    }
}

int inorder[MAXN], icnt;
void collect(int u) {
    if (!u) return;
    collect(ls[u]); inorder[++icnt] = u; collect(rs[u]);
}

bool check(int u) { // 堆性质
    if (!u) return true;
    if (ls[u] && arr[ls[u]] < arr[u]) return false;
    if (rs[u] && arr[rs[u]] < arr[u]) return false;
    return check(ls[u]) && check(rs[u]);
}

int main() {
    srand(20261018);
    int total = 0, fails = 0;
    for (int round = 0; round < 300; round++) {
        n = 1 + rand() % 20;
        for (int i = 1; i <= n; i++) arr[i] = rand() % 10;
        memset(ls, 0, sizeof(ls)); memset(rs, 0, sizeof(rs));
        build();
        icnt = 0; collect(sta[1]);
        bool ok = true;
        for (int i = 1; i <= n; i++) if (inorder[i] != i) ok = false; // BST 中序
        if (!check(sta[1])) ok = false;
        // 根：整段最小值里最左的（等值不弹 → 先出现者在上）
        int minv = arr[1], minpos = 1;
        for (int i = 2; i <= n; i++) if (arr[i] < minv) { minv = arr[i]; minpos = i; }
        if (sta[1] != minpos) ok = false;
        total++;
        if (!ok) {
            fails++;
            if (fails <= 3) {
                printf("FAIL round=%d n=%d arr:", round, n);
                for (int i = 1; i <= n; i++) printf("%d ", arr[i]);
                printf("root=%d expect=%d\n", sta[1], minpos);
            }
        }
    }
    printf("== descartes: total %d cases, fails %d ==\n", total, fails);
    return 0;
}

code/counting.cpp

#include <bits/stdc++.h>

using namespace std;

const int MOD = 1000000007;
const int MAXN = 1000001;

// 所有数字
int arr[MAXN];

// 笛卡尔树需要
int left_[MAXN];
int right_[MAXN];
int stack_[MAXN];

// tmp是动态规划的临时结果
long long tmp[MAXN];

int n, m;

// 动态规划表，n * m <= 10^6，所以临时申请
vector<vector<int>> dp;

void build() {
    for (int i = 1, top = 0, pos; i <= n; i++) {
        pos = top;
        while (pos > 0 && arr[stack_[pos]] < arr[i]) {
            pos--;
        }
        if (pos > 0) {
            right_[stack_[pos]] = i;
        }
        if (pos < top) {
            left_[i] = stack_[pos + 1];
        }
        stack_[++pos] = i;
        top = pos;
    }
}

void dfs(int u) {
    if (u == 0) {
        return;
    }
    dfs(left_[u]);
    dfs(right_[u]);
    for (int j = 1; j <= m; j++) {
        tmp[j] = 1LL * dp[left_[u]][j - 1] * dp[right_[u]][j] % MOD;
    }
    for (int j = 1; j <= m; j++) {
        dp[u][j] = (dp[u][j - 1] + tmp[j]) % MOD;
    }
}

long long compute() {
    build();
    // 虽然n * m <= 10^6，但是n和m，单独都是2 * 10^5规模
    // 所以提前准备固定大小的表是不可能的，空间根本接受不了
    // 所以根据此时具体的n和m的大小，临时申请动态规划表，总大小不超过10^6
    dp.assign(n + 1, vector<int>(m + 1, 0));
    for (int j = 0; j <= m; j++) {
        // 没有节点时，不管要求节点值是什么，都默认有1种形态
        // 因为没有节点时，根本不影响任何决策
        dp[0][j] = 1;
    }
    dfs(stack_[1]);
    return dp[stack_[1]][m];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int cases;
    cin >> cases;
    for (int t = 1; t <= cases; t++) {
        cin >> n;
        cin >> m;
        for (int i = 1; i <= n; i++) {
            cin >> arr[i];
        }
        cout << compute() << endl;
    }
    return 0;
}

code/followup2.cpp

#include <bits/stdc++.h>

using namespace std;

const int MAXN = 2000001;

int cnt = 0;
int head = 0;
int key[MAXN];
int key_count[MAXN];
int ls[MAXN];
int rs[MAXN];
int siz[MAXN];
double priority[MAXN];

void up(int i) {
    siz[i] = siz[ls[i]] + siz[rs[i]] + key_count[i];
}

int leftRotate(int i) {
    int r = rs[i];
    rs[i] = ls[r];
    ls[r] = i;
    up(i);
    up(r);
    return r;
}

int rightRotate(int i) {
    int l = ls[i];
    ls[i] = rs[l];
    rs[l] = i;
    up(i);
    up(l);
    return l;
}

int add(int i, int num) {
    if (i == 0) {
        key[++cnt] = num;
        key_count[cnt] = siz[cnt] = 1;
        priority[cnt] = static_cast<double>(rand()) / RAND_MAX;
        return cnt;
    }
    if (key[i] == num) {
        key_count[i]++;
    } else if (key[i] > num) {
        ls[i] = add(ls[i], num);
    } else {
        rs[i] = add(rs[i], num);
    }
    up(i);
    if (ls[i] != 0 && priority[ls[i]] > priority[i]) {
        return rightRotate(i);
    }
    if (rs[i] != 0 && priority[rs[i]] > priority[i]) {
        return leftRotate(i);
    }
    return i;
}

void add(int num) {
    head = add(head, num);
}

int small(int i, int num) {
    if (i == 0) {
        return 0;
    }
    if (key[i] >= num) {
        return small(ls[i], num);
    } else {
        return siz[ls[i]] + key_count[i] + small(rs[i], num);
    }
}

int getRank(int num) {
    return small(head, num) + 1;
}

int index(int i, int x) {
    if (siz[ls[i]] >= x) {
        return index(ls[i], x);
    } else if (siz[ls[i]] + key_count[i] < x) {
        return index(rs[i], x - siz[ls[i]] - key_count[i]);
    }
    return key[i];
}

int index(int x) {
    return index(head, x);
}

int pre(int i, int num) {
    if (i == 0) {
        return INT_MIN;
    }
    if (key[i] >= num) {
        return pre(ls[i], num);
    } else {
        return max(key[i], pre(rs[i], num));
    }
}

int pre(int num) {
    return pre(head, num);
}

int post(int i, int num) {
    if (i == 0) {
        return INT_MAX;
    }
    if (key[i] <= num) {
        return post(rs[i], num);
    } else {
        return min(key[i], post(ls[i], num));
    }
}

int post(int num) {
    return post(head, num);
}

int remove(int i, int num) {
    if (key[i] < num) {
        rs[i] = remove(rs[i], num);
    } else if (key[i] > num) {
        ls[i] = remove(ls[i], num);
    } else {
        if (key_count[i] > 1) {
            key_count[i]--;
        } else {
            if (ls[i] == 0 && rs[i] == 0) {
                return 0;
            } else if (ls[i] != 0 && rs[i] == 0) {
                i = ls[i];
            } else if (ls[i] == 0 && rs[i] != 0) {
                i = rs[i];
            } else {
                if (priority[ls[i]] >= priority[rs[i]]) {
                    i = rightRotate(i);
                    rs[i] = remove(rs[i], num);
                } else {
                    i = leftRotate(i);
                    ls[i] = remove(ls[i], num);
                }
            }
        }
    }
    up(i);
    return i;
}

void remove(int num) {
    if (getRank(num) != getRank(num + 1)) {
        head = remove(head, num);
    }
}

void clear() {
    memset(key + 1, 0, cnt * sizeof(int));
    memset(key_count + 1, 0, cnt * sizeof(int));
    memset(ls + 1, 0, cnt * sizeof(int));
    memset(rs + 1, 0, cnt * sizeof(int));
    memset(siz + 1, 0, cnt * sizeof(int));
    memset(priority + 1, 0, cnt * sizeof(int));
    cnt = 0;
    head = 0;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    srand(time(0));
    int n, m, lastAns = 0, ans = 0;
    cin >> n;
    cin >> m;
    for (int i = 1, num; i <= n; i++) {
        cin >> num;
        add(num);
    }
    for (int i = 1, op, x; i <= m; i++) {
        cin >> op >> x;
        x ^= lastAns;
        if (op == 1) {
            add(x);
        } else if (op == 2) {
            remove(x);
        } else if (op == 3) {
            lastAns = getRank(x);
            ans ^= lastAns;
        } else if (op == 4) {
            lastAns = index(x);
            ans ^= lastAns;
        } else if (op == 5) {
            lastAns = pre(x);
            ans ^= lastAns;
        } else {
            lastAns = post(x);
            ans ^= lastAns;
        }
    }
    cout << ans << endl;
    clear();
    return 0;
}

    srand(2026);
    int N = 100000;
    auto t0 = chrono::steady_clock::now();
    for (int i = 0; i < N; i++) add(rand() % 100000000);
    for (int i = 0; i < N; i++) getRank(rand() % 100000000);
    auto t1 = chrono::steady_clock::now();
    printf("treap 20万操作(10万插+10万查): %.3fs 节点=%d\n", chrono::duration<double>(t1-t0).count(), cnt);
    // 树深
    int maxd = 0;
    for (int i = 1; i <= cnt; i++) {
        int cur = i, dep = 0;
        while (ls[cur] || rs[cur]) { cur = ls[cur] ? ls[cur] : rs[cur]; dep++; }
        maxd = max(maxd, dep);
    }
    printf("treap max depth: %d\n", maxd);
    return 0;
}

code/periodni.cpp

#include <bits/stdc++.h>

using namespace std;

const int MOD = 1000000007;
const int MAXN = 501;
const int MAXH = 1000000;

// 所有数字
int arr[MAXN];

// 阶乘余数表
int fac[MAXH + 1];

// 阶乘逆元表
int inv[MAXH + 1];

// 笛卡尔树需要
int left_[MAXN];
int right_[MAXN];
int stack_[MAXN];

// dfs需要
int size_[MAXN];

// dp是动态规划表
int dp[MAXN][MAXN];

// tmp是动态规划的临时结果
int tmp[MAXN];

int n, k;

int power(long long x, long long p) {
    long long ans = 1;
    while (p > 0) {
        if ((p & 1) == 1) {
            ans = ans * x % MOD;
        }
        x = x * x % MOD;
        p >>= 1;
    }
    return (int) ans;
}

int c(int n, int k) {
    return n < k ? 0 : (int) ((long long) fac[n] * inv[k] % MOD * inv[n - k] % MOD);
}

void build() {
    fac[0] = fac[1] = inv[0] = 1;
    for (int i = 2; i <= MAXH; i++) {
        fac[i] = (int) ((long long) fac[i - 1] * i % MOD);
    }
    inv[MAXH] = power(fac[MAXH], MOD - 2);
    for (int i = MAXH - 1; i >= 1; i--) {
        inv[i] = (int) ((long long) inv[i + 1] * (i + 1) % MOD);
    }
    for (int i = 1, top = 0, pos; i <= n; i++) {
        pos = top;
        while (pos > 0 && arr[stack_[pos]] > arr[i]) {
            pos--;
        }
        if (pos > 0) {
            right_[stack_[pos]] = i;
        }
        if (pos < top) {
            left_[i] = stack_[pos + 1];
        }
        stack_[++pos] = i;
        top = pos;
    }
}

void dfs(int u, int fa) {
    if (u == 0) {
        return;
    }
    dfs(left_[u], u);
    dfs(right_[u], u);
    size_[u] = size_[left_[u]] + size_[right_[u]] + 1;
    for (int i = 0; i <= k; i++) {
        tmp[i] = 0;
    }
    // 所有dfs过程都算上，这一部分的总复杂度O(n^2)
    for (int l = 0; l <= min(size_[left_[u]], k); l++) {
        for (int r = 0; r <= min(size_[right_[u]], k - l); r++) {
            tmp[l + r] = (tmp[l + r] + (long long) dp[left_[u]][l] * dp[right_[u]][r] % MOD) % MOD;
        }
    }
    // 所有dfs过程都算上，这一部分的总复杂度O(min(n的3次方, n * k平方))
    for (int i = 0; i <= min(size_[u], k); i++) {
        for (int p = 0; p <= i; p++) {
            dp[u][i] = (dp[u][i] + (long long) c(size_[u] - p, i - p) * c(arr[u] - arr[fa], i - p) % MOD
                        * fac[i - p] % MOD * tmp[p] % MOD) % MOD;
        }
    }
}

int compute() {
    build();
    dp[0][0] = 1;
    dfs(stack_[1], 0);
    return dp[stack_[1]][k];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    cin >> k;
    for (int i = 1; i <= n; i++) {
        cin >> arr[i];
    }
    cout << compute() << endl;
    return 0;
}

code/removing.cpp

#include <bits/stdc++.h>

using namespace std;

const int MOD = 1000000007;
const int MAXN = 100001;

// 所有数字
int arr[MAXN];

// 连续数字逆元表
int inv[MAXN];

// sum[i] = (1/1 + 1/2 + 1/3 + ... + 1/i)，%MOD意义下的余数
int sum_[MAXN];

int n;

void build() {
    inv[1] = 1;
    for (int i = 2; i <= n; i++) {
        inv[i] = (int) (MOD - (long long) inv[MOD % i] * (MOD / i) % MOD);
    }
    for (int i = 1; i <= n; i++) {
        sum_[i] = (sum_[i - 1] + inv[i]) % MOD;
    }
}

long long compute() {
    build();
    long long ans = 0;
    for (int i = 1; i <= n; i++) {
        ans = (ans + (long long) (sum_[i] + sum_[n - i + 1] - 1) * arr[i]) % MOD;
    }
    for (int i = 1; i <= n; i++) {
        ans = ans * i % MOD;
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> arr[i];
    }
    cout << compute() << endl;
    return 0;
}

        ls[i] = remove(ls[i], num);
    } else {
        if (key_count[i] > 1) {
            key_count[i]--;
        } else {
            if (ls[i] == 0 && rs[i] == 0) {
                return 0;
            } else if (ls[i] != 0 && rs[i] == 0) {
                i = ls[i];
            } else if (ls[i] == 0 && rs[i] != 0) {
                i = rs[i];
            } else {
                if (priority[ls[i]] >= priority[rs[i]]) {
                    i = rightRotate(i);
                    rs[i] = remove(rs[i], num);
                } else {
                    i = leftRotate(i);
                    ls[i] = remove(ls[i], num);
                }
            }
        }
    }
    up(i);
    return i;
}

void remove(int num) {
    if (getRank(num) != getRank(num + 1)) {
        head = remove(head, num);
    }
}

void clear() {
    memset(key + 1, 0, cnt * sizeof(int));
    memset(key_count + 1, 0, cnt * sizeof(int));
    memset(ls + 1, 0, cnt * sizeof(int));
    memset(rs + 1, 0, cnt * sizeof(int));
    memset(siz + 1, 0, cnt * sizeof(int));
    memset(priority + 1, 0, cnt * sizeof(int));

    head = 0;
}

            pos--;
        }
        if (pos > 0) {
            right_[stack_[pos]] = i;
        }
        if (pos < top) {
            left_[i] = stack_[pos + 1];
        }
        stack_[++pos] = i;
        top = pos;
    }
}

            stack_[++size] = right_[cur];
        }
        if (left_[cur] != 0) {
            stack_[++size] = left_[cur];
        }
    }
}

// counting CF1748E 对拍：小 n,m 暴力枚举 B 验证最左端最大值位置一致
#include <bits/stdc++.h>
using namespace std;
const int MOD = 1000000007;
const int MAXN = 1005;
int arr[MAXN], left_[MAXN], right_[MAXN], stack_[MAXN];
long long tmp[MAXN];
int n, m;
vector<vector<int>> dp;

void build() {
    for (int i = 1, top = 0, pos; i <= n; i++) {
        pos = top;
        while (pos > 0 && arr[stack_[pos]] < arr[i]) pos--;
        if (pos > 0) right_[stack_[pos]] = i;
        if (pos < top) left_[i] = stack_[pos + 1];
        stack_[++pos] = i;
        top = pos;
    }
}
void dfs(int u) {
    if (u == 0) return;
    dfs(left_[u]); dfs(right_[u]);
    for (int j = 1; j <= m; j++) tmp[j] = 1LL * dp[left_[u]][j-1] * dp[right_[u]][j] % MOD;
    for (int j = 1; j <= m; j++) dp[u][j] = (dp[u][j-1] + tmp[j]) % MOD;
}
long long compute() {
    build();
    dp.assign(n + 1, vector<int>(m + 1, 0));
    for (int j = 0; j <= m; j++) dp[0][j] = 1;
    dfs(stack_[1]);
    return dp[stack_[1]][m];
}

    int best = l;
    for (int i = l + 1; i <= r; i++) if (v[i] > v[best]) best = i;
    return best;
}
int leftMaxPos(const int* v, int l, int r) {
    int best = l;
    for (int i = l + 1; i <= r; i++) if (v[i] > v[best]) best = i;
    return best;
}

// followup2 P6136 强制在线对拍：初始 n 个 + m 个操作（x^=lastAns），输出 ans 异或
#include <bits/stdc++.h>
using namespace std;

// ===== 课程代码（followup2.cpp 核心，去 main）=====
const int MAXN = 2000001;
int cnt = 0, head = 0;
int key[MAXN], key_count[MAXN], ls[MAXN], rs[MAXN], siz[MAXN];
double priority[MAXN];
void up(int i) { siz[i] = siz[ls[i]] + siz[rs[i]] + key_count[i]; }
int leftRotate(int i) { int r = rs[i]; rs[i] = ls[r]; ls[r] = i; up(i); up(r); return r; }
int rightRotate(int i) { int l = ls[i]; ls[i] = rs[l]; rs[l] = i; up(i); up(l); return l; }
int add(int i, int num) {
    if (i == 0) { key[++cnt] = num; key_count[cnt] = siz[cnt] = 1; priority[cnt] = (double)rand() / RAND_MAX; return cnt; }
    if (key[i] == num) key_count[i]++;
    else if (key[i] > num) ls[i] = add(ls[i], num);
    else rs[i] = add(rs[i], num);
    up(i);
    if (ls[i] != 0 && priority[ls[i]] > priority[i]) return rightRotate(i);
    if (rs[i] != 0 && priority[rs[i]] > priority[i]) return leftRotate(i);
    return i;
}
void add(int num) { head = add(head, num); }
int small(int i, int num) {
    if (i == 0) return 0;
    if (key[i] >= num) return small(ls[i], num);
    else return siz[ls[i]] + key_count[i] + small(rs[i], num);
}
int getRank(int num) { return small(head, num) + 1; }
int index(int i, int x) {
    if (siz[ls[i]] >= x) return index(ls[i], x);
    else if (siz[ls[i]] + key_count[i] < x) return index(rs[i], x - siz[ls[i]] - key_count[i]);
    return key[i];
}
int index(int x) { return index(head, x); }
int pre(int i, int num) {
    if (i == 0) return INT_MIN;
    if (key[i] >= num) return pre(ls[i], num);
    else return max(key[i], pre(rs[i], num));
}
int pre(int num) { return pre(head, num); }
int post(int i, int num) {
    if (i == 0) return INT_MAX;
    if (key[i] <= num) return post(rs[i], num);
    else return min(key[i], post(ls[i], num));
}
int post(int num) { return post(head, num); }
int remove_(int i, int num) {
    if (key[i] < num) rs[i] = remove_(rs[i], num);
    else if (key[i] > num) ls[i] = remove_(ls[i], num);
    else {
        if (key_count[i] > 1) key_count[i]--;
        else {
            if (ls[i] == 0 && rs[i] == 0) return 0;
            else if (ls[i] != 0 && rs[i] == 0) i = ls[i];
            else if (ls[i] == 0 && rs[i] != 0) i = rs[i];
            else {
                if (priority[ls[i]] >= priority[rs[i]]) { i = rightRotate(i); rs[i] = remove_(rs[i], num); }
                else { i = leftRotate(i); ls[i] = remove_(ls[i], num); }
            }
        }
    }
    up(i);
    return i;
}
void remove_(int num) { if (getRank(num) != getRank(num + 1)) head = remove_(head, num); }
void clear() {
    memset(key + 1, 0, cnt * sizeof(int));
    memset(key_count + 1, 0, cnt * sizeof(int));
    memset(ls + 1, 0, cnt * sizeof(int));
    memset(rs + 1, 0, cnt * sizeof(int));
    memset(siz + 1, 0, cnt * sizeof(int));
    memset(priority + 1, 0, cnt * sizeof(int));
    cnt = 0; head = 0;
}
// ===== 课程代码结束 =====

int main() {
    srand(20261019);
    int total = 0, fails = 0;
    for (int round = 0; round < 300; round++) {
        clear();
        int n = 50 + rand() % 100;
        int m = 200;
        vector<int> init;
        for (int i = 0; i < n; i++) init.push_back(rand() % 201 - 100);
        vector<pair<int,int>> ops; // 明文 (op, x)
        for (int i = 0; i < m; i++) {
            int op = 1 + rand() % 6;
            int x = rand() % 201 - 100;
            ops.push_back({op, x});
        }
        // 课程（交互编码）与暴力各维护自己的 lastAns / ans
        for (int v : init) add(v);
        multiset<int> ms;
        for (int v : init) ms.insert(v);
        long long curLast = 0, curAns = 0;
        long long brLast = 0, brAns = 0;
        for (auto& o : ops) {
            int op = o.first;
            int xc = o.second ^ curLast; // 课程端解码
            int xb = o.second ^ brLast;  // 暴力端解码（应为同一值）
            if (op == 1) { add(xc); ms.insert(xb); }
            else if (op == 2) {
                remove_(xc);
                auto it = ms.find(xb);
                if (it != ms.end()) ms.erase(it);
            } else if (op == 3) {
                curLast = getRank(xc); curAns ^= curLast;
                brLast = distance(ms.begin(), ms.lower_bound(xb)) + 1; brAns ^= brLast;
            } else if (op == 4) {
                if (xc < 1 || xc > (int)ms.size()) { curLast = 0; }
                else curLast = index(xc);
                curAns ^= curLast;
                if (xb < 1 || xb > (int)ms.size()) brLast = 0;
                else { auto it = ms.begin(); advance(it, xb - 1); brLast = *it; }
                brAns ^= brLast;
            } else if (op == 5) {
                curLast = pre(xc); curAns ^= curLast;
                auto it = ms.lower_bound(xb);
                if (it == ms.begin()) brLast = INT_MIN;
                else { --it; brLast = *it; }
                brAns ^= brLast;
            } else {
                curLast = post(xc); curAns ^= curLast;
                auto it = ms.upper_bound(xb);
                if (it == ms.end()) brLast = INT_MAX;
                else brLast = *it;
                brAns ^= brLast;
            }
        }
        total++;
        if (curAns != brAns) {
            fails++;
            if (fails <= 3) printf("FAIL round=%d cur=%lld brute=%lld\n", round, curAns, brAns);
        }
    }
    printf("== followup2: total %d cases, fails %d ==\n", total, fails);
    return 0;
}

// periodni P6453 对拍：小直方图暴力枚举填数 vs 课程 dp
#include <bits/stdc++.h>
using namespace std;
const int MOD = 1000000007;
const int MAXN = 501;
const int MAXH = 1000000;
int arr[MAXN];
int fac[MAXH + 1], inv[MAXH + 1];
int left_[MAXN], right_[MAXN], stack_[MAXN];
int size_[MAXN];
int dp[MAXN][MAXN];
int tmp[MAXN];
int n, k;

int power(long long x, long long p) {
    long long ans = 1;
    while (p > 0) { if (p & 1) ans = ans * x % MOD; x = x * x % MOD; p >>= 1; }
    return (int) ans;
}
int c(int n, int k) { return n < k ? 0 : (int)((long long)fac[n] * inv[k] % MOD * inv[n - k] % MOD); }
void build() {
    fac[0] = fac[1] = inv[0] = 1;
    for (int i = 2; i <= MAXH; i++) fac[i] = (int)((long long)fac[i-1] * i % MOD);
    inv[MAXH] = power(fac[MAXH], MOD - 2);
    for (int i = MAXH - 1; i >= 1; i--) inv[i] = (int)((long long)inv[i+1] * (i + 1) % MOD);
    for (int i = 1, top = 0, pos; i <= n; i++) {
        pos = top;
        while (pos > 0 && arr[stack_[pos]] > arr[i]) pos--;
        if (pos > 0) right_[stack_[pos]] = i;
        if (pos < top) left_[i] = stack_[pos + 1];
        stack_[++pos] = i; top = pos;
    }
}
void dfs(int u, int fa) {
    if (u == 0) return;
    dfs(left_[u], u); dfs(right_[u], u);
    size_[u] = size_[left_[u]] + size_[right_[u]] + 1;
    for (int i = 0; i <= k; i++) tmp[i] = 0;
    for (int l = 0; l <= min(size_[left_[u]], k); l++)
        for (int r = 0; r <= min(size_[right_[u]], k - l); r++)
            tmp[l + r] = (tmp[l + r] + (long long)dp[left_[u]][l] * dp[right_[u]][r] % MOD) % MOD;
    for (int i = 0; i <= min(size_[u], k); i++)
        for (int p = 0; p <= i; p++)
            dp[u][i] = (dp[u][i] + (long long)c(size_[u] - p, i - p) * c(arr[u] - arr[fa], i - p) % MOD
                        * fac[i - p] % MOD * tmp[p] % MOD) % MOD;
}
int compute() {
    build();
    dp[0][0] = 1;
    dfs(stack_[1], 0);
    return dp[stack_[1]][k];
}

// 暴力：格子 (x, y)，x=1..n, y=1..arr[x]
bool validPos(const vector<pair<int,int>>& cells) {
    int K = cells.size();
    for (int a = 0; a < K; a++)
        for (int b = a + 1; b < K; b++) {
            int ax = cells[a].first, ay = cells[a].second;
            int bx = cells[b].first, by = cells[b].second;
            if (ax == bx) return false; // 同列
            if (ay == by) { // 同行：中间断开不违规，连通才违规
                int lo = min(ax, bx), hi = max(ax, bx);
                bool connected = true;
                for (int x = lo; x <= hi; x++) if (arr[x] < ay) { connected = false; break; }
                if (connected) return false;
            }
        }
    return true;
}

    for (int round = 0; round < 200; round++) {

        memset(left_, 0, sizeof(left_)); memset(right_, 0, sizeof(right_));
        memset(dp, 0, sizeof(dp)); memset(size_, 0, sizeof(size_));
        long long got = compute();
        // 暴力：枚举所有 k 元格子组合
        vector<pair<int,int>> all;
        for (int x = 1; x <= n; x++)
            for (int y = 1; y <= arr[x]; y++)
                all.push_back({x, y});
        long long cnt = 0;
        if ((int)all.size() >= k) {
            vector<int> idx(all.size());
            for (size_t i = 0; i < all.size(); i++) idx[i] = i;
            vector<int> sel(k);
            function<void(int,int)> rec = [&](int start, int depth) {
                if (depth == k) {
                    vector<pair<int,int>> cells;
                    for (int i = 0; i < k; i++) cells.push_back(all[sel[i]]);
                    if (validPos(cells)) cnt++;
                    return;
                }
                for (size_t i = start; i + (k - depth) <= all.size(); i++) {
                    sel[depth] = i;
                    rec(i + 1, depth + 1);
                }
            };
            rec(0, 0);
        }
        cnt %= MOD;
        total++;
        if (got != cnt) {

// removing AGC028B 对拍：n 小枚举 n! 消除方案 vs 课程公式
#include <bits/stdc++.h>

const int MOD = 1000000007;
const int MAXN = 100001;
int arr[MAXN];
int inv[MAXN];
int sum_[MAXN];
int n;

void build() {
    inv[1] = 1;
    for (int i = 2; i <= n; i++) inv[i] = (int)(MOD - (long long)inv[MOD % i] * (MOD / i) % MOD);
    for (int i = 1; i <= n; i++) sum_[i] = (sum_[i-1] + inv[i]) % MOD;
}
long long compute() {
    build();
    long long ans = 0;
    for (int i = 1; i <= n; i++) ans = (ans + (long long)(sum_[i] + sum_[n - i + 1] - 1) * arr[i]) % MOD;
    for (int i = 1; i <= n; i++) ans = ans * i % MOD;
    return ans;
}

int main() {
    srand(20261021);

        vector<int> perm(n);
        for (int i = 0; i < n; i++) perm[i] = i + 1;
        long long brute = 0;
        vector<int> alive(n + 1, 1);
        sort(perm.begin(), perm.end());
        do {
            vector<int> al(n + 1, 1);
            long long gain = 0;
            for (int step = 0; step < n; step++) {
                int cur = perm[step];
                // 收益 = 消除前 cur 所在连续未消除段（含 cur）的重量和
                int l = cur; while (l > 1 && al[l - 1]) l--;
                int r = cur; while (r < n && al[r + 1]) r++;
                long long seg = 0;
                for (int x = l; x <= r; x++) if (al[x]) seg += arr[x];
                gain += seg;
                al[cur] = 0;
            }
            brute = (brute + gain) % MOD;
        } while (next_permutation(perm.begin(), perm.end()));
        total++;
        if (got != brute) {
            fails++;
            if (fails <= 3) {

            }
        }

// treap2 P3369 单进程对拍：课程实现内嵌 vs multiset，400 轮 × 300 步
#include <bits/stdc++.h>
using namespace std;

// ===== 课程代码（treap2.cpp 核心，去 main）=====
const int MAXN = 100001;
int cnt = 0, head = 0;
int key[MAXN], key_count[MAXN], ls[MAXN], rs[MAXN], siz[MAXN];
double priority[MAXN];
void up(int i) { siz[i] = siz[ls[i]] + siz[rs[i]] + key_count[i]; }
int leftRotate(int i) { int r = rs[i]; rs[i] = ls[r]; ls[r] = i; up(i); up(r); return r; }
int rightRotate(int i) { int l = ls[i]; ls[i] = rs[l]; rs[l] = i; up(i); up(l); return l; }
int add(int i, int num) {
    if (i == 0) { key[++cnt] = num; key_count[cnt] = siz[cnt] = 1; priority[cnt] = (double)rand() / RAND_MAX; return cnt; }
    if (key[i] == num) key_count[i]++;
    else if (key[i] > num) ls[i] = add(ls[i], num);
    else rs[i] = add(rs[i], num);
    up(i);
    if (ls[i] != 0 && priority[ls[i]] > priority[i]) return rightRotate(i);
    if (rs[i] != 0 && priority[rs[i]] > priority[i]) return leftRotate(i);
    return i;
}
void add(int num) { head = add(head, num); }
int small(int i, int num) {
    if (i == 0) return 0;
    if (key[i] >= num) return small(ls[i], num);
    else return siz[ls[i]] + key_count[i] + small(rs[i], num);
}
int getRank(int num) { return small(head, num) + 1; }
int index(int i, int x) {
    if (siz[ls[i]] >= x) return index(ls[i], x);
    else if (siz[ls[i]] + key_count[i] < x) return index(rs[i], x - siz[ls[i]] - key_count[i]);
    return key[i];
}
int index(int x) { return index(head, x); }
int pre(int i, int num) {
    if (i == 0) return INT_MIN;
    if (key[i] >= num) return pre(ls[i], num);
    else return max(key[i], pre(rs[i], num));
}
int pre(int num) { return pre(head, num); }
int post(int i, int num) {
    if (i == 0) return INT_MAX;
    if (key[i] <= num) return post(rs[i], num);
    else return min(key[i], post(ls[i], num));
}
int post(int num) { return post(head, num); }
int remove_(int i, int num) {
    if (key[i] < num) rs[i] = remove_(rs[i], num);
    else if (key[i] > num) ls[i] = remove_(ls[i], num);
    else {
        if (key_count[i] > 1) key_count[i]--;
        else {
            if (ls[i] == 0 && rs[i] == 0) return 0;
            else if (ls[i] != 0 && rs[i] == 0) i = ls[i];
            else if (ls[i] == 0 && rs[i] != 0) i = rs[i];
            else {
                if (priority[ls[i]] >= priority[rs[i]]) { i = rightRotate(i); rs[i] = remove_(rs[i], num); }
                else { i = leftRotate(i); ls[i] = remove_(ls[i], num); }
            }
        }
    }
    up(i);
    return i;
}
void remove_(int num) { if (getRank(num) != getRank(num + 1)) head = remove_(head, num); }
// ===== 课程代码结束 =====

int main() {
    srand(20261018);
    int total = 0, fails = 0;
    for (int round = 0; round < 400; round++) {
        cnt = 0; head = 0;
        memset(key, 0, sizeof(key)); memset(key_count, 0, sizeof(key_count));
        memset(ls, 0, sizeof(ls)); memset(rs, 0, sizeof(rs)); memset(siz, 0, sizeof(siz));
        multiset<int> ms;
        for (int step = 0; step < 300; step++) {
            int op = 1 + rand() % 6;
            int x;
            if (op == 4 && ms.empty()) op = 1; // 空集合时 op4 非法，换成插入
            if (op == 4) x = 1 + rand() % (int)ms.size();
            else x = rand() % 101 - 50;
            if (op == 1) { add(x); ms.insert(x); }
            else if (op == 2) {
                remove_(x);
                auto it = ms.find(x);
                if (it != ms.end()) ms.erase(it);
            } else if (op == 3) {
                long long g = getRank(x);
                long long e = distance(ms.begin(), ms.lower_bound(x)) + 1;
                if (g != e) { fails++; if (fails <= 3) printf("FAIL round=%d step=%d op3 x=%d got=%lld exp=%lld\n", round, step, x, g, e); }
            } else if (op == 4) {
                long long g = index(x);
                auto it = ms.begin(); advance(it, x - 1);
                if (g != *it) { fails++; if (fails <= 3) printf("FAIL round=%d step=%d op4 x=%d got=%lld exp=%lld\n", round, step, x, g, (long long)*it); }
            } else if (op == 5) {
                long long g = pre(x);
                auto it = ms.lower_bound(x);
                long long e = (it == ms.begin()) ? INT_MIN : (long long)(*prev(it));
                if (g != e) { fails++; if (fails <= 3) printf("FAIL round=%d step=%d op5 x=%d got=%lld exp=%lld\n", round, step, x, g, e); }
            } else {
                long long g = post(x);
                auto it = ms.upper_bound(x);
                long long e = (it == ms.end()) ? INT_MAX : (long long)*it;
                if (g != e) { fails++; if (fails <= 3) printf("FAIL round=%d step=%d op6 x=%d got=%lld exp=%lld\n", round, step, x, g, e); }
            }
            total++;
        }
    }
    printf("== treap2 inline: total %d cases, fails %d ==\n", total, fails);
    return 0;
}

// treeorder P1377 对拍：输入排列 → 课程输出先序 → 暴力验证字典序最小 + 形态一致
#include <bits/stdc++.h>
using namespace std;

int arr[15], ls[15], rs[15], sta[15], n;
int order[15]; // 课程输出

void build() {
    for (int i = 1, top = 0, pos = 0; i <= n; i++) {
        pos = top;
        while (pos > 0 && arr[sta[pos]] > arr[i]) pos--;
        if (pos > 0) rs[sta[pos]] = i;
        if (pos < top) ls[i] = sta[pos + 1];
        sta[++pos] = i;
        top = pos;
    }
}
void pre() {
    int size = 1, i = 0, cur;
    while (size > 0) {
        cur = sta[size--];
        arr[++i] = cur;
        if (rs[cur]) sta[++size] = rs[cur];
        if (ls[cur]) sta[++size] = ls[cur];
    }
}

// 用插入序列 seq 建 BST，返回形态编码（先序节点序列）
string buildBST(const vector<int>& seq) {
    vector<int> L(n + 1, 0), R(n + 1, 0);
    int root = 0;
    for (int v : seq) {
        if (root == 0) { root = v; continue; }
        int cur = root;
        while (true) {
            if (v < cur) {
                if (L[cur]) cur = L[cur];
                else { L[cur] = v; break; }
            } else {
                if (R[cur]) cur = R[cur];
                else { R[cur] = v; break; }
            }
        }
    }
    // 先序编码
    string s;
    vector<int> st = {root};
    while (!st.empty()) {
        int u = st.back(); st.pop_back();
        s += char('0' + u);
        if (R[u]) st.push_back(R[u]);
        if (L[u]) st.push_back(L[u]);
    }
    return s;
}

int main() {
    srand(20261018);
    int total = 0, fails = 0;
    for (int round = 0; round < 500; round++) {
        n = 1 + rand() % 8;
        // 生成输入排列（第 i 次插入的数值）
        vector<int> perm(n);
        for (int i = 0; i < n; i++) perm[i] = i + 1;
        random_shuffle(perm.begin(), perm.end());
        // 课程：arr[val] = 时间
        for (int i = 1; i <= n; i++) arr[i] = 0;
        for (int i = 0; i < n; i++) arr[perm[i]] = i + 1;
        memset(ls, 0, sizeof(ls)); memset(rs, 0, sizeof(rs));
        build();
        pre();
        vector<int> got;
        for (int i = 1; i <= n; i++) got.push_back(arr[i]);
        // 原序列对应形态
        string origShape = buildBST(perm);
        // 课程输出序列的形态
        string gotShape = buildBST(got);
        // 字典序最小：枚举所有排列，找形态相同且字典序最小的
        vector<int> best = got;
        vector<int> p(n);
        for (int i = 0; i < n; i++) p[i] = i + 1;
        do {
            string sh = buildBST(p);
            if (sh == origShape && p < best) best = p;
        } while (next_permutation(p.begin(), p.end()));
        total++;
        bool ok = (gotShape == origShape) && (got == best);
        if (!ok) {
            fails++;
            if (fails <= 3) {
                printf("FAIL round=%d n=%d perm:", round, n);
                for (int v : perm) printf("%d ", v);
                printf("\n  got:"); for (int v : got) printf("%d ", v);
                printf("\n  best:"); for (int v : best) printf("%d ", v);
                printf(" shapeOk=%d\n", gotShape == origShape);
            }
        }
    }
    printf("== treeorder: total %d cases, fails %d ==\n", total, fails);
    return 0;
}
