#include <bits/stdc++.h>

using namespace std;

using ll = long long;

const int MAXN = 200001;
const int MOD = 1000000007;
int n, cntb;

int head[MAXN];
int nxt[MAXN << 1];
int to[MAXN << 1];
int cntg;

int dfn[MAXN];
int cntd;

int from[MAXN];
int all[MAXN];
int cycle[MAXN];

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;
}

ll power(ll x, ll p) {
    ll ans = 1;
    while (p > 0) {
        if ((p & 1) == 1) {
            ans = (ans * x) % MOD;
        }
        x = (x * x) % MOD;
        p >>= 1;
    }
    return ans;
}

void dfs(int u) {
    dfn[u] = ++cntd;
    all[cntb]++;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (dfn[v] == 0) {
            from[v] = u;
            dfs(v);
        } else if (dfn[u] < dfn[v]) {
            cycle[cntb]++;
            for (int i = v; i != u; i = from[i]) {
                cycle[cntb]++;
            }
        }
    }
}

ll compute() {
    ll ans = 1;
    for (int i = 1; i <= n; i++) {
        if (dfn[i] == 0) {
            cntb++;
            dfs(i);
            ll a = power(2, all[cntb]);
            ll b = power(2, all[cntb] - cycle[cntb] + 1);
            ans = ans * ((a - b + MOD) % MOD) % MOD;
        }
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    for (int i = 1, x; i <= n; i++) {
        cin >> x;
        addEdge(i, x);
        addEdge(x, i);
    }
    ll ans = compute();
    cout << ans << "\n";
    return 0;
}

const int MAXN = 200001;
int t, n;

int head[MAXN];

int to[MAXN << 1];

int dfn[MAXN];
int cntd;

    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;

    dfn[u] = ++cntd;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (dfn[v] == 0) {
            from[v] = u;
            dfs(v);
        } else if (dfn[u] < dfn[v]) {
            cycle[u] = true;
            for (int i = v; i != u; i = from[i]) {
                cycle[i] = true;
            }
        }
    }
}

void dpOnTree(int u, int fa) {
    siz[u] = 1;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (v != fa && !cycle[v]) {
            dpOnTree(v, u);
            siz[u] += siz[v];
        }
    }
}

ll compute() {
    dfs(1);
    ll ans = 0;
    for (int i = 1; i <= n; i++) {
        if (cycle[i]) {
            dpOnTree(i, 0);
            ans += siz[i] * (siz[i] - 1) / 2 + siz[i] * (n - siz[i]);
        }
    }
    return ans;
}

    cntg = cntd = 0;
    for (int i = 1; i <= n; i++) {

#include <bits/stdc++.h>

using namespace std;

using ll = long long;

const int MAXN = 1000001;
int n;
int arr[MAXN];

int head[MAXN];
int nxt[MAXN << 1];
int to[MAXN << 1];
int cntg;

int dfn[MAXN];
int cntd;

int x, y, skip;

ll dp[MAXN][2];

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;
}

void dfs(int u) {
    dfn[u] = ++cntd;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (dfn[v] == 0) {
            dfs(v);
        } else if (dfn[u] < dfn[v]) {
            x = u;
            y = v;
            skip = (e + 1) >> 1;
        }
    }
}

void dpOnTree(int u, int fa) {
    dp[u][0] = 0;
    dp[u][1] = arr[u];
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (v != fa && (e + 1) >> 1 != skip) {
            dpOnTree(v, u);
            dp[u][0] += max(dp[v][0], dp[v][1]);
            dp[u][1] += dp[v][0];
        }
    }
}

ll compute() {
    ll ans = 0;
    for (int i = 1; i <= n; i++) {
        if (dfn[i] == 0) {
            x = y = 0;
            dfs(i);
            dpOnTree(x, 0);
            ll cur = dp[x][0];
            dpOnTree(y, 0);
            cur = max(cur, dp[y][0]);
            ans += cur;
        }
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    for (int x = 1, y; x <= n; x++) {
        cin >> arr[x];
        cin >> y;

    }
    ll ans = compute();
    cout << ans << "\n";
    return 0;
}

#include <bits/stdc++.h>

using namespace std;

using ll = long long;

const int MAXN = 200001;
int n;

int head[MAXN];
int nxt[MAXN << 1];
int to[MAXN << 1];
int weight[MAXN << 1];
int cntg;

int dfn[MAXN];
int cntd;

int fromNode[MAXN];
int fromWeight[MAXN];
bool cycle[MAXN];
int arr[MAXN];
int val[MAXN];
int cnta;

ll dist[MAXN];
ll diameter;

ll height[MAXN];

ll preMax[MAXN];
ll preDiameter[MAXN];

ll sufMax[MAXN];
ll sufDiameter[MAXN];

void addEdge(int u, int v, int w) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    weight[cntg] = w;
    head[u] = cntg;
}

void dfs(int u) {
    dfn[u] = ++cntd;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (dfn[v] == 0) {
            fromNode[v] = u;
            fromWeight[v] = weight[e];
            dfs(v);
        } else if (dfn[u] < dfn[v]) {
            cycle[u] = true;
            arr[++cnta] = u;
            val[cnta] = weight[e];
            for (int i = v; i != u; i = fromNode[i]) {
                cycle[i] = true;
                arr[++cnta] = i;
                val[cnta] = fromWeight[i];
            }
        }
    }
}

void dpOnTree(int u, int fa) {
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        int w = weight[e];
        if (v != fa && !cycle[v]) {
            dpOnTree(v, u);
            diameter = max(diameter, dist[u] + dist[v] + w);
            dist[u] = max(dist[u], dist[v] + w);
        }
    }
}

ll dpOnCycle() {
    ll best = 0;
    ll sum = 0;
    for (int i = 1; i <= cnta; i++) {
        preMax[i] = max(preMax[i - 1], height[i] + sum);
        preDiameter[i] = max(preDiameter[i - 1], sum + height[i] + best);
        best = max(best, height[i] - sum);
        sum += val[i];
    }
    sufMax[cnta] = height[cnta] + val[cnta];
    sufDiameter[cnta] = height[cnta];
    best = height[cnta] - val[cnta];
    sum = val[cnta] + val[cnta - 1];
    for (int i = cnta - 1; i >= 1; i--) {
        sufMax[i] = max(sufMax[i + 1], height[i] + sum);
        sufDiameter[i] = max(sufDiameter[i + 1], sum + height[i] + best);
        best = max(best, height[i] - sum);
        sum += val[i - 1];
    }
    ll ans = LLONG_MAX;
    for (int i = 1; i < cnta; i++) {
        ans = min(ans, max(preMax[i] + sufMax[i + 1], max(preDiameter[i], sufDiameter[i + 1])));
    }
    ans = min(ans, preDiameter[cnta]);
    return ans;
}

ll compute() {
    dfs(1);
    ll ans1 = 0;
    for (int i = 1; i <= cnta; i++) {
        diameter = 0;

        }
    }
}
int diameter() {
    memset(vis, 0, sizeof vis);
    int far = 1, maxd = 0;
    vis[1] = 1; dfsTree(1, 0, far, maxd, 0);
    memset(vis, 0, sizeof vis);
    int far2 = far; maxd = 0;
    vis[far] = 1; dfsTree(far, 0, far2, maxd, 0);
    return maxd;
}
bool connected(int skip) {
    memset(vis, 0, sizeof vis);
    vector<int> q; q.push_back(1); vis[1] = 1;
    for (size_t i = 0; i < q.size(); i++) {
        int u = q[i];
        for (size_t j = 0; j < es.size(); j++) {
            if ((int)j == skip) continue;
            int v = -1;
            if (es[j].u == u) v = es[j].v;
            else if (es[j].v == u) v = es[j].u;
            if (v >= 0 && !vis[v]) { vis[v] = 1; q.push_back(v); }
        }
    }
    for (int i = 1; i <= n; i++) if (!vis[i]) return false;
    return true;
}

    for (int i = 0; i < n; i++) {
        int u, v, w; cin >> u >> v >> w;
        es.push_back({u, v, w});

        if (!connected(skip)) continue;

        // 直径计算需要排除 skip 边：把 skip 边移走再求
        vector<E> saved = es;
        es.erase(es.begin() + skip);
        best = min(best, diameter());
        es = saved;
    }

}

import random, sys
def gen_block(start, size, want_ring=None):
    k = random.randint(3, min(size, 6)) if want_ring is None else want_ring
    ring = random.sample(range(start, start + size), k)
    edges = []
    for i in range(k):
        edges.append((ring[i], ring[(i + 1) % k]))
    used = set(ring)
    rest = [x for x in range(start, start + size) if x not in used]
    for x in rest:
        p = random.choice(list(used))
        edges.append((x, p))
        used.add(x)
    return edges

def gen(n, comps, seed):
    random.seed(seed)
    if comps > 1:
        sizes = [3] * comps
        rem = n - 3 * comps
        while rem > 0:
            i = random.randrange(comps)
            sizes[i] += 1
            rem -= 1
    else:
        sizes = [n]
    edges = []
    start = 1
    for s in sizes:
        edges += gen_block(start, s)
        start += s
    return edges, sizes

def gen_func(n, comps, seed):
    # 每个点恰好一个出边（函数图），底层无向图是基环树；返回 out[1..n]
    r = random.Random(seed * 7919)
    sizes = [3] * comps
    rem = n - 3 * comps
    while rem > 0:
        i = r.randrange(comps)
        sizes[i] += 1
        rem -= 1
    out = [0] * (n + 1)
    start = 1
    for size in sizes:
        k = r.randint(3, min(size, 6))
        ring = r.sample(range(start, start + size), k)
        for i in range(k):
            out[ring[i]] = ring[(i + 1) % k]
        used = set(ring)
        for x in range(start, start + size):
            if x not in used:
                p = r.choice(list(used))
                out[x] = p
                used.add(x)
        start += size
    return out
