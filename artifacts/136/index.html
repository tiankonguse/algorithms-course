线性基原理，只对非负数进行讨论线性基常指的是异或空间线性基，向量空间线性基是下期的内容一堆数字能得到的非0异或和的结果，能被元素个数尽量少的集合，不多不少的全部得到那么就说，这个元素个数尽量少的集合，是这一堆数字的异或空间线性基注意，只关心去重的非0异或和能否全部得到，并不关心每种异或和的个数如下几个结论是构建异或空间线性基的基础，课上重点图解1，一堆数字中，任意的a和b，用a^b的结果替代a、b中的一个数字，不会影响非0异或和的组成2，一堆数字中，任意的a和b，如果有a^b=0，那么舍弃掉a、b其中一个数字，不会影响非0异或和的组成3，一堆数字能否异或出0，在求出异或空间线性基之后，需要被单独标记

// 题1 P3812 最大异或和：普通消元建线性基，从高位贪心
#include <bits/stdc++.h>
using namespace std;
const int MAXN = 51;
const int BIT = 50;
long long arr[MAXN];
long long basis[BIT + 1];
int n;

bool insert(long long num) {
  for (int i = BIT; i >= 0; i--) {
    if (num >> i == 1) {
      if (basis[i] == 0) {
        basis[i] = num;
        return true;
      }
      num ^= basis[i];
    }
  }
  return false;
}

long long compute() {
  for (int i = 1; i <= n; i++) insert(arr[i]);
  long long ans = 0;
  for (int i = BIT; i >= 0; i--) {
    ans = max(ans, ans ^ basis[i]);
  }
  return ans;
}

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> arr[i];
  cout << compute() << endl;
  return 0;
}

// 题2 loj 114 第k小异或和：高斯消元标准基
#include <bits/stdc++.h>
using namespace std;
const int MAXN = 100001;
const int BIT = 50;
long long arr[MAXN];
int len, n;
bool zero;

void compute() {
  len = 1;
  for (long long i = BIT; i >= 0; i--) {
    for (int j = len; j <= n; j++) {
      if ((arr[j] & (1LL << i)) != 0) { swap(arr[j], arr[len]); break; }
    }
    if ((arr[len] & (1LL << i)) != 0) {
      for (int j = 1; j <= n; j++) {
        if (j != len && (arr[j] & (1LL << i)) != 0) arr[j] ^= arr[len];
      }
      len++;
    }
  }
  len--;
  zero = len != n;
}

long long query(long long k) {
  if (zero && k == 1) return 0;
  if (zero) k--;
  if (k >= 1LL << len) return -1;
  long long ans = 0;
  for (int i = len, j = 0; i >= 1; i--, j++) {
    if ((k & (1LL << j)) != 0) ans ^= arr[i];
  }
  return ans;
}

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> arr[i];
  compute();
  int q;
  cin >> q;
  for (int i = 1; i <= q; i++) {
    long long k;
    cin >> k;
    cout << query(k) << endl;
  }
  return 0;
}

// 题3 P4570 元素：魔力降序贪心，insert 成功累加
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 1001;
const int BIT = 60;
struct Ore { ll state, val; } arr[MAXN];
ll basis[BIT + 1];
int n;

bool insert(ll num) {
  for (int i = BIT; i >= 0; i--) {
    if (num >> i == 1) {
      if (basis[i] == 0) { basis[i] = num; return true; }
      num ^= basis[i];
    }
  }
  return false;
}

ll compute() {
  ll ans = 0;
  sort(arr + 1, arr + n + 1, [](const Ore& a, const Ore& b) { return a.val > b.val; });
  for (int i = 1; i <= n; i++) {
    if (insert(arr[i].state)) ans += arr[i].val;
  }
  return ans;
}

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> arr[i].state >> arr[i].val;
  cout << compute() << endl;
  return 0;
}

// 题4 P3857 彩灯：线性基大小 size → 2^size mod 2008
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 51;
const int MOD = 2008;
ll arr[MAXN];
ll basis[MAXN];
int n, m;

bool insert(ll num) {
  for (int i = n; i >= 0; i--) {
    if (num >> i == 1) {
      if (basis[i] == 0) { basis[i] = num; return true; }
      num ^= basis[i];
    }
  }
  return false;
}

int compute() {
  int size = 0;
  for (int i = 1; i <= m; i++) {
    if (insert(arr[i])) size++;
  }
  return size;
}

int main() {
  cin >> n >> m;
  n = n - 1;  // n 个灯泡 → 最高位 n-1
  string s;
  for (int i = 1; i <= m; i++) {
    cin >> s;
    ll num = 0;
    for (int j = 0; j <= n; j++) {
      if (s[j] == 'O') num |= 1LL << j;
    }
    arr[i] = num;
  }
  int size = compute();
  ll ans = 1;
  for (int i = 0; i < size; i++) ans = (ans * 2) % MOD;
  cout << ans << endl;
  return 0;
}

// ShowDetails 4 例：普通消元（例1 12,9,14,11 / 例2 2,5,11,6）与高斯消元（例3 6,37,35,33 / 例4 7,10,4）
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 101;
const int BIT = 60;
ll arr[MAXN];
int n;
ll basis1[BIT + 1];
bool zero1;
ll basis2[MAXN];
int len;
bool zero2;

bool insert(ll num) {
  for (int i = BIT; i >= 0; i--) {
    if (num >> i == 1) {
      if (basis1[i] == 0) { basis1[i] = num; return true; }
      num ^= basis1[i];
    }
  }
  return false;
}

void compute1() {
  zero1 = false;
  for (int i = 1; i <= n; i++) {
    if (!insert(arr[i])) zero1 = true;
  }
}

void compute2() {
  len = 1;
  for (ll i = BIT; i >= 0; i--) {
    for (int j = len; j <= n; j++) {
      if ((basis2[j] & (1LL << i)) != 0) { swap(basis2[j], basis2[len]); break; }
    }
    if ((basis2[len] & (1LL << i)) != 0) {
      for (int j = 1; j <= n; j++) {
        if (j != len && (basis2[j] & (1LL << i)) != 0) basis2[j] ^= basis2[len];
      }
      len++;
    }
  }
  len--;
  zero2 = len != n;
}

set<ll> collect;
void dfs(ll* nums, int nn, int i, bool pick, ll path) {
  if (i > nn) { if (pick) collect.insert(path); }
  else {
    dfs(nums, nn, i + 1, pick, path);
    dfs(nums, nn, i + 1, true, path ^ nums[i]);
  }
}
void printXor(ll* nums, int nn) {
  collect.clear();
  dfs(nums, nn, 1, false, 0);
  cout << "至少选择一个数字所有可能的异或和:";
  for (ll s : collect) cout << " " << s;
  cout << endl;
}

// 题1 对拍：课程 insert+贪心 vs 暴力全枚举
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll brute(ll* a, int n) {
  ll best = 0;
  for (int m = 1; m < (1 << n); m++) {
    ll x = 0;
    for (int i = 0; i < n; i++) if (m >> i & 1) x ^= a[i];
    best = max(best, x);
  }
  return best;
}

int main() {
  mt19937_64 rng(12345);
  int T = 3000, bad = 0;
  for (int t = 0; t < T; t++) {
    int n = 1 + rng() % 8;
    ll a[10];
    for (int i = 0; i < n; i++) a[i] = rng() & ((1LL << 50) - 1);
    // 课程
    ll basis[51] = {0};
    auto insert = [&](ll num) {
      for (int i = 50; i >= 0; i--) {
        if (num >> i == 1) {
          if (basis[i] == 0) { basis[i] = num; return true; }
          num ^= basis[i];
        }
      }
      return false;
    };
    for (int i = 0; i < n; i++) insert(a[i]);
    ll cur = 0;
    for (int i = 50; i >= 0; i--) cur = max(cur, cur ^ basis[i]);
    ll ref = brute(a, n);
    if (cur != ref) { bad++; if (bad <= 3) { cout << "BAD n=" << n << " cur=" << cur << " ref=" << ref << " a:"; for (int i = 0; i < n; i++) cout << " " << a[i]; cout << endl; } }
  }
  cout << (bad ? "FAIL" : "ALL OK") << " bad=" << bad << " T=" << T << endl;
  return 0;
}

// 题2 对拍：课程高斯消元 query vs 暴力枚举去重排序
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

vector<ll> bruteAll(ll* a, int n) {
  set<ll> s;
  for (int m = 1; m < (1 << n); m++) {
    ll x = 0;
    for (int i = 0; i < n; i++) if (m >> i & 1) x ^= a[i];
    s.insert(x);
  }
  return vector<ll>(s.begin(), s.end());
}

int main() {
  mt19937_64 rng(24680);
  int T = 3000, bad = 0;
  for (int t = 0; t < T; t++) {
    int n = 1 + rng() % 7;
    ll a[10];
    for (int i = 0; i < n; i++) a[i] = rng() & ((1LL << 50) - 1);
    // 课程
    ll arr[12] = {0};
    for (int i = 1; i <= n; i++) arr[i] = a[i - 1];
    int len = 1; bool zero = false;
    for (ll i = 50; i >= 0; i--) {
      for (int j = len; j <= n; j++) {
        if ((arr[j] & (1LL << i)) != 0) { swap(arr[j], arr[len]); break; }
      }
      if ((arr[len] & (1LL << i)) != 0) {
        for (int j = 1; j <= n; j++) {
          if (j != len && (arr[j] & (1LL << i)) != 0) arr[j] ^= arr[len];
        }
        len++;
      }
    }
    len--;
    zero = len != n;
    auto query = [&](ll k) {
      if (zero && k == 1) return 0LL;
      if (zero) k--;
      if (k >= 1LL << len) return -1LL;
      ll ans = 0;
      for (int i = len, j = 0; i >= 1; i--, j++) {
        if ((k & (1LL << j)) != 0) ans ^= arr[i];
      }
      return ans;
    };
    // 暴力
    vector<ll> all = bruteAll(a, n);
    int q = 1 + rng() % 8;
    for (int qi = 0; qi < q; qi++) {
      ll k = 1 + rng() % ((ll)all.size() + 3);
      ll cur = query(k);
      ll ref = (k <= (ll)all.size()) ? all[k - 1] : -1;
      if (cur != ref) {
        bad++;
        if (bad <= 3) {
          cout << "BAD n=" << n << " k=" << k << " cur=" << cur << " ref=" << ref << " zero=" << zero << " len=" << len << " a:";
          for (int i = 0; i < n; i++) cout << " " << a[i];
          cout << " all:";
          for (ll v : all) cout << " " << v;
          cout << endl;
        }
      }
    }
  }
  cout << (bad ? "FAIL" : "ALL OK") << " bad=" << bad << " T=" << T << endl;
  return 0;
}

// 题3 对拍：课程魔力降序贪心 vs 暴力枚举线性无关子集最大魔力
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

bool indep(ll* states, int cnt) {
  // cnt 个元素线性无关 ⇔ 无非空子集异或和为 0
  ll basis[61] = {0};
  for (int i = 0; i < cnt; i++) {
    ll num = states[i];
    for (int b = 60; b >= 0; b--) {
      if (num >> b == 1) {
        if (basis[b] == 0) { basis[b] = num; break; }
        num ^= basis[b];
      }
    }
    if (num == 0) return false;  // 插入失败 = 相关
  }
  return true;
}

ll brute(ll* st, ll* val, int n) {
  ll best = 0;
  for (int m = 1; m < (1 << n); m++) {
    vector<ll> s; ll sum = 0;
    for (int i = 0; i < n; i++) if (m >> i & 1) { s.push_back(st[i]); sum += val[i]; }
    if (indep(s.data(), s.size())) best = max(best, sum);
  }
  return best;
}

int main() {
  mt19937_64 rng(13579);
  int T = 2000, bad = 0;
  for (int t = 0; t < T; t++) {
    int n = 1 + rng() % 7;
    ll st[10], val[10];
    for (int i = 0; i < n; i++) { st[i] = 1 + rng() % ((1LL << 60) - 1); val[i] = 1 + rng() % 10000; }
    // 课程
    vector<pair<ll,ll>> v;
    for (int i = 0; i < n; i++) v.push_back({st[i], val[i]});
    sort(v.begin(), v.end(), [](const pair<ll,ll>& a, const pair<ll,ll>& b) { return a.second > b.second; });
    ll basis[61] = {0};
    auto insert = [&](ll num) {
      for (int i = 60; i >= 0; i--) {
        if (num >> i == 1) {
          if (basis[i] == 0) { basis[i] = num; return true; }
          num ^= basis[i];
        }
      }
      return false;
    };
    ll cur = 0;
    for (int i = 0; i < n; i++) if (insert(v[i].first)) cur += v[i].second;
    ll ref = brute(st, val, n);
    if (cur != ref) {
      bad++;
      if (bad <= 3) {
        cout << "BAD n=" << n << " cur=" << cur << " ref=" << ref << endl;
        for (int i = 0; i < n; i++) cout << "  st=" << st[i] << " val=" << val[i] << endl;
      }
    }
  }
  cout << (bad ? "FAIL" : "ALL OK") << " bad=" << bad << " T=" << T << endl;
  return 0;
}

// 题4 对拍：课程线性基大小 2^size vs 暴力枚举开关组合去重计数
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MOD = 2008;

int main() {
  mt19937_64 rng(86420);
  int T = 3000, bad = 0;
  for (int t = 0; t < T; t++) {
    int nn = 1 + rng() % 8;   // 灯泡数
    int m = 1 + rng() % 7;    // 开关数
    // 生成开关串
    vector<ll> sw(m);
    for (int i = 0; i < m; i++) {
      ll num = 0;
      for (int j = 0; j < nn; j++) if (rng() & 1) num |= 1LL << j;
      sw[i] = num;
    }
    // 课程
    ll basis[9] = {0};
    auto insert = [&](ll num) {
      for (int i = nn - 1; i >= 0; i--) {
        if (num >> i == 1) {
          if (basis[i] == 0) { basis[i] = num; return true; }
          num ^= basis[i];
        }
      }
      return false;
    };
    int size = 0;
    for (int i = 0; i < m; i++) if (insert(sw[i])) size++;
    ll cur = 1;
    for (int i = 0; i < size; i++) cur = (cur * 2) % MOD;
    // 暴力
    set<ll> states;
    for (int mask = 0; mask < (1 << m); mask++) {
      ll x = 0;
      for (int i = 0; i < m; i++) if (mask >> i & 1) x ^= sw[i];
      states.insert(x);
    }
    ll ref = states.size() % MOD;
    if (cur != ref) {
      bad++;
      if (bad <= 3) {
        cout << "BAD nn=" << nn << " m=" << m << " size=" << size << " cur=" << cur << " ref=" << ref << " states=" << states.size() << endl;
      }
    }
  }
  cout << (bad ? "FAIL" : "ALL OK") << " bad=" << bad << " T=" << T << endl;
  return 0;
}
