#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int rankRows(vector<vector<ll>> rows, int m) {
  int r = 0;
  for (int j = 0; j < m; j++) {
    int piv = -1;
    for (int i = r; i < (int)rows.size(); i++) if (rows[i][j] != 0) { piv = i; break; }
    if (piv == -1) continue;
    swap(rows[r], rows[piv]);
    for (int i = 0; i < (int)rows.size(); i++) {
      if (i != r && rows[i][j] != 0) {
        ll a = rows[r][j], b = rows[i][j];
        ll g = std::gcd(a < 0 ? -a : a, b < 0 ? -b : b);
        a /= g; b /= g;
        for (int k = 0; k < m; k++) rows[i][k] = rows[i][k] * a - rows[r][k] * b;
      }
    }
    r++;
  }
  return r;
}
int main() {
  vector<vector<ll>> rows = {
    {0,2,1,2}, {1,0,1,2}, {3,3,1,0}, {1,3,2,3}
  };
  printf("rank=%d\n", rankRows(rows, 4));
  // 只取物品2,1,3（前三个）
  vector<vector<ll>> rows2 = {{1,0,1,2},{0,2,1,2},{3,3,1,0}};
  printf("rank3=%d\n", rankRows(rows2, 4));
  return 0;
}

pair<int,int> brute(int n, int m, const vector<vector<int>>& attr, const vector<int>& price) {
  int bestCnt = 0, bestCost = INT_MAX;
  for (int mask = 1; mask < (1 << n); mask++) {
    vector<vector<ll>> rows;
    int costSum = 0;
    for (int i = 0; i < n; i++) if (mask >> i & 1) {
      vector<ll> row(m);
      for (int j = 0; j < m; j++) row[j] = attr[i + 1][j];
      rows.push_back(row);
      costSum += price[i + 1];
    }
    int sz = rows.size();
    int rk = rankRows(rows, m);
    if (mask == (1 << n) - 1) printf("全选 mask=%d sz=%d rank=%d\n", mask, sz, rk);
    if (rk != sz) continue;
    if (sz > bestCnt || (sz == bestCnt && costSum < bestCost)) { bestCnt = sz; bestCost = costSum; }
  }
  if (bestCnt == 0) bestCost = 0;
  return {bestCnt, bestCost};
}

const int MAXN = 50001, MAXM = 200002, BIT4 = 60;
int head4[MAXN], next4[MAXM], to4[MAXM], cnt4; ll weight4[MAXM];
ll basis4[BIT4+1]; bool visited4[MAXN]; ll path4[MAXN];
int ufe4[MAXN][3], stackSize4, u4, f4, e4; ll pstack4[MAXN], p4;
bool insert4(ll num){for(int i=BIT4;i>=0;i--)if(num>>i==1){if(basis4[i]==0){basis4[i]=num;return true;}num^=basis4[i];}return false;}
void push4(int u,int f,int e,ll p){ufe4[stackSize4][0]=u;ufe4[stackSize4][1]=f;ufe4[stackSize4][2]=e;pstack4[stackSize4]=p;stackSize4++;}
void pop4(){--stackSize4;u4=ufe4[stackSize4][0];f4=ufe4[stackSize4][1];e4=ufe4[stackSize4][2];p4=pstack4[stackSize4];}

#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
struct Course {
  double sml = 1e-5;
  vector<vector<double>> mat;
  vector<int> basis;
  int n, m, cnt, cost;
  bool insert(int i) {
    for (int j = 1; j <= m; j++) {
      if (fabs(mat[i][j]) >= sml) {
        if (basis[j] == 0) { basis[j] = i; return true; }
        double rate = mat[i][j] / mat[basis[j]][j];
        for (int k = j; k <= m; k++) mat[i][k] -= rate * mat[basis[j]][k];
      }
    }
    return false;
  }
  void compute() {
    cnt = cost = 0;
    sort(mat.begin() + 1, mat.end(), [&](const vector<double>& a, const vector<double>& b) { return a[m + 1] < b[m + 1]; });
    basis.assign(n + 1, 0);
    for (int i = 1; i <= n; i++) if (insert(i)) { cnt++; cost += (int)mat[i][m + 1]; }
  }
  pair<int,int> run(int nn, int mm, const vector<vector<int>>& attr, const vector<int>& price) {
    n = nn; m = mm;
    mat.assign(n + 1, vector<double>(m + 2, 0));
    for (int i = 1; i <= n; i++) { for (int j = 1; j <= m; j++) mat[i][j] = attr[i][j]; mat[i][m + 1] = price[i]; }
    compute();
    return {cnt, cost};
  }
};
int main() {
  int n = 4, m = 4;
  vector<vector<int>> attr(n + 1, vector<int>(m + 1));
  attr[1] = {0, 0, 2, 1, 2};
  attr[2] = {0, 1, 0, 1, 2};
  attr[3] = {0, 3, 3, 1, 0};
  attr[4] = {0, 1, 3, 2, 3};
  vector<int> price = {0, 3, 1, 5, 8};
  auto c = Course().run(n, m, attr, price);
  printf("课程=(%d,%d)\n", c.first, c.second);
  return 0;
}

// 题1 装备购买：课程（价格升序 + 实数消元贪心） vs 暴力（枚举子集：最大线性无关大小 + 最小价格）
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

struct Course {
  double sml = 1e-5;
  vector<vector<double>> mat;
  vector<int> basis;
  int n, m, cnt, cost;
  bool insert(int i) {
    for (int j = 1; j <= m; j++) {
      if (fabs(mat[i][j]) >= sml) {
        if (basis[j] == 0) { basis[j] = i; return true; }
        double rate = mat[i][j] / mat[basis[j]][j];
        for (int k = j; k <= m; k++) mat[i][k] -= rate * mat[basis[j]][k];
      }
    }
    return false;
  }
  void compute() {
    cnt = cost = 0;
    sort(mat.begin() + 1, mat.end(), [&](const vector<double>& a, const vector<double>& b) { return a[m + 1] < b[m + 1]; });
    basis.assign(m + 2, 0);   // 下标 j=1..m，m 可能 > n
    for (int i = 1; i <= n; i++) if (insert(i)) { cnt++; cost += (int)mat[i][m + 1]; }
  }
  pair<int,int> run(int nn, int mm, const vector<vector<int>>& attr, const vector<int>& price) {
    n = nn; m = mm;
    mat.assign(n + 1, vector<double>(m + 2, 0));
    for (int i = 1; i <= n; i++) { for (int j = 1; j <= m; j++) mat[i][j] = attr[i][j]; mat[i][m + 1] = price[i]; }
    compute();
    return {cnt, cost};
  }
};

  for (int j = 0; j < m; j++) {

    for (int i = r; i < (int)rows.size(); i++) if (rows[i][j] != 0) { piv = i; break; }
    if (piv == -1) continue;
    swap(rows[r], rows[piv]);
    for (int i = 0; i < (int)rows.size(); i++) {
      if (i != r && rows[i][j] != 0) {
        ll a = rows[r][j], b = rows[i][j];
        ll g = std::gcd(a < 0 ? -a : a, b < 0 ? -b : b);
        a /= g; b /= g;
        for (int k = 0; k < m; k++) rows[i][k] = rows[i][k] * a - rows[r][k] * b;
      }
    }
    r++;
  }
  return r;
}

pair<int,int> brute(int n, int m, const vector<vector<int>>& attr, const vector<int>& price) {
  int bestCnt = 0, bestCost = INT_MAX;
  for (int mask = 1; mask < (1 << n); mask++) {
    vector<vector<ll>> rows;
    int costSum = 0;
    for (int i = 0; i < n; i++) if (mask >> i & 1) {
      vector<ll> row(m);
      for (int j = 0; j < m; j++) row[j] = attr[i + 1][j + 1];   // 属性列 1-based
      rows.push_back(row);
      costSum += price[i + 1];
    }
    int sz = rows.size();
    if (rankRows(rows, m) != sz) continue;
    if (sz > bestCnt || (sz == bestCnt && costSum < bestCost)) { bestCnt = sz; bestCost = costSum; }
  }
  if (bestCnt == 0) bestCost = 0;
  return {bestCnt, bestCost};
}

int main() {
  mt19937 rng(1234567);
  int fails = 0;
  for (int T = 0; T < 5000; T++) {
    int n = 1 + rng() % 6, m = 1 + rng() % 4;
    vector<vector<int>> attr(n + 1, vector<int>(m + 1, 0));
    vector<int> price(n + 1);
    for (int i = 1; i <= n; i++) { for (int j = 1; j <= m; j++) attr[i][j] = rng() % 4; price[i] = 1 + rng() % 9; }
    auto c = Course().run(n, m, attr, price);
    auto b = brute(n, m, attr, price);
    if (c.first != b.first || c.second != b.second) {
      fails++;

      for (int i = 1; i <= n; i++) { printf("  物品%d 属性:", i); for (int j = 1; j <= m; j++) printf(" %d", attr[i][j]); printf(" 价格=%d\n", price[i]); }
      if (fails > 3) return 1;
    }
  }
  if (fails) return 1;
  printf("5000 组 ALL OK\n");
  return 0;
}

// 题2 P哥的桶：课程（线段树 + 线性基） vs 暴力（桶数组 + 区间合并线性基）
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int BIT2 = 30;

struct SegTree {
  static const int MAXN = 50001;
  int treeBasis[MAXN << 2][BIT2 + 1];
  int basis[BIT2 + 1];
  bool insert(int* bs, int num) {
    for (int i = BIT2; i >= 0; i--) {
      if (num >> i == 1) {
        if (bs[i] == 0) { bs[i] = num; return true; }
        num ^= bs[i];
      }
    }
    return false;
  }
  void add(int jobi, int jobv, int l, int r, int i) {
    insert(treeBasis[i], jobv);
    if (l < r) {
      int mid = (l + r) >> 1;
      if (jobi <= mid) add(jobi, jobv, l, mid, i << 1);
      else add(jobi, jobv, mid + 1, r, i << 1 | 1);
    }
  }
  void merge(int jobl, int jobr, int l, int r, int i) {
    if (jobl <= l && r <= jobr) {
      for (int j = BIT2; j >= 0; j--) if (treeBasis[i][j] != 0) insert(basis, treeBasis[i][j]);
    } else {
      int mid = (l + r) >> 1;
      if (jobl <= mid) merge(jobl, jobr, l, mid, i << 1);
      if (jobr > mid) merge(jobl, jobr, mid + 1, r, i << 1 | 1);
    }
  }
  int query(int jobl, int jobr, int m) {
    memset(basis, 0, sizeof(basis));
    merge(jobl, jobr, 1, m, 1);
    int ans = 0;
    for (int j = BIT2; j >= 0; j--) ans = max(ans, ans ^ basis[j]);
    return ans;
  }
  void reset() {
    memset(treeBasis, 0, sizeof(treeBasis));
  }
};

SegTree st;

int bruteMax(const vector<vector<int>>& bucket, int l, int r, int n) {
  int basis[BIT2 + 1];
  memset(basis, 0, sizeof(basis));
  for (int k = max(1, l); k <= min(r, n); k++) {
    for (int v : bucket[k]) {
      int num = v;
      for (int i = BIT2; i >= 0; i--) {
        if (num >> i == 1) {
          if (basis[i] == 0) { basis[i] = num; break; }
          num ^= basis[i];
        }
      }
    }
  }
  int ans = 0;
  for (int i = BIT2; i >= 0; i--) ans = max(ans, ans ^ basis[i]);
  return ans;
}

int main() {
  mt19937 rng(987654321);
  int fails = 0;

    int n = 1 + rng() % 8;                 // 桶数
    int m = n + rng() % 20;                // 操作数（≥ n，线段树范围 1..m 覆盖 1..n）
    st.reset();
    vector<vector<int>> bucket(n + 1);
    for (int opi = 1; opi <= m; opi++) {
      int op = 1 + rng() % 2;
      if (op == 1) {
        int k = 1 + rng() % n;             // 桶号 1..n
        int v = rng() % 16;                // 值域小，容易碰撞
        st.add(k, v, 1, m, 1);
        bucket[k].push_back(v);
      } else {
        int l = 1 + rng() % n, r = l + rng() % (n - l + 1);
        int c = st.query(l, r, m);
        int b = bruteMax(bucket, l, r, n);
        if (c != b) {
          fails++;
          printf("MISMATCH T=%d op2 l=%d r=%d 课程=%d 暴力=%d\n", T, l, r, c, b);
          if (fails > 3) return 1;
        }
      }
    }
  }
  if (fails) return 1;
  printf("3000 组 ALL OK\n");
  return 0;
}

// 题3 幸运数字：课程（树上倍增 + 带深度替换线性基） vs 暴力（路径点集线性基最大异或）
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int MAXN = 20002, LIMIT = 16, BIT3 = 60;

// ---- 课程（照 Code03_LuckyNumber2 迭代版）----
ll arr3[MAXN];
int head3[MAXN], next3[MAXN << 1], to3[MAXN << 1], cnt3;
int deep3[MAXN], stjump3[MAXN][LIMIT], power3;
ll bases3[MAXN][BIT3 + 1];
int levels3[MAXN][BIT3 + 1];
int ufe[MAXN][3], stackSize, u3, f3, e3;

void addEdge3(int u, int v) {
  next3[cnt3] = head3[u]; to3[cnt3] = v; head3[u] = cnt3++;
}
int log2(int n) { int ans = 0; while ((1 << ans) <= (n >> 1)) ans++; return ans; }
void push3(int u, int f, int e) { ufe[stackSize][0] = u; ufe[stackSize][1] = f; ufe[stackSize][2] = e; stackSize++; }
void pop3() { --stackSize; u3 = ufe[stackSize][0]; f3 = ufe[stackSize][1]; e3 = ufe[stackSize][2]; }
bool insertReplace3(ll curv, int curl, ll* basis, int* level) {
  for (int i = BIT3; i >= 0; i--) {
    if (curv >> i == 1) {
      if (basis[i] == 0) { basis[i] = curv; level[i] = curl; return true; }
      if (curl > level[i]) {
        ll tmp1 = curv; curv = basis[i]; basis[i] = tmp1;
        int tmp2 = level[i]; level[i] = curl; curl = tmp2;
      }
      curv ^= basis[i];
    }
  }
  return false;
}
void dfs3(int root, int n) {
  stackSize = 0;
  push3(root, 0, -1);
  while (stackSize > 0) {
    pop3();
    if (e3 == -1) {
      deep3[u3] = deep3[f3] + 1;
      stjump3[u3][0] = f3;
      for (int p = 1; p <= power3; p++) stjump3[u3][p] = stjump3[stjump3[u3][p - 1]][p - 1];
      for (int i = 0; i <= BIT3; i++) { bases3[u3][i] = bases3[f3][i]; levels3[u3][i] = levels3[f3][i]; }
      insertReplace3(arr3[u3], deep3[u3], bases3[u3], levels3[u3]);
      e3 = head3[u3];
    } else {
      e3 = next3[e3];
    }
    if (e3 != 0) {
      push3(u3, f3, e3);
      if (to3[e3] != f3) push3(to3[e3], u3, -1);
    }
  }
}
int lca3(int x, int y) {
  if (deep3[x] < deep3[y]) { int t = x; x = y; y = t; }
  for (int p = power3; p >= 0; p--) if (deep3[stjump3[x][p]] >= deep3[y]) x = stjump3[x][p];
  if (x == y) return x;
  for (int p = power3; p >= 0; p--) if (stjump3[x][p] != stjump3[y][p]) { x = stjump3[x][p]; y = stjump3[y][p]; }
  return stjump3[x][0];
}
ll basis3[BIT3 + 1];
ll query3(int x, int y) {
  int lc = lca3(x, y);
  ll* basisx = bases3[x]; int* levelx = levels3[x];
  ll* basisy = bases3[y]; int* levely = levels3[y];
  memset(basis3, 0, sizeof(basis3));
  for (int i = BIT3; i >= 0; i--) { ll num = basisx[i]; if (levelx[i] >= deep3[lc] && num != 0) basis3[i] = num; }
  for (int i = BIT3; i >= 0; i--) {
    ll num = basisy[i];
    if (levely[i] >= deep3[lc] && num != 0) {
      for (int j = i; j >= 0; j--) {
        if (num >> j == 1) {
          if (basis3[j] == 0) { basis3[j] = num; break; }
          num ^= basis3[j];
        }
      }
    }
  }
  ll ans = 0;
  for (int i = BIT3; i >= 0; i--) ans = max(ans, ans ^ basis3[i]);
  return ans;
}

// ---- 暴力：路径点集线性基 ----
vector<vector<int>> g;   // 1-based
vector<int> par, dep;
void dfsBuild(int u, int fa) {
  par[u] = fa;
  for (int v : g[u]) if (v != fa) { dep[v] = dep[u] + 1; dfsBuild(v, u); }
}
ll bruteMax(const vector<ll>& nums) {
  ll basis[BIT3 + 1]; memset(basis, 0, sizeof(basis));
  for (ll v : nums) {
    for (int i = BIT3; i >= 0; i--) {
      if (v >> i == 1) {
        if (basis[i] == 0) { basis[i] = v; break; }
        v ^= basis[i];
      }
    }
  }
  ll ans = 0;
  for (int i = BIT3; i >= 0; i--) ans = max(ans, ans ^ basis[i]);
  return ans;
}
ll brutePath(int x, int y, const vector<ll>& arr) {
  vector<ll> nums;
  vector<int> px, py;
  int a = x, b = y;
  while (dep[a] > dep[b]) { px.push_back(a); a = par[a]; }
  while (dep[b] > dep[a]) { py.push_back(b); b = par[b]; }
  while (a != b) { px.push_back(a); py.push_back(b); a = par[a]; b = par[b]; }
  px.push_back(a);
  for (int v : px) nums.push_back(arr[v]);
  for (int v : py) nums.push_back(arr[v]);
  return bruteMax(nums);
}

int main() {
  mt19937 rng(555123);
  int fails = 0;
  for (int T = 0; T < 3000; T++) {
    int n = 2 + rng() % 10;
    // 随机树
    g.assign(n + 1, {});
    par.assign(n + 1, 0); dep.assign(n + 1, 0);
    vector<ll> arr(n + 1);
    cnt3 = 1;
    memset(head3, 0, sizeof(head3));
    power3 = log2(n);
    for (int i = 1; i <= n; i++) arr[i] = rng() % 4096;   // 12 位小值域
    for (int i = 2; i <= n; i++) {
      int p = 1 + rng() % (i - 1);
      g[p].push_back(i); g[i].push_back(p);
      addEdge3(p, i); addEdge3(i, p);
    }
    for (int i = 1; i <= n; i++) arr3[i] = arr[i];
    dfsBuild(1, 0);
    dfs3(1, n);
    int q = 5 + rng() % 10;
    for (int qi = 0; qi < q; qi++) {
      int x = 1 + rng() % n, y = 1 + rng() % n;
      ll c = query3(x, y);
      ll b = brutePath(x, y, arr);
      if (c != b) {
        fails++;
        printf("MISMATCH T=%d q=%d x=%d y=%d 课程=%lld 暴力=%lld\n", T, qi, x, y, c, b);
        if (fails > 3) return 1;
      }
    }
  }
  if (fails) return 1;
  printf("3000 组 ALL OK\n");
  return 0;
}

// 题4 路径最大异或和：课程（DFS树 + 环线性基 + query(path[n])） vs 暴力（独立DFS树 + 枚举环组合）
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int MAXN = 50001, MAXM = 200002, BIT4 = 60;

// ---- 课程（照 Code04_MaximumXorOfPath2 迭代版）----
int head4[MAXN], next4[MAXM], to4[MAXM], cnt4;
ll weight4[MAXM];
ll basis4[BIT4 + 1];
bool visited4[MAXN];
ll path4[MAXN];
int ufe4[MAXN][3], stackSize4, u4, f4, e4;
ll pstack4[MAXN], p4;
bool insert4(ll num) {
  for (int i = BIT4; i >= 0; i--)
    if (num >> i == 1) {
      if (basis4[i] == 0) { basis4[i] = num; return true; }
      num ^= basis4[i];
    }
  return false;
}
void push4(int u, int f, int e, ll p) { ufe4[stackSize4][0] = u; ufe4[stackSize4][1] = f; ufe4[stackSize4][2] = e; pstack4[stackSize4] = p; stackSize4++; }
void pop4() { --stackSize4; u4 = ufe4[stackSize4][0]; f4 = ufe4[stackSize4][1]; e4 = ufe4[stackSize4][2]; p4 = pstack4[stackSize4]; }
void dfs4(int root, int n) {
  stackSize4 = 0;
  push4(root, 0, -1, 0);
  while (stackSize4 > 0) {
    pop4();
    if (e4 == -1) {
      path4[u4] = p4;
      visited4[u4] = true;
      e4 = head4[u4];
    } else {
      e4 = next4[e4];
    }
    if (e4 != 0) {
      push4(u4, f4, e4, p4);
      int v = to4[e4];
      ll xr = p4 ^ weight4[e4];
      if (visited4[v]) insert4(xr ^ path4[v]);
      else push4(v, u4, -1, xr);
    }
  }
}
ll query4(ll init) {
  for (int i = BIT4; i >= 0; i--) init = max(init, init ^ basis4[i]);
  return init;
}
void prepare4(int n) {
  cnt4 = 1;
  memset(head4, 0, sizeof(head4));
  memset(basis4, 0, sizeof(basis4));
  memset(visited4, 0, sizeof(visited4));
  memset(path4, 0, sizeof(path4));
}
void addEdge4(int u, int v, ll w) {
  next4[cnt4] = head4[u]; to4[cnt4] = v; weight4[cnt4] = w; head4[u] = cnt4++;
}

// ---- 暴力：独立 DFS 树 + 枚举环组合 ----
struct EdgeB { int v; ll w; };
vector<vector<EdgeB>> gb;
vector<ll> pathB;
vector<int> visB;
vector<ll> cycles;   // 环异或值
void dfsB(int u, int fa, ll p) {
  pathB[u] = p;
  visB[u] = 1;
  for (auto& e : gb[u]) {
    if (e.v == fa) continue;
    if (!visB[e.v]) dfsB(e.v, u, p ^ e.w);
    else cycles.push_back(p ^ e.w ^ pathB[e.v]);   // 任意已访问节点都构成环（含旁支子树）
  }
  visB[u] = 2;
}
ll bruteAns(int n) {
  cycles.clear();
  pathB.assign(n + 1, 0);
  visB.assign(n + 1, 0);
  dfsB(1, 0, 0);
  ll best = pathB[n];
  int k = cycles.size();
  for (int mask = 0; mask < (1 << k); mask++) {
    ll x = 0;
    for (int i = 0; i < k; i++) if (mask >> i & 1) x ^= cycles[i];
    best = max(best, pathB[n] ^ x);
  }
  return best;
}

int main() {
  mt19937 rng(20261017);
  int fails = 0;
  for (int T = 0; T < 3000; T++) {
    int n = 2 + rng() % 7;               // 2..8 个点
    // 先生成随机树保证连通
    gb.assign(n + 1, {});
    vector<pair<int,int>> edges;
    for (int i = 2; i <= n; i++) {
      int p = 1 + rng() % (i - 1);
      edges.push_back({p, i});
    }
    // 再加 0..4 条随机边（可能重复）
    int extra = rng() % 5;
    for (int i = 0; i < extra; i++) {
      int u = 1 + rng() % n, v = 1 + rng() % n;
      if (u == v) continue;
      edges.push_back({u, v});
    }
    prepare4(n);
    for (auto& e : edges) {
      ll w = rng() % 256;
      int u = e.first, v = e.second;
      gb[u].push_back({v, w}); gb[v].push_back({u, w});
      addEdge4(u, v, w); addEdge4(v, u, w);
    }
    dfs4(1, n);
    ll c = query4(path4[n]);
    ll b = bruteAns(n);
    if (c != b) {
      fails++;
      printf("MISMATCH T=%d 课程=%lld 暴力=%lld n=%d edges=%zu\n", T, c, b, n, edges.size());
      if (fails > 3) return 1;
    }
  }
  if (fails) return 1;
  printf("3000 组 ALL OK\n");
  return 0;
}
