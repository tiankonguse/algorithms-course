左程云
01分数规划最简化的问题给定n个数据，每个数据有(a, b)两个值，并且没有负数(a1, b1)、(a2, b2)、(a3, b3)、.. (an, bn)其中选出k个数据，希望让 : k个a的和 / k个b的和，这个比值尽量大，返回最大的比值错误的贪心：选择a/b最大的前k个数据，反例：(5, 5)  (0, 1)  (2, 6)暴力方法 : 2的n次方，每个数据都枚举要和不要两种选择计算机里关于小数的精度是有限的 或者 具体题目要求返回的比值精度也是规定好的那么此时没有必要精确求解，可以用二分答案法来逼近求解，保证规定精度内正确即可01分数规划的核心：数据转化成结余表达式的形式，当确定一个比值x，就在最优决策下，计算结余和最终希望获得在最优决策下，当结余和最接近0时，x的值手段：利用单调性进行二分
===== slide 3 =====
01分数规划
左程云
二分答案法找到最大比值的过程数据这么来看 : a1 - x * b1 = 结余1a2 - x * b2 = 结余2a3 - x * b3 = 结余3..规定精度的最小单位sml，找到x的可能范围[l, r]，然后二分逻辑如下：首先令x = (l + r) / 2，计算每一个结余，然后选出结余最大的前k名，然后加起来看看累加和1) 如果 结余累加和 >= 0，答案更新成此时的x，然后去右侧二分，范围变成[x + sml, r]2) 如果 结余累加和 <  0，答案不更新，然后去左侧二分，范围变成[l, x - sml]3) 如果发现范围[l,r]的大小 <= sml，过程停止，返回答案即可课上重点图解，时间复杂度O(n * logn * logv)

#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k;
    while (scanf("%d%d", &n, &k) == 2 && (n || k)) {
        vector<double> a(n), b(n);
        for (int i = 0; i < n; i++) scanf("%lf", &a[i]);
        for (int i = 0; i < n; i++) scanf("%lf", &b[i]);
        int keep = n - k;
        double best = 0;
        for (int mask = 0; mask < (1 << n); mask++) {
            if (__builtin_popcount(mask) != keep) continue;
            double sa = 0, sb = 0;
            for (int i = 0; i < n; i++)
                if (mask >> i & 1) { sa += a[i]; sb += b[i]; }
            if (sb <= 0) continue;
            best = max(best, sa / sb);
        }
        printf("%d\n", (int)(100 * best + 0.5));
    }
    return 0;
}

code/brute3.cpp

#include <bits/stdc++.h>
using namespace std;

// 暴力：枚举所有 Prufer 序列（Cayley），对每棵生成树算 sumCost/sumDist 取最小
int n;
vector<int> X, Y, Z;

void decode(vector<int>& pr, vector<pair<int,int>>& edges) {
    int m = n;
    vector<int> deg(n + 1, 1);
    for (int v : pr) deg[v]++;
    // 用集合维护叶子
    set<int> leaves;
    for (int v = 1; v <= n; v++) if (deg[v] == 1) leaves.insert(v);
    for (int v : pr) {
        int leaf = *leaves.begin();
        leaves.erase(leaves.begin());
        edges.push_back({leaf, v});
        if (--deg[v] == 1) leaves.insert(v);
    }
    // 剩两个
    int a = *leaves.begin(), b = *next(leaves.begin());
    edges.push_back({a, b});
}

int main() {
    while (scanf("%d", &n) == 1 && n != 0) {
        X.resize(n + 1); Y.resize(n + 1); Z.resize(n + 1);
        for (int i = 1; i <= n; i++) scanf("%d%d%d", &X[i], &Y[i], &Z[i]);
        double best = 1e100;
        int len = n - 2;
        int total = 1;
        for (int i = 0; i < len; i++) total *= n;
        for (int t = 0; t < total; t++) {
            int tmp = t;
            vector<int> pr(len);
            for (int i = 0; i < len; i++) { pr[i] = tmp % n + 1; tmp /= n; }
            vector<pair<int,int>> edges;
            decode(pr, edges);
            double sc = 0, sd = 0;
            for (auto& e : edges) {
                int a = e.first, b = e.second;
                sc += abs(Z[a] - Z[b]);
                sd += sqrt((double)(X[a] - X[b]) * (X[a] - X[b]) + (double)(Y[a] - Y[b]) * (Y[a] - Y[b]));
            }
            if (sd > 0) best = min(best, sc / sd);
        }
        printf("%.3f\n", best);
    }
    return 0;
}

code/brute4.cpp

#include <bits/stdc++.h>
using namespace std;

// 暴力：DFS 枚举所有简单环，求最小平均权
int n, m;
vector<vector<pair<int,double>>> g;
vector<int> pathv;
vector<double> pathw;  // pathw[i] = pathv[i] -> pathv[i+1] 的边权
vector<int> pos;       // inpath 下节点在 pathv 的位置
vector<char> inpath;
double best = 1e100;

void dfs(int u) {
    pos[u] = pathv.size();
    pathv.push_back(u);
    inpath[u] = 1;
    for (auto& e : g[u]) {
        int v = e.first;
        double w = e.second;
        if (inpath[v]) {
            int pv = pos[v];
            double s = 0;
            for (int i = pv; i < (int)pathw.size(); i++) s += pathw[i];
            s += w;  // 闭合边 u -> v
            int c = (int)pathv.size() - pv;
            if (c > 0) best = min(best, s / c);
        } else {
            pathw.push_back(w);
            dfs(v);
            pathw.pop_back();
        }
    }
    inpath[u] = 0;
    pathv.pop_back();
}

int main() {
    scanf("%d%d", &n, &m);
    g.resize(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v;
        double w;
        scanf("%d%d%lf", &u, &v, &w);
        g[u].push_back({v, w});
    }
    pos.assign(n + 1, -1);
    inpath.assign(n + 1, 0);
    for (int s = 1; s <= n; s++) {
        if (!inpath[s]) dfs(s);
    }
    if (best > 9e99) best = 1e7;  // 无环，对齐课程 ans 上限
    printf("%.8f\n", best);
    return 0;
}

code/brute5.cpp

#include <bits/stdc++.h>
using namespace std;

// 暴力：枚举 1..n 中恰好 k 个的闭包子集（选 v 必选其父），最大化战斗/花费
int main() {
    int k, n;
    scanf("%d%d", &k, &n);
    vector<int> c(n + 1), s(n + 1), pa(n + 1);
    for (int i = 1; i <= n; i++) scanf("%d%d%d", &c[i], &s[i], &pa[i]);
    double best = 0;
    for (int mask = 0; mask < (1 << n); mask++) {
        if (__builtin_popcount(mask) != k) continue;
        bool ok = true;
        for (int i = 1; i <= n && ok; i++)
            if (mask >> (i - 1) & 1) {
                int p = pa[i];
                if (p != 0 && !(mask >> (p - 1) & 1)) ok = false;
            }
        if (!ok) continue;
        long long sc = 0, ss = 0;
        for (int i = 1; i <= n; i++)
            if (mask >> (i - 1) & 1) { sc += c[i]; ss += s[i]; }
        if (sc > 0) best = max(best, (double)ss / sc);
    }
    printf("%.3f\n", best);
    return 0;
}

code/t1.cpp

#include <bits/stdc++.h>
using namespace std;

const double sml = 1e-6;

struct Node {
    double a, b, m;
};

vector<Node> arr;
int n, k;

bool check(double x) {
    for (int i = 1; i <= n; i++) arr[i].m = arr[i].a - x * arr[i].b;
    sort(arr.begin() + 1, arr.end(), [](const Node& u, const Node& v) {
        return u.m > v.m;
    });
    double sum = 0;
    for (int i = 1; i <= k; i++) sum += arr[i].m;
    return sum >= 0;
}

int main() {
    while (scanf("%d%d", &n, &k) == 2 && (n || k)) {
        arr.resize(n + 1);
        k = n - k;  // 舍弃 k 个 = 保留 n-k 个
        for (int i = 1; i <= n; i++) scanf("%lf", &arr[i].a);
        for (int i = 1; i <= n; i++) scanf("%lf", &arr[i].b);
        double l = 0, r = 0, ans = 0;
        for (int i = 1; i <= n; i++) r += arr[i].a;
        while (l < r && r - l >= sml) {
            double x = (l + r) / 2;
            if (check(x)) {
                ans = x;
                l = x + sml;
            } else {
                r = x - sml;
            }
        }
        printf("%d\n", (int)(100 * (ans + 0.005)));
    }
    return 0;
}

code/t2.cpp

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 251;
const int MAXW = 1001;
const double NA = -1e9;
const double sml = 1e-6;

int n, w;
int weight[MAXN], talent[MAXN];
double value[MAXN];
double dp[MAXW];

bool check(double x) {
    for (int i = 1; i <= n; i++) value[i] = (double)talent[i] - x * weight[i];
    dp[0] = 0;
    for (int j = 1; j <= w; j++) dp[j] = NA;
    for (int i = 1; i <= n; i++) {
        for (int p = w; p >= 0; p--) {
            int j = p + weight[i];
            if (j >= w) {
                dp[w] = max(dp[w], dp[p] + value[i]);
            } else {
                dp[j] = max(dp[j], dp[p] + value[i]);
            }
        }
    }
    return dp[w] >= 0;
}

int main() {
    scanf("%d%d", &n, &w);
    for (int i = 1; i <= n; i++) scanf("%d%d", &weight[i], &talent[i]);
    double l = 0, r = 0, ans = 0;
    for (int i = 1; i <= n; i++) r += talent[i];
    while (l < r && r - l >= sml) {
        double x = (l + r) / 2;
        if (check(x)) {
            ans = x;
            l = x + sml;
        } else {
            r = x - sml;
        }
    }
    printf("%d\n", (int)(ans * 1000));
    return 0;
}

code/t3.cpp

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1001;
const double sml = 1e-6;

int n;
int X[MAXN], Y[MAXN], Z[MAXN];
double dist[MAXN][MAXN], cost[MAXN][MAXN];
bool vis[MAXN];
double value[MAXN];

double prim(double x) {
    for (int i = 1; i <= n; i++) {
        vis[i] = false;
        value[i] = cost[1][i] - x * dist[1][i];
    }
    vis[1] = true;
    double sum = 0;
    for (int i = 1; i <= n - 1; i++) {
        double minDist = 1e100;
        int next = 0;
        for (int j = 1; j <= n; j++) {
            if (!vis[j] && value[j] < minDist) {
                minDist = value[j];
                next = j;
            }
        }
        sum += minDist;
        vis[next] = true;
        for (int j = 1; j <= n; j++) {
            if (!vis[j] && value[j] > cost[next][j] - x * dist[next][j]) {
                value[j] = cost[next][j] - x * dist[next][j];
            }
        }
    }
    return sum;
}

int main() {
    while (scanf("%d", &n) == 1 && n != 0) {
        for (int i = 1; i <= n; i++) scanf("%d%d%d", &X[i], &Y[i], &Z[i]);
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (i != j) {
                    dist[i][j] = sqrt((double)(X[i] - X[j]) * (X[i] - X[j]) +
                                      (double)(Y[i] - Y[j]) * (Y[i] - Y[j]));
                    cost[i][j] = abs(Z[i] - Z[j]);
                }
            }
        }
        double l = 0, r = 100, ans = 0;
        while (l < r && r - l >= sml) {
            double x = (l + r) / 2;
            if (prim(x) <= 0) {
                ans = x;

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 3001;
const int MAXM = 10001;
const double MAXE = 1e7;
const double sml = 1e-9;

int n, m;
int head[MAXN], nxt[MAXM], to[MAXM];
double wgt[MAXM];
int cnt;
double value[MAXN];
bool path[MAXN];

void prepare() {
    cnt = 1;
    fill(head + 1, head + n + 1, 0);
}

void addEdge(int u, int v, double w) {
    nxt[cnt] = head[u];
    to[cnt] = v;
    wgt[cnt] = w;
    head[u] = cnt++;
}

bool dfs(int u, double x) {
    if (u == 0) {
        for (int i = 1; i <= n; i++) {
            if (dfs(i, x)) return true;
        }
    } else {
        path[u] = true;
        for (int e = head[u]; e != 0; e = nxt[e]) {
            int v = to[e];
            double w = wgt[e] - x;
            if (value[v] > value[u] + w) {
                value[v] = value[u] + w;
                if (path[v] || dfs(v, x)) return true;
            }
        }
        path[u] = false;
    }
    return false;
}

bool check(double x) {
    fill(value + 1, value + n + 1, 0);
    fill(path + 1, path + n + 1, false);
    return dfs(0, x);
}

int main() {
    scanf("%d%d", &n, &m);
    prepare();
    for (int i = 0; i < m; i++) {
        int u, v;
        double w;
        scanf("%d%d%lf", &u, &v, &w);
        addEdge(u, v, w);
    }
    double l = -MAXE, r = MAXE, ans = 0;
    while (l < r && r - l >= sml) {
        double x = (l + r) / 2;
        if (check(x)) {
            r = x - sml;
        } else {
            ans = x;
            l = x + sml;
        }
    }
    printf("%.8f\n", ans);
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 3001;
const int LIMIT = 10000;
const double NA = -1e9;
const double sml = 1e-6;

int k, n;
int head[MAXN], nxt[MAXN], to[MAXN], edgeCnt;
int cost[MAXN], strength[MAXN];
int dfn[MAXN], dfnCnt;
double value[MAXN];
int sz[MAXN];
double dp[MAXN][MAXN];

void addEdge(int u, int v) {
    nxt[edgeCnt] = head[u];
    to[edgeCnt] = v;
    head[u] = edgeCnt++;
}

int dfs(int u) {
    int i = ++dfnCnt;
    dfn[u] = i;
    sz[i] = 1;
    for (int e = head[u]; e != 0; e = nxt[e]) {
        int v = to[e];
        sz[i] += dfs(v);
    }
    return sz[i];
}

bool check(double x) {
    for (int i = 0; i <= n; i++) value[dfn[i]] = (double)strength[i] - x * cost[i];
    for (int j = 1; j <= k; j++) dp[dfnCnt + 1][j] = NA;
    for (int i = dfnCnt; i >= 2; i--) {
        for (int j = 1; j <= k; j++) {
            dp[i][j] = max(dp[i + sz[i]][j], value[i] + dp[i + 1][j - 1]);
        }
    }
    return dp[2][k] >= 0;
}

int main() {
    scanf("%d%d", &k, &n);
    edgeCnt = 1;
    dfnCnt = 0;
    fill(head + 1, head + n + 1, 0);
    for (int i = 1; i <= n; i++) {
        scanf("%d%d", &cost[i], &strength[i]);
        int p;
        scanf("%d", &p);
        addEdge(p, i);
    }
    dfs(0);
    double l = 0, r = LIMIT, ans = 0;
    while (l < r && r - l >= sml) {
        double x = (l + r) / 2;
        if (check(x)) {
            ans = x;
            l = x + sml;
        } else {
            r = x - sml;
        }
    }
    printf("%.3f\n", ans);
    return 0;
}
