两次转圈法实现同余最短路给定一个基准数x，那么可能的余数为0..x-1假设当前余数是cur，当前出现数字为y，(cur + y) % x 所出现的新余数，最终会回到cur当前出现数字为y，0..x-1这些点，会形成gcd(x,y)个子环，所有子环的起点为：0 ~ gcd(x,y)-1每个子环为单位，进行同余最短路的更新，所有子环的所有点的数量 = x，即更新一轮的代价O(x)因为是最短路，所以每出现一个新的数字y，所形成的每个子环，只需要转一圈即可完成更新特别注意：子环的起点不一定是点权最小的点，所以实现中，往往用转两圈的方式如果一共有n个数字，基准数为x，那么同余最短路的更新代价为，O(n * x)spfa算法也可以实现同余最短路的更新，但是比赛中经常设计数据，让其时间复杂度高于O(n * x)为了顺利通过，所以更推荐两次绕圈法

code/cattle.cpp

#include <bits/stdc++.h>
using namespace std;
// 牛场围栏 P2662：同余最短路(dijkstra)，模数取最短可用长度
int n, m;
int arr[101];
long long dist[3001];
bool vis[3001];
int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> arr[i];
    bool usable[3001];
    fill(usable, usable + 3001, false);
    int x = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = max(1, arr[i] - m); j <= arr[i]; j++) {
            if (!usable[j]) {
                usable[j] = true;
                if (x == 0) x = j;
            }
        }
    }
    if (x == 1) { cout << -1 << endl; return 0; }
    for (int i = 0; i < x; i++) { dist[i] = LLONG_MAX; vis[i] = false; }
    dist[0] = 0;
    priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;
    pq.push({0, 0});
    while (!pq.empty()) {
        auto top = pq.top(); pq.pop();
        int u = top.second;
        if (vis[u]) continue;
        vis[u] = true;
        for (int j = 1; j <= 3000; j++) {
            if (!usable[j]) continue;
            int nxt = (u + j) % x;
            if (dist[nxt] > dist[u] + j) { dist[nxt] = dist[u] + j; pq.push({dist[nxt], nxt}); }
        }
    }
    long long ans = 0;
    for (int i = 1; i < x; i++) {
        if (dist[i] == LLONG_MAX) { cout << -1 << endl; return 0; }
        ans = max(ans, dist[i] - x);
    }
    cout << ans << endl;
    return 0;
}

code/elevator.cpp

#include <bits/stdc++.h>
using namespace std;
// 跳楼机 P3403：同余最短路(dijkstra)，模数取第一个数 x
long long h;
int x, y, z;
long long dist[100001];
bool vis[100001];
int main() {
    cin >> h >> x >> y >> z;
    h -= 1;
    for (int i = 0; i < x; i++) { dist[i] = LLONG_MAX; vis[i] = false; }
    dist[0] = 0;
    priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;
    pq.push({0, 0});
    while (!pq.empty()) {
        auto top = pq.top(); pq.pop();
        int u = top.second;
        if (vis[u]) continue;
        vis[u] = true;
        int nxt = (u + y) % x;
        if (dist[nxt] > dist[u] + y) { dist[nxt] = dist[u] + y; pq.push({dist[nxt], nxt}); }
        nxt = (u + z) % x;
        if (dist[nxt] > dist[u] + z) { dist[nxt] = dist[u] + z; pq.push({dist[nxt], nxt}); }
    }
    long long ans = 0;
    for (int i = 0; i < x; i++) {
        if (dist[i] <= h) ans += (h - dist[i]) / x + 1;
    }
    cout << ans << endl;
    return 0;
}

code/momo_dijk.cpp

#include <bits/stdc++.h>
using namespace std;
// 墨墨的等式 P2371 (dijkstra 版)：同余图最短路 + 区间计数
int n;
long long l, r;
int v[13];
long long dist[500001];
bool vis[500001];
long long cnt(long long bound) {
    long long s = 0;
    for (int i = 0; i < v[0]; i++) {
        if (dist[i] <= bound) s += (bound - dist[i]) / v[0] + 1;
    }
    return s;
}
int main() {
    cin >> n >> l >> r;
    for (int i = 0; i < n; i++) cin >> v[i];
    if (n == 0) { cout << 0 << endl; return 0; }
    l -= 1;
    for (int i = 0; i < v[0]; i++) { dist[i] = LLONG_MAX; vis[i] = false; }
    dist[0] = 0;
    priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;
    pq.push({0, 0});
    while (!pq.empty()) {
        auto top = pq.top(); pq.pop();
        int u = top.second;
        if (vis[u]) continue;
        vis[u] = true;
        for (int i = 0; i < n; i++) {
            if (v[i] == 0) continue;
            int nxt = (u + v[i]) % v[0];
            if (dist[nxt] > dist[u] + v[i]) { dist[nxt] = dist[u] + v[i]; pq.push({dist[nxt], nxt}); }
        }
    }
    cout << cnt(r) - cnt(l) << endl;
    return 0;
}

#include <bits/stdc++.h>
using namespace std;
// 最小数位和 AT_arc084_b：01bfs 同余图，补0权0放队首、加1权1放队尾
int k;
int main() {
    cin >> k;
    vector<bool> vis(k, false);
    deque<pair<int, int>> dq;
    dq.push_back({1, 1});
    while (!dq.empty()) {
        auto cur = dq.front(); dq.pop_front();
        int mod = cur.first, cost = cur.second;
        if (vis[mod]) continue;
        vis[mod] = true;
        if (mod == 0) { cout << cost << endl; return 0; }
        int f = (mod * 10) % k;
        dq.push_front({f, cost});
        int b = (mod + 1) % k;
        dq.push_back({b, cost + 1});
    }
    return 0;
}

code/twoloop.cpp

#include <bits/stdc++.h>
using namespace std;
// 背包 P9140 两次转圈法：基准=性价比最高物品，dp[r]=替换基准后的最大价值补偿
int n, m;
int x, y;
long long v[100001], c[100001];
long long dp[100001];
int gcd(int a, int b) { return b == 0 ? a : gcd(b, a % b); }
int main() {
    cin >> n >> m;
    double best = 0;
    for (int i = 1; i <= n; i++) {
        cin >> v[i] >> c[i];
        double ratio = (double)c[i] / v[i];
        if (ratio > best) { best = ratio; x = v[i]; y = c[i]; }
    }
    fill(dp, dp + x, LLONG_MIN);
    dp[0] = 0;
    for (int i = 1; i <= n; i++) {
        if (v[i] == x) continue;
        int d = gcd(v[i], x);
        for (int j = 0; j < d; j++) {
            for (int cur = j, next, circle = 0; circle < 2; circle += cur == j ? 1 : 0) {
                next = (cur + v[i]) % x;
                if (dp[cur] != LLONG_MIN) {
                    dp[next] = max(dp[next], dp[cur] - ((cur + v[i]) / x) * (long long)y + c[i]);
                }
                cur = next;
            }
        }
    }
    long long jobv;
    for (int i = 1; i <= m; i++) {
        cin >> jobv;
        int r = (int)(jobv % x);
        if (dp[r] == LLONG_MIN) cout << -1 << endl;
        else cout << jobv / x * (long long)y + dp[r] << endl;
    }
    return 0;
}
