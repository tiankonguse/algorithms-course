#include <bits/stdc++.h>
using namespace std;

// 题1：休息k分钟最大会议和
// 课程法：jump 数组预处理（滑动窗口）+ 倒序 dp
// 暴力法：直接枚举每个 i 向后累加直到 >= k

long long best1(const vector<int>& arr, int k) {  // 暴力 O(n^2)
    int n = arr.size();
    vector<long long> dp(n + 1, 0);
    for (int i = n - 1; i >= 0; i--) {
        int j = i + 1, sum = 0;
        while (j < n && sum < k) sum += arr[j++];
        dp[i] = max(dp[i + 1], dp[j] + (long long)arr[i]);
    }
    return dp[0];
}

long long best2(const vector<int>& arr, int k) {  // 课程：jump 预处理 O(n)
    int n = arr.size();
    vector<int> jump(n, 0);
    for (int i = 0, l = 1, r = 1, sum = 0; i < n - 1; i++, l++) {
        while (r < n && sum < k) sum += arr[r++];
        jump[i] = r;
        sum -= arr[l];
    }
    jump[n - 1] = n;
    vector<long long> dp(n + 1, 0);
    for (int i = n - 1; i >= 0; i--) {
        dp[i] = max(dp[i + 1], dp[jump[i]] + (long long)arr[i]);
    }
    return dp[0];
}

int main() {
    // 固定例
    {
        vector<int> arr = {200, 5, 6, 14, 7, 300};
        int k = 15;
        printf("固定例: best1=%lld best2=%lld (期望 500)\n", best1(arr, k), best2(arr, k));
        assert(best1(arr, k) == 500 && best2(arr, k) == 500);
    }
    {
        vector<int> arr = {1, 1, 1};
        printf("固定例 [1,1,1] k=1: %lld %lld\n", best1(arr, 1), best2(arr, 1));
    }
    mt19937 rnd(20261016);
    int T = 20000, bad = 0;
    for (int t = 0; t < T; t++) {
        int n = rnd() % 12 + 1;
        int v = rnd() % 15 + 1;
        int k = rnd() % 20 + 1;
        vector<int> arr(n);
        for (int i = 0; i < n; i++) arr[i] = rnd() % v + 1;
        long long a = best1(arr, k), b = best2(arr, k);
        if (a != b) {
            bad++;
            printf("MISMATCH n=%d k=%d\n", n, k);
            for (int x : arr) printf("%d ", x);
            printf("\na=%lld b=%lld\n", a, b);
            if (bad > 5) break;
        }
    }
    printf("随机 %d 组, bad=%d\n", T, bad);
    return bad ? 1 : 0;
}

code/t2.cpp

#include <bits/stdc++.h>
using namespace std;

// 题2：炮兵阵地
// 课程法：行内合法状态 DFS 生成 + 滚动 dp（memo/dp）
// 暴力法：逐行 DFS 枚举状态子集，检查行间十字攻击

const int MAXN = 100, MAXM = 10;

// ---- 课程法（同 Code02_SoldierPosition2）----
struct Course {
    int n, m, k;
    vector<vector<int>> grid;
    vector<int> sta;
    vector<vector<int>> memo, dp;
    void prepare(int j, int s) {
        if (j >= m) { sta.push_back(s); return; }
        prepare(j + 1, s);
        prepare(j + 3, s | (1 << j));
    }
    int cnt(int i, int s) {
        int ans = 0;
        for (int j = 0; j < m; j++)
            if (((s >> j) & 1) && grid[i][j]) ans++;
        return ans;
    }
    int compute() {
        k = sta.size();
        memo.assign(k, vector<int>(k, 0));
        dp.assign(k, vector<int>(k, 0));
        for (int a = 0; a < k; a++) memo[a][0] = cnt(0, sta[a]);
        for (int i = 1; i < n; i++) {
            for (int a = 0; a < k; a++) {
                int c = cnt(i, sta[a]);
                for (int b = 0; b < k; b++) {
                    dp[a][b] = 0;
                    if ((sta[a] & sta[b]) == 0) {
                        for (int cc = 0; cc < k; cc++) {
                            if ((sta[b] & sta[cc]) == 0 && (sta[a] & sta[cc]) == 0) {
                                dp[a][b] = max(dp[a][b], memo[b][cc] + c);
                            }
                        }
                    }
                }
            }
            swap(memo, dp);
        }
        int ans = 0;
        for (int a = 0; a < k; a++)
            for (int b = 0; b < k; b++)
                ans = max(ans, memo[a][b]);
        return ans;
    }
    int solve(const vector<string>& g) {
        n = g.size(); m = g[0].size();
        grid.assign(n, vector<int>(m, 0));
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                grid[i][j] = (g[i][j] == 'P' ? 1 : 0);
        sta.clear();
        prepare(0, 0);
        return compute();
    }
};

// ---- 暴力：逐行 DFS ----
int brute_n, brute_m;
vector<int> br_grid;  // 位表示
int best_cnt;

bool attack(int r1, int c1, int r2, int c2) {
    if (r1 == r2) return abs(c1 - c2) <= 2;
    if (c1 == c2) return abs(r1 - r2) <= 2;
    return false;
}

void dfs(int r, int prev1, int prev2, int cnt) {
    if (r == brute_n) { best_cnt = max(best_cnt, cnt); return; }
    for (int s = 0; s < (1 << brute_m); s++) {
        // 行内：与 grid 兼容 + 同行互不攻击（间隔 >= 3）
        bool ok = true;
        for (int c = 0; c < brute_m && ok; c++) {
            if (!((s >> c) & 1)) continue;
            if (!((br_grid[r] >> c) & 1)) { ok = false; break; }
            if (c >= 1 && ((s >> (c - 1)) & 1)) { ok = false; break; }
            if (c >= 2 && ((s >> (c - 2)) & 1)) { ok = false; break; }
        }
        if (!ok) continue;
        // 行间：与上一行、上上行十字攻击检查
        int pop = __builtin_popcount(s);
        bool ok2 = true;
        for (int c = 0; c < brute_m && ok2; c++) {
            if (!((s >> c) & 1)) continue;
            for (int c2 = 0; c2 < brute_m; c2++) {
                if ((prev1 >> c2) & 1) {
                    if (attack(r, c, r - 1, c2)) { ok2 = false; break; }
                }
                if ((prev2 >> c2) & 1) {
                    if (attack(r, c, r - 2, c2)) { ok2 = false; break; }
                }
            }
        }
        if (ok2) dfs(r + 1, s, prev1, cnt + pop);
    }
}

int brute(const vector<string>& g) {
    brute_n = g.size(); brute_m = g[0].size();
    br_grid.assign(brute_n, 0);
    for (int i = 0; i < brute_n; i++)
        for (int j = 0; j < brute_m; j++)
            if (g[i][j] == 'P') br_grid[i] |= (1 << j);
    best_cnt = 0;
    dfs(0, 0, 0, 0);
    return best_cnt;
}

int main() {
    {
        vector<string> g = {"P"};
        printf("固定例 1x1: course=%d brute=%d (期望 1)\n", Course().solve(g), brute(g));
        assert(Course().solve(g) == 1 && brute(g) == 1);
    }
    {
        vector<string> g = {"PP", "PP"};
        printf("固定例 2x2 全P: course=%d brute=%d (期望 2)\n", Course().solve(g), brute(g));
    }
    {
        vector<string> g = {"PPP", "PPP", "PPP"};
        printf("固定例 3x3 全P: course=%d brute=%d (期望 3)\n", Course().solve(g), brute(g));
    }
    mt19937 rnd(20261016);
    int T = 3000, bad = 0;
    for (int t = 0; t < T; t++) {
        int n = rnd() % 6 + 1, m = rnd() % 5 + 1;
        vector<string> g(n);
        for (int i = 0; i < n; i++) {
            g[i] = "";
            for (int j = 0; j < m; j++)
                g[i] += (rnd() % 100 < 80) ? 'P' : 'H';
        }
        int a = Course().solve(g), b = brute(g);
        if (a != b) {
            bad++;
            printf("MISMATCH n=%d m=%d course=%d brute=%d\n", n, m, a, b);
            for (auto& s : g) printf("%s\n", s.c_str());
            if (bad > 5) break;
        }
    }
    printf("随机 %d 组, bad=%d\n", T, bad);
    return bad ? 1 : 0;
}

code/t3.cpp

#include <bits/stdc++.h>
using namespace std;

// 题3：还原数组的方法数
// 约束：arr[i] <= max(arr[i-1], arr[i+1])，边界 arr[1]<=arr[2], arr[n]<=arr[n-1]
// 课程法：suf0/pre1 前缀和优化 + 滚动（m 参数化以便小值域对拍）
// 朴素法：三段枚举 cur（O(n*m^2)）
// 独立暴力：全排列填值，直接按题面约束检查

const int MOD = 998244353;

// 朴素：compute3 版（滚动 + 枚举）
int naive(const vector<int>& arr, int m) {
    int n = (int)arr.size() - 1;  // arr[1..n]，arr[0] 占位
    vector<vector<int>> memo(m + 1, vector<int>(2, 0)), dp(m + 1, vector<int>(2, 0));
    for (int v = 0; v <= m; v++) { memo[v][0] = 0; memo[v][1] = 1; }
    for (int i = 1; i <= n; i++) {
        for (int v = 0; v <= m; v++) {
            for (int s = 0; s <= 1; s++) {
                int ans = 0;
                if (arr[i] != 0) {
                    if (arr[i] >= v || s == 1) ans = memo[arr[i]][arr[i] > v ? 0 : 1];
                } else {
                    for (int cur = v + 1; cur <= m; cur++) ans = (ans + memo[cur][0]) % MOD;
                    if (v != 0) ans = (ans + memo[v][1]) % MOD;
                    if (s == 1) for (int cur = 1; cur < v; cur++) ans = (ans + memo[cur][1]) % MOD;
                }
                dp[v][s] = ans;
            }
        }
        swap(memo, dp);
    }
    return memo[0][1];
}

// 课程：suf0/pre1 优化 + 滚动
int course(const vector<int>& arr, int m) {
    int n = (int)arr.size() - 1;  // arr[1..n]
    vector<vector<int>> memo(m + 1, vector<int>(2, 0)), dp(m + 1, vector<int>(2, 0));
    vector<int> suf0(m + 1, 0), pre1(m + 1, 0);
    for (int v = 0; v <= m; v++) { memo[v][0] = 0; memo[v][1] = 1; }
    for (int i = 1; i <= n; i++) {
        suf0[m] = memo[m][0];
        for (int v = m - 1; v >= 0; v--) suf0[v] = (suf0[v + 1] + memo[v][0]) % MOD;
        pre1[1] = memo[1][1];
        for (int v = 2; v <= m; v++) pre1[v] = (pre1[v - 1] + memo[v][1]) % MOD;
        for (int v = 0; v <= m; v++) {
            for (int s = 0; s <= 1; s++) {
                int ans = 0;
                if (arr[i] != 0) {
                    if (arr[i] >= v || s == 1) ans = memo[arr[i]][arr[i] > v ? 0 : 1];
                } else {
                    if (v + 1 <= m) ans = (ans + suf0[v + 1]) % MOD;
                    if (v != 0) ans = (ans + memo[v][1]) % MOD;
                    if (s == 1 && v - 1 >= 0) ans = (ans + pre1[v - 1]) % MOD;
                }
                dp[v][s] = ans;
            }
        }
        swap(memo, dp);
    }
    return memo[0][1];
}

// 独立暴力：全排列填值
int brute(const vector<int>& arr, int m) {
    int n = (int)arr.size() - 1;  // arr[1..n]
    vector<int> a(n + 2, 0);  // 1..n，a[0]=a[n+1]=哨兵（约束只查 2..n-1？不，边界特殊）
    long long cnt = 0;
    function<void(int)> dfs = [&](int pos) {
        if (pos > n) {
            // 检查约束
            for (int i = 2; i <= n - 1; i++) {
                if (a[i] > max(a[i - 1], a[i + 1])) return;
            }
            if (a[1] > a[2]) return;
            if (a[n] > a[n - 1]) return;
            cnt++;
            return;
        }
        if (arr[pos] != 0) {
            a[pos] = arr[pos];
            dfs(pos + 1);
        } else {
            for (int v = 1; v <= m; v++) { a[pos] = v; dfs(pos + 1); }
        }
    };
    dfs(1);
    return (int)(cnt % MOD);
}

int main() {
    // 固定例
    {
        vector<int> arr = {0, 0, 0};
        printf("固定例 [0,0,0] m=3: naive=%d course=%d brute=%d\n", naive(arr, 3), course(arr, 3), brute(arr, 3));
    }
    {
        vector<int> arr = {1, 0, 0};
        printf("固定例 [1,0,0] m=3: naive=%d course=%d brute=%d\n", naive(arr, 3), course(arr, 3), brute(arr, 3));
    }
    mt19937 rnd(20261016);
    int T = 5000, bad = 0;
    for (int t = 0; t < T; t++) {
        int n = rnd() % 5 + 3;
        int m = rnd() % 6 + 2;
        vector<int> arr(n + 1, 0);
        for (int i = 1; i <= n; i++) {
            if (rnd() % 100 < 60) arr[i] = rnd() % m + 1;
        }
        int a = naive(arr, m), b = course(arr, m), c = brute(arr, m);
        if (a != b || a != c) {
            bad++;
            printf("MISMATCH n=%d m=%d naive=%d course=%d brute=%d\n", n, m, a, b, c);
            for (int i = 1; i <= n; i++) printf("%d ", arr[i]);
            printf("\n");
            if (bad > 5) break;
        }
    }
    printf("随机 %d 组, bad=%d\n", T, bad);
    return bad ? 1 : 0;
}

#include <bits/stdc++.h>
using namespace std;

// 题4：粉刷房子III
// 课程法：pre/suf 前缀最小值优化 + 滚动（minCost4）
// 朴素法：枚举 color（minCost2 逻辑）
// 独立暴力：未涂色房全排列颜色，统计街区

const int NA = INT_MAX;

// 朴素 O(n*t*c^2)
int naive(const vector<int>& house, const vector<vector<int>>& cost, int n, int c, int t) {
    t++;
    vector<vector<int>> memo(t + 1, vector<int>(c + 1, NA)), dp(t + 1, vector<int>(c + 1, NA));
    for (int v = 0; v <= c; v++) memo[0][v] = dp[0][v] = NA;
    for (int j = 1; j <= t; j++)
        for (int v = 0; v <= c; v++)
            memo[j][v] = (j == 1 ? 0 : NA);
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= t; j++) {
            for (int v = 0; v <= c; v++) {
                int ans = NA;
                if (house[i] != 0) {
                    if (house[i] == v) ans = memo[j][house[i]];
                    else ans = memo[j - 1][house[i]];
                } else {
                    for (int color = 1; color <= c; color++) {
                        int next = (color == v) ? memo[j][color] : memo[j - 1][color];
                        if (next != NA) ans = min(ans, next + cost[i][color]);
                    }
                }
                dp[j][v] = ans;
            }
        }
        swap(memo, dp);
    }
    int ans = memo[t][0];
    return ans == NA ? -1 : ans;
}

// 课程：pre/suf 优化 O(n*t*c)
int course(const vector<int>& house, const vector<vector<int>>& cost, int n, int c, int t) {
    t++;
    vector<vector<int>> memo(t + 1, vector<int>(c + 1, NA)), dp(t + 1, vector<int>(c + 1, NA));
    for (int v = 0; v <= c; v++) memo[0][v] = dp[0][v] = NA;
    for (int j = 1; j <= t; j++)
        for (int v = 0; v <= c; v++)
            memo[j][v] = (j == 1 ? 0 : NA);
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= t; j++) {
            vector<int> pre(c + 2, NA), suf(c + 2, NA);
            pre[0] = suf[c + 1] = NA;
            for (int v = 1; v <= c; v++) {
                pre[v] = pre[v - 1];
                if (memo[j - 1][v] != NA) pre[v] = min(pre[v], memo[j - 1][v] + cost[i][v]);
            }
            for (int v = c; v >= 1; v--) {
                suf[v] = suf[v + 1];
                if (memo[j - 1][v] != NA) suf[v] = min(suf[v], memo[j - 1][v] + cost[i][v]);
            }
            for (int v = 0; v <= c; v++) {
                int ans = NA;
                if (house[i] != 0) {
                    if (house[i] == v) ans = memo[j][house[i]];
                    else ans = memo[j - 1][house[i]];
                } else {
                    if (v == 0) ans = suf[1];
                    else {
                        ans = min(pre[v - 1], suf[v + 1]);
                        if (memo[j][v] != NA) ans = min(ans, memo[j][v] + cost[i][v]);
                    }
                }
                dp[j][v] = ans;
            }
        }
        swap(memo, dp);
    }
    int ans = memo[t][0];
    return ans == NA ? -1 : ans;
}

// 独立暴力：全排列颜色
int brute(const vector<int>& house, const vector<vector<int>>& cost, int n, int c, int t) {
    vector<int> col(n + 1, 0);
    long long best = LLONG_MAX;
    function<void(int)> dfs = [&](int i) {
        if (i > n) {
            int blocks = 1;
            for (int k = 2; k <= n; k++) if (col[k] != col[k - 1]) blocks++;
            if (blocks != t) return;
            long long s = 0;
            for (int k = 1; k <= n; k++) if (house[k] == 0) s += cost[k][col[k]];
            best = min(best, s);
            return;
        }
        if (house[i] != 0) { col[i] = house[i]; dfs(i + 1); }
        else {
            for (int v = 1; v <= c; v++) { col[i] = v; dfs(i + 1); }
        }
    };
    dfs(1);
    return best == LLONG_MAX ? -1 : (int)best;
}

#include <bits/stdc++.h>
using namespace std;

// 题5：从上到下挖砖块
// grid[i][j] = 原三角形（第 j 行，右数第 i 列）；挖 grid[i][j] 需正上 grid[i][j-1] 与右上 grid[i-1][j-1]
// 课程法：后缀 max 预处理 + 滚动（compute2）
// 朴素法：枚举上一列 p（compute1）
// 独立暴力：每列枚举挖的块数 j（递减支撑链），直接按题面规则检查

int n, m;
vector<vector<int>> grid;

// 朴素 O(n^3*m)
int naive() {
    vector<vector<vector<int>>> dp(n + 1, vector<vector<int>>(n + 1, vector<int>(m + 1, 0)));
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        int cur = 0;
        for (int j = 0; j <= i; j++) {
            cur += grid[i][j];
            for (int k = (j + 1) * j / 2; k <= m; k++) {
                int mx = 0;
                for (int p = max(0, j - 1); p <= i - 1; p++)
                    mx = max(mx, dp[i - 1][p][k - j]);
                dp[i][j][k] = mx + cur;
                ans = max(ans, dp[i][j][k]);
            }
        }
    }
    return ans;
}

// 课程：后缀 max 优化 + 滚动 O(n^2*m)
int course() {
    vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));
    vector<vector<int>> mx(n + 1, vector<int>(m + 1, 0));
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int col = 0; col <= m; col++) {
            int suf = 0;
            for (int row = i - 1; row >= 0; row--) {
                suf = max(suf, dp[row][col]);
                mx[row][col] = suf;
            }
        }
        int cur = 0;
        for (int j = 0; j <= i; j++) {
            cur += grid[i][j];
            for (int k = (j + 1) * j / 2; k <= m; k++) {
                dp[j][k] = mx[max(0, j - 1)][k - j] + cur;
                ans = max(ans, dp[j][k]);
            }
        }
    }
    return ans;
}

// 独立暴力：枚举每列挖 j 块（j_i 递减支撑：j_{i-1} >= j_i - 1），收益 = 该列前 j 块和，总块数 <= m
int brute() {
    vector<int> colPick(n + 1, 0);
    long long best = 0;
    function<void(int, int, int, long long)> dfs = [&](int i, int prev, int cnt, long long sum) {
        // i: 当前列（从底 i=1 到顶 n）
        if (i > n) {
            if (cnt <= m) best = max(best, sum);
            return;
        }
        for (int j = 0; j <= i; j++) {
            // 挖第 i 列前 j 块，需要第 i-1 列前 max(0,j-1) 块已挖
            if (j - 1 > prev) continue;
            int c = 0;
            for (int k = 1; k <= j; k++) c += grid[i][k];
            dfs(i + 1, j, cnt + j, sum + c);
        }
    };
    dfs(1, 0, 0, 0);
    return (int)best;
}
