2-SAT（Satisfiability）问题每个变量的取值只有0和1两种，为了满足所有给定的限制，确定每个变量的值2-SAT求解过程1，每个变量x，有两个互斥的取值假设，x和¬x2，每个限制，形如 x 或 y，转化为两条关系，¬x -> y 和 ¬y -> x3，每个限制转化为关系时，要满足 取反反向的对称性，若 u -> v 必有 ¬v -> ¬u4，每个取值假设认为是点，每条关系认为是边，就构成了图，求解图中的强连通分量5，如果某个变量的互斥取值假设，对应图中的两个点，在同一个强连通分量里，则无解6，否则有解，每个变量选择SCC缩点后的DAG中，拓扑序更靠后的取值假设，作为变量的值7，变量取值时注意，拓扑序越靠后的点，归属的强连通分量的序号越小课上重点图解，核心是第6点，为什么总能得到可行解

#include <bits/stdc++.h>

using namespace std;

const int MAXN = 2000001;
int n, m;

int head[MAXN];
int nxt[MAXN];
int to[MAXN];
int cntg;

int dfn[MAXN];
int low[MAXN];
int cntd;

int sta[MAXN];
int top;

int belong[MAXN];
int sccCnt;

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;
}

void tarjan(int u) {
    dfn[u] = low[u] = ++cntd;
    sta[++top] = u;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (dfn[v] == 0) {
            tarjan(v);
            low[u] = min(low[u], low[v]);
        } else {
            if (belong[v] == 0) {
                low[u] = min(low[u], dfn[v]);
            }
        }
    }
    if (dfn[u] == low[u]) {
        sccCnt++;
        int pop;
        do {
            pop = sta[top--];
            belong[pop] = sccCnt;
        } while (pop != u);
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m;
    for (int i = 1, x1, v1, x2, v2; i <= m; i++) {
        cin >> x1 >> v1 >> x2 >> v2;
        if (v1 == 1 && v2 == 1) {
            addEdge(x1 + n, x2);
            addEdge(x2 + n, x1);
        } else if (v1 == 0 && v2 == 1) {
            addEdge(x1, x2);
            addEdge(x2 + n, x1 + n);
        } else if (v1 == 1 && v2 == 0) {
            addEdge(x2, x1);
            addEdge(x1 + n, x2 + n);
        } else {
            addEdge(x1, x2 + n);
            addEdge(x2, x1 + n);
        }
    }
    for (int i = 1; i <= n << 1; i++) {
        if (dfn[i] == 0) {
            tarjan(i);
        }
    }
    bool check = true;
    for (int i = 1; i <= n; i++) {
        if (belong[i] == belong[i + n]) {
            check = false;
            break;
        }
    }
    if (check) {
        cout << "POSSIBLE" << "\n";
        for (int i = 1; i <= n; i++) {
            if (belong[i] < belong[i + n]) {
                cout << "1 ";
            } else {
                cout << "0 ";
            }
        }
    } else {
        cout << "IMPOSSIBLE" << "\n";
    }
    return 0;
}

#include <bits/stdc++.h>

using namespace std;

const int MAXN = 20001;
const int MAXM = 2000001;
const int MAXK = 12;
const int MAXT = MAXN * MAXK * 2;
int t, n, m, k;

int id[MAXN][MAXK][2];
int cntt;

int head[MAXT];
int nxt[MAXM];

int cntg;

int dfn[MAXT];
int low[MAXT];
int cntd;

int sta[MAXT];
int top;

int belong[MAXT];
int sccCnt;

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;
}

void tarjan(int u) {
    dfn[u] = low[u] = ++cntd;
    sta[++top] = u;
    for (int e = head[u]; e > 0; e = nxt[e]) {
        int v = to[e];
        if (dfn[v] == 0) {
            tarjan(v);
            low[u] = min(low[u], low[v]);
        } else {
            if (belong[v] == 0) {
                low[u] = min(low[u], dfn[v]);

    }
}

void prepare() {
    cntg = cntd = sccCnt = 0;
    for (int i = 1; i <= cntt; i++) {
        head[i] = dfn[i] = low[i] = belong[i] = 0;
    }
    cntt = 0;
    for (int i = 1; i <= n; i++) {
        for (int v = 1; v <= k + 1; v++) {
            id[i][v][0] = ++cntt;
            id[i][v][1] = ++cntt;
        }
    }
    for (int i = 1; i <= n; i++) {
        for (int v = 1; v <= k; v++) {
            addEdge(id[i][v][0], id[i][v + 1][0]);
            addEdge(id[i][v + 1][1], id[i][v][1]);
        }
    }
    for (int i = 1; i < n; i++) {
        for (int v = 1; v <= k; v++) {
            addEdge(id[i + 1][v][0], id[i][v][0]);
            addEdge(id[i][v][1], id[i + 1][v][1]);
        }
    }
    for (int i = 1; i <= n; i++) {
        addEdge(id[i][1][0], id[i][1][1]);
        addEdge(id[i][k + 1][1], id[i][k + 1][0]);
    }
}

void notEqual(int i, int x) {
    addEdge(id[i][x][1], id[i][x + 1][1]);
    addEdge(id[i][x + 1][0], id[i][x][0]);
}

void lessEqual(int i, int j, int x) {
    for (int w = 1; w <= k; w++) {
        if (w >= x) {
            notEqual(i, w);
        } else {
            if (x - w + 1 <= k) {
                addEdge(id[i][w][1], id[j][x - w + 1][0]);
                addEdge(id[j][x - w + 1][1], id[i][w][0]);
            }
        }
    }
}

void moreEqual(int i, int j, int x) {
    for (int w = 1; w <= k; w++) {
        if (w <= x - k - 1) {
            notEqual(i, w);
        } else {
            if (x - w > 1) {
                addEdge(id[i][w + 1][0], id[j][x - w][1]);
                addEdge(id[j][x - w][0], id[i][w + 1][1]);
            }
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> t;
    for (int c = 1; c <= t; c++) {
        cin >> n >> m >> k;
        prepare();
        for (int i = 1, op, x, y, v; i <= m; i++) {
            cin >> op;
            if (op == 1) {
                cin >> x >> v;
                notEqual(x, v);
            } else if (op == 2) {
                cin >> x >> y >> v;
                lessEqual(x, y, v);
                swap(x, y);
                lessEqual(x, y, v);
            } else {
                cin >> x >> y >> v;
                moreEqual(x, y, v);
                swap(x, y);
                moreEqual(x, y, v);
            }
        }
        for (int i = 1; i <= cntt; i++) {
            if (dfn[i] == 0) {
                tarjan(i);
            }
        }
        bool check = true;
        for (int i = 1; i <= n; i++) {
            for (int v = 1; v <= k + 1; v++) {
                if (belong[id[i][v][0]] == belong[id[i][v][1]]) {
                    check = false;
                    break;
                }

    }

}

static void p8() {
    int n, p, f, m;
    cin >> n >> p >> f >> m;
    vector<pair<int,int>> req;
    for (int i = 0; i < n; i++) { int u, v; cin >> u >> v; req.push_back({u, v}); }
    vector<pair<int,int>> lr(p + 1);
    for (int i = 1; i <= p; i++) cin >> lr[i].first >> lr[i].second;
    vector<pair<int,int>> ban;
    for (int i = 0; i < m; i++) { int u, v; cin >> u >> v; ban.push_back({u, v}); }
    for (int ax = 1; ax <= f; ax++) {
        for (int mask = 0; mask < (1 << p); mask++) {
            vector<int> pick;
            for (int i = 1; i <= p; i++) if ((mask >> (i - 1)) & 1) pick.push_back(i);
            bool ok = true;
            // 签约电台必须含 ax
            for (int i : pick) if (lr[i].first > ax || lr[i].second < ax) { ok = false; break; }
            if (!ok) continue;
            for (auto &r : req) {
                bool in1 = (mask >> (r.first - 1)) & 1, in2 = (mask >> (r.second - 1)) & 1;
                if (!(in1 || in2)) { ok = false; break; }
            }
            if (!ok) continue;
            for (auto &b : ban) {
                bool in1 = (mask >> (b.first - 1)) & 1, in2 = (mask >> (b.second - 1)) & 1;
                if (in1 && in2) { ok = false; break; }
            }
            if (ok) {
                cout << pick.size() << " " << ax << "\n";
                for (int i : pick) cout << i << " ";
                cout << "\n";
                return;
            }
        }
    }
    cout << -1 << "\n";
}

#!/usr/bin/env python3
"""dui196.py：8 题 × N 种子对拍。课程程序 *.run vs brute196。
比较策略：题1/2/3/4/8 比对可行性 + 验证课程方案合法性；题5 数直比；题6/7 判词直比。

import subprocess, sys, os

def run(cmd, inp):
    p = subprocess.run(cmd, input=inp.encode(), capture_output=True)
    return p.stdout.decode().strip()

def norm_lines(s):
    return [l.strip() for l in s.strip().split('\n') if l.strip()]

def verify_solution(no, inp, out):
    """对课程输出做独立合法性验证（可行性 + 方案合法）。返回 True/False + 说明。"""
    lines = norm_lines(out)
    nums = inp.strip().split('\n')
    if no == 1:
        # POSSIBLE / IMPOSSIBLE + 赋值
        if lines[0] == 'IMPOSSIBLE': return True, 'IMPOSSIBLE'
        val = [int(x) for x in lines[1].split()]
        n = int(nums[0].split()[0])
        if len(val) != n: return False, f'赋值长度 {len(val)} ≠ n {n}'
        for l in nums[1:]:
            x1, v1, x2, v2 = map(int, l.split())
            b1 = val[x1 - 1] == v1; b2 = val[x2 - 1] == v2
            if not (b1 or b2): return False, f'子句不满足: {l}'
        return True, 'POSSIBLE'
    if no == 2:
        if lines[0] == 'NIE': return True, 'NIE'
        sel = [int(x) for x in lines]
        n = int(nums[0].split()[0])
        if len(sel) != n: return False, f'人数 {len(sel)} ≠ n {n}'
        for i in range(1, n + 1):
            if not (sel.count(2 * i - 1) == 1 or sel.count(2 * i) == 1):
                return False, f'团体 {i} 选择异常: {sel}'
        for l in nums[1:]:
            u, v = map(int, l.split())
            if u in sel and v in sel: return False, f'讨厌同席: {l}'
        return True, 'OK'
    if no == 3:
        if 'No solution.' in out: return True, 'No solution.'
        tasks = lines
        n = int(nums[0].split()[0])
        if len(tasks) != n: return False, f'任务数 {len(tasks)} ≠ n {n}'
        ages = list(map(int, nums[1].split()))
        s = sum(ages)
        for i in range(1, n + 1):
            older = ages[i - 1] * n >= s
            t = tasks[i - 1]
            if t == 'A' and not older: return False, f'{i} 年龄不够做 A'
            if t == 'B' and older: return False, f'{i} 年龄够却做 B'
        for l in nums[2:]:
            parts = l.split()
            if len(parts) < 2: continue
            x, y = int(parts[0]), int(parts[1])
            if x == 0 and y == 0: continue
            if tasks[x - 1] == tasks[y - 1]: return False, f'讨厌同任务: {l}'
        return True, 'OK'
    if no == 4:
        it_in = iter(nums)
        t = int(next(it_in))
        out_lines = iter(lines)
        for _ in range(t):
            n, m, k = map(int, next(it_in).split())
            cons = []
            for _c in range(m):
                parts = list(map(int, next(it_in).split()))
                cons.append(parts)
            out_line = next(out_lines)
            if out_line == '-1': continue
            arr = [int(x) for x in out_line.split()]
            if len(arr) != n: return False, f'arr 长 {len(arr)} ≠ n {n}'
            for i in range(n - 1):
                if arr[i] > arr[i + 1]: return False, f'非降破坏: {arr}'
            for v in arr:
                if v < 1 or v > k: return False, f'值越界: {v}'
            for parts in cons:
                if parts[0] == 1:
                    if arr[parts[1] - 1] == parts[2]: return False, f'不等于破坏: {parts}'
                elif parts[0] == 2:
                    if arr[parts[1] - 1] + arr[parts[2] - 1] > parts[3]: return False, f'≤破坏: {parts}'
                else:
                    if arr[parts[1] - 1] + arr[parts[2] - 1] < parts[3]: return False, f'≥破坏: {parts}'
        return True, 'OK'
    if no == 8:
        if lines[0] == '-1': return True, '-1'
        cnt, ax = map(int, lines[0].split())
        pick = [int(x) for x in lines[1].split()]
        if len(pick) != cnt: return False, f'数量 {len(pick)} ≠ {cnt}'
        it = iter(nums)
        n, p, f, m = map(int, next(it).split())
        lr = {}
        req = []
        for _ in range(n):
            u, v = map(int, next(it).split()); req.append((u, v))
        for i in range(1, p + 1):
            l, r = map(int, next(it).split()); lr[i] = (l, r)
        ban = []
        for _ in range(m):
            u, v = map(int, next(it).split()); ban.append((u, v))
        if ax < 1 or ax > f: return False, f'ansx 越界: {ax}'
        for i in pick:
            l, r = lr[i]
            if not (l <= ax <= r): return False, f'电台 {i} 不含 {ax}'
        for u, v in req:
            if u not in pick and v not in pick: return False, f'要求未满足: {u} {v}'
        for u, v in ban:
            if u in pick and v in pick: return False, f'限制被破: {u} {v}'
        return True, 'OK'
    return False, '未知题号'

def compare(no, brute_out, course_out):
    if no in (5, 6, 7):
        return brute_out.strip() == course_out.strip(), f'暴力={brute_out!r} 课程={course_out!r}'
    if no in (1, 2, 3, 4, 8):
        # 可行性比较：POSSIBLE/IMPOSSIBLE、NIE/方案、No solution./方案、-1/方案
        def feasible(s, no):
            if no == 1: return 'POSSIBLE' in s
            if no == 2: return s != 'NIE'
            if no == 3: return 'No solution.' not in s
            if no in (4, 8): return not (s.startswith('-1'))
        bf, cf = feasible(brute_out, no), feasible(course_out, no)
        if bf != cf:
            return False, f'可行性不一致 暴力={brute_out[:60]!r} 课程={course_out[:60]!r}'
        return True, f'可行性一致'
    return False, '?'
