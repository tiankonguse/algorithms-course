左程云
各种求解LCA的算法对比1，树上倍增，预处理的空间和时间O(n * log n)，单次查询O(log n)2，tarjan算法，预处理的空间和时间O(n + m)，单次均摊O(1)，只能处理批量离线任务3，重链剖分，预处理的空间和时间O(n)，单次查询O(log n)4，欧拉序结合倍增表，预处理的空间和时间O(n * log n)，单次查询O(1)5，dfn序结合倍增表，预处理的空间和时间O(n * log n)，单次查询O(1)，常数是欧拉序的一半以上总结都是针对静态树动态树求最低公共祖先，LCT是经典方案之一，后续课程会安排讲述这个数据结构

#include <bits/stdc++.h>

using namespace std;

const int MAXN = 500001;
const int MAXP = 20;
int n, m, root;

int headg[MAXN];
int nextg[MAXN << 1];
int tog[MAXN << 1];
int cntg;

int first[MAXN];
int lg2[MAXN << 1];
int rmq[MAXN << 1][MAXP];
int cntEuler;

void addEdge(int u, int v) {
    nextg[++cntg] = headg[u];
    tog[cntg] = v;
    headg[u] = cntg;
}

int getUp(int x, int y) {
    return first[x] < first[y] ? x : y;
}

void dfs(int u, int fa) {
    first[u] = ++cntEuler;
    rmq[first[u]][0] = u;
    for (int e = headg[u]; e > 0; e = nextg[e]) {
        int v = tog[e];
        if (v != fa) {
            dfs(v, u);
            rmq[++cntEuler][0] = u;
        }
    }
}

void buildRmq() {
    dfs(root, 0);
    for (int i = 2; i <= cntEuler; i++) {
        lg2[i] = lg2[i >> 1] + 1;
    }
    for (int pre = 0, cur = 1; cur <= lg2[cntEuler]; pre++, cur++) {
        for (int i = 1; i + (1 << cur) - 1 <= cntEuler; i++) {
            rmq[i][cur] = getUp(rmq[i][pre], rmq[i + (1 << pre)][pre]);
        }
    }
}

int getLCA(int x, int y) {
    x = first[x];
    y = first[y];
    if (x > y) {
        swap(x, y);
    }
    int k = lg2[y - x + 1];
    return getUp(rmq[x][k], rmq[y - (1 << k) + 1][k]);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m >> root;
    for (int i = 1, u, v; i < n; i++) {
        cin >> u >> v;
        addEdge(u, v);
        addEdge(v, u);
    }
    buildRmq();
    for (int i = 1, u, v; i <= m; i++) {
        cin >> u >> v;
        cout << getLCA(u, v) << '\n';
    }
    return 0;
}

#include <bits/stdc++.h>

using namespace std;

const int MAXN = 500001;
const int MAXP = 20;
int n, m, root;

int headg[MAXN];
int nextg[MAXN << 1];
int tog[MAXN << 1];
int cntg;

int dfn[MAXN];
int lg2[MAXN];
int rmq[MAXN][MAXP];
int cntDfn;

void addEdge(int u, int v) {
    nextg[++cntg] = headg[u];
    tog[cntg] = v;
    headg[u] = cntg;
}

int getUp(int x, int y) {
    return dfn[x] < dfn[y] ? x : y;
}

void dfs(int u, int fa) {
    dfn[u] = ++cntDfn;
    rmq[dfn[u]][0] = fa;
    for (int e = headg[u]; e > 0; e = nextg[e]) {
        int v = tog[e];
        if (v != fa) {
            dfs(v, u);
        }
    }
}

void buildRmq() {
    dfs(root, 0);
    for (int i = 2; i <= n; i++) {
        lg2[i] = lg2[i >> 1] + 1;
    }
    for (int pre = 0, cur = 1; cur <= lg2[n]; pre++, cur++) {
        for (int i = 1; i + (1 << cur) - 1 <= n; i++) {
            rmq[i][cur] = getUp(rmq[i][pre], rmq[i + (1 << pre)][pre]);
        }
    }
}

int getLCA(int x, int y) {
    if (x == y) {
        return x;
    }
    x = dfn[x];
    y = dfn[y];
    if (x > y) {
        swap(x, y);
    }
    x++;
    int k = lg2[y - x + 1];
    return getUp(rmq[x][k], rmq[y - (1 << k) + 1][k]);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m >> root;
    for (int i = 1, u, v; i < n; i++) {
        cin >> u >> v;
        addEdge(u, v);
        addEdge(v, u);
    }
    buildRmq();
    for (int i = 1, u, v; i <= m; i++) {
        cin >> u >> v;
        cout << getLCA(u, v) << '\n';
    }
    return 0;
}

code/brute.cpp

#include <bits/stdc++.h>
using namespace std;
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m, root;
    cin >> n >> m >> root;
    vector<vector<int>> adj(n + 1);
    for (int i = 1, u, v; i < n; i++) {
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    vector<int> dep(n + 1), par(n + 1);
    function<void(int, int)> dfs = [&](int u, int fa) {
        par[u] = fa;
        for (int v : adj[u]) if (v != fa) { dep[v] = dep[u] + 1; dfs(v, u); }
    };
    dfs(root, 0);
    auto lca = [&](int u, int v) -> int {
        if (u == v) return u;
        if (dep[u] < dep[v]) swap(u, v);
        while (dep[u] > dep[v]) u = par[u];
        while (u != v) { u = par[u]; v = par[v]; }
        return u;
    };
    for (int i = 1, u, v; i <= m; i++) {
        cin >> u >> v;
        cout << lca(u, v) << '\n';
    }
    return 0;
}

code/gen.py

#!/usr/bin/env python3
# 生成随机树 + 随机 LCA 查询（P3379 格式：n m root, n-1 边, m 查询）
import random, sys

seed = int(sys.argv[1])
r = random.Random(seed)
n = r.randint(2, 60)
m = r.randint(1, 80)
root = r.randint(1, n)
out = [f"{n} {m} {root}"]
par = [0] * (n + 1)
edges = []
for i in range(2, n + 1):
    p = r.randint(1, i - 1)
    par[i] = p
    edges.append((p, i))
r.shuffle(edges)
for u, v in edges:
    out.append(f"{u} {v}")
# 树邻接 + 深度（供暴力）
adj = [[] for _ in range(n + 1)]
for u, v in edges:
    adj[u].append(v); adj[v].append(u)
dep = [0] * (n + 1)
def dfs(u, fa):
    for v in adj[u]:
        if v != fa:
            dep[v] = dep[u] + 1
            dfs(v, u)
dfs(root, 0)
for _ in range(m):
    u = r.randint(1, n)
    v = r.randint(1, n)
    out.append(f"{u} {v}")
print('\n'.join(out))
